1"""Train the intent classifier: python -m training.train_intents 

2 

3Writes agent/models/intent_nb.json and eval/reports/intents.md with: 

4 * 5-fold cross-validation on eval/intents/train.yaml (logistic regression vs naive Bayes baseline), 

5 * accuracy on the held-out eval/intents/test.yaml for the classifier alone and for the full router 

6 (guardrails + rules + classifier), which is what runs in production. 

7Runs in seconds on any laptop. Extend the dataset (or augment it with training/augment_with_llm.py) and retrain. 

8""" 

9 

10from __future__ import annotations 

11 

12import argparse 

13import random 

14from collections import Counter, defaultdict 

15from pathlib import Path 

16 

17import yaml 

18 

19from agent.config import REPO_ROOT, SERVICE_ROOT 

20from agent.nlu.classifier import IntentModel, NaiveBayesBaseline 

21from agent.nlu.router import IntentRouter 

22 

23 

24def load_samples(path: Path) -> list[tuple[str, str]]: 

25 data = yaml.safe_load(path.read_text(encoding="utf-8")) 

26 return [(text, label) for label, texts in data.items() for text in texts] 

27 

28 

29def cross_validate(samples: list[tuple[str, str]], folds: int = 5, seed: int = 1) -> tuple[float, float]: 

30 idx = list(range(len(samples))) 

31 random.Random(seed).shuffle(idx) 

32 parts = [idx[i::folds] for i in range(folds)] 

33 lr_ok = nb_ok = 0 

34 for part in parts: 

35 held = set(part) 

36 train = [samples[i] for i in idx if i not in held] 

37 lr, nb = IntentModel.train(train), NaiveBayesBaseline(train) 

38 for i in part: 

39 text, label = samples[i] 

40 lr_ok += lr.predict(text).intent == label 

41 nb_ok += nb.predict(text) == label 

42 return lr_ok / len(samples), nb_ok / len(samples) 

43 

44 

45def evaluate(model: IntentModel, test: list[tuple[str, str]]) -> tuple[float, float, dict[str, Counter[str]]]: 

46 router = IntentRouter(model, 0.3) 

47 confusion: dict[str, Counter[str]] = defaultdict(Counter) 

48 clf_ok = router_ok = 0 

49 for text, label in test: 

50 clf_ok += model.predict(text).intent == label 

51 got = router.route(text).intent 

52 confusion[label][got] += 1 

53 router_ok += got == label 

54 return clf_ok / len(test), router_ok / len(test), confusion 

55 

56 

57def main(argv: list[str] | None = None) -> int: 

58 ap = argparse.ArgumentParser() 

59 ap.add_argument("--train", default=str(REPO_ROOT / "eval" / "intents" / "train.yaml")) 

60 ap.add_argument("--test", default=str(REPO_ROOT / "eval" / "intents" / "test.yaml")) 

61 ap.add_argument("--out", default=str(SERVICE_ROOT / "agent" / "models" / "intent_model.json")) 

62 ap.add_argument("--report", default=str(REPO_ROOT / "eval" / "reports" / "intents.md")) 

63 ap.add_argument("--skip-cv", action="store_true") 

64 args = ap.parse_args(argv) 

65 

66 samples = load_samples(Path(args.train)) 

67 test = load_samples(Path(args.test)) 

68 model = IntentModel.train(samples) 

69 Path(args.out).parent.mkdir(parents=True, exist_ok=True) 

70 model.save(Path(args.out)) 

71 clf_acc, router_acc, confusion = evaluate(model, test) 

72 labels = sorted({lbl for _, lbl in samples}) 

73 lines = [ 

74 "# Intent classifier report", 

75 "", 

76 f"Train: {len(samples)} phrases, {len(labels)} intents · held-out test: {len(test)} phrases · " 

77 f"model: softmax logistic regression, {model.meta['vocab']} features (stems + char 3–4-grams)", 

78 "", 

79 ] 

80 if not args.skip_cv: 

81 lr_cv, nb_cv = cross_validate(samples) 

82 lines += [ 

83 "| metric | value |", 

84 "|---|---|", 

85 f"| 5-fold CV, logistic regression | {lr_cv:.1%} |", 

86 f"| 5-fold CV, naive Bayes baseline | {nb_cv:.1%} |", 

87 ] 

88 print(f"cv: lr={lr_cv:.3f} nb={nb_cv:.3f}") 

89 else: 

90 lines += ["| metric | value |", "|---|---|"] 

91 lines += [ 

92 f"| held-out test, classifier only | {clf_acc:.1%} |", 

93 f"| held-out test, full router (guardrails + rules + classifier) | **{router_acc:.1%}** |", 

94 "", 

95 "## Router confusion on the held-out set", 

96 "", 

97 "| intent | n | correct | routed instead |", 

98 "|---|---|---|---|", 

99 ] 

100 for lbl in sorted(confusion): 

101 row = confusion[lbl] 

102 wrong = [(k, v) for k, v in row.most_common() if k != lbl] 

103 lines.append( 

104 f"| {lbl} | {sum(row.values())} | {row[lbl]} | {', '.join(f'{k} ({v})' for k, v in wrong) or '—'} |" 

105 ) 

106 lines += [ 

107 "", 

108 "Write scenarios (S3, S4, approvals) are matched by high-precision rules before the classifier, and " 

109 "refusals by deterministic guardrails, so classifier mistakes mostly land in `kb_question` " 

110 "(an answer from the knowledge base or an escalation draft) — never in a write action.", 

111 ] 

112 Path(args.report).parent.mkdir(parents=True, exist_ok=True) 

113 Path(args.report).write_text("\n".join(lines) + "\n", encoding="utf-8") 

114 print(f"test: classifier={clf_acc:.3f} router={router_acc:.3f}\nmodel → {args.out}\nreport → {args.report}") 

115 return 0 

116 

117 

118if __name__ == "__main__": # pragma: no cover 

119 raise SystemExit(main())
