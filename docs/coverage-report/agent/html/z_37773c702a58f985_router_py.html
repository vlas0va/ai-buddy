208class Route: 

209 intent: str

217def detect_other_person(original: str, own_name: str = "") -> str | None: 

218 """Return the detected foreign person (or 'colleague') when the user asks about another person's data/actions.""" 

219 own_first = (own_name.split() or [""])[0][:4].lower() 

220 for rx in (OTHER_ACTION, OTHER_DATA): 

221 for m in rx.finditer(original): 

222 target = next((g for g in reversed(m.groups()) if g and g.strip()), "") 

223 word = target.split()[-1] if target else "" 

224 if word[:1].isupper(): 

225 if word in NOT_NAMES or (own_first and word[:4].lower() == own_first): 

226 continue 

227 return word 

228 return "colleague" 

229 return None

232def guard(text: str, original: str, own_name: str = "") -> Route | None: 

233 if INJECTION.search(original): 

234 return Route("refusal", 1.0, "guardrail", reason="prompt_injection") 

235 if SENSITIVE.search(text): 

236 return Route("refusal", 1.0, "guardrail", reason="sensitive_data", redirect_role="hrbp") 

237 if HIRING_FIRING.search(text): 

238 return Route("refusal", 1.0, "guardrail", reason="out_of_scope", redirect_role="hrbp", detail={"topic": "hr"}) 

239 if TRIP_OPS.search(text): 

240 return Route("refusal", 1.0, "guardrail", reason="out_of_scope", redirect_role="lead", detail={"topic": "trip"}) 

241 if BENEFIT_OPS.search(text): # no money operations, but explain how to do it yourself 

242 return Route("benefit", 1.0, "guardrail", detail={"topic": "benefit"}) 

243 other = detect_other_person(original, own_name) 

244 if other: 

245 return Route("refusal", 1.0, "guardrail", reason="no_rights", detail={"person": other}) 

246 if HUMAN_ONLY.search(text): 

247 return Route("human_only", 1.0, "guardrail") 

248 if DELEGATE_WORK.search(text): # "сделай за меня" — coach, do not do the intern's work 

249 return Route("learning", 1.0, "guardrail") 

250 if looks_like_log(original): 

251 return Route("troubleshoot", 0.95, "rule") 

252 return None

255class IntentRouter: 

256 def __init__(self, model: IntentModel | None, min_confidence: float = 0.4) -> None: 

257 self.model = model 

258 self.min_confidence = min_confidence 

259 

260 def route(self, message: str, own_name: str = "") -> Route: 

261 text = normalize(message) 

262 if not text: 

263 return Route("help", 1.0, "rule") 

264 g = guard(text, message, own_name) 

265 if g: 

266 return g 

267 for intent, rx in RULES: 

268 if rx.search(text): 

269 if intent in ("absence", "access", "channels") and self._is_info_question(text, intent): 

270 return Route("kb_question", 0.9, "rule", detail={"topic": intent}) 

271 return Route(intent, 0.95, "rule") 

272 if self.model is not None: 

273 p = self.model.predict(text) 

274 if p.confidence >= self.min_confidence: 

275 if p.intent == "escalate" and not ESCALATE_VERB.search(text): 

276 return Route("kb_question", p.confidence, "classifier", detail={"ranking": p.ranking}) 

277 if p.intent in WRITE_INTENTS and p.confidence < 0.6: 

278 # write scenarios start from the classifier only when it is confident; rules catch the usual phrasing 

279 return Route("kb_question", p.confidence, "classifier", detail={"ranking": p.ranking}) 

280 return Route(p.intent, p.confidence, "classifier", detail={"ranking": p.ranking}) 

281 ranking = self.model.predict(text).ranking if self.model is not None else [] 

282 return Route("kb_question", 0.3, "fallback", detail={"ranking": ranking})

284 @staticmethod 

285 def _is_info_question(text: str, intent: str) -> bool: 

286 """«Сколько дней отпуска у стажёра?» → knowledge question, not a request flow.""" 

287 if intent == "absence": 

288 return bool(re.search(r"(больничн|заболел)", text)) 

289 if intent == "access": 

290 # «как склонировать репозиторий?» is engineering onboarding (KB), not an access request 

291 return bool( 

292 re.search(r"(клонир|ssh|ветк|pull\s+request|пулл?\s*реквест|коммит|запуш|структур)", text) 

293 ) and not (re.search(r"доступ|права\s+на|запроси|оформи|выдай", text)) 

294 return False
