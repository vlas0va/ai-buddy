type Face = 'open' | 'wink' | 'glasses' | 'happy'
type SmallFace = 'smile' | 'wow' | 'wink'
type Accessory = 'sprout' | 'question' | 'antenna' | 'headset' | 'star'
 
interface Variant {
  big: Face
  small: SmallFace

}

export function Logo({ size = 32, variant = BUDDY_VARIANT }: { size?: number; variant?: string }) {

  return (

  )
}
