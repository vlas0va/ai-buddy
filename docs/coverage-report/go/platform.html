// Package apierr is the error type shared by the HTTP API (contracts/openapi.yaml, schema Error) and the MCP tool
// results (contracts/mcp-tools.md, "Errors"). Messages are Russian, user-facing, neutral; they never reveal whether
// a foreign resource exists (R71).
package apierr

import (
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "log/slog"
        "net/http"
)

// Code is a machine-readable error code.
type Code string

// Codes of the HTTP API (openapi Error.code).
const (
        CodeUnauthorized        Code = "unauthorized"
        CodeForbidden           Code = "forbidden"
        CodeNotFound            Code = "not_found"
        CodeValidation          Code = "validation_error"
        CodeConflict            Code = "conflict"
        CodeExpired             Code = "expired"
        CodeRateLimited         Code = "rate_limited"
        CodeUpstreamUnavailable Code = "upstream_unavailable"
        CodeInternal            Code = "internal"
)

// Codes that exist only in MCP tool results (forbidden, not_found, validation_error and conflict are shared).
const (
        CodeInvalidConfirmation Code = "invalid_confirmation"
        CodeUnavailable         Code = "unavailable"
)

// Error is an API or tool error. Status is the HTTP status used by WriteJSON; it is not serialised.
type Error struct {
        Code      Code   `json:"code"`
        Message   string `json:"message"`
        Retryable bool   `json:"retryable"`
        Status    int    `json:"-"`
}

type spec struct {
        status    int
        retryable bool
        message   string
}

var specs = map[Code]spec{
        CodeUnauthorized:        {http.StatusUnauthorized, false, "Требуется вход в систему"},
        CodeForbidden:           {http.StatusForbidden, false, "Недостаточно прав для этого действия"},
        CodeNotFound:            {http.StatusNotFound, false, "Не найдено"},
        CodeValidation:          {http.StatusBadRequest, false, "Проверьте введённые данные"},
        CodeConflict:            {http.StatusConflict, false, "Действие уже выполнено или недоступно в текущем состоянии"},
        CodeExpired:             {http.StatusConflict, false, "Срок действия истёк, повторите действие"},
        CodeRateLimited:         {http.StatusTooManyRequests, true, "Слишком много запросов, попробуйте чуть позже"},
        CodeUpstreamUnavailable: {http.StatusBadGateway, true, "Сервис временно недоступен, попробуйте ещё раз"},
        CodeInternal:            {http.StatusInternalServerError, false, "Внутренняя ошибка, попробуйте позже"},
        CodeInvalidConfirmation: {http.StatusForbidden, false, "Подтверждение недействительно или устарело, подтвердите действие ещё раз"},
        CodeUnavailable:         {http.StatusServiceUnavailable, true, "Сервис временно недоступен, попробуйте ещё раз"},
}

// New returns an error with the given code; an empty message means the default Russian message of the code.
// Status and Retryable come from the code (unknown codes: 500, not retryable).
func New(code Code, message string) *Error {
        s, ok := specs[code]
        if !ok {
                s = spec{status: http.StatusInternalServerError, message: specs[CodeInternal].message}
        }
        if message == "" {
                message = s.message
        }
        return &Error{Code: code, Message: message, Retryable: s.retryable, Status: s.status}
}

// Unauthorized returns the error for a missing or invalid JWT (401).
func Unauthorized() *Error { return New(CodeUnauthorized, "") }

// Forbidden returns the error for an action or resource the caller may not access, including other users' data (403).
func Forbidden() *Error { return New(CodeForbidden, "") }

// NotFound returns the error for a missing resource of the caller (404).
func NotFound() *Error { return New(CodeNotFound, "") }

// Validation returns the error for invalid request data or tool arguments (400); usually followed by WithMessage.
func Validation() *Error { return New(CodeValidation, "") }

// Conflict returns the error for a state conflict, e.g. an already decided approval (409).
func Conflict() *Error { return New(CodeConflict, "") }

// Expired returns the error for an expired or cancelled proposal (409).
func Expired() *Error { return New(CodeExpired, "") }

// RateLimited returns the error for too many requests (429, retryable).
func RateLimited() *Error { return New(CodeRateLimited, "") }

// UpstreamUnavailable returns the error for an agent or MCP server that did not answer (502, retryable).
func UpstreamUnavailable() *Error { return New(CodeUpstreamUnavailable, "") }

// Internal returns the error for an unexpected server failure (500).
func Internal() *Error { return New(CodeInternal, "") }

// InvalidConfirmation returns the tool error for a write without a valid confirmation token (403 if sent over HTTP).
func InvalidConfirmation() *Error { return New(CodeInvalidConfirmation, "") }

// Unavailable returns the tool error for an MCP server whose dependency is down (503 if sent over HTTP, retryable).
func Unavailable() *Error { return New(CodeUnavailable, "") }

// Error implements error.
func (e *Error) Error() string { return fmt.Sprintf("%s: %s", e.Code, e.Message) }

// Is reports whether target is an *Error with the same code, so errors.Is(err, apierr.Forbidden()) works.
func (e *Error) Is(target error) bool {
        t, ok := target.(*Error)
        return ok && t.Code == e.Code
}

// WithMessage returns a copy of e with a different user-facing message.
func (e *Error) WithMessage(msg string) *Error {
        c := *e
        c.Message = msg
        return &c
}

// As returns the *Error in err's chain, if any.
func As(err error) (*Error, bool) {
        var e *Error
        if errors.As(err, &e) && e != nil {
                return e, true
        }
        return nil, false
}

// From converts any error to an *Error: an *Error in the chain is returned as is, context deadline or
// cancellation becomes UpstreamUnavailable, anything else becomes Internal. From(nil) is nil.
func From(err error) *Error {
        if err == nil {
                return nil
        }
        if e, ok := As(err); ok {
                return e
        }
        if errors.Is(err, context.DeadlineExceeded) || errors.Is(err, context.Canceled) {
                return UpstreamUnavailable()
        }
        return Internal()
}

// HasCode reports whether err is (or wraps) an *Error with the given code.
func HasCode(err error, code Code) bool {
        e, ok := As(err)
        return ok && e.Code == code
}

// ToAPI maps MCP-only codes onto openapi Error.code values, keeping message and retryable:
// unavailable → upstream_unavailable (502), invalid_confirmation → expired (409). Other errors are returned as is.
func ToAPI(e *Error) *Error {
        switch e.Code {
        case CodeUnavailable:
                return remap(e, CodeUpstreamUnavailable)
        case CodeInvalidConfirmation:
                return remap(e, CodeExpired)
        }
        if _, ok := specs[e.Code]; !ok {
                return Internal()
        }
        return e
}

// ToMCP maps API-only codes onto MCP tool error codes, keeping message and retryable:
// unauthorized → forbidden, expired → invalid_confirmation, rate_limited/upstream_unavailable/internal and
// unknown codes → unavailable. Shared codes are returned as is.
func ToMCP(e *Error) *Error {
        switch e.Code {
        case CodeForbidden, CodeNotFound, CodeValidation, CodeConflict, CodeInvalidConfirmation, CodeUnavailable:
                return e
        case CodeUnauthorized:
                return remap(e, CodeForbidden)
        case CodeExpired:
                return remap(e, CodeInvalidConfirmation)
        default:
                return remap(e, CodeUnavailable)
        }
}

func remap(e *Error, code Code) *Error {
        c := *e
        c.Code = code
        c.Status = specs[code].status
        return &c
}

// WriteJSON writes err as the openapi Error body with its HTTP status (MCP-only codes mapped by ToAPI).
// Errors that are not *Error are logged with slog and answered as Internal, so details never leak to clients.
func WriteJSON(w http.ResponseWriter, err error) {
        e, ok := As(err)
        if !ok {
                e = From(err)
                if err != nil && e.Code == CodeInternal {
                        slog.Default().Error("internal error", "err", err)
                }
        }
        if e == nil {
                e = Internal()
        }
        e = ToAPI(e)
        status := e.Status
        if status == 0 {
                status = New(e.Code, "").Status
        }
        body, _ := json.Marshal(e) // fields are strings and a bool: cannot fail
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.Header().Set("X-Content-Type-Options", "nosniff")
        w.WriteHeader(status)
        _, _ = w.Write(body)
}

// Package authn issues and verifies the HS256 JWTs of the mock IdP and puts verified claims into the request
// context. Identity used anywhere in the backend comes only from these claims (R71).
//
// Token: base64url (no padding) of header {"alg":"HS256","typ":"JWT"}, payload Claims and the HMAC-SHA256 of
// "<header>.<payload>" with JWT_SECRET. Verification is strict: exactly HS256 and typ JWT (alg "none" and every
// other algorithm are rejected, as is a "crit" header), constant-time MAC comparison, exp required,
// sub and role required, iat not in the future.
package authn

import (
        "context"
        "crypto/hmac"
        "crypto/sha256"
        "encoding/base64"
        "encoding/json"
        "errors"
        "fmt"
        "log/slog"
        "net/http"
        "strings"
        "time"

        "aibuddy/services/internal/platform/apierr"
)

// DefaultLeeway is the clock skew tolerated for exp and iat by Verify and NewVerifier.
const DefaultLeeway = 30 * time.Second

// Claims are the JWT claims of a user. Times are Unix seconds.
type Claims struct {
        Sub         string `json:"sub"`
        Role        string `json:"role"`
        TeamID      string `json:"team_id"`
        CohortStart string `json:"cohort_start,omitempty"` // YYYY-MM-DD, empty for users without a cohort
        Iat         int64  `json:"iat,omitempty"`
        Exp         int64  `json:"exp"`
}

// Verification errors. Middleware answers all of them with the same 401 and does not tell the client which one.
var (
        ErrNoSecret      = errors.New("authn: empty secret")
        ErrMalformed     = errors.New("authn: malformed token")
        ErrAlgorithm     = errors.New("authn: unsupported token header")
        ErrSignature     = errors.New("authn: invalid signature")
        ErrExpired       = errors.New("authn: token expired")
        ErrNotYetValid   = errors.New("authn: token issued in the future")
        ErrMissingClaims = errors.New("authn: required claim missing")
)

var (
        b64        = base64.RawURLEncoding
        b64Strict  = base64.RawURLEncoding.Strict()
        headerPart = b64.EncodeToString([]byte(`{"alg":"HS256","typ":"JWT"}`))
)

// Issue signs c with Iat = now and Exp = now + ttl. ttl must be positive.
func Issue(secret []byte, c Claims, ttl time.Duration) (string, error) {
        if ttl <= 0 {
                return "", fmt.Errorf("authn: issue: ttl must be positive, got %s", ttl)
        }
        now := time.Now()
        c.Iat = now.Unix()
        c.Exp = now.Add(ttl).Unix()
        return Sign(secret, c)
}

// Sign signs c exactly as given (tests use it to build expired tokens). Sub, Role and Exp are required.
func Sign(secret []byte, c Claims) (string, error) {
        if len(secret) == 0 {
                return "", ErrNoSecret
        }
        if c.Sub == "" || c.Role == "" || c.Exp == 0 {
                return "", ErrMissingClaims
        }
        payload, err := json.Marshal(c)
        if err != nil {
                return "", fmt.Errorf("authn: marshal claims: %w", err)
        }
        signingInput := headerPart + "." + b64.EncodeToString(payload)
        return signingInput + "." + b64.EncodeToString(mac(secret, signingInput)), nil
}

// Verify checks token at time now with DefaultLeeway and returns its claims.
func Verify(secret []byte, token string, now time.Time) (Claims, error) {
        return verify(secret, token, now, DefaultLeeway)
}

func verify(secret []byte, token string, now time.Time, leeway time.Duration) (Claims, error) {
        if len(secret) == 0 {
                return Claims{}, ErrNoSecret
        }
        parts := strings.Split(token, ".")
        if len(parts) != 3 || parts[0] == "" || parts[1] == "" || parts[2] == "" {
                return Claims{}, ErrMalformed
        }
        // A map, not a struct: encoding/json matches struct fields case-insensitively ("ALG" would pass).
        var h map[string]any
        if err := decodePart(parts[0], &h); err != nil {
                return Claims{}, err
        }
        if _, crit := h["crit"]; h["alg"] != "HS256" || h["typ"] != "JWT" || crit {
                return Claims{}, ErrAlgorithm
        }
        sig, err := b64Strict.DecodeString(parts[2])
        if err != nil {
                return Claims{}, ErrMalformed
        }
        if !hmac.Equal(sig, mac(secret, parts[0]+"."+parts[1])) {
                return Claims{}, ErrSignature
        }
        var c Claims
        if err := decodePart(parts[1], &c); err != nil {
                return Claims{}, err
        }
        if c.Sub == "" || c.Role == "" || c.Exp == 0 {
                return Claims{}, ErrMissingClaims
        }
        if !now.Before(time.Unix(c.Exp, 0).Add(leeway)) {
                return Claims{}, ErrExpired
        }
        if c.Iat != 0 && time.Unix(c.Iat, 0).After(now.Add(leeway)) {
                return Claims{}, ErrNotYetValid
        }
        return c, nil
}

func decodePart(part string, dst any) error {
        raw, err := b64Strict.DecodeString(part)
        if err != nil {
                return ErrMalformed
        }
        if err := json.Unmarshal(raw, dst); err != nil {
                return ErrMalformed
        }
        return nil
}

func mac(secret []byte, signingInput string) []byte {
        m := hmac.New(sha256.New, secret)
        m.Write([]byte(signingInput))
        return m.Sum(nil)
}

// Verifier verifies tokens with a fixed secret, leeway and clock; it is safe for concurrent use.
type Verifier struct {
        Secret []byte
        Leeway time.Duration    // tolerated clock skew; NewVerifier sets DefaultLeeway
        Now    func() time.Time // nil means time.Now
}

// NewVerifier returns a Verifier with DefaultLeeway and the system clock.
func NewVerifier(secret []byte) *Verifier {
        return &Verifier{Secret: secret, Leeway: DefaultLeeway, Now: time.Now}
}

// Verify checks token and returns its claims.
func (v *Verifier) Verify(token string) (Claims, error) {
        now := time.Now
        if v.Now != nil {
                now = v.Now
        }
        return verify(v.Secret, token, now(), v.Leeway)
}

// Middleware requires a valid `Authorization: Bearer <JWT>`; on success it stores the claims and the raw token
// in the request context (FromContext, TokenFromContext), otherwise it answers apierr 401 with
// `WWW-Authenticate: Bearer`.
func (v *Verifier) Middleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                token, ok := BearerToken(r)
                if !ok {
                        unauthorized(w)
                        return
                }
                c, err := v.Verify(token)
                if err != nil {
                        slog.Default().DebugContext(r.Context(), "jwt rejected", "err", err)
                        unauthorized(w)
                        return
                }
                next.ServeHTTP(w, r.WithContext(WithToken(NewContext(r.Context(), c), token)))
        })
}

// Middleware is NewVerifier(secret).Middleware.
func Middleware(secret []byte) func(http.Handler) http.Handler {
        return NewVerifier(secret).Middleware
}

func unauthorized(w http.ResponseWriter) {
        w.Header().Set("WWW-Authenticate", "Bearer")
        apierr.WriteJSON(w, apierr.Unauthorized())
}

// BearerToken returns the token of an `Authorization: Bearer <token>` header (scheme case-insensitive).
func BearerToken(r *http.Request) (string, bool) {
        scheme, token, ok := strings.Cut(r.Header.Get("Authorization"), " ")
        if !ok || !strings.EqualFold(scheme, "Bearer") {
                return "", false
        }
        token = strings.TrimSpace(token)
        return token, token != ""
}

type claimsKey struct{}

type tokenKey struct{}

// NewContext returns ctx carrying c (Middleware does this; tests of handlers may call it directly).
func NewContext(ctx context.Context, c Claims) context.Context {
        return context.WithValue(ctx, claimsKey{}, c)
}

// FromContext returns the verified claims stored by Middleware or NewContext.
func FromContext(ctx context.Context) (Claims, bool) {
        c, ok := ctx.Value(claimsKey{}).(Claims)
        return c, ok
}

// TokenFromContext returns the raw bearer token accepted by Middleware, for forwarding to the agent and MCP servers.
func TokenFromContext(ctx context.Context) (string, bool) {
        t, ok := ctx.Value(tokenKey{}).(string)
        return t, ok && t != ""
}

// WithToken returns ctx carrying a raw bearer token, for code paths that call MCP without an HTTP request
// (e.g. tests or core acting on behalf of a user with a token it issued).
func WithToken(ctx context.Context, token string) context.Context {
        return context.WithValue(ctx, tokenKey{}, token)
}

// Package canonjson produces canonical JSON: the byte-exact form that core, the MCP servers and the Python agent
// hash and sign (contracts/mcp-tools.md: "keys sorted, no whitespace, UTF-8").
//
// The output is identical to Python's
//
//        json.dumps(json.loads(text), sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")
//
// for every value made of objects, arrays, strings, integers, booleans and null:
//   - object keys are sorted by Unicode code point (byte order of UTF-8, which is the same order);
//     duplicate keys in JSON text keep the last value, like Python;
//   - no whitespace between tokens;
//   - strings are raw UTF-8; only '"', '\\' and control characters U+0000–U+001F are escaped:
//     \b \f \n \r \t use the short form, the rest \u00XX with lowercase hex; '/', '<', '>', '&', U+007F,
//     U+2028 and U+2029 are written as is;
//   - invalid UTF-8 in Go strings is replaced by U+FFFD (as encoding/json does).
//
// Numbers: number text is preserved as given (json.Number, json.RawMessage) or as encoding/json formats Go
// numbers. Integers therefore match Python exactly (any size); the only rewrite is "-0" → "0", as in Python.
// Non-integer numbers are NOT normalised: Python re-formats floats with repr ("1.0", "1e+16", 1e2 → "100.0"),
// Go keeps "1", "1e2" or whatever the producer wrote. Do not put non-integer numbers into signed arguments;
// the current write-tool schemas contain none.
//
// Invariant: Marshal(v) == Marshal(json.RawMessage(json.Marshal(v))) for every v encoding/json accepts.
package canonjson

import (
        "bytes"
        "crypto/sha256"
        "encoding/hex"
        "encoding/json"
        "fmt"
        "slices"
)

// Marshal returns the canonical JSON encoding of v. v may be any value accepted by encoding/json (maps, structs
// with json tags, slices, json.Marshaler implementations, json.RawMessage, json.Number). A []byte is encoded as a
// base64 string, as encoding/json does; wrap JSON text in json.RawMessage instead.
func Marshal(v any) ([]byte, error) {
        raw, err := json.Marshal(v)
        if err != nil {
                return nil, fmt.Errorf("canonjson: marshal: %w", err)
        }
        dec := json.NewDecoder(bytes.NewReader(raw))
        dec.UseNumber()
        var tree any
        if err := dec.Decode(&tree); err != nil {
                return nil, fmt.Errorf("canonjson: decode: %w", err)
        }
        var buf bytes.Buffer
        buf.Grow(len(raw))
        if err := encode(&buf, tree); err != nil {
                return nil, err
        }
        return buf.Bytes(), nil
}

// SHA256Hex returns the lowercase hex SHA-256 of Marshal(v), the form of `args_hash` and `draft_hash`.
func SHA256Hex(v any) (string, error) {
        b, err := Marshal(v)
        if err != nil {
                return "", err
        }
        sum := sha256.Sum256(b)
        return hex.EncodeToString(sum[:]), nil
}

func encode(buf *bytes.Buffer, v any) error {
        switch x := v.(type) {
        case nil:
                buf.WriteString("null")
        case bool:
                if x {
                        buf.WriteString("true")
                } else {
                        buf.WriteString("false")
                }
        case json.Number:
                if x == "-0" {
                        x = "0"
                }
                buf.WriteString(string(x))
        case string:
                writeString(buf, x)
        case []any:
                buf.WriteByte('[')
                for i, e := range x {
                        if i > 0 {
                                buf.WriteByte(',')
                        }
                        if err := encode(buf, e); err != nil {
                                return err
                        }
                }
                buf.WriteByte(']')
        case map[string]any:
                keys := make([]string, 0, len(x))
                for k := range x {
                        keys = append(keys, k)
                }
                slices.Sort(keys)
                buf.WriteByte('{')
                for i, k := range keys {
                        if i > 0 {
                                buf.WriteByte(',')
                        }
                        writeString(buf, k)
                        buf.WriteByte(':')
                        if err := encode(buf, x[k]); err != nil {
                                return err
                        }
                }
                buf.WriteByte('}')
        default:
                return fmt.Errorf("canonjson: unexpected decoded type %T", v)
        }
        return nil
}

const hexDigits = "0123456789abcdef"

// writeString writes s as a JSON string. s is valid UTF-8 (it comes from encoding/json), and every byte of a
// multi-byte sequence is >= 0x80, so escaping byte by byte is safe.
func writeString(buf *bytes.Buffer, s string) {
        buf.WriteByte('"')
        start := 0
        for i := 0; i < len(s); i++ {
                c := s[i]
                if c >= 0x20 && c != '"' && c != '\\' {
                        continue
                }
                buf.WriteString(s[start:i])
                switch c {
                case '"':
                        buf.WriteString(`\"`)
                case '\\':
                        buf.WriteString(`\\`)
                case '\b':
                        buf.WriteString(`\b`)
                case '\f':
                        buf.WriteString(`\f`)
                case '\n':
                        buf.WriteString(`\n`)
                case '\r':
                        buf.WriteString(`\r`)
                case '\t':
                        buf.WriteString(`\t`)
                default:
                        buf.WriteString(`\u00`)
                        buf.WriteByte(hexDigits[c>>4])
                        buf.WriteByte(hexDigits[c&0xf])
                }
                start = i + 1
        }
        buf.WriteString(s[start:])
        buf.WriteByte('"')
}

// Package config reads service settings from environment variables. An Env collects every missing or invalid
// variable, so a service reports all configuration problems at once and refuses to start (docs/backend §7).
//
//        env := config.New(os.Getenv)
//        cfg := Config{Port: env.Int("CORE_PORT", 8080), JWTSecret: env.Required("JWT_SECRET")}
//        if err := env.Err(); err != nil { return err }
//
// Values are trimmed; an empty value counts as unset (compose often passes VAR= ).
package config

import (
        "fmt"
        "os"
        "strconv"
        "strings"
        "time"
)

// Env reads variables through a getenv function and records problems for Err. Not safe for concurrent use.
type Env struct {
        getenv  func(string) string
        missing []string
        invalid []string
}

// New returns an Env over getenv; nil means os.Getenv.
func New(getenv func(string) string) *Env {
        if getenv == nil {
                getenv = os.Getenv
        }
        return &Env{getenv: getenv}
}

func (e *Env) get(key string) string { return strings.TrimSpace(e.getenv(key)) }

// String returns the value of key, or def when it is unset.
func (e *Env) String(key, def string) string {
        if v := e.get(key); v != "" {
                return v
        }
        return def
}

// Required returns the value of key; when it is unset the key is recorded as missing and "" is returned.
func (e *Env) Required(key string) string {
        v := e.get(key)
        if v == "" {
                e.missing = append(e.missing, key)
        }
        return v
}

// Int returns key as a decimal integer, def when unset; an unparsable value is recorded as invalid (def returned).
func (e *Env) Int(key string, def int) int {
        v := e.get(key)
        if v == "" {
                return def
        }
        n, err := strconv.Atoi(v)
        if err != nil {
                e.invalid = append(e.invalid, fmt.Sprintf("%s=%q (want an integer)", key, v))
                return def
        }
        return n
}

// Duration returns key as a Go duration ("24h", "90s") or a bare number of seconds ("30"), def when unset;
// an unparsable or negative value is recorded as invalid (def returned).
func (e *Env) Duration(key string, def time.Duration) time.Duration {
        v := e.get(key)
        if v == "" {
                return def
        }
        d, err := time.ParseDuration(v)
        if err != nil {
                secs, aerr := strconv.Atoi(v)
                if aerr != nil {
                        e.invalid = append(e.invalid, fmt.Sprintf("%s=%q (want a duration like 30s or 24h)", key, v))
                        return def
                }
                d = time.Duration(secs) * time.Second
        }
        if d < 0 {
                e.invalid = append(e.invalid, fmt.Sprintf("%s=%q (must not be negative)", key, v))
                return def
        }
        return d
}

// List returns key split on commas with blanks dropped (e.g. CORS_ORIGINS), def when unset or empty.
func (e *Env) List(key string, def []string) []string {
        var out []string
        for _, part := range strings.Split(e.get(key), ",") {
                if p := strings.TrimSpace(part); p != "" {
                        out = append(out, p)
                }
        }
        if len(out) == 0 {
                return def
        }
        return out
}

// Err returns nil, or one error listing every missing required variable and every invalid value read so far.
func (e *Env) Err() error {
        var parts []string
        if len(e.missing) > 0 {
                parts = append(parts, "missing required environment variables: "+strings.Join(e.missing, ", "))
        }
        if len(e.invalid) > 0 {
                parts = append(parts, "invalid values: "+strings.Join(e.invalid, ", "))
        }
        if len(parts) == 0 {
                return nil
        }
        return fmt.Errorf("config: %s", strings.Join(parts, "; "))
}

// Package confirm issues and verifies confirmation tokens for MCP write tools (contracts/mcp-tools.md, R71, R79).
//
// Token = P + "." + S, where
//
//        P = base64url(payload JSON), S = base64url(HMAC-SHA256(CONFIRMATION_HMAC_SECRET, P))
//
// base64url is RFC 4648 §5 WITHOUT padding; the MAC is computed over the ASCII bytes of P (not the decoded JSON).
// The payload is the canonical JSON (canonjson, keys sorted) of {"args_hash","exp","key","sub"}: exp is Unix
// seconds, args_hash is the lowercase hex SHA-256 of the canonical arguments (ArgsHash). Python produces the same P
// with json.dumps(payload, sort_keys=True, separators=(",", ":"), ensure_ascii=False).
package confirm

import (
        "bytes"
        "crypto/hmac"
        "crypto/sha256"
        "encoding/base64"
        "encoding/json"
        "errors"
        "fmt"
        "strings"
        "time"

        "aibuddy/services/internal/platform/canonjson"
)

// Argument keys excluded from the hash: they are not part of what the user confirmed.
const (
        KeyIdempotency  = "idempotency_key"
        KeyConfirmation = "confirmation_token"
)

// ErrInvalid is wrapped by every verification failure; MCP servers answer all of them with invalid_confirmation.
var ErrInvalid = errors.New("confirm: invalid confirmation token")

// Specific verification failures; each satisfies errors.Is(err, ErrInvalid).
var (
        ErrMalformed = fmt.Errorf("%w: malformed", ErrInvalid)
        ErrSignature = fmt.Errorf("%w: bad signature", ErrInvalid)
        ErrExpired   = fmt.Errorf("%w: expired", ErrInvalid)
        ErrSubject   = fmt.Errorf("%w: subject mismatch", ErrInvalid)
        ErrKey       = fmt.Errorf("%w: idempotency key mismatch", ErrInvalid)
        ErrArgsHash  = fmt.Errorf("%w: arguments mismatch", ErrInvalid)
)

// ErrNoSecret is a configuration error (empty secret), deliberately not wrapping ErrInvalid.
var ErrNoSecret = errors.New("confirm: empty secret")

type payload struct {
        ArgsHash string `json:"args_hash"`
        Exp      int64  `json:"exp"`
        Key      string `json:"key"`
        Sub      string `json:"sub"`
}

var b64 = base64.RawURLEncoding.Strict()

// Issue returns a token for sub (JWT subject), key (idempotency_key) and argsHash (from ArgsHash) that expires
// ttl from now. ttl must be positive.
func Issue(secret []byte, sub, key, argsHash string, ttl time.Duration) (string, error) {
        if ttl <= 0 {
                return "", fmt.Errorf("confirm: issue: ttl must be positive, got %s", ttl)
        }
        return IssueUntil(secret, sub, key, argsHash, time.Now().Add(ttl))
}

// IssueUntil is Issue with an absolute expiry (truncated to seconds); tests use it for expired tokens.
func IssueUntil(secret []byte, sub, key, argsHash string, exp time.Time) (string, error) {
        if len(secret) == 0 {
                return "", ErrNoSecret
        }
        if sub == "" || key == "" {
                return "", errors.New("confirm: issue: sub and key are required")
        }
        if !isHexSHA256(argsHash) {
                return "", fmt.Errorf("confirm: issue: args hash must be 64 lowercase hex chars, got %q", argsHash)
        }
        body, err := canonjson.Marshal(payload{ArgsHash: argsHash, Exp: exp.Unix(), Key: key, Sub: sub})
        if err != nil {
                return "", fmt.Errorf("confirm: issue: %w", err)
        }
        p := base64.RawURLEncoding.EncodeToString(body)
        return p + "." + base64.RawURLEncoding.EncodeToString(sign(secret, p)), nil
}

// Verify checks token against the caller's JWT subject, the idempotency_key of the call and the arguments the
// server actually received (idempotency_key and confirmation_token are ignored, see ArgsHash), at time now.
// Checks run in order signature, exp, sub, key, args; the first failure is returned.
func Verify(secret []byte, token, sub, key string, args any, now time.Time) error {
        if len(secret) == 0 {
                return ErrNoSecret
        }
        p, s, ok := strings.Cut(token, ".")
        if !ok || p == "" || s == "" || strings.Contains(s, ".") {
                return ErrMalformed
        }
        sig, err := b64.DecodeString(s)
        if err != nil {
                return ErrMalformed
        }
        if !hmac.Equal(sig, sign(secret, p)) {
                return ErrSignature
        }
        raw, err := b64.DecodeString(p)
        if err != nil {
                return ErrMalformed
        }
        var pl payload
        dec := json.NewDecoder(bytes.NewReader(raw))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&pl); err != nil || pl.Sub == "" || pl.Key == "" || pl.Exp == 0 || pl.ArgsHash == "" {
                return ErrMalformed
        }
        if !now.Before(time.Unix(pl.Exp, 0)) {
                return ErrExpired
        }
        if pl.Sub != sub {
                return ErrSubject
        }
        if pl.Key != key {
                return ErrKey
        }
        h, err := ArgsHash(args)
        if err != nil {
                return fmt.Errorf("%w: %v", ErrArgsHash, err)
        }
        if !hmac.Equal([]byte(h), []byte(pl.ArgsHash)) {
                return ErrArgsHash
        }
        return nil
}

// ArgsHash returns the lowercase hex SHA-256 of the canonical JSON of args (a JSON object: map, struct or
// json.RawMessage) after removing the top-level keys idempotency_key and confirmation_token.
// It equals canonjson.SHA256Hex(draft.args) for the draft the user confirmed.
func ArgsHash(args any) (string, error) {
        raw, err := json.Marshal(args)
        if err != nil {
                return "", fmt.Errorf("confirm: args: %w", err)
        }
        dec := json.NewDecoder(bytes.NewReader(raw))
        dec.UseNumber()
        var obj map[string]any
        if err := dec.Decode(&obj); err != nil || obj == nil {
                return "", errors.New("confirm: args must be a JSON object")
        }
        delete(obj, KeyIdempotency)
        delete(obj, KeyConfirmation)
        return canonjson.SHA256Hex(obj)
}

func sign(secret []byte, p string) []byte {
        m := hmac.New(sha256.New, secret)
        m.Write([]byte(p))
        return m.Sum(nil)
}

func isHexSHA256(s string) bool {
        if len(s) != 64 {
                return false
        }
        for i := 0; i < len(s); i++ {
                c := s[i]
                if (c < '0' || c > '9') && (c < 'a' || c > 'f') {
                        return false
                }
        }
        return true
}

// DefaultMaxBody is the request body limit of ReadJSON (1 MiB).
const DefaultMaxBody int64 = 1 << 20

// ReadJSON decodes the request body into dst with DefaultMaxBody; see ReadJSONLimit.
func ReadJSON(w http.ResponseWriter, r *http.Request, dst any) error {
        return ReadJSONLimit(w, r, dst, DefaultMaxBody)
}

// ReadJSONLimit decodes exactly one JSON value from the request body into dst. Unknown object fields, trailing
// data, an empty body and a body over limit bytes are rejected. Every failure is an *apierr.Error with code
// validation_error and a Russian message.
func ReadJSONLimit(w http.ResponseWriter, r *http.Request, dst any, limit int64) error {
        if r.Body == nil {
                return apierr.Validation().WithMessage("Пустое тело запроса")
        }
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, limit))
        dec.DisallowUnknownFields()
        if err := dec.Decode(dst); err != nil {
                return decodeError(err)
        }
        if err := dec.Decode(&struct{}{}); !errors.Is(err, io.EOF) {
                var tooBig *http.MaxBytesError
                if errors.As(err, &tooBig) {
                        return decodeError(err)
                }
                return apierr.Validation().WithMessage("Лишние данные после JSON в теле запроса")
        }
        return nil
}

func decodeError(err error) *apierr.Error {
        var (
                tooBig  *http.MaxBytesError
                typeErr *json.UnmarshalTypeError
        )
        switch {
        case errors.Is(err, io.EOF):
                return apierr.Validation().WithMessage("Пустое тело запроса")
        case errors.As(err, &tooBig):
                return apierr.Validation().WithMessage("Слишком большой запрос")
        case errors.As(err, &typeErr):
                if typeErr.Field != "" {
                        return apierr.Validation().WithMessage(fmt.Sprintf("Неверный тип поля %q", typeErr.Field))
                }
                return apierr.Validation().WithMessage("Неверный тип данных в теле запроса")
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                field := strings.TrimPrefix(err.Error(), "json: unknown field ")
                return apierr.Validation().WithMessage("Неизвестное поле " + field)
        default:
                return apierr.Validation().WithMessage("Некорректный JSON в теле запроса")
        }
}

// WriteJSON writes v as JSON with the given status. If v cannot be encoded, an internal error is written instead.
func WriteJSON(w http.ResponseWriter, status int, v any) {
        body, err := json.Marshal(v)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("httpx: encode response: %w", err))
                return
        }
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _, _ = w.Write(body)
}

// HeaderRequestID is the request id header, accepted from clients and forwarded to the agent.
const HeaderRequestID = "X-Request-Id"

// Middleware wraps an http.Handler.
type Middleware = func(http.Handler) http.Handler

// Chain composes middleware; the first one is the outermost: Chain(a, b)(h) == a(b(h)).
func Chain(mws ...Middleware) Middleware {
        return func(h http.Handler) http.Handler {
                for i := len(mws) - 1; i >= 0; i-- {
                        h = mws[i](h)
                }
                return h
        }
}

type requestIDKey struct{}

// RequestID keeps a well-formed incoming X-Request-Id (1–128 chars of [A-Za-z0-9._:-]) or generates a UUID v4,
// stores it in the context (RequestIDFromContext), on the request header and on the response header.
func RequestID(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                id := r.Header.Get(HeaderRequestID)
                if !validRequestID(id) {
                        id = newUUID()
                }
                r.Header.Set(HeaderRequestID, id)
                w.Header().Set(HeaderRequestID, id)
                next.ServeHTTP(w, r.WithContext(WithRequestID(r.Context(), id)))
        })
}

// WithRequestID returns ctx carrying id (RequestID does this for incoming requests).
func WithRequestID(ctx context.Context, id string) context.Context {
        return context.WithValue(ctx, requestIDKey{}, id)
}

// RequestIDFromContext returns the request id stored by RequestID, or "".
func RequestIDFromContext(ctx context.Context) string {
        id, _ := ctx.Value(requestIDKey{}).(string)
        return id
}

func validRequestID(id string) bool {
        if id == "" || len(id) > 128 {
                return false
        }
        for i := 0; i < len(id); i++ {
                c := id[i]
                ok := c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || strings.IndexByte("._:-", c) >= 0
                if !ok {
                        return false
                }
        }
        return true
}

func newUUID() string {
        var b [16]byte
        _, _ = rand.Read(b[:]) // crypto/rand.Read never fails on supported platforms
        b[6] = b[6]&0x0f | 0x40
        b[8] = b[8]&0x3f | 0x80
        return fmt.Sprintf("%x-%x-%x-%x-%x", b[0:4], b[4:6], b[6:8], b[8:10], b[10:16])
}

// Recover turns a panic into an apierr internal response (if nothing was written yet) and logs it with the stack.
// http.ErrAbortHandler is re-panicked, as net/http expects. A nil log means slog.Default().
func Recover(log *slog.Logger) Middleware {
        if log == nil {
                log = slog.Default()
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        sw := wrap(w)
                        defer func() {
                                v := recover()
                                if v == nil {
                                        return
                                }
                                if v == http.ErrAbortHandler {
                                        panic(v)
                                }
                                log.ErrorContext(r.Context(), "panic in handler", "panic", fmt.Sprint(v),
                                        "request_id", RequestIDFromContext(r.Context()), "stack", string(debug.Stack()))
                                if !sw.wroteHeader {
                                        apierr.WriteJSON(sw, apierr.Internal())
                                }
                        }()
                        next.ServeHTTP(sw, r)
                })
        }
}

// AccessLog logs one line per request: method, path, status, bytes, duration_ms, request_id.
// GET /healthz is logged at debug level to keep container health checks out of the log. A nil log means slog.Default().
func AccessLog(log *slog.Logger) Middleware {
        if log == nil {
                log = slog.Default()
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        sw := wrap(w)
                        next.ServeHTTP(sw, r)
                        level := slog.LevelInfo

                                level = slog.LevelDebug

}

// statusWriter records the status and size of a response. It keeps flushing working for SSE
// (Flush, FlushError and Unwrap for http.ResponseController).
type statusWriter struct {
        http.ResponseWriter
        status      int
        bytes       int64
        wroteHeader bool
}

func wrap(w http.ResponseWriter) *statusWriter {
        if sw, ok := w.(*statusWriter); ok {
                return sw
        }
        return &statusWriter{ResponseWriter: w}
}

func (s *statusWriter) WriteHeader(code int) {
        if !s.wroteHeader && code >= 200 {
                s.status, s.wroteHeader = code, true
        }
        s.ResponseWriter.WriteHeader(code)
}

func (s *statusWriter) Write(b []byte) (int, error) {
        if !s.wroteHeader {
                s.status, s.wroteHeader = http.StatusOK, true
        }
        n, err := s.ResponseWriter.Write(b)
        s.bytes += int64(n)
        return n, err
}

// FlushError flushes the underlying writer, reporting http.ErrNotSupported when it cannot flush.
func (s *statusWriter) FlushError() error {
        err := http.NewResponseController(s.ResponseWriter).Flush()
        if err == nil && !s.wroteHeader {
                s.status, s.wroteHeader = http.StatusOK, true // flushing sends an implicit 200
        }
        return err
}

// Flush implements http.Flusher.
func (s *statusWriter) Flush() { _ = s.FlushError() }

// Unwrap exposes the underlying writer to http.ResponseController.
func (s *statusWriter) Unwrap() http.ResponseWriter { return s.ResponseWriter }

func (s *statusWriter) statusCode() int {
        if s.status == 0 {
                return http.StatusOK
        }
        return s.status
}

// CORS headers sent for allowed origins.
const (
        corsMethods = "GET, POST, PUT, PATCH, DELETE"
        corsHeaders = "Authorization, Content-Type, Idempotency-Key, X-Request-Id, Last-Event-ID"
        corsMaxAge  = 600
)

// CORS allows cross-origin requests from the exact origins listed (scheme://host[:port], trailing "/" ignored;
// "*" allows any origin). Credentials are never allowed: the API uses bearer tokens, not cookies.
// Preflight requests (OPTIONS with Access-Control-Request-Method) are answered here: 204 for allowed
// origins, apierr forbidden otherwise. Requests without Origin pass through untouched.
func CORS(origins []string) Middleware {
        allowed := make(map[string]bool, len(origins))
        anyOrigin := false
        for _, o := range origins {
                o = strings.TrimRight(strings.TrimSpace(o), "/")
                switch o {
                case "":
                case "*":
                        anyOrigin = true
                default:
                        allowed[o] = true
                }
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        origin := r.Header.Get("Origin")
                        if origin == "" {
                                next.ServeHTTP(w, r)
                                return
                        }
                        h := w.Header()
                        h.Add("Vary", "Origin")
                        ok := anyOrigin || allowed[origin]
                        if r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != "" {
                                h.Add("Vary", "Access-Control-Request-Method")
                                h.Add("Vary", "Access-Control-Request-Headers")
                                if !ok {
                                        apierr.WriteJSON(w, apierr.Forbidden())
                                        return
                                }
                                setAllowOrigin(h, origin, anyOrigin)
                                h.Set("Access-Control-Allow-Methods", corsMethods)
                                h.Set("Access-Control-Allow-Headers", corsHeaders)
                                h.Set("Access-Control-Max-Age", strconv.Itoa(corsMaxAge))
                                w.WriteHeader(http.StatusNoContent)
                                return
                        }
                        if ok {
                                setAllowOrigin(h, origin, anyOrigin)
                                h.Set("Access-Control-Expose-Headers", HeaderRequestID)
                        }
                        next.ServeHTTP(w, r)
                })
        }
}

func setAllowOrigin(h http.Header, origin string, anyOrigin bool) {
        if anyOrigin {
                h.Set("Access-Control-Allow-Origin", "*")
                return
        }
        h.Set("Access-Control-Allow-Origin", origin)
}

// DefaultKeepAlive is the interval of `: ping` comments on an SSE stream (contracts/agent-stream.md).
const DefaultKeepAlive = 15 * time.Second

// ErrSSEClosed is returned by SSEWriter.Send after Close.
var ErrSSEClosed = errors.New("httpx: sse stream closed")

// SSEWriter writes Server-Sent Events (R78): every event gets an incrementing `id:` (from 1), an `event:` line and
// one `data:` line of compact JSON, and is flushed immediately. A goroutine writes `: ping` comments every
// keep-alive interval until Close or until the context is done. All methods are safe for concurrent use.
// Always `defer sse.Close()` in the handler: the ping goroutine must stop before the handler returns.
type SSEWriter struct {
        ctx    context.Context
        w      http.ResponseWriter
        rc     *http.ResponseController
        mu     sync.Mutex
        id     int64
        err    error
        closed bool
        stop   chan struct{}
        wg     sync.WaitGroup
}

// NewSSEWriter sets the SSE headers, sends the 200 status, flushes and starts the keep-alive goroutine
// (keepAlive <= 0 means DefaultKeepAlive). ctx is normally r.Context(). It fails, without writing anything,
// when w cannot flush. The server's write deadline is cleared for the stream when the writer supports it.
func NewSSEWriter(ctx context.Context, w http.ResponseWriter, keepAlive time.Duration) (*SSEWriter, error) {
        if keepAlive <= 0 {
                keepAlive = DefaultKeepAlive
        }
        rc := http.NewResponseController(w)
        h := w.Header()
        h.Set("Content-Type", "text/event-stream; charset=utf-8")
        h.Set("Cache-Control", "no-cache")
        h.Set("X-Accel-Buffering", "no")
        if err := rc.Flush(); err != nil {
                return nil, fmt.Errorf("httpx: sse: response writer cannot flush: %w", err)
        }
        _ = rc.SetWriteDeadline(time.Time{}) // long-lived stream; ErrNotSupported is fine
        s := &SSEWriter{ctx: ctx, w: w, rc: rc, stop: make(chan struct{})}
        s.wg.Add(1)
        go s.keepAlive(keepAlive)
        return s, nil
}

func (s *SSEWriter) keepAlive(every time.Duration) {
        defer s.wg.Done()
        t := time.NewTicker(every)
        defer t.Stop()
        for {
                select {
                case <-s.ctx.Done():
                        return
                case <-s.stop:
                        return
                case <-t.C:
                        if err := s.write(func(b *bytes.Buffer) { b.WriteString(": ping\n\n") }); err != nil {
                                return
                        }
                }
        }
}

// Send writes one event with the given name and data encoded as single-line JSON (HTML characters are not
// escaped; json.RawMessage is compacted, so relayed agent events keep their meaning). It returns the first
// write error of the stream, ctx.Err() once the context is done, or ErrSSEClosed after Close.
func (s *SSEWriter) Send(event string, data any) error {
        if event == "" || strings.ContainsAny(event, "\r\n") {
                return fmt.Errorf("httpx: sse: invalid event name %q", event)
        }
        var payload bytes.Buffer
        enc := json.NewEncoder(&payload)
        enc.SetEscapeHTML(false)
        if err := enc.Encode(data); err != nil {
                return fmt.Errorf("httpx: sse: encode %s: %w", event, err)
        }
        line := bytes.TrimRight(payload.Bytes(), "\n")
        return s.write(func(b *bytes.Buffer) {
                s.id++
                b.WriteString("id: ")
                b.WriteString(strconv.FormatInt(s.id, 10))
                b.WriteString("\nevent: ")
                b.WriteString(event)
                b.WriteString("\ndata: ")
                b.Write(line)
                b.WriteString("\n\n")
        })
}

// write builds a frame under the lock (so ids are assigned in write order), writes and flushes it.
func (s *SSEWriter) write(frame func(*bytes.Buffer)) error {
        s.mu.Lock()
        defer s.mu.Unlock()
        switch {
        case s.closed:
                return ErrSSEClosed
        case s.err != nil:
                return s.err
        case s.ctx.Err() != nil:
                return s.ctx.Err()
        }
        var b bytes.Buffer
        frame(&b)
        if _, err := s.w.Write(b.Bytes()); err != nil {
                s.err = fmt.Errorf("httpx: sse: write: %w", err)
                return s.err
        }
        if err := s.rc.Flush(); err != nil {
                s.err = fmt.Errorf("httpx: sse: flush: %w", err)
                return s.err
        }
        return nil
}

// LastID returns the id of the last event sent (0 before the first one).
func (s *SSEWriter) LastID() int64 {
        s.mu.Lock()
        defer s.mu.Unlock()
        return s.id
}

// Close stops the keep-alive goroutine and waits for it; later Sends return ErrSSEClosed. It is idempotent and
// does not close the underlying connection (the handler returning does).
func (s *SSEWriter) Close() error {
        s.mu.Lock()
        if s.closed {
                s.mu.Unlock()
                return nil
        }
        s.closed = true
        close(s.stop)
        s.mu.Unlock()
        s.wg.Wait()
        return nil
}

// sseLineSlack is room for the field name ("data: ") on a line of a limited reader.
const sseLineSlack = 64

// NewSSEReaderLimit is NewSSEReader with a limit of maxEvent bytes on the data of one event (all its data lines
// together) and on one line (plus room for the field name), for streams from less trusted peers; maxEvent <= 0
// means no event limit. A larger event makes Next return ErrSSETooLarge and stop.
func NewSSEReaderLimit(r io.Reader, maxEvent int) *SSEReader {
        maxLine := MaxSSELine
        if maxEvent > 0 && maxEvent+sseLineSlack < maxLine {
                maxLine = maxEvent + sseLineSlack
        } else if maxEvent < 0 {
                maxEvent = 0
        }
        sc := bufio.NewScanner(r)
        sc.Buffer(make([]byte, 0, min(4096, maxLine)), maxLine)
        sc.Split(splitLines)
        return &SSEReader{sc: sc, maxEvent: maxEvent}
}

// Next returns the next event, io.EOF at the end of the stream, a read error, or the event together with an
// error wrapping ErrSSEData when its data is not valid JSON (Next may be called again after that).
func (r *SSEReader) Next() (Event, error) {
        if r.failed != nil {
                return Event{}, r.failed
        }
        var (
                name    string
                data    []byte
                hasData bool
        )
        for r.sc.Scan() {
                line := r.sc.Bytes()
                if !r.started {
                        r.started = true
                        line = bytes.TrimPrefix(line, []byte("\xef\xbb\xbf"))
                }
                if len(line) == 0 {
                        if !hasData {
                                name = ""
                                continue
                        }
                        return r.dispatch(name, data)
                }
                if line[0] == ':' {
                        continue
                }
                field, value, _ := bytes.Cut(line, []byte(":"))
                value = bytes.TrimPrefix(value, []byte(" "))
                switch string(field) {
                case "event":
                        name = string(value)
                case "data":
                        if r.maxEvent > 0 && len(data)+1+len(value) > r.maxEvent {
                                r.failed = fmt.Errorf("%w: more than %d bytes of data", ErrSSETooLarge, r.maxEvent)
                                return Event{}, r.failed
                        }
                        if hasData {
                                data = append(data, '\n')
                        }
                        data = append(data, value...)
                        hasData = true
                case "id":
                        if bytes.IndexByte(value, 0) < 0 {
                                r.lastID = string(value)
                        }
                }
        }
        if err := r.sc.Err(); err != nil {
                if errors.Is(err, bufio.ErrTooLong) {
                        r.failed = fmt.Errorf("%w: line too long", ErrSSETooLarge)
                        return Event{}, r.failed
                }
                return Event{}, fmt.Errorf("httpx: sse: read: %w", err)
        }
        if hasData {
                return r.dispatch(name, data)
        }
        return Event{}, io.EOF
}

func (r *SSEReader) dispatch(name string, data []byte) (Event, error) {
        if name == "" {
                name = "message"
        }
        ev := Event{ID: r.lastID, Name: name, Data: json.RawMessage(data)}
        if !json.Valid(data) {
                return ev, fmt.Errorf("%w: event %q", ErrSSEData, name)
        }
        return ev, nil
}

// splitLines is a bufio.SplitFunc for SSE line ends: "\r\n", "\n" or "\r".
func splitLines(data []byte, atEOF bool) (int, []byte, error) {
        if atEOF && len(data) == 0 {
                return 0, nil, nil
        }
        if i := bytes.IndexAny(data, "\r\n"); i >= 0 {
                if data[i] == '\n' {
                        return i + 1, data[:i], nil
                }
                if i+1 < len(data) {
                        if data[i+1] == '\n' {
                                return i + 2, data[:i], nil
                        }
                        return i + 1, data[:i], nil
                }
                if !atEOF {
                        return 0, nil, nil // a "\n" may follow in the next read
                }
                return i + 1, data[:i], nil
        }
        if atEOF {
                return len(data), data, nil
        }
        return 0, nil, nil
}

// Client calls tools of the mock MCP servers on behalf of a user (core: approvals.*, kb.add_faq; tests).
// The zero value is not usable: set BaseURL. Safe for concurrent use.
type Client struct {
        BaseURL    string       // e.g. http://mcp-mocks:8100/mcp; the server name is appended as /<server>
        HTTPClient *http.Client // nil means a client with http.DefaultTransport and no timeout (use ctx deadlines)
}

// Call opens a session to BaseURL/<server> with `Authorization: Bearer userToken`, calls tool with args (any
// JSON-object value; nil means {}), closes the session and returns exactly one of:
//   - the structured result (JSON object) on success;
//   - a tool error (*apierr.Error with an MCP code: forbidden, not_found, validation_error, conflict,
//     invalid_confirmation, unavailable);
//   - an error for transport and protocol failures. That error wraps an *apierr.Error (find it with apierr.As):
//     unauthorized when the server rejected the token, internal for protocol errors (e.g. unknown tool),
//     upstream_unavailable otherwise. It also wraps the cause (context errors keep working with errors.Is).
//
// X-Request-Id from httpx.RequestIDFromContext(ctx) is forwarded.
func (c *Client) Call(ctx context.Context, userToken, server, tool string, args any) (json.RawMessage, *apierr.Error, error) {
        rt := &bearerTransport{token: userToken, base: http.DefaultTransport}
        hc := &http.Client{Transport: rt}
        if c.HTTPClient != nil {
                if c.HTTPClient.Transport != nil {
                        rt.base = c.HTTPClient.Transport
                }
                hc.Timeout = c.HTTPClient.Timeout
        }
        transport := &mcp.StreamableClientTransport{
                Endpoint:   strings.TrimRight(c.BaseURL, "/") + "/" + server,
                HTTPClient: hc,
                MaxRetries: -1,
        }
        client := mcp.NewClient(&mcp.Implementation{Name: "aibuddy-core", Version: version}, nil)
        cs, err := client.Connect(ctx, transport, nil)
        if err != nil {
                return nil, nil, rt.wrap(ctx, fmt.Sprintf("connect %s", server), err)
        }
        defer cs.Close()
        res, err := cs.CallTool(ctx, &mcp.CallToolParams{Name: tool, Arguments: args})
        if err != nil {
                return nil, nil, rt.wrap(ctx, fmt.Sprintf("call %s.%s", server, tool), err)
        }
        if res.IsError {
                return nil, toolError(res), nil
        }
        out, err := structured(res)
        if err != nil {
                return nil, nil, fmt.Errorf("mcpx: call %s.%s: %w: %w", server, tool, apierr.Internal(), err)
        }
        return out, nil, nil
}

// structured returns the structured content of a successful result, falling back to JSON text content.
func structured(res *mcp.CallToolResult) (json.RawMessage, error) {
        if res.StructuredContent != nil {
                raw, err := json.Marshal(res.StructuredContent)
                if err != nil {
                        return nil, fmt.Errorf("encode structured content: %w", err)
                }
                return raw, nil
        }
        if text := firstText(res); json.Valid([]byte(text)) && strings.HasPrefix(strings.TrimSpace(text), "{") {
                return json.RawMessage(text), nil
        }
        return nil, errors.New("result has no structured content")
}

// toolError decodes {code, message, retryable} from an isError result; unknown shapes become unavailable.
func toolError(res *mcp.CallToolResult) *apierr.Error {
        var wire struct {
                Code      apierr.Code `json:"code"`
                Message   string      `json:"message"`
                Retryable bool        `json:"retryable"`
        }
        raw, _ := json.Marshal(res.StructuredContent)
        if res.StructuredContent == nil {
                raw = []byte(firstText(res))
        }
        if err := json.Unmarshal(raw, &wire); err != nil || wire.Code == "" {
                e := apierr.Unavailable()
                if text := firstText(res); text != "" {
                        e = e.WithMessage(text)
                }
                e.Retryable = false
                return e
        }
        e := apierr.New(wire.Code, wire.Message)
        e.Retryable = wire.Retryable
        return e
}

func firstText(res *mcp.CallToolResult) string {
        for _, c := range res.Content {
                if t, ok := c.(*mcp.TextContent); ok {
                        return t.Text
                }
        }
        return ""
}

// bearerTransport adds the user's token and the request id to every request of a session and remembers
// whether any HTTP exchange failed, to classify errors.
type bearerTransport struct {
        token string
        base  http.RoundTripper

        mu           sync.Mutex
        unauthorized bool
        httpFailed   bool
}

func (t *bearerTransport) RoundTrip(req *http.Request) (*http.Response, error) {
        r := req.Clone(req.Context())
        r.Header.Set("Authorization", "Bearer "+t.token)
        if id := httpx.RequestIDFromContext(req.Context()); id != "" {
                r.Header.Set(httpx.HeaderRequestID, id)
        }
        resp, err := t.base.RoundTrip(r)
        t.mu.Lock()
        defer t.mu.Unlock()
        switch {
        case err != nil:
                t.httpFailed = true
        case resp.StatusCode == http.StatusUnauthorized:
                t.unauthorized = true
        case resp.StatusCode >= 400 && !(req.Method == http.MethodGet && resp.StatusCode == http.StatusMethodNotAllowed) &&
                req.Method != http.MethodDelete:
                t.httpFailed = true // a 405 for the optional hanging GET and the DELETE on close are expected
        }
        return resp, err
}

func (t *bearerTransport) wrap(ctx context.Context, op string, err error) error {
        t.mu.Lock()
        unauthorized, httpFailed := t.unauthorized, t.httpFailed
        t.mu.Unlock()
        kind := apierr.UpstreamUnavailable()
        switch {
        case unauthorized:
                kind = apierr.Unauthorized()
        case !httpFailed && ctx.Err() == nil:
                kind = apierr.Internal() // the server answered with a JSON-RPC error
        }
        if cerr := ctx.Err(); cerr != nil && !errors.Is(err, cerr) {
                err = fmt.Errorf("%w: %w", cerr, err)
        }
        return fmt.Errorf("mcpx: %s: %w: %w", op, kind, err)
}

// PathPrefix is where Mount serves the servers: /mcp/<name>.
const PathPrefix = "/mcp/"

// MaxRequestBody limits one MCP HTTP request body (1 MiB).
const MaxRequestBody = 1 << 20

const (
        callerKey = "mcpx.caller" // key in auth.TokenInfo.Extra
        version   = "0.1.0"
)

// Mount validates every server and serves it at PathPrefix+name on mux. It fails on the first invalid server.
func Mount(mux *http.ServeMux, servers ...*Server) error {
        for _, s := range servers {
                h, err := s.Handler()
                if err != nil {
                        return err
                }
                mux.Handle(PathPrefix+s.name, h)
        }
        return nil
}

// Handler validates the server (see Validate) and returns its HTTP handler: JWT verification, then the go-sdk
// streamable HTTP handler in stateless mode with JSON responses (no session id, no SSE). Missing or invalid JWTs
// get 401 with `WWW-Authenticate: Bearer` and a JSON-RPC error body whose data is the apierr JSON.
func (s *Server) Handler() (http.Handler, error) {
        if err := s.Validate(); err != nil {
                return nil, err
        }
        srv := mcp.NewServer(&mcp.Implementation{Name: "aibuddy-" + s.name, Version: version}, &mcp.ServerOptions{
                HasTools:     true,
                GetSessionID: func() string { return "" }, // stateless: no Mcp-Session-Id
        })
        for _, n := range s.order {
                t := s.tools[n]
                mt := &mcp.Tool{Name: t.def.Name, Description: t.def.Description, InputSchema: t.def.InputSchema}
                if len(t.def.OutputSchema) > 0 {
                        mt.OutputSchema = t.def.OutputSchema
                }
                srv.AddTool(mt, s.toolHandler(t))
        }
        srv.AddReceivingMiddleware(s.filterToolList)
        streamable := mcp.NewStreamableHTTPHandler(func(*http.Request) *mcp.Server { return srv },
                &mcp.StreamableHTTPOptions{Stateless: true, JSONResponse: true})
        // RequireBearerToken is how the go-sdk carries identity: the TokenInfo it puts into the request context is
        // copied by the streamable transport into CallToolRequest.Extra.TokenInfo for every JSON-RPC message.
        withTokenInfo := auth.RequireBearerToken(tokenInfo, nil)(streamable)
        verifier := authn.NewVerifier(s.cfg.JWTSecret)
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                token, ok := authn.BearerToken(r)
                if !ok {
                        writeUnauthorized(w)
                        return
                }
                claims, err := verifier.Verify(token)
                if err != nil {
                        s.log.DebugContext(r.Context(), "mcp jwt rejected", "server", s.name, "err", err)
                        writeUnauthorized(w)
                        return
                }
                r.Body = http.MaxBytesReader(w, r.Body, MaxRequestBody)
                ctx := authn.WithToken(authn.NewContext(r.Context(), claims), token)
                withTokenInfo.ServeHTTP(w, r.WithContext(ctx))
        }), nil
}

// tokenInfo is the auth.TokenVerifier: the JWT was already verified, so it only wraps the caller.
func tokenInfo(ctx context.Context, _ string, _ *http.Request) (*auth.TokenInfo, error) {
        c, ok := callerFromContext(ctx)
        if !ok {
                return nil, auth.ErrInvalidToken
        }
        return &auth.TokenInfo{
                Expiration: time.Unix(c.Exp, 0).Add(authn.DefaultLeeway),
                Extra:      map[string]any{callerKey: c},
        }, nil
}

func callerFromContext(ctx context.Context) (Caller, bool) {
        claims, ok := authn.FromContext(ctx)
        if !ok {
                return Caller{}, false
        }
        token, _ := authn.TokenFromContext(ctx)
        return Caller{Claims: claims, Token: token}, true
}

// CallerFrom returns the caller of an MCP request: from req.Extra.TokenInfo (set by Handler) or, failing that,
// from the context. Tool handlers receive the Caller directly; this is for custom go-sdk handlers and tests.
func CallerFrom(ctx context.Context, req mcp.Request) (Caller, bool) {
        if req != nil {
                if extra := req.GetExtra(); extra != nil && extra.TokenInfo != nil {
                        if c, ok := extra.TokenInfo.Extra[callerKey].(Caller); ok {
                                return c, true
                        }
                }
        }
        return callerFromContext(ctx)
}

func (s *Server) toolHandler(t *tool) mcp.ToolHandler {
        return func(ctx context.Context, req *mcp.CallToolRequest) (*mcp.CallToolResult, error) {
                c, ok := CallerFrom(ctx, req)
                if !ok {
                        return errorResult(apierr.ToMCP(apierr.Unauthorized())), nil
                }
                out, aerr := s.run(ctx, c, t, req.Params.Arguments)
                if aerr != nil {
                        return errorResult(aerr), nil
                }
                return &mcp.CallToolResult{
                        StructuredContent: out,
                        Content:           []mcp.Content{&mcp.TextContent{Text: string(out)}},
                }, nil
        }
}

func errorResult(e *apierr.Error) *mcp.CallToolResult {
        body, _ := json.Marshal(e) // strings and a bool: cannot fail
        return &mcp.CallToolResult{
                IsError:           true,
                StructuredContent: json.RawMessage(body),
                Content:           []mcp.Content{&mcp.TextContent{Text: string(body)}},
        }
}

// filterToolList hides from tools/list the tools the caller's role may not call (direct calls get forbidden).
func (s *Server) filterToolList(next mcp.MethodHandler) mcp.MethodHandler {
        return func(ctx context.Context, method string, req mcp.Request) (mcp.Result, error) {
                res, err := next(ctx, method, req)
                list, ok := res.(*mcp.ListToolsResult)
                if err != nil || !ok {
                        return res, err
                }
                c, _ := CallerFrom(ctx, req)
                visible := make([]*mcp.Tool, 0, len(list.Tools))
                for _, mt := range list.Tools {
                        if t := s.tools[mt.Name]; t != nil && t.allows(c.Role) {
                                visible = append(visible, mt)
                        }
                }
                filtered := *list
                filtered.Tools = visible
                return &filtered, nil
        }
}

// rpcErrUnauthorized is the JSON-RPC server error code sent with HTTP 401.
const rpcErrUnauthorized = -32001

func writeUnauthorized(w http.ResponseWriter) {
        e := apierr.Unauthorized()
        body, _ := json.Marshal(map[string]any{
                "jsonrpc": "2.0",
                "id":      nil,
                "error":   map[string]any{"code": rpcErrUnauthorized, "message": e.Message, "data": e},
        })
        w.Header().Set("WWW-Authenticate", "Bearer")
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(http.StatusUnauthorized)
        _, _ = fmt.Fprintf(w, "%s", body)
}

// ToolDef is one tool of contracts/schemas/<server>.json.
type ToolDef struct {
        Name         string          `json:"name"`
        Description  string          `json:"description"`
        Roles        []string        `json:"roles"`
        Write        bool            `json:"write"`
        InputSchema  json.RawMessage `json:"inputSchema"`
        OutputSchema json.RawMessage `json:"outputSchema"`
}

type schemaFile struct {
        Server string    `json:"server"`
        Tools  []ToolDef `json:"tools"`
}

// LoadTools reads the tool definitions of server from contracts.Schemas (schemas/<server>.json).
func LoadTools(server string) ([]ToolDef, error) {
        return loadTools(contracts.Schemas, server)
}

func loadTools(fsys fs.FS, server string) ([]ToolDef, error) {
        raw, err := fs.ReadFile(fsys, "schemas/"+server+".json")
        if err != nil {
                return nil, fmt.Errorf("mcpx: load schema %q: %w", server, err)
        }
        var f schemaFile
        if err := json.Unmarshal(raw, &f); err != nil {
                return nil, fmt.Errorf("mcpx: parse schema %q: %w", server, err)
        }
        if f.Server != server {
                return nil, fmt.Errorf("mcpx: schema file %q declares server %q", server, f.Server)
        }
        return f.Tools, nil
}

// tool is a compiled ToolDef.
type tool struct {
        def     ToolDef
        input   *jsonschema.Resolved
        output  *jsonschema.Resolved // nil when the contract has no output schema
        formats map[string]string    // top-level argument -> format (date, date-time, uuid)
}

func compile(def ToolDef) (*tool, error) {
        if def.Name == "" {
                return nil, fmt.Errorf("mcpx: tool without name")
        }
        if len(def.Roles) == 0 {
                return nil, fmt.Errorf("mcpx: tool %q: no roles", def.Name)
        }
        in, props, err := resolve(def.InputSchema)
        if err != nil {
                return nil, fmt.Errorf("mcpx: tool %q: input schema: %w", def.Name, err)
        }
        if def.Write {
                for _, k := range []string{"idempotency_key", "confirmation_token"} {
                        if _, ok := props[k]; !ok {
                                return nil, fmt.Errorf("mcpx: write tool %q: input schema lacks %s", def.Name, k)
                        }
                }
        }
        t := &tool{def: def, input: in, formats: map[string]string{}}
        for name, p := range props {
                if p != nil && formatCheckers[p.Format] != nil {
                        t.formats[name] = p.Format
                }
        }
        if len(def.OutputSchema) > 0 {
                if t.output, _, err = resolve(def.OutputSchema); err != nil {
                        return nil, fmt.Errorf("mcpx: tool %q: output schema: %w", def.Name, err)
                }
        }
        return t, nil
}

func resolve(raw json.RawMessage) (*jsonschema.Resolved, map[string]*jsonschema.Schema, error) {
        var s jsonschema.Schema
        if err := json.Unmarshal(raw, &s); err != nil {
                return nil, nil, err
        }
        if s.Type != "object" {
                return nil, nil, fmt.Errorf(`type must be "object", got %q`, s.Type)
        }
        r, err := s.Resolve(nil)
        if err != nil {
                return nil, nil, err
        }
        return r, s.Properties, nil
}

func (t *tool) allows(role string) bool { return slices.Contains(t.def.Roles, role) }

var uuidRE = regexp.MustCompile(`^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$`)

// formatCheckers enforce the "format" keywords the contracts use (jsonschema-go treats format as annotation only).
var formatCheckers = map[string]func(string) bool{
        "date":      func(s string) bool { _, err := time.Parse(time.DateOnly, s); return err == nil },
        "date-time": func(s string) bool { _, err := time.Parse(time.RFC3339, s); return err == nil },
        "uuid":      uuidRE.MatchString,
}

var formatNames = map[string]string{
        "date":      "дата в формате ГГГГ-ММ-ДД",
        "date-time": "дата и время в формате RFC 3339",
        "uuid":      "UUID",
}

// checkFormats validates the format of top-level string arguments.
func (t *tool) checkFormats(args map[string]any) error {
        for name, format := range t.formats {
                s, ok := args[name].(string)
                if ok && !formatCheckers[format](s) {
                        return fmt.Errorf("поле %s: ожидается %s", name, formatNames[format])
                }
        }
        return nil
}

// Package mcpx builds the MCP servers of mcp-mocks from the contracts and calls them from core
// (docs/backend/IMPLEMENTATION.md §4, contracts/mcp-tools.md).
//
// Server side: NewServer loads contracts/schemas/<name>.json, Handle binds a plain Go func to every tool, and
// Handler serves the MCP streamable HTTP transport (stateless, JSON responses) behind JWT verification.
// For every tools/call mcpx enforces, in this order, before the handler runs:
//  1. role allow-list of the tool (`roles`) → forbidden
//  2. write tools: confirmation_token and idempotency_key present and non-empty → else invalid_confirmation
//  3. inputSchema (unknown fields such as employee_id, types, enums, patterns, required; format date,
//     date-time, uuid of top-level arguments) after applying schema defaults → validation_error
//  4. write tools: confirm.Verify(sub = caller, key = idempotency_key, args as received) → invalid_confirmation
//
// Identity reaches handlers through the go-sdk mechanism for bearer auth: Handler verifies the JWT, wraps it in an
// auth.TokenInfo (auth.RequireBearerToken), and the streamable transport copies it into CallToolRequest.Extra.
//
// Handler errors of type *apierr.Error become tool results with isError=true and structuredContent
// {code, message, retryable} (MCP codes, see apierr.ToMCP); other errors and panics are logged and reported as a
// generic, non-retryable error. Successful results must match the tool's outputSchema; they are sent as
// structuredContent plus the same JSON as text content. tools/list shows only the tools the caller's role may use.

// Caller is the authenticated user of an MCP request: verified JWT claims plus the raw bearer token.
// Identity used by tool handlers comes only from here, never from tool arguments (R71).
type Caller struct {

        Token string
}

// HandlerFunc implements a tool. args are the validated tool arguments as JSON (including idempotency_key and
// confirmation_token for write tools; schema defaults applied). The result must marshal to a JSON object that
// matches the tool's outputSchema. Return *apierr.Error for expected failures.
type HandlerFunc func(ctx context.Context, c Caller, args json.RawMessage) (any, error)

// Config holds the secrets and logger of a Server.

        ConfirmationSecret []byte       // CONFIRMATION_HMAC_SECRET: verifies write confirmations
        Logger             *slog.Logger // nil means slog.Default()
}

// Server is one mock MCP server (absence, approvals, access, checklist, directory, kb).

        errs     []error

// NewServer returns a server whose tools are defined by contracts/schemas/<name>.json. It fails when the schema
// is missing or invalid. Bind every tool with Handle, then call Handler (or Mount).
func NewServer(name string, cfg Config) (*Server, error) {
        defs, err := LoadTools(name)
        if err != nil {
                return nil, err
        }
        return newServer(name, defs, cfg)
}

func newServer(name string, defs []ToolDef, cfg Config) (*Server, error) {
        if name == "" || strings.ContainsAny(name, "/ ") {
                return nil, fmt.Errorf("mcpx: invalid server name %q", name)
        }
        s := &Server{name: name, cfg: cfg, log: cfg.Logger, tools: map[string]*tool{}, handlers: map[string]HandlerFunc{}}
        if s.log == nil {
                s.log = slog.Default()
        }
        for _, d := range defs {
                t, err := compile(d)
                if err != nil {
                        return nil, fmt.Errorf("server %q: %w", name, err)
                }
                if _, dup := s.tools[d.Name]; dup {
                        return nil, fmt.Errorf("mcpx: server %q: duplicate tool %q", name, d.Name)
                }
                s.tools[d.Name] = t
                s.order = append(s.order, d.Name)
        }
        return s, nil
}

// Name returns the server name; it is served at /mcp/<name>.
func (s *Server) Name() string { return s.name }

// Tools returns the tool definitions in contract order.
func (s *Server) Tools() []ToolDef {
        out := make([]ToolDef, 0, len(s.order))
        for _, n := range s.order {
                out = append(out, s.tools[n].def)
        }
        return out
}

// Handle binds h to the tool called name. Unknown names, duplicates and nil handlers are reported by Validate.
func (s *Server) Handle(name string, h HandlerFunc) {
        switch {
        case s.tools[name] == nil:
                s.errs = append(s.errs, fmt.Errorf("mcpx: server %q: tool %q is not in the contract", s.name, name))
        case h == nil:
                s.errs = append(s.errs, fmt.Errorf("mcpx: server %q: nil handler for %q", s.name, name))
        case s.handlers[name] != nil:
                s.errs = append(s.errs, fmt.Errorf("mcpx: server %q: tool %q registered twice", s.name, name))
        default:
                s.handlers[name] = h
        }
}

// Validate reports every registration problem: unknown or duplicate tools, contract tools without a handler,
// missing secrets. Handler calls it, so a misconfigured server fails at startup.
func (s *Server) Validate() error {
        errs := slices.Clone(s.errs)
        for _, n := range s.order {
                if s.handlers[n] == nil {
                        errs = append(errs, fmt.Errorf("mcpx: server %q: no handler for contract tool %q", s.name, n))
                }
        }
        if len(s.cfg.JWTSecret) == 0 {
                errs = append(errs, fmt.Errorf("mcpx: server %q: empty JWT secret", s.name))
        }
        if len(s.cfg.ConfirmationSecret) == 0 && slices.ContainsFunc(s.order, func(n string) bool { return s.tools[n].def.Write }) {
                errs = append(errs, fmt.Errorf("mcpx: server %q: empty confirmation secret", s.name))
        }
        return errors.Join(errs...)
}

// Invoke runs tool for c through the same checks as an MCP tools/call (role, schema, confirmation, handler,
// output schema) without HTTP. It returns the structured result or a tool error; mock servers use it in tests.
func (s *Server) Invoke(ctx context.Context, c Caller, name string, args any) (json.RawMessage, *apierr.Error) {
        raw, err := json.Marshal(args)
        if err != nil {
                return nil, apierr.ToMCP(apierr.Validation().WithMessage("Аргументы инструмента не кодируются в JSON"))
        }
        t := s.tools[name]
        if t == nil || s.handlers[name] == nil {
                return nil, apierr.ToMCP(apierr.NotFound().WithMessage(fmt.Sprintf("Инструмент %s.%s не найден", s.name, name)))
        }
        return s.run(ctx, c, t, raw)
}

func (s *Server) run(ctx context.Context, c Caller, t *tool, raw json.RawMessage) (json.RawMessage, *apierr.Error) {
        if !t.allows(c.Role) {
                return nil, apierr.Forbidden()
        }
        if t.def.Write && !hasConfirmationFields(raw) {
                // The rights matrix wants invalid_confirmation (not validation_error) for a write without a token.
                return nil, apierr.InvalidConfirmation()
        }
        args, aerr := s.checkArgs(t, raw)
        if aerr != nil {
                return nil, aerr
        }
        if t.def.Write {
                key, _ := args[confirm.KeyIdempotency].(string)
                token, _ := args[confirm.KeyConfirmation].(string)
                if err := confirm.Verify(s.cfg.ConfirmationSecret, token, c.Sub, key, raw, time.Now()); err != nil {
                        s.log.InfoContext(ctx, "mcp confirmation rejected", "server", s.name, "tool", t.def.Name, "sub", c.Sub, "reason", err)
                        return nil, apierr.InvalidConfirmation()
                }
        }
        withDefaults, err := json.Marshal(args)
        if err != nil {
                return nil, s.internal(ctx, t, fmt.Errorf("re-encode args: %w", err))
        }
        out, err := s.call(ctx, c, t, withDefaults)
        if err != nil {
                if e, ok := apierr.As(err); ok {
                        return nil, apierr.ToMCP(e)
                }
                if ctx.Err() != nil {
                        return nil, apierr.ToMCP(apierr.From(ctx.Err()))
                }
                return nil, s.internal(ctx, t, err)
        }
        result, err := s.checkOutput(t, out)
        if err != nil {
                return nil, s.internal(ctx, t, err)
        }
        return result, nil
}

// checkArgs decodes the arguments, applies schema defaults and validates them.
func (s *Server) checkArgs(t *tool, raw json.RawMessage) (map[string]any, *apierr.Error) {
        args := map[string]any{}
        if len(bytes.TrimSpace(raw)) > 0 && !bytes.Equal(bytes.TrimSpace(raw), []byte("null")) {
                if err := json.Unmarshal(raw, &args); err != nil {
                        return nil, apierr.Validation().WithMessage("Аргументы инструмента должны быть JSON-объектом")
                }
        }
        if err := t.input.ApplyDefaults(&args); err != nil {
                return nil, apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        if err := t.input.Validate(&args); err != nil {
                return nil, apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        if err := t.checkFormats(args); err != nil {
                return nil, apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        return args, nil
}

// hasConfirmationFields reports whether raw is an object with non-empty string confirmation_token and
// idempotency_key.
func hasConfirmationFields(raw json.RawMessage) bool {
        var f struct {
                Token *string `json:"confirmation_token"`
                Key   *string `json:"idempotency_key"`
        }
        if err := json.Unmarshal(raw, &f); err != nil {
                return false
        }
        return f.Token != nil && *f.Token != "" && f.Key != nil && *f.Key != ""
}

// call runs the handler, turning a panic into an error.
func (s *Server) call(ctx context.Context, c Caller, t *tool, args json.RawMessage) (out any, err error) {
        defer func() {
                if v := recover(); v != nil {
                        err = fmt.Errorf("panic: %v\n%s", v, debug.Stack())
                }
        }()
        return s.handlers[t.def.Name](ctx, c, args)
}

func (s *Server) checkOutput(t *tool, out any) (json.RawMessage, error) {
        raw, err := json.Marshal(out)
        if err != nil {
                return nil, fmt.Errorf("encode result: %w", err)
        }
        var obj map[string]any
        if err := json.Unmarshal(raw, &obj); err != nil || obj == nil {
                return nil, fmt.Errorf("result is not a JSON object: %s", truncate(raw))
        }
        if t.output != nil {
                if err := t.output.Validate(&obj); err != nil {
                        return nil, fmt.Errorf("result violates outputSchema: %w", err)
                }
        }
        return raw, nil
}

func (s *Server) internal(ctx context.Context, t *tool, err error) *apierr.Error {
        s.log.ErrorContext(ctx, "mcp tool failed", "server", s.name, "tool", t.def.Name, "err", err)
        return apierr.ToMCP(apierr.Internal())
}

func truncate(b []byte) string {
        if len(b) > 200 {
                return string(b[:200]) + "…"
        }
        return string(b)
}

// DecodeArgs unmarshals validated tool arguments into dst; a failure is a validation_error.
func DecodeArgs(args json.RawMessage, dst any) error {
        if err := json.Unmarshal(args, dst); err != nil {
                return apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        return nil
}

// WithConfirmation returns a copy of args with idempotency_key = key and a confirmation_token signed for sub,
// key and the hash of args (core uses it for approvals.decide and kb.add_faq, where the button press is the
// confirmation; tests use it for write tools).
func WithConfirmation(secret []byte, sub, key string, args map[string]any, ttl time.Duration) (map[string]any, error) {
        hash, err := confirm.ArgsHash(args)
        if err != nil {
                return nil, err
        }
        token, err := confirm.Issue(secret, sub, key, hash, ttl)
        if err != nil {
                return nil, err
        }
        out := make(map[string]any, len(args)+2)
        for k, v := range args {
                out[k] = v
        }
        out[confirm.KeyIdempotency] = key
        out[confirm.KeyConfirmation] = token
        return out, nil
}

// Package period maps an intern's cohort start date to the periods of the Avito checklist
// (data/checklist/checklist.yaml: week_1 … after_3_months) and gives each period its due offset and Russian label.
package period

import (
        "fmt"
        "slices"
        "time"
)

// Period is a checklist period id (openapi Period).
type Period string

// The checklist periods in order.
const (
        Week1        Period = "week_1"
        Week2        Period = "week_2"
        Month1       Period = "month_1"
        Month2       Period = "month_2"
        After3Months Period = "after_3_months"
)

type info struct {
        p       Period
        before  int // Current returns p while days since cohort start < before
        dueDays int
        label   string
}

// Same table as data/checklist/checklist.yaml (labels) and migrations/0003_seed_demo.sql (due offsets).
var table = []info{
        {Week1, 7, 7, "1 неделя"},
        {Week2, 14, 14, "2 неделя"},
        {Month1, 30, 30, "1 месяц"},
        {Month2, 90, 60, "2 месяц"},
        {After3Months, -1, 100, "После 3-х месяцев"},
}

// All returns the periods in checklist order (a fresh slice).
func All() []Period {
        out := make([]Period, len(table))
        for i, t := range table {
                out[i] = t.p
        }
        return out
}

// Parse returns s as a Period, or an error for an unknown id.
func Parse(s string) (Period, error) {
        p := Period(s)
        if Order(p) < 0 {
                return "", fmt.Errorf("period: unknown period %q", s)
        }
        return p, nil
}

// Order returns the 0-based position of p in checklist order, or -1 for an unknown period.
func Order(p Period) int {
        return slices.IndexFunc(table, func(t info) bool { return t.p == p })
}

// Label returns the Russian label («1 неделя» … «После 3-х месяцев»), or string(p) for an unknown period.
func Label(p Period) string {
        if i := Order(p); i >= 0 {
                return table[i].label
        }
        return string(p)
}

// DueOffsetDays returns the due date offset from the cohort start: 7, 14, 30, 60, 100; 0 for an unknown period.
func DueOffsetDays(p Period) int {
        if i := Order(p); i >= 0 {
                return table[i].dueDays
        }
        return 0
}

// DaysSince returns the number of calendar days from cohortStart to today (negative before the start).
// Each value is reduced to its calendar date in its own location; the time of day is ignored.
func DaysSince(cohortStart, today time.Time) int {
        return int(date(today).Sub(date(cohortStart)).Hours() / 24)
}

// Current returns the period of an intern on today: week_1 for days < 7 (also before the start), week_2 < 14,
// month_1 < 30, month_2 < 90, after_3_months otherwise. Days are counted by DaysSince.
func Current(cohortStart, today time.Time) Period {
        days := DaysSince(cohortStart, today)
        for _, t := range table {
                if t.before < 0 || days < t.before {
                        return t.p
                }
        }
        return After3Months // unreachable: the last row has no bound
}

// DueDate returns cohortStart + DueOffsetDays(p) as a date (midnight UTC), like the seed's cohort_start + days.
func DueDate(cohortStart time.Time, p Period) time.Time {
        return date(cohortStart).AddDate(0, 0, DueOffsetDays(p))
}

// ParseDate parses a YYYY-MM-DD date (JWT claim cohort_start, DB dates) as midnight UTC.
func ParseDate(s string) (time.Time, error) {
        t, err := time.Parse(time.DateOnly, s)
        if err != nil {
                return time.Time{}, fmt.Errorf("period: parse date %q: %w", s, err)
        }
        return t, nil
}

func date(t time.Time) time.Time {
        y, m, d := t.Date()
        return time.Date(y, m, d, 0, 0, 0, 0, time.UTC)
}

// Package pg opens PostgreSQL connection pools (pgx/v5) for core and mcp-mocks.
package pg

import (
        "context"
        "fmt"
        "time"

        "github.com/jackc/pgx/v5/pgxpool"
)

// Options tune Open. Zero values mean defaults.
type Options struct {
        MaxConns     int32         // default 10
        StartupWait  time.Duration // how long to retry while Postgres is still starting; default 30s
        RetryBackoff time.Duration // pause between attempts; default 1s
}

// Open connects to url and pings it, retrying until opts.StartupWait elapses (Postgres may still be
// running init scripts when the service starts).
func Open(ctx context.Context, url string, opts Options) (*pgxpool.Pool, error) {
        cfg, err := pgxpool.ParseConfig(url)
        if err != nil {
                return nil, fmt.Errorf("parse postgres url: %w", err)
        }
        cfg.MaxConns = opts.MaxConns
        if cfg.MaxConns == 0 {
                cfg.MaxConns = 10
        }
        cfg.MaxConnIdleTime = 5 * time.Minute
        wait, backoff := opts.StartupWait, opts.RetryBackoff
        if wait == 0 {
                wait = 30 * time.Second
        }
        if backoff == 0 {
                backoff = time.Second
        }
        deadline := time.Now().Add(wait)
        for {
                pool, err := pgxpool.NewWithConfig(ctx, cfg)
                if err == nil {
                        if err = pool.Ping(ctx); err == nil {
                                return pool, nil
                        }
                        pool.Close()
                }
                if time.Now().After(deadline) {
                        return nil, fmt.Errorf("connect postgres: %w", err)
                }
                select {
                case <-ctx.Done():
                        return nil, ctx.Err()
                case <-time.After(backoff):
                }
        }
}

// Package pgtest gives tests a fresh PostgreSQL database built from the real migrations (schema + demo seed).
//
// Set TEST_DATABASE_URL to a superuser connection (scripts/test-db.sh prints one). Each New call creates
// database t_<random>, applies migrations.FS in lexical order and drops the database when the test ends.
package pgtest

import (
        "context"
        "crypto/rand"
        "encoding/hex"
        "fmt"
        "io/fs"
        "net/url"
        "os"
        "sort"
        "testing"
        "time"

        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgxpool"

        "aibuddy/migrations"
)

// EnvVar is the variable holding the admin connection string.
const EnvVar = "TEST_DATABASE_URL"

// New returns a pool connected to a fresh migrated database, or skips the test when TEST_DATABASE_URL is unset.
func New(t testing.TB) *pgxpool.Pool {
        t.Helper()
        admin := os.Getenv(EnvVar)
        if admin == "" {
                t.Skipf("%s is not set; run `bash scripts/test-db.sh up` or `make test`", EnvVar)
        }
        ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
        defer cancel()
        name := newName()
        adminConn, err := pgx.Connect(ctx, admin)
        if err != nil {
                t.Fatalf("pgtest: connect admin: %v", err)
        }
        defer adminConn.Close(context.Background())
        if _, err := adminConn.Exec(ctx, "CREATE DATABASE "+name); err != nil {
                t.Fatalf("pgtest: create database: %v", err)
        }
        dbURL, err := withDatabase(admin, name)
        if err != nil {
                t.Fatalf("pgtest: %v", err)
        }
        if err := applyMigrations(ctx, dbURL); err != nil {
                dropDatabase(admin, name)
                t.Fatalf("pgtest: %v", err)
        }
        pool, err := pgxpool.New(ctx, dbURL)
        if err != nil {
                dropDatabase(admin, name)
                t.Fatalf("pgtest: open pool: %v", err)
        }
        t.Cleanup(func() {
                pool.Close()
                dropDatabase(admin, name)
        })
        return pool
}

// URL returns the connection string of the database behind pool (useful for services that open their own pool).
func URL(t testing.TB, pool *pgxpool.Pool) string {
        t.Helper()
        return pool.Config().ConnString()
}

func applyMigrations(ctx context.Context, dbURL string) error {
        conn, err := pgx.Connect(ctx, dbURL)
        if err != nil {
                return fmt.Errorf("connect test database: %w", err)
        }
        defer conn.Close(context.Background())
        names, err := fs.Glob(migrations.FS, "*.sql")
        if err != nil {
                return fmt.Errorf("list migrations: %w", err)
        }
        sort.Strings(names)
        for _, n := range names {
                sql, err := migrations.FS.ReadFile(n)
                if err != nil {
                        return fmt.Errorf("read %s: %w", n, err)
                }
                // Simple protocol: a migration file is many statements in one string.
                if _, err := conn.Exec(ctx, string(sql), pgx.QueryExecModeSimpleProtocol); err != nil {
                        return fmt.Errorf("apply %s: %w", n, err)
                }
        }
        return nil
}

func withDatabase(raw, name string) (string, error) {
        u, err := url.Parse(raw)
        if err != nil {
                return "", fmt.Errorf("parse %s: %w", EnvVar, err)
        }
        u.Path = "/" + name
        return u.String(), nil
}

func dropDatabase(admin, name string) {
        ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
        defer cancel()
        conn, err := pgx.Connect(ctx, admin)
        if err != nil {
                return
        }
        defer conn.Close(context.Background())
        _, _ = conn.Exec(ctx, "DROP DATABASE IF EXISTS "+name+" WITH (FORCE)")
}

func randomHex(n int) string {
        b := make([]byte, n)
        if _, err := rand.Read(b); err != nil {
                panic(err)
        }
        return hex.EncodeToString(b)
}
