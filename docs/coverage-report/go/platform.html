// Пакет apierr — тип ошибки, общий для HTTP API (contracts/openapi.yaml, схема Error) и результатов инструментов MCP
// (contracts/mcp-tools.md, «Errors»). Сообщения на русском, для пользователя, нейтральные; никогда не раскрывают,
// существует ли чужой ресурс (R71).
package apierr

import (
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "log/slog"
        "net/http"
)

// Code — машиночитаемый код ошибки.
type Code string

// Коды HTTP API (openapi Error.code).
const (
        CodeUnauthorized        Code = "unauthorized"
        CodeForbidden           Code = "forbidden"
        CodeNotFound            Code = "not_found"
        CodeValidation          Code = "validation_error"
        CodeConflict            Code = "conflict"
        CodeExpired             Code = "expired"
        CodeRateLimited         Code = "rate_limited"
        CodeUpstreamUnavailable Code = "upstream_unavailable"
        CodeInternal            Code = "internal"
)

// Коды, которые есть только в результатах инструментов MCP (forbidden, not_found, validation_error и conflict — общие).
const (
        CodeInvalidConfirmation Code = "invalid_confirmation"
        CodeUnavailable         Code = "unavailable"
)

// Error — ошибка API или инструмента. Status — HTTP-статус, который использует WriteJSON; не сериализуется.
type Error struct {
        Code      Code   `json:"code"`
        Message   string `json:"message"`
        Retryable bool   `json:"retryable"`
        Status    int    `json:"-"`
}

type spec struct {
        status    int
        retryable bool
        message   string
}

var specs = map[Code]spec{
        CodeUnauthorized:        {http.StatusUnauthorized, false, "Требуется вход в систему"},
        CodeForbidden:           {http.StatusForbidden, false, "Недостаточно прав для этого действия"},
        CodeNotFound:            {http.StatusNotFound, false, "Не найдено"},
        CodeValidation:          {http.StatusBadRequest, false, "Проверьте введённые данные"},
        CodeConflict:            {http.StatusConflict, false, "Действие уже выполнено или недоступно в текущем состоянии"},
        CodeExpired:             {http.StatusConflict, false, "Срок действия истёк, повторите действие"},
        CodeRateLimited:         {http.StatusTooManyRequests, true, "Слишком много запросов, попробуйте чуть позже"},
        CodeUpstreamUnavailable: {http.StatusBadGateway, true, "Сервис временно недоступен, попробуйте ещё раз"},
        CodeInternal:            {http.StatusInternalServerError, false, "Внутренняя ошибка, попробуйте позже"},
        CodeInvalidConfirmation: {http.StatusForbidden, false, "Подтверждение недействительно или устарело, подтвердите действие ещё раз"},
        CodeUnavailable:         {http.StatusServiceUnavailable, true, "Сервис временно недоступен, попробуйте ещё раз"},
}

// New возвращает ошибку с указанным кодом; пустое сообщение означает русское сообщение кода по умолчанию. Status и
// Retryable берутся из кода (неизвестные коды: 500, без повтора).
func New(code Code, message string) *Error {
        s, ok := specs[code]
        if !ok {
                s = spec{status: http.StatusInternalServerError, message: specs[CodeInternal].message}
        }
        if message == "" {
                message = s.message
        }
        return &Error{Code: code, Message: message, Retryable: s.retryable, Status: s.status}
}

// Unauthorized возвращает ошибку для отсутствующего или некорректного JWT (401).
func Unauthorized() *Error { return New(CodeUnauthorized, "") }

// Forbidden возвращает ошибку для действия или ресурса, к которому у вызывающего нет доступа, включая чужие данные
// (403).
func Forbidden() *Error { return New(CodeForbidden, "") }

// NotFound возвращает ошибку для отсутствующего ресурса вызывающего (404).
func NotFound() *Error { return New(CodeNotFound, "") }

// Validation возвращает ошибку для некорректных данных запроса или аргументов инструмента (400); обычно дальше идёт
// WithMessage.
func Validation() *Error { return New(CodeValidation, "") }

// Conflict возвращает ошибку для конфликта состояния, например уже решённого согласования (409).
func Conflict() *Error { return New(CodeConflict, "") }

// Expired возвращает ошибку для истёкшего или отменённого черновика (409).
func Expired() *Error { return New(CodeExpired, "") }

// RateLimited возвращает ошибку для слишком частых запросов (429, можно повторить).
func RateLimited() *Error { return New(CodeRateLimited, "") }

// UpstreamUnavailable возвращает ошибку для агента или MCP-сервера, который не ответил (502, можно повторить).
func UpstreamUnavailable() *Error { return New(CodeUpstreamUnavailable, "") }

// Internal возвращает ошибку для неожиданного сбоя сервера (500).
func Internal() *Error { return New(CodeInternal, "") }

// InvalidConfirmation возвращает ошибку инструмента для записи без корректного токена подтверждения (403 при отправке
// по HTTP).
func InvalidConfirmation() *Error { return New(CodeInvalidConfirmation, "") }

// Unavailable возвращает ошибку инструмента для MCP-сервера, у которого недоступна зависимость (503 при отправке по
// HTTP, можно повторить).
func Unavailable() *Error { return New(CodeUnavailable, "") }

// Error реализует error.
func (e *Error) Error() string { return fmt.Sprintf("%s: %s", e.Code, e.Message) }

// Is сообщает, является ли target ошибкой *Error с тем же кодом, чтобы работал errors.Is(err, apierr.Forbidden()).
func (e *Error) Is(target error) bool {
        t, ok := target.(*Error)
        return ok && t.Code == e.Code
}

// WithMessage возвращает копию e с другим сообщением для пользователя.
func (e *Error) WithMessage(msg string) *Error {
        c := *e
        c.Message = msg
        return &c
}

// As возвращает *Error из цепочки err, если она есть.
func As(err error) (*Error, bool) {
        var e *Error
        if errors.As(err, &e) && e != nil {
                return e, true
        }
        return nil, false
}

// From превращает любую ошибку в *Error: *Error из цепочки возвращается как есть, дедлайн или отмена контекста
// становятся UpstreamUnavailable, всё остальное — Internal. From(nil) — это nil.
func From(err error) *Error {
        if err == nil {
                return nil
        }
        if e, ok := As(err); ok {
                return e
        }
        if errors.Is(err, context.DeadlineExceeded) || errors.Is(err, context.Canceled) {
                return UpstreamUnavailable()
        }
        return Internal()
}

// HasCode сообщает, является ли err ошибкой *Error с указанным кодом (или оборачивает её).
func HasCode(err error, code Code) bool {
        e, ok := As(err)
        return ok && e.Code == code
}

// ToAPI переводит коды, которые есть только в MCP, в значения openapi Error.code, сохраняя сообщение и retryable:
// unavailable → upstream_unavailable (502), invalid_confirmation → expired (409). Остальные ошибки возвращаются как
// есть.
func ToAPI(e *Error) *Error {
        switch e.Code {
        case CodeUnavailable:
                return remap(e, CodeUpstreamUnavailable)
        case CodeInvalidConfirmation:
                return remap(e, CodeExpired)
        }
        if _, ok := specs[e.Code]; !ok {
                return Internal()
        }
        return e
}

// ToMCP переводит коды, которые есть только в API, в коды ошибок инструментов MCP, сохраняя сообщение и retryable:
// unauthorized → forbidden, expired → invalid_confirmation, rate_limited/upstream_unavailable/internal и неизвестные
// коды → unavailable. Общие коды возвращаются как есть.
func ToMCP(e *Error) *Error {
        switch e.Code {
        case CodeForbidden, CodeNotFound, CodeValidation, CodeConflict, CodeInvalidConfirmation, CodeUnavailable:
                return e
        case CodeUnauthorized:
                return remap(e, CodeForbidden)
        case CodeExpired:
                return remap(e, CodeInvalidConfirmation)
        default:
                return remap(e, CodeUnavailable)
        }
}

func remap(e *Error, code Code) *Error {
        c := *e
        c.Code = code
        c.Status = specs[code].status
        return &c
}

// WriteJSON пишет err как тело openapi Error с его HTTP-статусом (коды только MCP переводятся через ToAPI). Ошибки,
// которые не *Error, логируются через slog и отдаются как Internal, чтобы подробности не утекали клиентам.
func WriteJSON(w http.ResponseWriter, err error) {
        e, ok := As(err)
        if !ok {
                e = From(err)
                if err != nil && e.Code == CodeInternal {
                        slog.Default().Error("internal error", "err", err)
                }
        }
        if e == nil {
                e = Internal()
        }
        e = ToAPI(e)
        status := e.Status
        if status == 0 {
                status = New(e.Code, "").Status
        }
        body, _ := json.Marshal(e) // поля — строки и bool: ошибки быть не может
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.Header().Set("X-Content-Type-Options", "nosniff")
        w.WriteHeader(status)
        _, _ = w.Write(body)
}

// Пакет authn выпускает и проверяет JWT HS256 мок-IdP и кладёт проверенные claims в контекст запроса. Личность в любом
// месте бэкенда берётся только из этих claims (R71).
//
// Токен: base64url (без выравнивания) от заголовка {"alg":"HS256","typ":"JWT"}, payload Claims и HMAC-SHA256 от
// "<header>.<payload>" на JWT_SECRET. Проверка строгая: ровно HS256 и typ JWT (alg "none" и любые другие алгоритмы
// отклоняются, как и заголовок "crit"), сравнение MAC за постоянное время, exp обязателен, sub и role обязательны, iat
// не в будущем.
package authn

import (
        "context"
        "crypto/hmac"
        "crypto/sha256"
        "encoding/base64"
        "encoding/json"
        "errors"
        "fmt"
        "log/slog"
        "net/http"
        "strings"
        "time"

        "aibuddy/services/internal/platform/apierr"
)

// DefaultLeeway — допустимое расхождение часов для exp и iat в Verify и NewVerifier.
const DefaultLeeway = 30 * time.Second

// Claims — claims JWT пользователя. Время — в секундах Unix.
type Claims struct {
        Sub         string `json:"sub"`
        Role        string `json:"role"`
        TeamID      string `json:"team_id"`
        CohortStart string `json:"cohort_start,omitempty"` // YYYY-MM-DD, пусто для пользователей без когорты
        Iat         int64  `json:"iat,omitempty"`
        Exp         int64  `json:"exp"`
}

// Ошибки проверки. Middleware на все отвечает одинаковым 401 и не говорит клиенту, какая именно.
var (
        ErrNoSecret      = errors.New("authn: empty secret")
        ErrMalformed     = errors.New("authn: malformed token")
        ErrAlgorithm     = errors.New("authn: unsupported token header")
        ErrSignature     = errors.New("authn: invalid signature")
        ErrExpired       = errors.New("authn: token expired")
        ErrNotYetValid   = errors.New("authn: token issued in the future")
        ErrMissingClaims = errors.New("authn: required claim missing")
)

var (
        b64        = base64.RawURLEncoding
        b64Strict  = base64.RawURLEncoding.Strict()
        headerPart = b64.EncodeToString([]byte(`{"alg":"HS256","typ":"JWT"}`))
)

// Issue подписывает c с Iat = now и Exp = now + ttl. ttl должен быть положительным.
func Issue(secret []byte, c Claims, ttl time.Duration) (string, error) {
        if ttl <= 0 {
                return "", fmt.Errorf("authn: issue: ttl must be positive, got %s", ttl)
        }
        now := time.Now()
        c.Iat = now.Unix()
        c.Exp = now.Add(ttl).Unix()
        return Sign(secret, c)
}

// Sign подписывает c ровно как передано (тесты собирают им истёкшие токены). Sub, Role и Exp обязательны.
func Sign(secret []byte, c Claims) (string, error) {
        if len(secret) == 0 {
                return "", ErrNoSecret
        }
        if c.Sub == "" || c.Role == "" || c.Exp == 0 {
                return "", ErrMissingClaims
        }
        payload, err := json.Marshal(c)
        if err != nil {
                return "", fmt.Errorf("authn: marshal claims: %w", err)
        }
        signingInput := headerPart + "." + b64.EncodeToString(payload)
        return signingInput + "." + b64.EncodeToString(mac(secret, signingInput)), nil
}

// Verify проверяет token на момент now с DefaultLeeway и возвращает его claims.
func Verify(secret []byte, token string, now time.Time) (Claims, error) {
        return verify(secret, token, now, DefaultLeeway)
}

func verify(secret []byte, token string, now time.Time, leeway time.Duration) (Claims, error) {
        if len(secret) == 0 {
                return Claims{}, ErrNoSecret
        }
        parts := strings.Split(token, ".")
        if len(parts) != 3 || parts[0] == "" || parts[1] == "" || parts[2] == "" {
                return Claims{}, ErrMalformed
        }
        // Map, а не структура: encoding/json сопоставляет поля структуры без учёта регистра ("ALG" прошёл бы).
        var h map[string]any
        if err := decodePart(parts[0], &h); err != nil {
                return Claims{}, err
        }
        if _, crit := h["crit"]; h["alg"] != "HS256" || h["typ"] != "JWT" || crit {
                return Claims{}, ErrAlgorithm
        }
        sig, err := b64Strict.DecodeString(parts[2])
        if err != nil {
                return Claims{}, ErrMalformed
        }
        if !hmac.Equal(sig, mac(secret, parts[0]+"."+parts[1])) {
                return Claims{}, ErrSignature
        }
        var c Claims
        if err := decodePart(parts[1], &c); err != nil {
                return Claims{}, err
        }
        if c.Sub == "" || c.Role == "" || c.Exp == 0 {
                return Claims{}, ErrMissingClaims
        }
        if !now.Before(time.Unix(c.Exp, 0).Add(leeway)) {
                return Claims{}, ErrExpired
        }
        if c.Iat != 0 && time.Unix(c.Iat, 0).After(now.Add(leeway)) {
                return Claims{}, ErrNotYetValid
        }
        return c, nil
}

func decodePart(part string, dst any) error {
        raw, err := b64Strict.DecodeString(part)
        if err != nil {
                return ErrMalformed
        }
        if err := json.Unmarshal(raw, dst); err != nil {
                return ErrMalformed
        }
        return nil
}

func mac(secret []byte, signingInput string) []byte {
        m := hmac.New(sha256.New, secret)
        m.Write([]byte(signingInput))
        return m.Sum(nil)
}

// Verifier проверяет токены с фиксированными секретом, допуском и часами; безопасен для конкурентного использования.
type Verifier struct {
        Secret []byte
        Leeway time.Duration    // допустимое расхождение часов; NewVerifier ставит DefaultLeeway
        Now    func() time.Time // nil означает time.Now
}

// NewVerifier возвращает Verifier с DefaultLeeway и системными часами.
func NewVerifier(secret []byte) *Verifier {
        return &Verifier{Secret: secret, Leeway: DefaultLeeway, Now: time.Now}
}

// Verify проверяет token и возвращает его claims.
func (v *Verifier) Verify(token string) (Claims, error) {
        now := time.Now
        if v.Now != nil {
                now = v.Now
        }
        return verify(v.Secret, token, now(), v.Leeway)
}

// Middleware требует корректный `Authorization: Bearer <JWT>`; при успехе кладёт claims и исходный токен в контекст
// запроса (FromContext, TokenFromContext), иначе отвечает apierr 401 с `WWW-Authenticate: Bearer`.
func (v *Verifier) Middleware(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                token, ok := BearerToken(r)
                if !ok {
                        unauthorized(w)
                        return
                }
                c, err := v.Verify(token)
                if err != nil {
                        slog.Default().DebugContext(r.Context(), "jwt rejected", "err", err)
                        unauthorized(w)
                        return
                }
                next.ServeHTTP(w, r.WithContext(WithToken(NewContext(r.Context(), c), token)))
        })
}

// Middleware — это NewVerifier(secret).Middleware.
func Middleware(secret []byte) func(http.Handler) http.Handler {
        return NewVerifier(secret).Middleware
}

func unauthorized(w http.ResponseWriter) {
        w.Header().Set("WWW-Authenticate", "Bearer")
        apierr.WriteJSON(w, apierr.Unauthorized())
}

// BearerToken возвращает токен из заголовка `Authorization: Bearer <token>` (схема без учёта регистра).
func BearerToken(r *http.Request) (string, bool) {
        scheme, token, ok := strings.Cut(r.Header.Get("Authorization"), " ")
        if !ok || !strings.EqualFold(scheme, "Bearer") {
                return "", false
        }
        token = strings.TrimSpace(token)
        return token, token != ""
}

type claimsKey struct{}

type tokenKey struct{}

// NewContext возвращает ctx с c (так делает Middleware; тесты обработчиков могут вызывать напрямую).
func NewContext(ctx context.Context, c Claims) context.Context {
        return context.WithValue(ctx, claimsKey{}, c)
}

// FromContext возвращает проверенные claims, сохранённые Middleware или NewContext.
func FromContext(ctx context.Context) (Claims, bool) {
        c, ok := ctx.Value(claimsKey{}).(Claims)
        return c, ok
}

// TokenFromContext возвращает исходный bearer-токен, принятый Middleware, для передачи агенту и MCP-серверам.
func TokenFromContext(ctx context.Context) (string, bool) {
        t, ok := ctx.Value(tokenKey{}).(string)
        return t, ok && t != ""
}

// WithToken возвращает ctx с исходным bearer-токеном для путей кода, которые вызывают MCP без HTTP-запроса (например,
// тесты или ядро, действующее от имени пользователя с выпущенным им токеном).
func WithToken(ctx context.Context, token string) context.Context {
        return context.WithValue(ctx, tokenKey{}, token)
}

// Пакет canonjson выдаёт канонический JSON: побайтно точную форму, которую хешируют и подписывают ядро, MCP-серверы и
// агент на Python (contracts/mcp-tools.md: «ключи отсортированы, без пробелов, UTF-8»).
//
// Результат совпадает с Python
//
//        json.dumps(json.loads(text), sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")
//
// для любого значения из объектов, массивов, строк, целых чисел, булевых значений и null:
//   - ключи объектов отсортированы по кодовым точкам Unicode (это тот же порядок, что байтовый порядок UTF-8);
//     при дубликатах ключей в тексте JSON остаётся последнее значение, как в Python;
//   - между токенами нет пробелов;
//   - строки — «сырой» UTF-8; экранируются только '"', '\\' и управляющие символы U+0000–U+001F:
//     \b \f \n \r \t — в короткой форме, остальные — \u00XX со строчными hex; '/', '<', '>', '&', U+007F,
//     U+2028 и U+2029 пишутся как есть;
//   - некорректный UTF-8 в строках Go заменяется на U+FFFD (как делает encoding/json).
//
// Числа: текст числа сохраняется как передан (json.Number, json.RawMessage) или как encoding/json форматирует числа Go.
// Поэтому целые совпадают с Python в точности (любого размера); единственная правка — "-0" → "0", как в Python. Нецелые
// числа НЕ нормализуются: Python переформатирует float через repr ("1.0", "1e+16", 1e2 → "100.0"), Go оставляет "1",
// "1e2" или то, что записал производитель. Не кладите нецелые числа в подписываемые аргументы; в текущих схемах
// инструментов записи их нет.
//
// Инвариант: Marshal(v) == Marshal(json.RawMessage(json.Marshal(v))) для любого v, которое принимает encoding/json.
package canonjson

import (
        "bytes"
        "crypto/sha256"
        "encoding/hex"
        "encoding/json"
        "fmt"
        "slices"
)

// Marshal возвращает каноническое JSON-представление v. v может быть любым значением, которое принимает encoding/json
// (map, структуры с json-тегами, срезы, реализации json.Marshaler, json.RawMessage, json.Number). []byte кодируется
// строкой base64, как в encoding/json; текст JSON вместо этого оборачивайте в json.RawMessage.
func Marshal(v any) ([]byte, error) {
        raw, err := json.Marshal(v)
        if err != nil {
                return nil, fmt.Errorf("canonjson: marshal: %w", err)
        }
        dec := json.NewDecoder(bytes.NewReader(raw))
        dec.UseNumber()
        var tree any
        if err := dec.Decode(&tree); err != nil {
                return nil, fmt.Errorf("canonjson: decode: %w", err)
        }
        var buf bytes.Buffer
        buf.Grow(len(raw))
        if err := encode(&buf, tree); err != nil {
                return nil, err
        }
        return buf.Bytes(), nil
}

// SHA256Hex возвращает SHA-256 от Marshal(v) строчным hex — форма `args_hash` и `draft_hash`.
func SHA256Hex(v any) (string, error) {
        b, err := Marshal(v)
        if err != nil {
                return "", err
        }
        sum := sha256.Sum256(b)
        return hex.EncodeToString(sum[:]), nil
}

func encode(buf *bytes.Buffer, v any) error {
        switch x := v.(type) {
        case nil:
                buf.WriteString("null")
        case bool:
                if x {
                        buf.WriteString("true")
                } else {
                        buf.WriteString("false")
                }
        case json.Number:
                if x == "-0" {
                        x = "0"
                }
                buf.WriteString(string(x))
        case string:
                writeString(buf, x)
        case []any:
                buf.WriteByte('[')
                for i, e := range x {
                        if i > 0 {
                                buf.WriteByte(',')
                        }
                        if err := encode(buf, e); err != nil {
                                return err
                        }
                }
                buf.WriteByte(']')
        case map[string]any:
                keys := make([]string, 0, len(x))
                for k := range x {
                        keys = append(keys, k)
                }
                slices.Sort(keys)
                buf.WriteByte('{')
                for i, k := range keys {
                        if i > 0 {
                                buf.WriteByte(',')
                        }
                        writeString(buf, k)
                        buf.WriteByte(':')
                        if err := encode(buf, x[k]); err != nil {
                                return err
                        }
                }
                buf.WriteByte('}')
        default:
                return fmt.Errorf("canonjson: unexpected decoded type %T", v)
        }
        return nil
}

const hexDigits = "0123456789abcdef"

// writeString пишет s как строку JSON. s — корректный UTF-8 (приходит из encoding/json), и каждый байт многобайтовой
// последовательности >= 0x80, поэтому экранировать побайтно безопасно.
func writeString(buf *bytes.Buffer, s string) {
        buf.WriteByte('"')
        start := 0
        for i := 0; i < len(s); i++ {
                c := s[i]
                if c >= 0x20 && c != '"' && c != '\\' {
                        continue
                }
                buf.WriteString(s[start:i])
                switch c {
                case '"':
                        buf.WriteString(`\"`)
                case '\\':
                        buf.WriteString(`\\`)
                case '\b':
                        buf.WriteString(`\b`)
                case '\f':
                        buf.WriteString(`\f`)
                case '\n':
                        buf.WriteString(`\n`)
                case '\r':
                        buf.WriteString(`\r`)
                case '\t':
                        buf.WriteString(`\t`)
                default:
                        buf.WriteString(`\u00`)
                        buf.WriteByte(hexDigits[c>>4])
                        buf.WriteByte(hexDigits[c&0xf])
                }
                start = i + 1
        }
        buf.WriteString(s[start:])
        buf.WriteByte('"')
}

// Пакет config читает настройки сервиса из переменных окружения. Env собирает все отсутствующие или некорректные
// переменные, чтобы сервис сразу сообщил обо всех проблемах конфигурации и отказался стартовать (docs/backend §7).
//
//        env := config.New(os.Getenv)
//        cfg := Config{Port: env.Int("CORE_PORT", 8080), JWTSecret: env.Required("JWT_SECRET")}
//        if err := env.Err(); err != nil { return err }
//
// Значения обрезаются по краям; пустое значение считается незаданным (compose часто передаёт VAR= ).
package config

import (
        "fmt"
        "os"
        "strconv"
        "strings"
        "time"

// Env читает переменные через функцию getenv и записывает проблемы для Err. Не безопасен для конкурентного
// использования.
type Env struct {
        getenv  func(string) string
        missing []string
        invalid []string
}

// New возвращает Env поверх getenv; nil означает os.Getenv.
func New(getenv func(string) string) *Env {
        if getenv == nil {
                getenv = os.Getenv
        }
        return &Env{getenv: getenv}
}

func (e *Env) get(key string) string { return strings.TrimSpace(e.getenv(key)) }

// String возвращает значение key или def, если оно не задано.
func (e *Env) String(key, def string) string {
        if v := e.get(key); v != "" {
                return v
        }
        return def
}

// Required возвращает значение key; если оно не задано, ключ записывается как отсутствующий и возвращается "".
func (e *Env) Required(key string) string {
        v := e.get(key)
        if v == "" {
                e.missing = append(e.missing, key)
        }
        return v
}

// Int возвращает key как десятичное целое, def — если не задано; неразбираемое значение записывается как некорректное
// (возвращается def).
func (e *Env) Int(key string, def int) int {
        v := e.get(key)
        if v == "" {
                return def
        }
        n, err := strconv.Atoi(v)
        if err != nil {
                e.invalid = append(e.invalid, fmt.Sprintf("%s=%q (want an integer)", key, v))
                return def
        }
        return n
}

// Duration возвращает key как длительность Go ("24h", "90s") или просто число секунд ("30"), def — если не задано;
// неразбираемое или отрицательное значение записывается как некорректное (возвращается def).
func (e *Env) Duration(key string, def time.Duration) time.Duration {
        v := e.get(key)
        if v == "" {
                return def
        }
        d, err := time.ParseDuration(v)
        if err != nil {
                secs, aerr := strconv.Atoi(v)
                if aerr != nil {
                        e.invalid = append(e.invalid, fmt.Sprintf("%s=%q (want a duration like 30s or 24h)", key, v))
                        return def
                }
                d = time.Duration(secs) * time.Second
        }
        if d < 0 {
                e.invalid = append(e.invalid, fmt.Sprintf("%s=%q (must not be negative)", key, v))
                return def
        }
        return d
}

// List возвращает key, разделённый по запятым, без пустых элементов (например, CORS_ORIGINS), def — если не задано или
// пусто.
func (e *Env) List(key string, def []string) []string {
        var out []string
        for _, part := range strings.Split(e.get(key), ",") {
                if p := strings.TrimSpace(part); p != "" {
                        out = append(out, p)
                }
        }
        if len(out) == 0 {
                return def
        }
        return out
}

// Err возвращает nil или одну ошибку со списком всех отсутствующих обязательных переменных и всех некорректных
// значений, прочитанных к этому моменту.
func (e *Env) Err() error {
        var parts []string
        if len(e.missing) > 0 {
                parts = append(parts, "missing required environment variables: "+strings.Join(e.missing, ", "))
        }
        if len(e.invalid) > 0 {
                parts = append(parts, "invalid values: "+strings.Join(e.invalid, ", "))
        }
        if len(parts) == 0 {
                return nil
        }
        return fmt.Errorf("config: %s", strings.Join(parts, "; "))
}

// Пакет confirm выпускает и проверяет токены подтверждения для инструментов записи MCP (contracts/mcp-tools.md, R71,
// R79).
//
// Токен = P + "." + S, где
//
//        P = base64url(payload JSON), S = base64url(HMAC-SHA256(CONFIRMATION_HMAC_SECRET, P))
//
// base64url — RFC 4648 §5 БЕЗ выравнивания; MAC считается по ASCII-байтам P (а не по декодированному JSON). Payload —
// канонический JSON (canonjson, ключи отсортированы) от {"args_hash","exp","key","sub"}: exp — секунды Unix, args_hash
// — SHA-256 канонических аргументов строчным hex (ArgsHash). Python выдаёт тот же P через json.dumps(payload,
// sort_keys=True, separators=(",", ":"), ensure_ascii=False).
package confirm

import (
        "bytes"
        "crypto/hmac"
        "crypto/sha256"
        "encoding/base64"
        "encoding/json"
        "errors"
        "fmt"
        "strings"
        "time"

        "aibuddy/services/internal/platform/canonjson"
)

// Ключи аргументов, исключённые из хеша: они не входят в то, что подтвердил пользователь.
const (
        KeyIdempotency  = "idempotency_key"
        KeyConfirmation = "confirmation_token"
)

// ErrInvalid оборачивается каждой ошибкой проверки; MCP-серверы на все отвечают invalid_confirmation.
var ErrInvalid = errors.New("confirm: invalid confirmation token")

// Конкретные ошибки проверки; каждая удовлетворяет errors.Is(err, ErrInvalid).
var (
        ErrMalformed = fmt.Errorf("%w: malformed", ErrInvalid)
        ErrSignature = fmt.Errorf("%w: bad signature", ErrInvalid)
        ErrExpired   = fmt.Errorf("%w: expired", ErrInvalid)
        ErrSubject   = fmt.Errorf("%w: subject mismatch", ErrInvalid)
        ErrKey       = fmt.Errorf("%w: idempotency key mismatch", ErrInvalid)
        ErrArgsHash  = fmt.Errorf("%w: arguments mismatch", ErrInvalid)
)

// ErrNoSecret — ошибка конфигурации (пустой секрет), намеренно не оборачивает ErrInvalid.
var ErrNoSecret = errors.New("confirm: empty secret")

type payload struct {
        ArgsHash string `json:"args_hash"`
        Exp      int64  `json:"exp"`
        Key      string `json:"key"`
        Sub      string `json:"sub"`
}

var b64 = base64.RawURLEncoding.Strict()

// Issue возвращает токен для sub (субъект JWT), key (idempotency_key) и argsHash (из ArgsHash), который истекает через
// ttl. ttl должен быть положительным.
func Issue(secret []byte, sub, key, argsHash string, ttl time.Duration) (string, error) {
        if ttl <= 0 {
                return "", fmt.Errorf("confirm: issue: ttl must be positive, got %s", ttl)
        }
        return IssueUntil(secret, sub, key, argsHash, time.Now().Add(ttl))
}

// IssueUntil — это Issue с абсолютным сроком (обрезается до секунд); тесты используют его для истёкших токенов.
func IssueUntil(secret []byte, sub, key, argsHash string, exp time.Time) (string, error) {
        if len(secret) == 0 {
                return "", ErrNoSecret
        }
        if sub == "" || key == "" {
                return "", errors.New("confirm: issue: sub and key are required")
        }
        if !isHexSHA256(argsHash) {
                return "", fmt.Errorf("confirm: issue: args hash must be 64 lowercase hex chars, got %q", argsHash)
        }
        body, err := canonjson.Marshal(payload{ArgsHash: argsHash, Exp: exp.Unix(), Key: key, Sub: sub})
        if err != nil {
                return "", fmt.Errorf("confirm: issue: %w", err)
        }
        p := base64.RawURLEncoding.EncodeToString(body)
        return p + "." + base64.RawURLEncoding.EncodeToString(sign(secret, p)), nil
}

// Verify проверяет token по субъекту JWT вызывающего, idempotency_key вызова и аргументам, которые сервер действительно
// получил (idempotency_key и confirmation_token игнорируются, см. ArgsHash), на момент now. Проверки идут по порядку:
// подпись, exp, sub, key, args; возвращается первая неудача.
func Verify(secret []byte, token, sub, key string, args any, now time.Time) error {
        if len(secret) == 0 {
                return ErrNoSecret
        }
        p, s, ok := strings.Cut(token, ".")
        if !ok || p == "" || s == "" || strings.Contains(s, ".") {
                return ErrMalformed
        }
        sig, err := b64.DecodeString(s)
        if err != nil {
                return ErrMalformed
        }
        if !hmac.Equal(sig, sign(secret, p)) {
                return ErrSignature
        }
        raw, err := b64.DecodeString(p)
        if err != nil {
                return ErrMalformed
        }
        var pl payload
        dec := json.NewDecoder(bytes.NewReader(raw))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&pl); err != nil || pl.Sub == "" || pl.Key == "" || pl.Exp == 0 || pl.ArgsHash == "" {
                return ErrMalformed
        }
        if !now.Before(time.Unix(pl.Exp, 0)) {
                return ErrExpired
        }
        if pl.Sub != sub {
                return ErrSubject
        }
        if pl.Key != key {
                return ErrKey
        }
        h, err := ArgsHash(args)
        if err != nil {
                return fmt.Errorf("%w: %v", ErrArgsHash, err)
        }
        if !hmac.Equal([]byte(h), []byte(pl.ArgsHash)) {
                return ErrArgsHash
        }
        return nil
}

// ArgsHash возвращает SHA-256 канонического JSON от args (JSON-объект: map, структура или json.RawMessage) строчным hex
// после удаления ключей верхнего уровня idempotency_key и confirmation_token. Равен canonjson.SHA256Hex(draft.args) для
// черновика, который подтвердил пользователь.
func ArgsHash(args any) (string, error) {
        raw, err := json.Marshal(args)
        if err != nil {
                return "", fmt.Errorf("confirm: args: %w", err)
        }
        dec := json.NewDecoder(bytes.NewReader(raw))
        dec.UseNumber()
        var obj map[string]any
        if err := dec.Decode(&obj); err != nil || obj == nil {
                return "", errors.New("confirm: args must be a JSON object")
        }
        delete(obj, KeyIdempotency)
        delete(obj, KeyConfirmation)
        return canonjson.SHA256Hex(obj)
}

func sign(secret []byte, p string) []byte {
        m := hmac.New(sha256.New, secret)
        m.Write([]byte(p))
        return m.Sum(nil)
}

func isHexSHA256(s string) bool {
        if len(s) != 64 {
                return false
        }
        for i := 0; i < len(s); i++ {
                c := s[i]
                if (c < '0' || c > '9') && (c < 'a' || c > 'f') {
                        return false
                }
        }
        return true
}

// DefaultMaxBody — ограничение тела запроса в ReadJSON (1 МиБ).
const DefaultMaxBody int64 = 1 << 20

// ReadJSON декодирует тело запроса в dst с DefaultMaxBody; см. ReadJSONLimit.
func ReadJSON(w http.ResponseWriter, r *http.Request, dst any) error {
        return ReadJSONLimit(w, r, dst, DefaultMaxBody)
}

// ReadJSONLimit декодирует ровно одно JSON-значение из тела запроса в dst. Неизвестные поля объекта, лишние данные в
// конце, пустое тело и тело больше limit байт отклоняются. Любая неудача — *apierr.Error с кодом validation_error и
// русским сообщением.
func ReadJSONLimit(w http.ResponseWriter, r *http.Request, dst any, limit int64) error {
        if r.Body == nil {
                return apierr.Validation().WithMessage("Пустое тело запроса")
        }
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, limit))
        dec.DisallowUnknownFields()
        if err := dec.Decode(dst); err != nil {
                return decodeError(err)
        }
        if err := dec.Decode(&struct{}{}); !errors.Is(err, io.EOF) {
                var tooBig *http.MaxBytesError
                if errors.As(err, &tooBig) {
                        return decodeError(err)
                }
                return apierr.Validation().WithMessage("Лишние данные после JSON в теле запроса")
        }
        return nil
}

func decodeError(err error) *apierr.Error {
        var (
                tooBig  *http.MaxBytesError
                typeErr *json.UnmarshalTypeError
        )
        switch {
        case errors.Is(err, io.EOF):
                return apierr.Validation().WithMessage("Пустое тело запроса")
        case errors.As(err, &tooBig):
                return apierr.Validation().WithMessage("Слишком большой запрос")
        case errors.As(err, &typeErr):
                if typeErr.Field != "" {
                        return apierr.Validation().WithMessage(fmt.Sprintf("Неверный тип поля %q", typeErr.Field))
                }
                return apierr.Validation().WithMessage("Неверный тип данных в теле запроса")
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                field := strings.TrimPrefix(err.Error(), "json: unknown field ")
                return apierr.Validation().WithMessage("Неизвестное поле " + field)
        default:
                return apierr.Validation().WithMessage("Некорректный JSON в теле запроса")
        }
}

// WriteJSON пишет v как JSON с указанным статусом. Если v не кодируется, вместо него пишется внутренняя ошибка.
func WriteJSON(w http.ResponseWriter, status int, v any) {
        body, err := json.Marshal(v)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("httpx: encode response: %w", err))
                return
        }
        w.Header().Set("Content-Type", "application/json; charset=utf-8")
        w.WriteHeader(status)
        _, _ = w.Write(body)
}

// HeaderRequestID — заголовок id запроса; принимается от клиентов и передаётся агенту.
const HeaderRequestID = "X-Request-Id"

// Middleware оборачивает http.Handler.
type Middleware = func(http.Handler) http.Handler

// Chain составляет middleware; первый — самый внешний: Chain(a, b)(h) == a(b(h)).
func Chain(mws ...Middleware) Middleware {
        return func(h http.Handler) http.Handler {
                for i := len(mws) - 1; i >= 0; i-- {
                        h = mws[i](h)
                }
                return h
        }
}

type requestIDKey struct{}

// RequestID сохраняет корректный входящий X-Request-Id (1–128 символов из [A-Za-z0-9._:-]) или генерирует UUID v4,
// кладёт его в контекст (RequestIDFromContext), в заголовок запроса и в заголовок ответа.
func RequestID(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                id := r.Header.Get(HeaderRequestID)
                if !validRequestID(id) {
                        id = newUUID()
                }
                r.Header.Set(HeaderRequestID, id)
                w.Header().Set(HeaderRequestID, id)
                next.ServeHTTP(w, r.WithContext(WithRequestID(r.Context(), id)))
        })
}

// WithRequestID возвращает ctx с id (RequestID делает это для входящих запросов).
func WithRequestID(ctx context.Context, id string) context.Context {
        return context.WithValue(ctx, requestIDKey{}, id)
}

// RequestIDFromContext возвращает id запроса, сохранённый RequestID, или "".
func RequestIDFromContext(ctx context.Context) string {
        id, _ := ctx.Value(requestIDKey{}).(string)
        return id
}

func validRequestID(id string) bool {
        if id == "" || len(id) > 128 {
                return false
        }
        for i := 0; i < len(id); i++ {
                c := id[i]
                ok := c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || strings.IndexByte("._:-", c) >= 0
                if !ok {
                        return false
                }
        }
        return true
}

func newUUID() string {
        var b [16]byte
        _, _ = rand.Read(b[:]) // crypto/rand.Read никогда не падает на поддерживаемых платформах
        b[6] = b[6]&0x0f | 0x40
        b[8] = b[8]&0x3f | 0x80
        return fmt.Sprintf("%x-%x-%x-%x-%x", b[0:4], b[4:6], b[6:8], b[8:10], b[10:16])
}

// Recover превращает панику во внутреннюю ошибку apierr (если ещё ничего не записано) и логирует её со стеком.
// http.ErrAbortHandler пробрасывается паникой дальше, как ожидает net/http. Nil log означает slog.Default().
func Recover(log *slog.Logger) Middleware {
        if log == nil {
                log = slog.Default()
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        sw := wrap(w)
                        defer func() {
                                v := recover()
                                if v == nil {
                                        return
                                }
                                if v == http.ErrAbortHandler {
                                        panic(v)
                                }
                                log.ErrorContext(r.Context(), "panic in handler", "panic", fmt.Sprint(v),
                                        "request_id", RequestIDFromContext(r.Context()), "stack", string(debug.Stack()))
                                if !sw.wroteHeader {
                                        apierr.WriteJSON(sw, apierr.Internal())
                                }
                        }()
                        next.ServeHTTP(sw, r)
                })
        }
}

// AccessLog пишет одну строку на запрос: method, path, status, bytes, duration_ms, request_id. GET /healthz логируется
// на уровне debug, чтобы проверки здоровья контейнера не засоряли журнал. Nil log означает slog.Default().
func AccessLog(log *slog.Logger) Middleware {
        if log == nil {
                log = slog.Default()
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        sw := wrap(w)
                        next.ServeHTTP(sw, r)
                        level := slog.LevelInfo
                        if r.URL.Path == "/healthz" {
                                level = slog.LevelDebug
                        }
                        log.Log(r.Context(), level, "http request",
                                "method", r.Method, "path", r.URL.Path, "status", sw.statusCode(), "bytes", sw.bytes,
                                "duration_ms", time.Since(start).Milliseconds(), "request_id", RequestIDFromContext(r.Context()))
                })
        }
}

// statusWriter запоминает статус и размер ответа. Сохраняет работу flush для SSE (Flush, FlushError и Unwrap для
// http.ResponseController).
type statusWriter struct {
        http.ResponseWriter
        status      int
        bytes       int64
        wroteHeader bool
}

func wrap(w http.ResponseWriter) *statusWriter {
        if sw, ok := w.(*statusWriter); ok {
                return sw
        }
        return &statusWriter{ResponseWriter: w}
}

func (s *statusWriter) WriteHeader(code int) {
        if !s.wroteHeader && code >= 200 {
                s.status, s.wroteHeader = code, true
        }
        s.ResponseWriter.WriteHeader(code)
}

func (s *statusWriter) Write(b []byte) (int, error) {
        if !s.wroteHeader {
                s.status, s.wroteHeader = http.StatusOK, true
        }
        n, err := s.ResponseWriter.Write(b)
        s.bytes += int64(n)
        return n, err
}

// FlushError сбрасывает нижележащий writer и сообщает http.ErrNotSupported, если сброс невозможен.
func (s *statusWriter) FlushError() error {
        err := http.NewResponseController(s.ResponseWriter).Flush()
        if err == nil && !s.wroteHeader {
                s.status, s.wroteHeader = http.StatusOK, true // сброс неявно отправляет 200
        }
        return err
}

// Flush реализует http.Flusher.
func (s *statusWriter) Flush() { _ = s.FlushError() }

// Unwrap открывает нижележащий writer для http.ResponseController.
func (s *statusWriter) Unwrap() http.ResponseWriter { return s.ResponseWriter }

func (s *statusWriter) statusCode() int {
        if s.status == 0 {
                return http.StatusOK
        }
        return s.status
}

// Заголовки CORS, отправляемые для разрешённых origin.
const (
        corsMethods = "GET, POST, PUT, PATCH, DELETE"
        corsHeaders = "Authorization, Content-Type, Idempotency-Key, X-Request-Id, Last-Event-ID"
        corsMaxAge  = 600
)

// CORS разрешает кросс-доменные запросы с перечисленных origin в точности (scheme://host[:port], завершающий "/"
// игнорируется; "*" разрешает любой origin). Credentials никогда не разрешаются: API использует bearer-токены, а не
// cookies. Preflight-запросы (OPTIONS с Access-Control-Request-Method) отвечаются здесь: 204 для разрешённых origin,
// иначе apierr forbidden. Запросы без Origin проходят без изменений.
func CORS(origins []string) Middleware {
        allowed := make(map[string]bool, len(origins))
        anyOrigin := false
        for _, o := range origins {
                o = strings.TrimRight(strings.TrimSpace(o), "/")
                switch o {
                case "":
                case "*":
                        anyOrigin = true
                default:
                        allowed[o] = true
                }
        }
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        origin := r.Header.Get("Origin")
                        if origin == "" {
                                next.ServeHTTP(w, r)
                                return
                        }
                        h := w.Header()
                        h.Add("Vary", "Origin")
                        ok := anyOrigin || allowed[origin]
                        if r.Method == http.MethodOptions && r.Header.Get("Access-Control-Request-Method") != "" {
                                h.Add("Vary", "Access-Control-Request-Method")
                                h.Add("Vary", "Access-Control-Request-Headers")
                                if !ok {
                                        apierr.WriteJSON(w, apierr.Forbidden())
                                        return
                                }
                                setAllowOrigin(h, origin, anyOrigin)
                                h.Set("Access-Control-Allow-Methods", corsMethods)
                                h.Set("Access-Control-Allow-Headers", corsHeaders)
                                h.Set("Access-Control-Max-Age", strconv.Itoa(corsMaxAge))
                                w.WriteHeader(http.StatusNoContent)
                                return
                        }
                        if ok {
                                setAllowOrigin(h, origin, anyOrigin)
                                h.Set("Access-Control-Expose-Headers", HeaderRequestID)
                        }
                        next.ServeHTTP(w, r)
                })
        }
}

func setAllowOrigin(h http.Header, origin string, anyOrigin bool) {
        if anyOrigin {
                h.Set("Access-Control-Allow-Origin", "*")
                return
        }
        h.Set("Access-Control-Allow-Origin", origin)
}

// DefaultKeepAlive — интервал комментариев `: ping` в потоке SSE (contracts/agent-stream.md).
const DefaultKeepAlive = 15 * time.Second

// ErrSSEClosed возвращает SSEWriter.Send после Close.
var ErrSSEClosed = errors.New("httpx: sse stream closed")

// SSEWriter пишет Server-Sent Events (R78): каждое событие получает возрастающий `id:` (с 1), строку `event:` и одну
// строку `data:` с компактным JSON и сразу сбрасывается. Горутина пишет комментарии `: ping` раз в интервал keep-alive
// до Close или завершения контекста. Все методы безопасны для конкурентного использования. Всегда пишите `defer
// sse.Close()` в обработчике: горутина пингов должна остановиться до возврата из обработчика.
type SSEWriter struct {
        ctx    context.Context
        w      http.ResponseWriter
        rc     *http.ResponseController
        mu     sync.Mutex
        id     int64
        err    error
        closed bool
        stop   chan struct{}
        wg     sync.WaitGroup
}

// NewSSEWriter ставит заголовки SSE, отправляет статус 200, сбрасывает и запускает горутину keep-alive (keepAlive <= 0
// — DefaultKeepAlive). ctx обычно r.Context(). Падает, ничего не записав, если w не умеет сбрасывать. Дедлайн записи
// сервера для потока снимается, если writer это поддерживает.
func NewSSEWriter(ctx context.Context, w http.ResponseWriter, keepAlive time.Duration) (*SSEWriter, error) {
        if keepAlive <= 0 {
                keepAlive = DefaultKeepAlive
        }
        rc := http.NewResponseController(w)
        h := w.Header()
        h.Set("Content-Type", "text/event-stream; charset=utf-8")
        h.Set("Cache-Control", "no-cache")
        h.Set("X-Accel-Buffering", "no")
        if err := rc.Flush(); err != nil {
                return nil, fmt.Errorf("httpx: sse: response writer cannot flush: %w", err)
        }
        _ = rc.SetWriteDeadline(time.Time{}) // долгоживущий поток; ErrNotSupported допустим
        s := &SSEWriter{ctx: ctx, w: w, rc: rc, stop: make(chan struct{})}
        s.wg.Add(1)
        go s.keepAlive(keepAlive)
        return s, nil
}

func (s *SSEWriter) keepAlive(every time.Duration) {
        defer s.wg.Done()
        t := time.NewTicker(every)
        defer t.Stop()
        for {
                select {
                case <-s.ctx.Done():
                        return
                case <-s.stop:
                        return
                case <-t.C:
                        if err := s.write(func(b *bytes.Buffer) { b.WriteString(": ping\n\n") }); err != nil {
                                return
                        }
                }
        }
}

// Send пишет одно событие с указанным именем и данными в виде однострочного JSON (HTML-символы не экранируются;
// json.RawMessage уплотняется, поэтому пересылаемые события агента сохраняют смысл). Возвращает первую ошибку записи
// потока, ctx.Err() после завершения контекста или ErrSSEClosed после Close.
func (s *SSEWriter) Send(event string, data any) error {
        if event == "" || strings.ContainsAny(event, "\r\n") {
                return fmt.Errorf("httpx: sse: invalid event name %q", event)
        }
        var payload bytes.Buffer
        enc := json.NewEncoder(&payload)
        enc.SetEscapeHTML(false)
        if err := enc.Encode(data); err != nil {
                return fmt.Errorf("httpx: sse: encode %s: %w", event, err)
        }
        line := bytes.TrimRight(payload.Bytes(), "\n")
        return s.write(func(b *bytes.Buffer) {
                s.id++
                b.WriteString("id: ")
                b.WriteString(strconv.FormatInt(s.id, 10))
                b.WriteString("\nevent: ")
                b.WriteString(event)
                b.WriteString("\ndata: ")
                b.Write(line)
                b.WriteString("\n\n")
        })
}

// write собирает кадр под блокировкой (чтобы id назначались в порядке записи), пишет и сбрасывает его.
func (s *SSEWriter) write(frame func(*bytes.Buffer)) error {
        s.mu.Lock()
        defer s.mu.Unlock()
        switch {
        case s.closed:
                return ErrSSEClosed
        case s.err != nil:
                return s.err
        case s.ctx.Err() != nil:
                return s.ctx.Err()
        }
        var b bytes.Buffer
        frame(&b)
        if _, err := s.w.Write(b.Bytes()); err != nil {
                s.err = fmt.Errorf("httpx: sse: write: %w", err)
                return s.err
        }
        if err := s.rc.Flush(); err != nil {
                s.err = fmt.Errorf("httpx: sse: flush: %w", err)
                return s.err
        }
        return nil
}

// LastID возвращает id последнего отправленного события (0 до первого).
func (s *SSEWriter) LastID() int64 {
        s.mu.Lock()
        defer s.mu.Unlock()
        return s.id
}

// Close останавливает горутину keep-alive и ждёт её; последующие Send возвращают ErrSSEClosed. Идемпотентен и не
// закрывает нижележащее соединение (это делает возврат из обработчика).
func (s *SSEWriter) Close() error {
        s.mu.Lock()
        if s.closed {
                s.mu.Unlock()
                return nil
        }
        s.closed = true
        close(s.stop)
        s.mu.Unlock()
        s.wg.Wait()
        return nil
}

// MaxSSELine — самая длинная строка, которую принимает SSEReader (1 МиБ).
const MaxSSELine = 1 << 20

// ErrSSEData отмечает событие, у которого data — некорректный JSON; читатель может продолжить со следующего события.
var ErrSSEData = errors.New("httpx: sse: event data is not valid JSON")

// ErrSSETooLarge отмечает строку длиннее MaxSSELine или событие сверх лимита читателя (NewSSEReaderLimit). Дальше поток
// читать нельзя.
var ErrSSETooLarge = errors.New("httpx: sse: event too large")

// Event — одно Server-Sent Event. Name по умолчанию "message"; ID — последний id, встреченный в потоке (SSE «last event
// ID»), поэтому он сохраняется и для событий без строки id.
type Event struct {
        ID   string
        Name string
        Data json.RawMessage
}

// SSEReader разбирает text/event-stream (правила WHATWG): концы строк LF, CR и CRLF, BOM в начале, строки-комментарии,
// `field:value` с одним необязательным пробелом, несколько строк data склеиваются через "\n", неизвестные поля и retry
// игнорируются. В отличие от браузера, последнее событие без следующей пустой строки всё равно доставляется.
type SSEReader struct {
        sc       *bufio.Scanner
        lastID   string
        started  bool
        maxEvent int // байт data на событие; 0 — без лимита, кроме MaxSSELine на строку
        failed   error
}

// NewSSEReader возвращает читатель поверх r (обычно тела HTTP-ответа).
func NewSSEReader(r io.Reader) *SSEReader {
        return NewSSEReaderLimit(r, 0)
}

// sseLineSlack — запас под имя поля ("data: ") в строке ограниченного читателя.
const sseLineSlack = 64

// NewSSEReaderLimit — это NewSSEReader с лимитом maxEvent байт на data одного события (все его строки data вместе) и на
// одну строку (плюс запас под имя поля), для потоков от менее доверенных сторон; maxEvent <= 0 — без лимита на событие.
// Большее событие заставляет Next вернуть ErrSSETooLarge и остановиться.
func NewSSEReaderLimit(r io.Reader, maxEvent int) *SSEReader {
        maxLine := MaxSSELine
        if maxEvent > 0 && maxEvent+sseLineSlack < maxLine {
                maxLine = maxEvent + sseLineSlack
        } else if maxEvent < 0 {
                maxEvent = 0
        }
        sc := bufio.NewScanner(r)
        sc.Buffer(make([]byte, 0, min(4096, maxLine)), maxLine)
        sc.Split(splitLines)
        return &SSEReader{sc: sc, maxEvent: maxEvent}
}

// Next возвращает следующее событие, io.EOF в конце потока, ошибку чтения или событие вместе с ошибкой, оборачивающей
// ErrSSEData, если его data — некорректный JSON (после этого Next можно вызывать снова).
func (r *SSEReader) Next() (Event, error) {
        if r.failed != nil {
                return Event{}, r.failed
        }
        var (
                name    string
                data    []byte
                hasData bool
        )
        for r.sc.Scan() {
                line := r.sc.Bytes()
                if !r.started {
                        r.started = true
                        line = bytes.TrimPrefix(line, []byte("\xef\xbb\xbf"))
                }
                if len(line) == 0 {
                        if !hasData {
                                name = ""
                                continue
                        }
                        return r.dispatch(name, data)
                }
                if line[0] == ':' {
                        continue
                }
                field, value, _ := bytes.Cut(line, []byte(":"))
                value = bytes.TrimPrefix(value, []byte(" "))
                switch string(field) {
                case "event":
                        name = string(value)
                case "data":
                        if r.maxEvent > 0 && len(data)+1+len(value) > r.maxEvent {
                                r.failed = fmt.Errorf("%w: more than %d bytes of data", ErrSSETooLarge, r.maxEvent)
                                return Event{}, r.failed
                        }
                        if hasData {
                                data = append(data, '\n')
                        }
                        data = append(data, value...)
                        hasData = true
                case "id":
                        if bytes.IndexByte(value, 0) < 0 {
                                r.lastID = string(value)
                        }
                }
        }
        if err := r.sc.Err(); err != nil {
                if errors.Is(err, bufio.ErrTooLong) {
                        r.failed = fmt.Errorf("%w: line too long", ErrSSETooLarge)
                        return Event{}, r.failed
                }
                return Event{}, fmt.Errorf("httpx: sse: read: %w", err)
        }
        if hasData {
                return r.dispatch(name, data)
        }
        return Event{}, io.EOF
}

func (r *SSEReader) dispatch(name string, data []byte) (Event, error) {
        if name == "" {
                name = "message"
        }
        ev := Event{ID: r.lastID, Name: name, Data: json.RawMessage(data)}
        if !json.Valid(data) {
                return ev, fmt.Errorf("%w: event %q", ErrSSEData, name)
        }
        return ev, nil
}

// splitLines — bufio.SplitFunc для концов строк SSE: "\r\n", "\n" или "\r".
func splitLines(data []byte, atEOF bool) (int, []byte, error) {
        if atEOF && len(data) == 0 {
                return 0, nil, nil
        }
        if i := bytes.IndexAny(data, "\r\n"); i >= 0 {
                if data[i] == '\n' {
                        return i + 1, data[:i], nil
                }
                if i+1 < len(data) {
                        if data[i+1] == '\n' {
                                return i + 2, data[:i], nil
                        }
                        return i + 1, data[:i], nil
                }
                if !atEOF {
                        return 0, nil, nil // "\n" может прийти в следующем чтении
                }
                return i + 1, data[:i], nil
        }
        if atEOF {
                return len(data), data, nil
        }
        return 0, nil, nil
}

// Client вызывает инструменты мок-серверов MCP от имени пользователя (ядро: approvals.*, kb.add_faq; тесты). Нулевое
// значение непригодно: задайте BaseURL. Безопасен для конкурентного использования.
type Client struct {
        BaseURL    string       // например http://mcp-mocks:8100/mcp; имя сервера добавляется как /<server>
        HTTPClient *http.Client // nil означает клиента с http.DefaultTransport без таймаута (используйте дедлайны ctx)
}

// Call открывает сессию к BaseURL/<server> с `Authorization: Bearer userToken`, вызывает tool с args (любое
// значение-JSON-объект; nil — {}), закрывает сессию и возвращает ровно одно из:
//   - структурированный результат (JSON-объект) при успехе;
//   - ошибку инструмента (*apierr.Error с кодом MCP: forbidden, not_found, validation_error, conflict,
//     invalid_confirmation, unavailable);
//   - ошибку при сбоях транспорта и протокола. Она оборачивает *apierr.Error (ищите через apierr.As):
//     unauthorized, если сервер отклонил токен, internal для ошибок протокола (например, неизвестный инструмент),
//     иначе upstream_unavailable. Она же оборачивает причину (ошибки контекста работают с errors.Is).
//
// X-Request-Id из httpx.RequestIDFromContext(ctx) передаётся дальше.
func (c *Client) Call(ctx context.Context, userToken, server, tool string, args any) (json.RawMessage, *apierr.Error, error) {
        rt := &bearerTransport{token: userToken, base: http.DefaultTransport}
        hc := &http.Client{Transport: rt}
        if c.HTTPClient != nil {
                if c.HTTPClient.Transport != nil {
                        rt.base = c.HTTPClient.Transport
                }
                hc.Timeout = c.HTTPClient.Timeout
        }
        transport := &mcp.StreamableClientTransport{
                Endpoint:   strings.TrimRight(c.BaseURL, "/") + "/" + server,
                HTTPClient: hc,
                MaxRetries: -1,
        }
        client := mcp.NewClient(&mcp.Implementation{Name: "aibuddy-core", Version: version}, nil)
        cs, err := client.Connect(ctx, transport, nil)
        if err != nil {
                return nil, nil, rt.wrap(ctx, fmt.Sprintf("connect %s", server), err)
        }
        defer cs.Close()
        res, err := cs.CallTool(ctx, &mcp.CallToolParams{Name: tool, Arguments: args})
        if err != nil {
                return nil, nil, rt.wrap(ctx, fmt.Sprintf("call %s.%s", server, tool), err)
        }
        if res.IsError {
                return nil, toolError(res), nil
        }
        out, err := structured(res)
        if err != nil {
                return nil, nil, fmt.Errorf("mcpx: call %s.%s: %w: %w", server, tool, apierr.Internal(), err)
        }
        return out, nil, nil
}

// structured возвращает структурированное содержимое успешного результата, а если его нет — текстовое содержимое JSON.
func structured(res *mcp.CallToolResult) (json.RawMessage, error) {
        if res.StructuredContent != nil {
                raw, err := json.Marshal(res.StructuredContent)
                if err != nil {
                        return nil, fmt.Errorf("encode structured content: %w", err)
                }
                return raw, nil
        }
        if text := firstText(res); json.Valid([]byte(text)) && strings.HasPrefix(strings.TrimSpace(text), "{") {
                return json.RawMessage(text), nil
        }
        return nil, errors.New("result has no structured content")
}

// toolError декодирует {code, message, retryable} из результата с isError; неизвестные формы становятся unavailable.
func toolError(res *mcp.CallToolResult) *apierr.Error {
        var wire struct {
                Code      apierr.Code `json:"code"`
                Message   string      `json:"message"`
                Retryable bool        `json:"retryable"`
        }
        raw, _ := json.Marshal(res.StructuredContent)
        if res.StructuredContent == nil {
                raw = []byte(firstText(res))
        }
        if err := json.Unmarshal(raw, &wire); err != nil || wire.Code == "" {
                e := apierr.Unavailable()
                if text := firstText(res); text != "" {
                        e = e.WithMessage(text)
                }
                e.Retryable = false
                return e
        }
        e := apierr.New(wire.Code, wire.Message)
        e.Retryable = wire.Retryable
        return e
}

func firstText(res *mcp.CallToolResult) string {
        for _, c := range res.Content {
                if t, ok := c.(*mcp.TextContent); ok {
                        return t.Text
                }
        }
        return ""
}

// bearerTransport добавляет токен пользователя и id запроса к каждому запросу сессии и запоминает, был ли сбой
// какого-либо HTTP-обмена, чтобы классифицировать ошибки.
type bearerTransport struct {
        token string
        base  http.RoundTripper

        mu           sync.Mutex
        unauthorized bool
        httpFailed   bool
}

func (t *bearerTransport) RoundTrip(req *http.Request) (*http.Response, error) {
        r := req.Clone(req.Context())
        r.Header.Set("Authorization", "Bearer "+t.token)
        if id := httpx.RequestIDFromContext(req.Context()); id != "" {
                r.Header.Set(httpx.HeaderRequestID, id)
        }
        resp, err := t.base.RoundTrip(r)
        t.mu.Lock()
        defer t.mu.Unlock()
        switch {
        case err != nil:
                t.httpFailed = true
        case resp.StatusCode == http.StatusUnauthorized:
                t.unauthorized = true
        case resp.StatusCode >= 400 && !(req.Method == http.MethodGet && resp.StatusCode == http.StatusMethodNotAllowed) &&
                req.Method != http.MethodDelete:
                t.httpFailed = true // 405 на необязательный висящий GET и на DELETE при закрытии ожидаемы
        }
        return resp, err
}

func (t *bearerTransport) wrap(ctx context.Context, op string, err error) error {
        t.mu.Lock()
        unauthorized, httpFailed := t.unauthorized, t.httpFailed
        t.mu.Unlock()
        kind := apierr.UpstreamUnavailable()
        switch {
        case unauthorized:
                kind = apierr.Unauthorized()
        case !httpFailed && ctx.Err() == nil:
                kind = apierr.Internal() // сервер ответил ошибкой JSON-RPC
        }
        if cerr := ctx.Err(); cerr != nil && !errors.Is(err, cerr) {
                err = fmt.Errorf("%w: %w", cerr, err)
        }
        return fmt.Errorf("mcpx: %s: %w: %w", op, kind, err)
}

// PathPrefix — где Mount обслуживает серверы: /mcp/<name>.
const PathPrefix = "/mcp/"

// MaxRequestBody ограничивает тело одного HTTP-запроса MCP (1 МиБ).
const MaxRequestBody = 1 << 20

const (
        callerKey = "mcpx.caller" // ключ в auth.TokenInfo.Extra
        version   = "0.1.0"
)

// Mount проверяет каждый сервер и обслуживает его по PathPrefix+name на mux. Падает на первом некорректном сервере.
func Mount(mux *http.ServeMux, servers ...*Server) error {
        for _, s := range servers {
                h, err := s.Handler()
                if err != nil {
                        return err
                }
                mux.Handle(PathPrefix+s.name, h)
        }
        return nil
}

// Handler проверяет сервер (см. Validate) и возвращает его HTTP-обработчик: проверка JWT, затем streamable
// HTTP-обработчик go-sdk в режиме без состояния с JSON-ответами (без id сессии, без SSE). Отсутствующий или
// некорректный JWT получает 401 с `WWW-Authenticate: Bearer` и телом ошибки JSON-RPC, в data которого JSON apierr.
func (s *Server) Handler() (http.Handler, error) {
        if err := s.Validate(); err != nil {
                return nil, err
        }
        srv := mcp.NewServer(&mcp.Implementation{Name: "aibuddy-" + s.name, Version: version}, &mcp.ServerOptions{
                HasTools:     true,
                GetSessionID: func() string { return "" }, // без состояния: без Mcp-Session-Id
        })
        for _, n := range s.order {
                t := s.tools[n]
                mt := &mcp.Tool{Name: t.def.Name, Description: t.def.Description, InputSchema: t.def.InputSchema}
                if len(t.def.OutputSchema) > 0 {
                        mt.OutputSchema = t.def.OutputSchema
                }
                srv.AddTool(mt, s.toolHandler(t))
        }
        srv.AddReceivingMiddleware(s.filterToolList)
        streamable := mcp.NewStreamableHTTPHandler(func(*http.Request) *mcp.Server { return srv },
                &mcp.StreamableHTTPOptions{Stateless: true, JSONResponse: true})
        // RequireBearerToken — так go-sdk переносит личность: TokenInfo, который он кладёт в контекст запроса,
        // streamable-транспорт копирует в CallToolRequest.Extra.TokenInfo для каждого сообщения JSON-RPC.
        withTokenInfo := auth.RequireBearerToken(tokenInfo, nil)(streamable)
        verifier := authn.NewVerifier(s.cfg.JWTSecret)
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                token, ok := authn.BearerToken(r)
                if !ok {
                        writeUnauthorized(w)
                        return
                }
                claims, err := verifier.Verify(token)
                if err != nil {
                        s.log.DebugContext(r.Context(), "mcp jwt rejected", "server", s.name, "err", err)
                        writeUnauthorized(w)
                        return
                }
                r.Body = http.MaxBytesReader(w, r.Body, MaxRequestBody)
                ctx := authn.WithToken(authn.NewContext(r.Context(), claims), token)
                withTokenInfo.ServeHTTP(w, r.WithContext(ctx))
        }), nil
}

// tokenInfo — это auth.TokenVerifier: JWT уже проверен, поэтому он только оборачивает вызывающего.
func tokenInfo(ctx context.Context, _ string, _ *http.Request) (*auth.TokenInfo, error) {
        c, ok := callerFromContext(ctx)
        if !ok {
                return nil, auth.ErrInvalidToken
        }
        return &auth.TokenInfo{
                Expiration: time.Unix(c.Exp, 0).Add(authn.DefaultLeeway),
                Extra:      map[string]any{callerKey: c},
        }, nil
}

func callerFromContext(ctx context.Context) (Caller, bool) {
        claims, ok := authn.FromContext(ctx)
        if !ok {
                return Caller{}, false
        }
        token, _ := authn.TokenFromContext(ctx)
        return Caller{Claims: claims, Token: token}, true
}

// CallerFrom возвращает вызывающего MCP-запроса: из req.Extra.TokenInfo (ставит Handler), а если его нет — из
// контекста. Обработчики инструментов получают Caller напрямую; это для собственных обработчиков go-sdk и тестов.
func CallerFrom(ctx context.Context, req mcp.Request) (Caller, bool) {
        if req != nil {
                if extra := req.GetExtra(); extra != nil && extra.TokenInfo != nil {
                        if c, ok := extra.TokenInfo.Extra[callerKey].(Caller); ok {
                                return c, true
                        }
                }
        }
        return callerFromContext(ctx)
}

func (s *Server) toolHandler(t *tool) mcp.ToolHandler {
        return func(ctx context.Context, req *mcp.CallToolRequest) (*mcp.CallToolResult, error) {
                c, ok := CallerFrom(ctx, req)
                if !ok {
                        return errorResult(apierr.ToMCP(apierr.Unauthorized())), nil
                }
                out, aerr := s.run(ctx, c, t, req.Params.Arguments)
                if aerr != nil {
                        return errorResult(aerr), nil
                }
                return &mcp.CallToolResult{
                        StructuredContent: out,
                        Content:           []mcp.Content{&mcp.TextContent{Text: string(out)}},
                }, nil
        }
}

func errorResult(e *apierr.Error) *mcp.CallToolResult {
        body, _ := json.Marshal(e) // строки и bool: ошибки быть не может
        return &mcp.CallToolResult{
                IsError:           true,
                StructuredContent: json.RawMessage(body),
                Content:           []mcp.Content{&mcp.TextContent{Text: string(body)}},
        }
}

// filterToolList скрывает из tools/list инструменты, которые роли вызывающего вызывать нельзя (прямые вызовы получают
// forbidden).
func (s *Server) filterToolList(next mcp.MethodHandler) mcp.MethodHandler {
        return func(ctx context.Context, method string, req mcp.Request) (mcp.Result, error) {
                res, err := next(ctx, method, req)
                list, ok := res.(*mcp.ListToolsResult)
                if err != nil || !ok {
                        return res, err
                }
                c, _ := CallerFrom(ctx, req)
                visible := make([]*mcp.Tool, 0, len(list.Tools))
                for _, mt := range list.Tools {
                        if t := s.tools[mt.Name]; t != nil && t.allows(c.Role) {
                                visible = append(visible, mt)
                        }
                }
                filtered := *list
                filtered.Tools = visible
                return &filtered, nil
        }
}

// ToolDef — один инструмент из contracts/schemas/<server>.json.
type ToolDef struct {
        Name         string          `json:"name"`
        Description  string          `json:"description"`
        Roles        []string        `json:"roles"`
        Write        bool            `json:"write"`
        InputSchema  json.RawMessage `json:"inputSchema"`
        OutputSchema json.RawMessage `json:"outputSchema"`
}

type schemaFile struct {
        Server string    `json:"server"`
        Tools  []ToolDef `json:"tools"`
}

// LoadTools читает определения инструментов server из contracts.Schemas (schemas/<server>.json).
func LoadTools(server string) ([]ToolDef, error) {
        return loadTools(contracts.Schemas, server)
}

func loadTools(fsys fs.FS, server string) ([]ToolDef, error) {
        raw, err := fs.ReadFile(fsys, "schemas/"+server+".json")
        if err != nil {
                return nil, fmt.Errorf("mcpx: load schema %q: %w", server, err)
        }
        var f schemaFile
        if err := json.Unmarshal(raw, &f); err != nil {
                return nil, fmt.Errorf("mcpx: parse schema %q: %w", server, err)
        }
        if f.Server != server {
                return nil, fmt.Errorf("mcpx: schema file %q declares server %q", server, f.Server)
        }
        return f.Tools, nil
}

// tool — скомпилированный ToolDef.
type tool struct {
        def     ToolDef
        input   *jsonschema.Resolved
        output  *jsonschema.Resolved // nil, если в контракте нет выходной схемы
        formats map[string]string    // аргумент верхнего уровня -> формат (date, date-time, uuid)
}

func compile(def ToolDef) (*tool, error) {
        if def.Name == "" {
                return nil, fmt.Errorf("mcpx: tool without name")
        }
        if len(def.Roles) == 0 {
                return nil, fmt.Errorf("mcpx: tool %q: no roles", def.Name)
        }
        in, props, err := resolve(def.InputSchema)
        if err != nil {
                return nil, fmt.Errorf("mcpx: tool %q: input schema: %w", def.Name, err)
        }
        if def.Write {
                for _, k := range []string{"idempotency_key", "confirmation_token"} {
                        if _, ok := props[k]; !ok {
                                return nil, fmt.Errorf("mcpx: write tool %q: input schema lacks %s", def.Name, k)
                        }
                }
        }
        t := &tool{def: def, input: in, formats: map[string]string{}}
        for name, p := range props {
                if p != nil && formatCheckers[p.Format] != nil {
                        t.formats[name] = p.Format
                }
        }
        if len(def.OutputSchema) > 0 {
                if t.output, _, err = resolve(def.OutputSchema); err != nil {
                        return nil, fmt.Errorf("mcpx: tool %q: output schema: %w", def.Name, err)
                }
        }
        return t, nil
}

func resolve(raw json.RawMessage) (*jsonschema.Resolved, map[string]*jsonschema.Schema, error) {
        var s jsonschema.Schema
        if err := json.Unmarshal(raw, &s); err != nil {
                return nil, nil, err
        }
        if s.Type != "object" {
                return nil, nil, fmt.Errorf(`type must be "object", got %q`, s.Type)
        }
        r, err := s.Resolve(nil)
        if err != nil {
                return nil, nil, err
        }
        return r, s.Properties, nil
}

func (t *tool) allows(role string) bool { return slices.Contains(t.def.Roles, role) }

var uuidRE = regexp.MustCompile(`^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$`)

// formatCheckers проверяют ключевые слова "format", которые используют контракты (jsonschema-go считает format только
// аннотацией).
var formatCheckers = map[string]func(string) bool{
        "date":      func(s string) bool { _, err := time.Parse(time.DateOnly, s); return err == nil },
        "date-time": func(s string) bool { _, err := time.Parse(time.RFC3339, s); return err == nil },
        "uuid":      uuidRE.MatchString,
}

var formatNames = map[string]string{
        "date":      "дата в формате ГГГГ-ММ-ДД",
        "date-time": "дата и время в формате RFC 3339",
        "uuid":      "UUID",
}

// checkFormats проверяет формат строковых аргументов верхнего уровня.
func (t *tool) checkFormats(args map[string]any) error {
        for name, format := range t.formats {
                s, ok := args[name].(string)
                if ok && !formatCheckers[format](s) {
                        return fmt.Errorf("поле %s: ожидается %s", name, formatNames[format])
                }
        }
        return nil
}

// Пакет mcpx собирает MCP-серверы mcp-mocks из контрактов и вызывает их из ядра (docs/backend/IMPLEMENTATION.md §4,
// contracts/mcp-tools.md).
//
// Серверная сторона: NewServer загружает contracts/schemas/<name>.json, Handle привязывает обычную функцию Go к каждому
// инструменту, а Handler обслуживает streamable HTTP-транспорт MCP (без состояния, JSON-ответы) за проверкой JWT. Для
// каждого tools/call mcpx до запуска обработчика проверяет по порядку:
//  1. белый список ролей инструмента (`roles`) → forbidden
//  2. инструменты записи: confirmation_token и idempotency_key есть и не пустые → иначе invalid_confirmation
//  3. inputSchema (неизвестные поля, например employee_id, типы, перечисления, шаблоны, required; формат date,
//     date-time, uuid у аргументов верхнего уровня) после применения значений по умолчанию → validation_error
//  4. инструменты записи: confirm.Verify(sub = вызывающий, key = idempotency_key, args как получены) →
//     invalid_confirmation
//
// Личность доходит до обработчиков через механизм bearer-авторизации go-sdk: Handler проверяет JWT, оборачивает его в
// auth.TokenInfo (auth.RequireBearerToken), а streamable-транспорт копирует его в CallToolRequest.Extra.
//
// Ошибки обработчика типа *apierr.Error становятся результатами инструмента с isError=true и structuredContent {code,
// message, retryable} (коды MCP, см. apierr.ToMCP); прочие ошибки и паники логируются и отдаются как общая ошибка без
// повтора. Успешные результаты должны соответствовать outputSchema инструмента; они отправляются как structuredContent
// плюс тот же JSON в текстовом содержимом. tools/list показывает только инструменты, доступные роли вызывающего.
package mcpx

import (
        "bytes"
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "log/slog"
        "runtime/debug"
        "slices"

// Caller — аутентифицированный пользователь MCP-запроса: проверенные claims JWT плюс исходный bearer-токен. Личность в
// обработчиках инструментов берётся только отсюда, никогда из аргументов инструмента (R71).
type Caller struct {
        authn.Claims
        Token string
}

// HandlerFunc реализует инструмент. args — проверенные аргументы инструмента в JSON (с idempotency_key и
// confirmation_token для инструментов записи; значения по умолчанию из схемы применены). Результат должен
// сериализоваться в JSON-объект, соответствующий outputSchema инструмента. Для ожидаемых сбоев возвращайте
// *apierr.Error.
type HandlerFunc func(ctx context.Context, c Caller, args json.RawMessage) (any, error)

// Config хранит секреты и логгер Server.
type Config struct {
        JWTSecret          []byte       // JWT_SECRET: проверяет bearer-токен вызывающего
        ConfirmationSecret []byte       // CONFIRMATION_HMAC_SECRET: проверяет подтверждения записи
        Logger             *slog.Logger // nil означает slog.Default()
}

// Server — один мок-сервер MCP (absence, approvals, access, checklist, directory, kb).
type Server struct {
        name     string
        cfg      Config
        log      *slog.Logger
        order    []string // имена инструментов в порядке контракта
        tools    map[string]*tool
        handlers map[string]HandlerFunc
        errs     []error
}

// NewServer возвращает сервер, инструменты которого определены в contracts/schemas/<name>.json. Падает, если схемы нет
// или она некорректна. Привяжите каждый инструмент через Handle, затем вызовите Handler (или Mount).
func NewServer(name string, cfg Config) (*Server, error) {
        defs, err := LoadTools(name)
        if err != nil {
                return nil, err
        }
        return newServer(name, defs, cfg)
}

func newServer(name string, defs []ToolDef, cfg Config) (*Server, error) {
        if name == "" || strings.ContainsAny(name, "/ ") {
                return nil, fmt.Errorf("mcpx: invalid server name %q", name)
        }
        s := &Server{name: name, cfg: cfg, log: cfg.Logger, tools: map[string]*tool{}, handlers: map[string]HandlerFunc{}}
        if s.log == nil {
                s.log = slog.Default()
        }
        for _, d := range defs {
                t, err := compile(d)
                if err != nil {
                        return nil, fmt.Errorf("server %q: %w", name, err)
                }
                if _, dup := s.tools[d.Name]; dup {
                        return nil, fmt.Errorf("mcpx: server %q: duplicate tool %q", name, d.Name)
                }
                s.tools[d.Name] = t
                s.order = append(s.order, d.Name)
        }
        return s, nil
}

// Name возвращает имя сервера; он обслуживается по /mcp/<name>.
func (s *Server) Name() string { return s.name }

// Tools возвращает определения инструментов в порядке контракта.
func (s *Server) Tools() []ToolDef {
        out := make([]ToolDef, 0, len(s.order))
        for _, n := range s.order {
                out = append(out, s.tools[n].def)
        }
        return out
}

// Handle привязывает h к инструменту с именем name. О неизвестных именах, дубликатах и nil-обработчиках сообщает
// Validate.
func (s *Server) Handle(name string, h HandlerFunc) {
        switch {
        case s.tools[name] == nil:
                s.errs = append(s.errs, fmt.Errorf("mcpx: server %q: tool %q is not in the contract", s.name, name))
        case h == nil:
                s.errs = append(s.errs, fmt.Errorf("mcpx: server %q: nil handler for %q", s.name, name))
        case s.handlers[name] != nil:
                s.errs = append(s.errs, fmt.Errorf("mcpx: server %q: tool %q registered twice", s.name, name))
        default:
                s.handlers[name] = h
        }
}

// Validate сообщает обо всех проблемах регистрации: неизвестные или повторные инструменты, инструменты контракта без
// обработчика, отсутствующие секреты. Handler вызывает его, поэтому неправильно настроенный сервер падает при старте.
func (s *Server) Validate() error {
        errs := slices.Clone(s.errs)
        for _, n := range s.order {
                if s.handlers[n] == nil {
                        errs = append(errs, fmt.Errorf("mcpx: server %q: no handler for contract tool %q", s.name, n))
                }
        }
        if len(s.cfg.JWTSecret) == 0 {
                errs = append(errs, fmt.Errorf("mcpx: server %q: empty JWT secret", s.name))
        }
        if len(s.cfg.ConfirmationSecret) == 0 && slices.ContainsFunc(s.order, func(n string) bool { return s.tools[n].def.Write }) {
                errs = append(errs, fmt.Errorf("mcpx: server %q: empty confirmation secret", s.name))
        }
        return errors.Join(errs...)
}

// Invoke выполняет tool для c через те же проверки, что и MCP tools/call (роль, схема, подтверждение, обработчик,
// выходная схема), без HTTP. Возвращает структурированный результат или ошибку инструмента; мок-серверы используют его
// в тестах.
func (s *Server) Invoke(ctx context.Context, c Caller, name string, args any) (json.RawMessage, *apierr.Error) {
        raw, err := json.Marshal(args)
        if err != nil {
                return nil, apierr.ToMCP(apierr.Validation().WithMessage("Аргументы инструмента не кодируются в JSON"))
        }
        t := s.tools[name]
        if t == nil || s.handlers[name] == nil {
                return nil, apierr.ToMCP(apierr.NotFound().WithMessage(fmt.Sprintf("Инструмент %s.%s не найден", s.name, name)))
        }
        return s.run(ctx, c, t, raw)
}

func (s *Server) run(ctx context.Context, c Caller, t *tool, raw json.RawMessage) (json.RawMessage, *apierr.Error) {
        if !t.allows(c.Role) {
                return nil, apierr.Forbidden()
        }
        if t.def.Write && !hasConfirmationFields(raw) {
                // Матрица прав требует invalid_confirmation (а не validation_error) для записи без токена.
                return nil, apierr.InvalidConfirmation()
        }
        args, aerr := s.checkArgs(t, raw)
        if aerr != nil {
                return nil, aerr
        }
        if t.def.Write {
                key, _ := args[confirm.KeyIdempotency].(string)
                token, _ := args[confirm.KeyConfirmation].(string)
                if err := confirm.Verify(s.cfg.ConfirmationSecret, token, c.Sub, key, raw, time.Now()); err != nil {
                        s.log.InfoContext(ctx, "mcp confirmation rejected", "server", s.name, "tool", t.def.Name, "sub", c.Sub, "reason", err)
                        return nil, apierr.InvalidConfirmation()
                }
        }
        withDefaults, err := json.Marshal(args)
        if err != nil {
                return nil, s.internal(ctx, t, fmt.Errorf("re-encode args: %w", err))
        }
        out, err := s.call(ctx, c, t, withDefaults)
        if err != nil {
                if e, ok := apierr.As(err); ok {
                        return nil, apierr.ToMCP(e)
                }
                if ctx.Err() != nil {
                        return nil, apierr.ToMCP(apierr.From(ctx.Err()))
                }
                return nil, s.internal(ctx, t, err)
        }
        result, err := s.checkOutput(t, out)
        if err != nil {
                return nil, s.internal(ctx, t, err)
        }
        return result, nil
}

// checkArgs декодирует аргументы, применяет значения по умолчанию из схемы и проверяет их.
func (s *Server) checkArgs(t *tool, raw json.RawMessage) (map[string]any, *apierr.Error) {
        args := map[string]any{}
        if len(bytes.TrimSpace(raw)) > 0 && !bytes.Equal(bytes.TrimSpace(raw), []byte("null")) {
                if err := json.Unmarshal(raw, &args); err != nil {
                        return nil, apierr.Validation().WithMessage("Аргументы инструмента должны быть JSON-объектом")
                }
        }
        if err := t.input.ApplyDefaults(&args); err != nil {
                return nil, apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        if err := t.input.Validate(&args); err != nil {
                return nil, apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        if err := t.checkFormats(args); err != nil {
                return nil, apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        return args, nil
}

// hasConfirmationFields сообщает, является ли raw объектом с непустыми строками confirmation_token и idempotency_key.
func hasConfirmationFields(raw json.RawMessage) bool {
        var f struct {
                Token *string `json:"confirmation_token"`
                Key   *string `json:"idempotency_key"`
        }
        if err := json.Unmarshal(raw, &f); err != nil {
                return false
        }
        return f.Token != nil && *f.Token != "" && f.Key != nil && *f.Key != ""
}

// call запускает обработчик и превращает панику в ошибку.
func (s *Server) call(ctx context.Context, c Caller, t *tool, args json.RawMessage) (out any, err error) {
        defer func() {
                if v := recover(); v != nil {
                        err = fmt.Errorf("panic: %v\n%s", v, debug.Stack())
                }
        }()
        return s.handlers[t.def.Name](ctx, c, args)
}

func (s *Server) checkOutput(t *tool, out any) (json.RawMessage, error) {
        raw, err := json.Marshal(out)
        if err != nil {
                return nil, fmt.Errorf("encode result: %w", err)
        }
        var obj map[string]any
        if err := json.Unmarshal(raw, &obj); err != nil || obj == nil {
                return nil, fmt.Errorf("result is not a JSON object: %s", truncate(raw))
        }
        if t.output != nil {
                if err := t.output.Validate(&obj); err != nil {
                        return nil, fmt.Errorf("result violates outputSchema: %w", err)
                }
        }
        return raw, nil
}

func (s *Server) internal(ctx context.Context, t *tool, err error) *apierr.Error {
        s.log.ErrorContext(ctx, "mcp tool failed", "server", s.name, "tool", t.def.Name, "err", err)
        return apierr.ToMCP(apierr.Internal())
}

func truncate(b []byte) string {
        if len(b) > 200 {
                return string(b[:200]) + "…"
        }
        return string(b)
}

// DecodeArgs разбирает проверенные аргументы инструмента в dst; сбой — validation_error.
func DecodeArgs(args json.RawMessage, dst any) error {
        if err := json.Unmarshal(args, dst); err != nil {
                return apierr.Validation().WithMessage("Некорректные аргументы инструмента: " + err.Error())
        }
        return nil
}

// WithConfirmation возвращает копию args с idempotency_key = key и confirmation_token, подписанным на sub, key и хеш
// args (ядро использует его для approvals.decide и kb.add_faq, где подтверждением служит нажатие кнопки; тесты — для
// инструментов записи).
func WithConfirmation(secret []byte, sub, key string, args map[string]any, ttl time.Duration) (map[string]any, error) {
        hash, err := confirm.ArgsHash(args)
        if err != nil {
                return nil, err
        }
        token, err := confirm.Issue(secret, sub, key, hash, ttl)
        if err != nil {
                return nil, err
        }
        out := make(map[string]any, len(args)+2)
        for k, v := range args {
                out[k] = v
        }
        out[confirm.KeyIdempotency] = key
        out[confirm.KeyConfirmation] = token
        return out, nil
}

        {Week1, 7, 7, "1 неделя"},
        {Week2, 14, 14, "2 неделя"},
        {Month1, 30, 30, "1 месяц"},
        {Month2, 90, 60, "2 месяц"},
        {After3Months, -1, 100, "После 3-х месяцев"},
}

// All возвращает этапы в порядке чек-листа (новый срез).
func All() []Period {
        out := make([]Period, len(table))
        for i, t := range table {
                out[i] = t.p
        }
        return out
}

// Parse возвращает s как Period или ошибку для неизвестного id.
func Parse(s string) (Period, error) {
        p := Period(s)
        if Order(p) < 0 {
                return "", fmt.Errorf("period: unknown period %q", s)
        }
        return p, nil
}

// Order возвращает позицию p в порядке чек-листа (с 0) или -1 для неизвестного этапа.
func Order(p Period) int {
        return slices.IndexFunc(table, func(t info) bool { return t.p == p })
}

// Label возвращает русскую подпись («1 неделя» … «После 3-х месяцев») или string(p) для неизвестного этапа.
func Label(p Period) string {
        if i := Order(p); i >= 0 {
                return table[i].label
        }
        return string(p)
}

// DueOffsetDays возвращает смещение срока от начала когорты: 7, 14, 30, 60, 100; 0 для неизвестного этапа.
func DueOffsetDays(p Period) int {
        if i := Order(p); i >= 0 {
                return table[i].dueDays
        }
        return 0
}

// DaysSince возвращает число календарных дней от cohortStart до today (отрицательное до начала). Каждое значение
// приводится к календарной дате в своём часовом поясе; время суток игнорируется.
func DaysSince(cohortStart, today time.Time) int {
        return int(date(today).Sub(date(cohortStart)).Hours() / 24)
}

// Current возвращает этап стажёра на today: week_1 при днях < 7 (и до начала), week_2 < 14, month_1 < 30, month_2 < 90,
// иначе after_3_months. Дни считаются через DaysSince.
func Current(cohortStart, today time.Time) Period {
        days := DaysSince(cohortStart, today)
        for _, t := range table {
                if t.before < 0 || days < t.before {
                        return t.p
                }
        }
        return After3Months // недостижимо: у последней строки нет границы
}

// DueDate возвращает cohortStart + DueOffsetDays(p) как дату (полночь UTC), как cohort_start + days в демо-данных.
func DueDate(cohortStart time.Time, p Period) time.Time {
        return date(cohortStart).AddDate(0, 0, DueOffsetDays(p))
}

// ParseDate разбирает дату YYYY-MM-DD (claim JWT cohort_start, даты из БД) как полночь UTC.
func ParseDate(s string) (time.Time, error) {
        t, err := time.Parse(time.DateOnly, s)
        if err != nil {
                return time.Time{}, fmt.Errorf("period: parse date %q: %w", s, err)
        }
        return t, nil
}

func date(t time.Time) time.Time {
        y, m, d := t.Date()
        return time.Date(y, m, d, 0, 0, 0, 0, time.UTC)
}

// Пакет pg открывает пулы соединений PostgreSQL (pgx/v5) для ядра и mcp-mocks.
package pg

import (
        "context"
        "fmt"
        "time"

        "github.com/jackc/pgx/v5/pgxpool"
)

// Options настраивают Open. Нулевые значения — значения по умолчанию.
type Options struct {
        MaxConns     int32         // по умолчанию 10
        StartupWait  time.Duration // сколько повторять, пока Postgres ещё стартует; по умолчанию 30 с
        RetryBackoff time.Duration // пауза между попытками; по умолчанию 1 с
}

// Open подключается к url и пингует его, повторяя, пока не истечёт opts.StartupWait (при старте сервиса Postgres может
// ещё выполнять init-скрипты).
func Open(ctx context.Context, url string, opts Options) (*pgxpool.Pool, error) {
        cfg, err := pgxpool.ParseConfig(url)
        if err != nil {
                return nil, fmt.Errorf("parse postgres url: %w", err)
        }
        cfg.MaxConns = opts.MaxConns
        if cfg.MaxConns == 0 {
                cfg.MaxConns = 10
        }
        cfg.MaxConnIdleTime = 5 * time.Minute
        wait, backoff := opts.StartupWait, opts.RetryBackoff
        if wait == 0 {
                wait = 30 * time.Second
        }
        if backoff == 0 {
                backoff = time.Second
        }
        deadline := time.Now().Add(wait)
        for {
                pool, err := pgxpool.NewWithConfig(ctx, cfg)
                if err == nil {
                        if err = pool.Ping(ctx); err == nil {
                                return pool, nil
                        }
                        pool.Close()
                }
                if time.Now().After(deadline) {
                        return nil, fmt.Errorf("connect postgres: %w", err)
                }
                select {
                case <-ctx.Done():
                        return nil, ctx.Err()
                case <-time.After(backoff):
                }
        }
}

// Пакет pgtest даёт тестам свежую базу PostgreSQL, собранную из настоящих миграций (схема + демо-данные).
//
// Задайте TEST_DATABASE_URL — подключение суперпользователя (его печатает scripts/test-db.sh). Каждый вызов New создаёт
// базу t_<random>, применяет migrations.FS в лексикографическом порядке и удаляет базу после теста.
package pgtest

import (
        "context"
        "crypto/rand"
        "encoding/hex"
        "fmt"
        "io/fs"
        "net/url"
        "os"
        "sort"
        "testing"
        "time"

        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgxpool"

        "aibuddy/migrations"
)

// EnvVar — переменная со строкой подключения администратора.
const EnvVar = "TEST_DATABASE_URL"

// New возвращает пул, подключённый к свежей базе с миграциями, или пропускает тест, если TEST_DATABASE_URL не задан.
func New(t testing.TB) *pgxpool.Pool {
        t.Helper()
        admin := os.Getenv(EnvVar)
        if admin == "" {
                t.Skipf("%s is not set; run `bash scripts/test-db.sh up` or `make test`", EnvVar)
        }
        ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
        defer cancel()
        name := newName()
        adminConn, err := pgx.Connect(ctx, admin)
        if err != nil {
                t.Fatalf("pgtest: connect admin: %v", err)
        }
        defer adminConn.Close(context.Background())
        if _, err := adminConn.Exec(ctx, "CREATE DATABASE "+name); err != nil {
                t.Fatalf("pgtest: create database: %v", err)
        }
        dbURL, err := withDatabase(admin, name)
        if err != nil {
                t.Fatalf("pgtest: %v", err)
        }
        if err := applyMigrations(ctx, dbURL); err != nil {
                dropDatabase(admin, name)
                t.Fatalf("pgtest: %v", err)
        }
        pool, err := pgxpool.New(ctx, dbURL)
        if err != nil {
                dropDatabase(admin, name)
                t.Fatalf("pgtest: open pool: %v", err)
        }
        t.Cleanup(func() {
                pool.Close()
                dropDatabase(admin, name)
        })
        return pool
}

// URL возвращает строку подключения к базе за pool (полезно для сервисов, открывающих свой пул).
func URL(t testing.TB, pool *pgxpool.Pool) string {
        t.Helper()
        return pool.Config().ConnString()
}

func applyMigrations(ctx context.Context, dbURL string) error {
        conn, err := pgx.Connect(ctx, dbURL)
        if err != nil {
                return fmt.Errorf("connect test database: %w", err)
        }
        defer conn.Close(context.Background())
        names, err := fs.Glob(migrations.FS, "*.sql")
        if err != nil {
                return fmt.Errorf("list migrations: %w", err)
        }
        sort.Strings(names)
        for _, n := range names {
                sql, err := migrations.FS.ReadFile(n)
                if err != nil {
                        return fmt.Errorf("read %s: %w", n, err)
                }
                // Простой протокол: файл миграции — много запросов в одной строке.
                if _, err := conn.Exec(ctx, string(sql), pgx.QueryExecModeSimpleProtocol); err != nil {
                        return fmt.Errorf("apply %s: %w", n, err)
                }
        }
        return nil
}

func withDatabase(raw, name string) (string, error) {
        u, err := url.Parse(raw)
        if err != nil {
                return "", fmt.Errorf("parse %s: %w", EnvVar, err)
        }
        u.Path = "/" + name
        return u.String(), nil
}

func dropDatabase(admin, name string) {
        ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
        defer cancel()
        conn, err := pgx.Connect(ctx, admin)
        if err != nil {
                return
        }
        defer conn.Close(context.Background())
        _, _ = conn.Exec(ctx, "DROP DATABASE IF EXISTS "+name+" WITH (FORCE)")
}

func randomHex(n int) string {
        b := make([]byte, n)
        if _, err := rand.Read(b); err != nil {
                panic(err)
        }
        return hex.EncodeToString(b)
}

var newName = func() string { return "t_" + randomHex(8) }
