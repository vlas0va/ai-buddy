// ErrNotFound is returned when no action matches (ApplyDecision for a request made outside the chat, ByProposal).

// Action is the openapi Action plus fields core keeps for itself.
type Action struct {
        ID                  string  `json:"id"`
        Kind                string  `json:"kind"`
        ExternalID          *string `json:"external_id"`
        Status              string  `json:"status"`

        ApproverDisplayName *string `json:"approver_display_name"`
        TaskID              *string `json:"task_id"`

        UpdatedAt           string  `json:"updated_at"`

        ProposalID string `json:"-"`
        MessageID  string `json:"-"` // assistant message saved with the action (chat replays it in done)

// Result is the action_result event of contracts/agent-stream.md.

// Result returns the action_result event of a.
func (a Action) Result() Result {

                ApproverDisplayName: a.ApproverDisplayName, TaskID: a.TaskID}

// SaveInput is a successful MCP write to record.
type SaveInput struct {
        Claims     authn.Claims    // owner (verified JWT claims)
        ProposalID string          // idempotency key of the write (= proposal id)
        Kind       string          // KindAbsence or KindAccess
        Tool       string          // draft.tool
        Args       json.RawMessage // draft.args
        Result     json.RawMessage // structured result of the tool
        MessageID  string          // optional: assistant message that announces the result
        ProposedAt time.Time       // optional: for duration_ms_since_proposal
        Attempt    int             // optional: 1 for the first confirmation, 2+ after a failure
        // Optional hints from the agent's tool_result, used when the MCP result lacks them: the checklist item of the
        // plan task to link (task_checklist_item_id) and the approver's display name.
        ChecklistItemID string
        ApproverName    string
}

// Service implements the action use cases. Emitter and Notifier may be nil (events/notifications dropped).
type Service struct {
        emitter  Emitter
        notifier Notifier
        log      *slog.Logger
        // Now is the clock used for the current period of new tasks; tests may replace it.
        Now func() time.Time
}

// NewService returns a Service.
func NewService(emitter Emitter, notifier Notifier, log *slog.Logger) *Service {
        if log == nil {
                log = slog.Default()
        }
        return &Service{emitter: emitter, notifier: notifier, log: log, Now: time.Now}
}

const actionCols = `a.id::text, a.user_id, a.kind, a.external_id, a.status, a.summary,
        a.payload->>'approver_display_name', a.task_id::text,
        to_char(a.created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
        to_char(a.updated_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"'),
        a.idempotency_key::text, COALESCE(a.payload->>'message_id', '')`

func scanAction(row pgx.Row) (Action, error) {
        var a Action
        err := row.Scan(&a.ID, &a.UserID, &a.Kind, &a.ExternalID, &a.Status, &a.Summary, &a.ApproverDisplayName, &a.TaskID,
                &a.CreatedAt, &a.UpdatedAt, &a.ProposalID, &a.MessageID)
        if errors.Is(err, pgx.ErrNoRows) {
                return a, ErrNotFound
        }
        if err != nil {
                return a, fmt.Errorf("scan action: %w", err)
        }
        return a, nil
}

// description is what Save derives from a tool result.
type description struct {
        externalID    string
        status        string
        summary       string
        approverName  string
        taskTitle     string
        checklistItem string  // access: link the checklist task with this item
        dueAt         *string // absence: the first day off
}

// Save records a successful write in q (run it in the transaction that marks the proposal executed): inserts
// core.actions with idempotency_key = ProposalID, links a plan task (access → the intern-plan task of the resource's
// checklist item, otherwise a new task with source = action in the current period), sets linked_action_id and emits
// action.created. Saving the same ProposalID again returns the stored action unchanged.
func (s *Service) Save(ctx context.Context, q DBTX, in SaveInput) (Action, error) {
        d, err := s.describe(ctx, q, in)
        if err != nil {
                return Action{}, err
        }
        payload, err := json.Marshal(map[string]any{
                "proposal_id": in.ProposalID, "tool": in.Tool, "args": in.Args, "result": in.Result,
                "approver_display_name": nullable(d.approverName), "message_id": nullable(in.MessageID),
        })
        if err != nil {
                return Action{}, fmt.Errorf("encode action payload: %w", err)
        }
        var id string
        err = q.QueryRow(ctx, `INSERT INTO core.actions (user_id, kind, external_id, status, summary, payload, idempotency_key)
                VALUES ($1, $2, $3, $4, $5, $6, $7) ON CONFLICT (idempotency_key) DO NOTHING RETURNING id::text`,
                in.Claims.Sub, in.Kind, nullable(d.externalID), d.status, d.summary, payload, in.ProposalID).Scan(&id)
        if errors.Is(err, pgx.ErrNoRows) {
                a, err := s.ByProposal(ctx, q, in.Claims.Sub, in.ProposalID)
                if errors.Is(err, ErrNotFound) {
                        return Action{}, apierr.Conflict()
                }
                return a, err
        }
        if err != nil {
                return Action{}, fmt.Errorf("insert action: %w", err)
        }
        taskID, err := s.linkTask(ctx, q, in.Claims.Sub, id, d)
        if err != nil {
                return Action{}, err
        }
        if taskID != nil {
                if _, err := q.Exec(ctx, `UPDATE core.actions SET task_id = $2 WHERE id = $1`, id, *taskID); err != nil {
                        return Action{}, fmt.Errorf("link action task: %w", err)
                }
        }
        props := map[string]any{"kind": in.Kind, "action_id": id, "proposal_id": in.ProposalID}
        if !in.ProposedAt.IsZero() {
                props["duration_ms_since_proposal"] = s.Now().Sub(in.ProposedAt).Milliseconds()
        }
        if in.Attempt > 0 {
                props["attempt"] = in.Attempt
        }
        if err := s.emit(ctx, q, analytics.Event{Name: "action.created", UserID: in.Claims.Sub, Role: in.Claims.Role,
                TeamID: in.Claims.TeamID, Props: props}); err != nil {
                return Action{}, err
        }
        return scanAction(q.QueryRow(ctx, `SELECT `+actionCols+` FROM core.actions a WHERE a.id = $1`, id))
}

func (s *Service) describe(ctx context.Context, q DBTX, in SaveInput) (description, error) {
        switch in.Kind {
        case KindAbsence:
                var r mcpclient.AbsenceRequest
                if err := json.Unmarshal(in.Result, &r); err != nil || r.ID == "" {
                        return description{}, fmt.Errorf("absence result without id: %w", apierr.Internal())
                }
                name, err := s.displayName(ctx, q, firstNonEmpty(r.ApproverDisplayName, in.ApproverName), &r.ApproverID)
                if err != nil {
                        return description{}, err
                }
                dates := ruDate(r.StartDate)
                if r.EndDate != "" && r.EndDate != r.StartDate {
                        dates += "–" + ruDate(r.EndDate)
                }
                d := description{externalID: r.ID, status: absenceStatus(r.Status), summary: "Отгул " + dates,
                        taskTitle: "Отгул " + dates, approverName: name}
                if r.StartDate != "" {
                        d.dueAt = &r.StartDate
                }
                // S3: never a checklist task (the agent's task_checklist_item_id hint is ignored); a source = action task is created.
                if name != "" {
                        d.summary += ", согласующий — " + name
                }
                return d, nil
        case KindAccess:
                var r mcpclient.AccessRequest
                if err := json.Unmarshal(in.Result, &r); err != nil || r.ID == "" {
                        return description{}, fmt.Errorf("access result without id: %w", apierr.Internal())
                }
                name, err := s.displayName(ctx, q, in.ApproverName, r.ApproverID)
                if err != nil {
                        return description{}, err
                }
                resource := r.ResourceName
                if resource == "" {
                        resource = r.ResourceID
                }
                d := description{externalID: r.ID, status: accessStatus(r.Status), taskTitle: "Доступ: " + resource, approverName: name,
                        checklistItem: in.ChecklistItemID}
                if r.ChecklistItemID != nil && *r.ChecklistItemID != "" {
                        d.checklistItem = *r.ChecklistItemID
                }
                switch {
                case d.status == StatusGranted:
                        d.summary = "Доступ: " + resource + " — выдан"
                case d.status == StatusRejected:
                        d.summary = "Доступ: " + resource + " — отклонён"
                case name != "":
                        d.summary = "Доступ: " + resource + ", согласующий — " + name
                default:
                        d.summary = "Доступ: " + resource + " — на согласовании"
                }
                return d, nil
        default:
                return description{}, apierr.Validation().WithMessage("Неизвестный тип действия")
        }
}

// displayName returns given, or the display name of the user id (empty when unknown).
func (s *Service) displayName(ctx context.Context, q DBTX, given string, id *string) (string, error) {
        if given != "" || id == nil || *id == "" {
                return given, nil
        }
        var name string
        err := q.QueryRow(ctx, `SELECT display_name FROM core.users WHERE id = $1`, *id).Scan(&name)
        if err != nil && !errors.Is(err, pgx.ErrNoRows) {
                return "", fmt.Errorf("load approver name: %w", err)
        }
        return name, nil
}

// linkTask links action id to a plan task of userID and returns the task id (nil when the user has no plan).
func (s *Service) linkTask(ctx context.Context, q DBTX, userID, actionID string, d description) (*string, error) {
        status := taskStatus(d.status)
        if d.checklistItem != "" {
                var taskID string
                err := q.QueryRow(ctx, `SELECT t.id::text FROM core.plan_tasks t JOIN core.plans p ON p.id = t.plan_id
                        WHERE p.user_id = $1 AND p.plan_role = 'intern' AND t.parent_task_id IS NULL AND t.checklist_item_id = $2
                                AND NOT t.human_only -- buddy only reminds about human-only items, an action never completes them (R05)
                        ORDER BY t.position, t.id LIMIT 1`, userID, d.checklistItem).Scan(&taskID)
                switch {
                case err == nil:
                        // A done task stays done and keeps its link: a later request for another resource of the same item
                        // must not reopen it.
                        _, err = q.Exec(ctx, `UPDATE core.plan_tasks SET
                                linked_action_id = CASE WHEN status = 'done' THEN linked_action_id ELSE $2::uuid END,
                                status = CASE WHEN status = 'done' THEN status ELSE $3 END,
                                updated_at = now()
                                WHERE id = $1`, taskID, actionID, status)
                        if err != nil {
                                return nil, fmt.Errorf("link checklist task: %w", err)
                        }
                        return &taskID, nil
                case !errors.Is(err, pgx.ErrNoRows):
                        return nil, fmt.Errorf("find checklist task: %w", err)
                }
        }
        var planID string
        var cohort *string
        err := q.QueryRow(ctx, `SELECT p.id::text, u.cohort_start::text FROM core.plans p LEFT JOIN core.users u ON u.id = p.intern_id
                WHERE p.user_id = $1 ORDER BY (p.plan_role = 'intern') DESC, p.created_at, p.id LIMIT 1`, userID).Scan(&planID, &cohort)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, nil
        }
        if err != nil {
                return nil, fmt.Errorf("find own plan: %w", err)
        }
        cur := period.Week1
        if cohort != nil {
                if start, err := period.ParseDate(*cohort); err == nil {
                        cur = period.Current(start, s.Now())
                }
        }
        var taskID string
        err = q.QueryRow(ctx, `INSERT INTO core.plan_tasks (plan_id, source, period, position, title, description, status, due_at, linked_action_id)
                VALUES ($1::uuid, 'action', $2,
                        COALESCE((SELECT MAX(position) FROM core.plan_tasks WHERE plan_id = $1::uuid AND period = $2), 0) + 1,
                        $3, $4, $5, $6::date, $7::uuid)
                RETURNING id::text`,
                planID, string(cur), d.taskTitle, "Создано ассистентом по заявке "+d.externalID, status, d.dueAt, actionID).Scan(&taskID)
        if err != nil {
                return nil, fmt.Errorf("create action task: %w", err)
        }
        return &taskID, nil
}

// ByProposal returns the action of userID saved for proposalID, or ErrNotFound.
func (s *Service) ByProposal(ctx context.Context, q DBTX, userID, proposalID string) (Action, error) {
        return scanAction(q.QueryRow(ctx, `SELECT `+actionCols+` FROM core.actions a
                WHERE a.idempotency_key = $2::uuid AND a.user_id = $1`, userID, proposalID))
}

// listLimit bounds GET /api/actions.
const listLimit = 100

// List returns userID's actions, newest first.
func (s *Service) List(ctx context.Context, q DBTX, userID string) ([]Action, error) {
        rows, err := q.Query(ctx, `SELECT `+actionCols+` FROM core.actions a WHERE a.user_id = $1
                ORDER BY a.created_at DESC, a.id DESC LIMIT $2`, userID, listLimit)
        if err != nil {
                return nil, fmt.Errorf("list actions: %w", err)
        }
        defer rows.Close()
        out := []Action{}
        for rows.Next() {
                a, err := scanAction(rows)
                if err != nil {
                        return nil, err
                }
                out = append(out, a)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("list actions: %w", err)
        }
        return out, nil
}

// Scope restricts ApplyDecision to the action of one requester whose request names one approver (R71). Both come
// from the approvals service's decide result and the deciding lead's verified claims, never from the agent.
type Scope struct {
        RequesterID string // owner of the action (approvals requester_id)
        ApproverID  string // approver_id in the saved MCP result of the request (the deciding lead)
}

// ApplyDecision applies the approver's decision ("approved" or "rejected", as returned by approvals.decide) to the
// action of kind with externalID (ABS-…/ACC-…) owned by scope.RequesterID whose request names scope.ApproverID as
// approver: action status approved/granted/rejected, the linked task done or not_started (never a human_only task),
// and a status_change notification for the owner, all through q. It reports whether anything changed (a repeated
// decision is a no-op) and returns ErrNotFound when core has no such action; both scope fields are required.
func (s *Service) ApplyDecision(ctx context.Context, q DBTX, kind, externalID, decision string, scope Scope) (Action, bool, error) {
        status, err := decisionStatus(kind, decision)
        if err != nil {
                return Action{}, false, err
        }
        if scope.RequesterID == "" || scope.ApproverID == "" {
                return Action{}, false, ErrNotFound
        }
        a, err := scanAction(q.QueryRow(ctx, `SELECT `+actionCols+` FROM core.actions a
                WHERE a.kind = $1 AND a.external_id = $2 AND a.user_id = $3 AND a.payload->'result'->>'approver_id' = $4
                ORDER BY a.created_at LIMIT 1 FOR UPDATE`, kind, externalID, scope.RequesterID, scope.ApproverID))
        if err != nil {
                return Action{}, false, err
        }
        if a.Status == status {
                return a, false, nil
        }
        if _, err := q.Exec(ctx, `UPDATE core.actions SET status = $2, updated_at = now() WHERE id = $1`, a.ID, status); err != nil {
                return Action{}, false, fmt.Errorf("update action status: %w", err)
        }
        if a.TaskID != nil {
                if _, err := q.Exec(ctx, `UPDATE core.plan_tasks SET status = $2, updated_at = now()
                        WHERE id = $1 AND linked_action_id = $3 AND NOT human_only`,
                        *a.TaskID, taskStatus(status), a.ID); err != nil {
                        return Action{}, false, fmt.Errorf("update linked task: %w", err)
                }
        }
        if s.notifier != nil {
                if err := s.notifier.Create(ctx, q, notify.Notification{UserID: a.UserID, Kind: notify.KindStatusChange,
                        Title: decisionTitle(kind, status), Body: a.Summary, DedupeKey: "action:" + a.ID + ":" + status}); err != nil {
                        return Action{}, false, fmt.Errorf("notify owner: %w", err)
                }
        }
        a, err = scanAction(q.QueryRow(ctx, `SELECT `+actionCols+` FROM core.actions a WHERE a.id = $1`, a.ID))
        return a, err == nil, err
}

func (s *Service) emit(ctx context.Context, q DBTX, ev analytics.Event) error {
        if s.emitter == nil {
                return nil
        }
        if err := s.emitter.Emit(ctx, q, ev); err != nil {
                return fmt.Errorf("emit %s: %w", ev.Name, err)
        }
        return nil
}

func decisionStatus(kind, decision string) (string, error) {
        switch {
        case decision == "rejected" && (kind == KindAbsence || kind == KindAccess):
                return StatusRejected, nil
        case decision == "approved" && kind == KindAbsence:
                return StatusApproved, nil
        case (decision == "approved" || decision == "granted") && kind == KindAccess:
                return StatusGranted, nil
        }
        return "", apierr.Validation().WithMessage("Неизвестное решение по заявке")
}

func decisionTitle(kind, status string) string {
        switch {
        case kind == KindAbsence && status == StatusApproved:
                return "Отгул согласован"
        case kind == KindAbsence:
                return "Отгул отклонён"
        case status == StatusGranted:
                return "Доступ выдан"
        default:
                return "Запрос доступа отклонён"
        }
}

func absenceStatus(s string) string {
        switch s {
        case StatusPendingApproval, StatusApproved, StatusRejected:
                return s
        }
        return StatusCreated
}

func accessStatus(s string) string {
        switch s {
        case StatusPendingApproval, StatusGranted, StatusRejected:
                return s
        }
        return StatusCreated
}

// taskStatus maps an action status onto the status of its plan task.
func taskStatus(actionStatus string) string {
        switch actionStatus {
        case StatusApproved, StatusGranted:
                return "done"
        case StatusRejected, StatusFailed:
                return "not_started"
        case StatusPendingApproval:
                return "waiting_approval"
        }
        return "in_progress"
}

// ruDate formats YYYY-MM-DD as DD.MM.YYYY (input returned as is when it is not a date).
func ruDate(d string) string {
        t, err := time.Parse(time.DateOnly, d)
        if err != nil {
                return d
        }
        return t.Format("02.01.2006")
}

// RuDate is ruDate for other packages (chat summaries).
func RuDate(d string) string { return ruDate(d) }

func firstNonEmpty(a, b string) string {
        if a != "" {
                return a
        }
        return b
}

                apierr.WriteJSON(w, apierr.Unauthorized())
                return

        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        httpx.WriteJSON(w, http.StatusOK, items)
}

// Turn is a user message (Kind = message, Text) or the confirmation of a stored proposal (Kind = confirm,
// ProposalID, Draft exactly as stored, ConfirmationToken signed by core): the agent then calls draft.tool and
// answers with a tool_result event.
type Turn struct {
        Kind              string          `json:"kind"`

// ErrUnavailable is matched (errors.Is) by every error meaning the agent could not be reached, answered with a
// non-200 status, broke the stream or ran past the timeout.
var ErrUnavailable = errors.New("agentclient: agent unavailable")

// UnavailableError is the typed error for an unreachable or failing agent.
type UnavailableError struct {
        Op     string // connect, status 503, read
        Err    error  // cause, may be nil
        Status int    // HTTP status of a non-200 answer, 0 otherwise
        // Code, Message and Retryable come from the agent's JSON error body {"error": {code, message, retryable}}
        // of a non-200 answer, when it sent one.
        Code      string
        Message   string
        Retryable bool
}

// Error implements error.
func (e *UnavailableError) Error() string {
        if e.Err == nil {
                return fmt.Sprintf("agentclient: %s: agent unavailable", e.Op)
        }
        return fmt.Sprintf("agentclient: %s: agent unavailable: %v", e.Op, e.Err)
}

// Unwrap returns the cause.
func (e *UnavailableError) Unwrap() error { return e.Err }

// Is makes errors.Is(err, ErrUnavailable) true.
func (e *UnavailableError) Is(target error) bool { return target == ErrUnavailable }

// Client calls the agent. Fields may be changed before first use.
type Client struct {
        BaseURL      string        // AGENT_URL, e.g. http://agent:8090
        HTTPClient   *http.Client  // nil means http.DefaultClient (no client timeout: Timeout bounds the call)
        Timeout      time.Duration // whole call including the stream; <= 0 means none
        ServiceToken string        // AGENT_SERVICE_TOKEN, sent as X-Service-Token when set
        // MaxEventBytes bounds one event, MaxStreamBytes the whole answer; <= 0 means the defaults.
        MaxEventBytes  int
        MaxStreamBytes int64
}

// New returns a client for baseURL whose calls are cancelled after llmTimeout + TimeoutMargin.
func New(baseURL string, llmTimeout time.Duration) *Client {
        return &Client{BaseURL: strings.TrimRight(baseURL, "/"), Timeout: llmTimeout + TimeoutMargin}
}

// Turn posts req with `Authorization: Bearer userToken` and X-Request-Id (from ctx) and returns the open event
// stream. Cancelling ctx (client disconnect) or reaching Timeout aborts the call. Connection failures and
// non-200 answers are *UnavailableError. The caller must Close the stream.
func (c *Client) Turn(ctx context.Context, userToken string, req TurnRequest) (*Stream, error) {
        if req.History == nil {
                req.History = []HistoryItem{}
        }
        if req.UserContext.OpenTasks == nil {
                req.UserContext.OpenTasks = []OpenTask{}
        }
        body, err := json.Marshal(req)
        if err != nil {
                return nil, fmt.Errorf("agentclient: encode turn: %w", err)
        }
        cancel := context.CancelFunc(func() {})
        if c.Timeout > 0 {
                ctx, cancel = context.WithTimeout(ctx, c.Timeout)
        }
        hreq, err := http.NewRequestWithContext(ctx, http.MethodPost, strings.TrimRight(c.BaseURL, "/")+"/v1/turn", bytes.NewReader(body))
        if err != nil {
                cancel()
                return nil, &UnavailableError{Op: "build request", Err: err}
        }
        hreq.Header.Set("Content-Type", "application/json")
        hreq.Header.Set("Accept", "text/event-stream")
        hreq.Header.Set("Authorization", "Bearer "+userToken)
        if id := httpx.RequestIDFromContext(ctx); id != "" {
                hreq.Header.Set(httpx.HeaderRequestID, id)
        }
        if c.ServiceToken != "" {
                hreq.Header.Set("X-Service-Token", c.ServiceToken)
        }
        hc := c.HTTPClient
        if hc == nil {
                hc = http.DefaultClient
        }
        resp, err := hc.Do(hreq)
        if err != nil {
                cancel()
                return nil, &UnavailableError{Op: "connect", Err: err}
        }
        if resp.StatusCode != http.StatusOK {
                ue := &UnavailableError{Op: fmt.Sprintf("status %d", resp.StatusCode), Status: resp.StatusCode}
                var body struct {
                        Error struct {
                                Code      string `json:"code"`
                                Message   string `json:"message"`
                                Retryable bool   `json:"retryable"`
                        } `json:"error"`
                }
                if json.NewDecoder(io.LimitReader(resp.Body, 4096)).Decode(&body) == nil {
                        ue.Code, ue.Message, ue.Retryable = body.Error.Code, body.Error.Message, body.Error.Retryable
                }
                _ = resp.Body.Close()
                cancel()
                return nil, ue
        }
        maxEvent, maxStream := c.MaxEventBytes, c.MaxStreamBytes
        if maxEvent <= 0 {
                maxEvent = DefaultMaxEventBytes
        }
        if maxStream <= 0 {
                maxStream = DefaultMaxStreamBytes
        }
        limited := &limitedBody{r: resp.Body, left: maxStream}
        return &Stream{body: resp.Body, r: httpx.NewSSEReaderLimit(limited, maxEvent), cancel: cancel}, nil
}

// limitedBody fails with ErrTooLarge once more than left bytes were read.
type limitedBody struct {
        r    io.Reader
        left int64
}

func (l *limitedBody) Read(p []byte) (int, error) {
        if l.left <= 0 {
                return 0, ErrTooLarge
        }
        if int64(len(p)) > l.left {
                p = p[:l.left]
        }
        n, err := l.r.Read(p)
        l.left -= int64(n)
        return n, err
}

// Stream is an open agent event stream. It is not safe for concurrent use.
type Stream struct {
        body   io.ReadCloser
        r      *httpx.SSEReader
        cancel context.CancelFunc
}

// Next returns the next event, io.EOF when the agent closed the stream normally, or an *UnavailableError when the
// stream broke or timed out. Events whose data is not JSON are skipped.
func (s *Stream) Next() (Event, error) {
        for {
                ev, err := s.r.Next()
                switch {
                case err == nil:
                        return Event{Name: ev.Name, Data: ev.Data}, nil
                case errors.Is(err, httpx.ErrSSEData):
                        continue
                case errors.Is(err, io.EOF):
                        return Event{}, io.EOF
                case errors.Is(err, httpx.ErrSSETooLarge) && !errors.Is(err, ErrTooLarge):
                        return Event{}, &UnavailableError{Op: "read", Err: fmt.Errorf("%w: %w", ErrTooLarge, err)}
                default:
                        return Event{}, &UnavailableError{Op: "read", Err: err}
                }
        }
}

// Close aborts the call and releases the connection; it is safe to call more than once.
func (s *Stream) Close() error {
        s.cancel()
        return s.body.Close()
}

func (a *Agent) serve(w http.ResponseWriter, r *http.Request) {
        if r.Method != http.MethodPost || r.URL.Path != "/v1/turn" {
                http.NotFound(w, r)
                return
        }
        var req Request
        var raw json.RawMessage
        if err := json.NewDecoder(r.Body).Decode(&raw); err != nil {
                http.Error(w, err.Error(), http.StatusBadRequest)
                return
        }
        _ = json.Unmarshal(raw, &req.TurnRequest)
        _ = json.Unmarshal(raw, &req.Raw)
        req.Auth = r.Header.Get("Authorization")
        a.mu.Lock()
        a.reqs = append(a.reqs, req)
        status, hang, onMessage, onConfirm := a.status, a.hang, a.onMessage, a.onConfirm
        a.mu.Unlock()
        if status != 0 {
                w.Header().Set("Content-Type", "application/json")
                w.WriteHeader(status)
                _ = json.NewEncoder(w).Encode(map[string]any{"error": map[string]any{
                        "code": "context_mismatch", "message": "Контекст пользователя не совпадает с токеном.", "retryable": false}})
                return
        }
        w.Header().Set("Content-Type", "text/event-stream")
        out := &writer{w: w}
        if hang {
                out.send(Event{"status", map[string]string{"stage": "thinking"}})
                <-r.Context().Done()
                return
        }
        var events []Event
        switch {
        case req.Turn.Kind == agentclient.KindConfirm && onConfirm != nil:
                events = onConfirm(req)
        case req.Turn.Kind == agentclient.KindConfirm:
                events = a.Confirm(r.Context(), req)
        case onMessage != nil:
                events = onMessage(req)
        default:
                events = Message(req)
        }
        for _, ev := range events {
                out.send(ev)
        }
}

type writer struct {
        w  http.ResponseWriter
        id int
}

func (o *writer) send(ev Event) {
        o.id++
        var data []byte
        if raw, ok := ev.Data.(json.RawMessage); ok {
                data = raw
        } else {
                data, _ = json.Marshal(ev.Data)
        }
        fmt.Fprintf(o.w, "id: %d\nevent: %s\ndata: %s\n\n", o.id, ev.Name, data)
        if f, ok := o.w.(http.Flusher); ok {
                f.Flush()
        }
}

// Confirm is the default script of a confirm turn: it calls draft.tool via MCP as the real agent does.
func (a *Agent) Confirm(ctx context.Context, req Request) []Event {
        var d struct {
                Tool string         `json:"tool"`
                Args map[string]any `json:"args"`
        }
        if err := json.Unmarshal(req.Turn.Draft, &d); err != nil || d.Args == nil {
                return []Event{{"error", map[string]any{"code": "bad_proposal", "message": "Некорректный черновик", "retryable": false}}}
        }
        server, tool, _ := strings.Cut(d.Tool, ".")
        args := map[string]any{}
        for k, v := range d.Args {
                args[k] = v
        }
        args[confirm.KeyIdempotency] = req.Turn.ProposalID
        args[confirm.KeyConfirmation] = req.Turn.ConfirmationToken
        token, _ := strings.CutPrefix(req.Auth, "Bearer ")
        out, toolErr, err := (&mcpx.Client{BaseURL: a.MCPBaseURL}).Call(ctx, token, server, tool, args)
        switch {
        case err != nil:
                return []Event{{"error", map[string]any{"code": "service_unavailable", "message": "Сервис недоступен", "retryable": true}}}
        case toolErr != nil:
                return []Event{{"error", map[string]any{"code": string(toolErr.Code), "message": toolErr.Message, "retryable": toolErr.Retryable}}}
        }
        kind := map[string]string{"absence.create_request": "absence_request", "access.request_access": "access_request",
                "approvals.decide": "approval_decision"}[d.Tool]
        var res struct {
                ID              string  `json:"id"`
                Status          string  `json:"status"`
                ChecklistItemID *string `json:"checklist_item_id"`
                Approver        string  `json:"approver_display_name"`
        }
        _ = json.Unmarshal(out, &res)
        return []Event{
                {"status", map[string]string{"stage": "calling_service"}},
                {"status", map[string]string{"stage": "saving"}},
                {"tool_result", map[string]any{"proposal_id": req.Turn.ProposalID, "kind": kind, "tool": d.Tool,
                        "external_id": res.ID, "status": res.Status, "approver_display_name": res.Approver,
                        "task_checklist_item_id": res.ChecklistItemID, "replayed": false,
                        "followup": "Заявка отправлена на согласование.", "result": json.RawMessage(out)}},
                Done(),
        }
}

// unknownKeys is the union of all whitelisted keys; events with a name outside the contract keep only these.
var unknownKeys = func() map[string]bool {
        m := map[string]bool{}
        for _, ks := range propKeys {
                for _, k := range ks {
                        m[k] = true
                }
        }
        return m
}()

const maxPropString = 100

// SanitizeProps keeps only whitelisted keys of the event with scalar values (short strings, bools, numbers).
func SanitizeProps(name string, props map[string]any) map[string]any {
        allowed := map[string]bool{}
        if ks, ok := propKeys[name]; ok {
                for _, k := range ks {
                        allowed[k] = true
                }
        } else {
                allowed = unknownKeys
        }
        out := map[string]any{}
        for k, v := range props {
                if !allowed[k] {
                        continue
                }
                switch x := v.(type) {
                case string:
                        if len(x) <= maxPropString {
                                out[k] = x
                        }
                case bool, int, int8, int16, int32, int64, uint, uint8, uint16, uint32, uint64, float32, float64:
                        out[k] = x
                case json.Number:
                        out[k] = x
                }
        }
        return out
}

func nullable(s string) *string {
        if s == "" {
                return nil

func newUUID() string {
        var b [16]byte
        _, _ = rand.Read(b[:]) // crypto/rand does not fail on supported platforms
        b[6] = b[6]&0x0f | 0x40
        b[8] = b[8]&0x3f | 0x80
        return fmt.Sprintf("%x-%x-%x-%x-%x", b[0:4], b[4:6], b[6:8], b[8:10], b[10:16])
}

// Emit stores ev through q. With Kafka enabled the envelope is also queued in core.outbox by the same statement.
func (e *Emitter) Emit(ctx context.Context, q DBTX, ev Event) error {
        if e == nil {
                return nil
        }
        if !nameRE.MatchString(ev.Name) {
                return fmt.Errorf("%w: name %q", ErrInvalid, ev.Name)
        }
        ts := ev.TS
        if ts.IsZero() {
                ts = e.now()
        }
        ts = ts.UTC()
        id := newUUID()
        props := SanitizeProps(ev.Name, ev.Props)
        propsJSON, err := json.Marshal(props)
        if err != nil {
                return fmt.Errorf("encode props: %w", err)
        }
        if !e.kafka {
                _, err = q.Exec(ctx, `INSERT INTO core.analytics_events (event_id, ts, name, user_id, role, team_id, props)
                        VALUES ($1, $2, $3, $4, $5, $6, $7)`,
                        id, ts, ev.Name, nullable(ev.UserID), nullable(ev.Role), nullable(ev.TeamID), propsJSON)
                if err != nil {
                        return fmt.Errorf("insert analytics event: %w", err)
                }
                return nil
        }
        topic := TopicDomain
        if strings.HasPrefix(ev.Name, "ui.") {
                topic = TopicClickstream
        }
        envelope, err := json.Marshal(map[string]any{
                "event_id": id, "name": ev.Name, "ts": ts.Format(time.RFC3339Nano),
                "user_id": nullable(ev.UserID), "role": nullable(ev.Role), "team_id": nullable(ev.TeamID), "props": props,
        })
        if err != nil {
                return fmt.Errorf("encode envelope: %w", err)
        }
        _, err = q.Exec(ctx, `WITH e AS (
                        INSERT INTO core.analytics_events (event_id, ts, name, user_id, role, team_id, props)
                        VALUES ($1, $2, $3, $4, $5, $6, $7))
                INSERT INTO core.outbox (topic, payload) VALUES ($8, $9)`,
                id, ts, ev.Name, nullable(ev.UserID), nullable(ev.Role), nullable(ev.TeamID), propsJSON, topic, envelope)
        if err != nil {
                return fmt.Errorf("insert analytics event with outbox: %w", err)
        }
        return nil
}

var (
        uiNameRE = regexp.MustCompile(`^ui\.[a-z_.]+$`)
        uuidRE   = regexp.MustCompile(`^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$`)

const maxBatch = 100

type clientEvent struct {
        Name  string         `json:"name"`
        TS    string         `json:"ts"`
        Props map[string]any `json:"props"`
}

func (h *Handler) events(w http.ResponseWriter, r *http.Request) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        var body struct {
                Events []clientEvent `json:"events"`
        }
        if err := httpx.ReadJSON(w, r, &body); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        if len(body.Events) > maxBatch {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Слишком много событий в одном запросе (не более 100)"))
                return
        }
        now := time.Now()
        events := make([]Event, 0, len(body.Events))
        for _, ce := range body.Events {
                if !uiNameRE.MatchString(ce.Name) {
                        apierr.WriteJSON(w, apierr.Validation().WithMessage("Недопустимое имя события"))
                        return
                }
                ts, err := time.Parse(time.RFC3339, ce.TS)
                if err != nil {
                        apierr.WriteJSON(w, apierr.Validation().WithMessage("Время события должно быть в формате RFC 3339"))
                        return
                }
                // A client clock far in the future would corrupt time series: fall back to server time.
                if ts.After(now.Add(5 * time.Minute)) {
                        ts = now
                }
                events = append(events, Event{Name: ce.Name, UserID: c.Sub, Role: c.Role, TeamID: c.TeamID, Props: ce.Props, TS: ts})
        }
        if err := h.emitAll(r, events); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        w.WriteHeader(http.StatusAccepted)
}

func (h *Handler) emitAll(r *http.Request, events []Event) error {
        tx, err := h.pool.Begin(r.Context())
        if err != nil {
                return fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(r.Context()) }()
        for _, ev := range events {
                if err := h.emitter.Emit(r.Context(), tx, ev); err != nil {
                        return err
                }
        }
        if err := tx.Commit(r.Context()); err != nil {
                return fmt.Errorf("commit: %w", err)
        }
        return nil
}

func (h *Handler) feedback(w http.ResponseWriter, r *http.Request) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        var body struct {
                MessageID string  `json:"message_id"`
                Useful    *bool   `json:"useful"`
                Comment   *string `json:"comment"`
        }
        if err := httpx.ReadJSON(w, r, &body); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        if !uuidRE.MatchString(body.MessageID) || body.Useful == nil {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Укажите message_id и useful"))
                return
        }
        comment := ""
        if body.Comment != nil {
                comment = strings.TrimSpace(*body.Comment)
        }
        if len([]rune(comment)) > 1000 {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Комментарий не длиннее 1000 символов"))
                return
        }
        ctx := r.Context()
        // Only assistant messages of the caller's own sessions can be rated; anything else is 403 (R71).
        var owned bool
        err := h.pool.QueryRow(ctx, `SELECT true FROM core.chat_messages m JOIN core.chat_sessions s ON s.id = m.session_id
                WHERE m.id = $1 AND s.user_id = $2 AND m.author = 'assistant'`, body.MessageID, c.Sub).Scan(&owned)
        if errors.Is(err, pgx.ErrNoRows) {
                apierr.WriteJSON(w, apierr.Forbidden())
                return
        }
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("check message owner: %w", err))
                return
        }
        tx, err := h.pool.Begin(ctx)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("begin: %w", err))
                return
        }
        defer func() { _ = tx.Rollback(ctx) }()
        if _, err := tx.Exec(ctx, `INSERT INTO core.feedback (message_id, user_id, useful, comment)
                VALUES ($1, $2, $3, NULLIF($4, ''))
                ON CONFLICT (message_id, user_id) DO UPDATE SET useful = EXCLUDED.useful, comment = EXCLUDED.comment, created_at = now()`,
                body.MessageID, c.Sub, *body.Useful, comment); err != nil {
                apierr.WriteJSON(w, fmt.Errorf("save feedback: %w", err))
                return
        }
        if err := h.emitter.Emit(ctx, tx, Event{Name: "chat.feedback", UserID: c.Sub, Role: c.Role, TeamID: c.TeamID,
                Props: map[string]any{"message_id": body.MessageID, "useful": *body.Useful}}); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        if err := tx.Commit(ctx); err != nil {
                apierr.WriteJSON(w, fmt.Errorf("commit: %w", err))
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

// Producer sends one message to Kafka and returns when it is acknowledged. Tests use a fake.
type Producer interface {
        Produce(ctx context.Context, topic, key string, value []byte) error
}

// DefaultInterval is how often Publisher.Run drains the outbox.
const DefaultInterval = 2 * time.Second

const drainBatch = 200

// Publisher moves core.outbox rows to Kafka in id order and stamps published_at.
type Publisher struct {
        pool     *pgxpool.Pool
        prod     Producer
        log      *slog.Logger
        Interval time.Duration
}

// NewPublisher returns a Publisher draining every DefaultInterval.
func NewPublisher(pool *pgxpool.Pool, prod Producer, log *slog.Logger) *Publisher {
        if log == nil {
                log = slog.Default()
        }
        return &Publisher{pool: pool, prod: prod, log: log, Interval: DefaultInterval}
}

// Run drains the outbox every Interval until ctx is cancelled. Errors are logged and retried on the next tick.
func (p *Publisher) Run(ctx context.Context) {
        interval := p.Interval
        if interval <= 0 {
                interval = DefaultInterval
        }
        t := time.NewTicker(interval)
        defer t.Stop()
        for {
                select {
                case <-ctx.Done():
                        return
                case <-t.C:
                        if n, err := p.DrainOnce(ctx); err != nil {
                                p.log.Warn("outbox drain failed", "published", n, "err", err)
                        }
                }
        }
}

// DrainOnce publishes up to one batch of unpublished rows in id order. It stops at the first producer error (order
// is kept) after stamping the rows already sent, and returns the number published.
func (p *Publisher) DrainOnce(ctx context.Context) (int, error) {
        tx, err := p.pool.Begin(ctx)
        if err != nil {
                return 0, fmt.Errorf("begin drain: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        rows, err := tx.Query(ctx, `SELECT id, topic, COALESCE(payload->>'user_id', payload->>'event_id', ''), payload::text
                FROM core.outbox WHERE published_at IS NULL ORDER BY id LIMIT $1 FOR UPDATE SKIP LOCKED`, drainBatch)
        if err != nil {
                return 0, fmt.Errorf("select outbox: %w", err)
        }
        type msg struct {
                id         int64
                topic, key string
                payload    []byte
        }
        var msgs []msg
        for rows.Next() {
                var m msg
                var payload string
                if err := rows.Scan(&m.id, &m.topic, &m.key, &payload); err != nil {
                        rows.Close()
                        return 0, fmt.Errorf("scan outbox: %w", err)
                }
                m.payload = []byte(payload)
                msgs = append(msgs, m)
        }
        rows.Close()
        if err := rows.Err(); err != nil {
                return 0, fmt.Errorf("read outbox: %w", err)
        }
        var sent []int64
        var produceErr error
        for _, m := range msgs {
                if produceErr = p.prod.Produce(ctx, m.topic, m.key, m.payload); produceErr != nil {
                        break
                }
                sent = append(sent, m.id)
        }
        if len(sent) > 0 {
                if _, err := tx.Exec(ctx, `UPDATE core.outbox SET published_at = now() WHERE id = ANY($1)`, sent); err != nil {
                        return 0, fmt.Errorf("stamp outbox: %w", err)
                }
                if err := tx.Commit(ctx); err != nil {
                        return 0, fmt.Errorf("commit drain: %w", err)
                }
        }
        if produceErr != nil {
                return len(sent), fmt.Errorf("produce: %w", produceErr)
        }
        return len(sent), nil
}

// KafkaProducer is the franz-go implementation of Producer.
type KafkaProducer struct {
        cl      *kgo.Client
        timeout time.Duration
}

// NewKafkaProducer connects lazily to the seed brokers (host:port).
func NewKafkaProducer(brokers []string) (*KafkaProducer, error) {
        if len(brokers) == 0 {
                return nil, errors.New("analytics: no kafka brokers")
        }
        cl, err := kgo.NewClient(kgo.SeedBrokers(brokers...), kgo.AllowAutoTopicCreation())
        if err != nil {
                return nil, fmt.Errorf("create kafka client: %w", err)
        }
        return &KafkaProducer{cl: cl, timeout: 10 * time.Second}, nil
}

// Produce sends the message synchronously.
func (k *KafkaProducer) Produce(ctx context.Context, topic, key string, value []byte) error {
        ctx, cancel := context.WithTimeout(ctx, k.timeout)
        defer cancel()
        res := k.cl.ProduceSync(ctx, &kgo.Record{Topic: topic, Key: []byte(key), Value: value})
        return res.FirstErr()
}

// Close releases the client.
func (k *KafkaProducer) Close() { k.cl.Close() }

// healthz answers 200 when Postgres answers a ping, 503 otherwise (the agent is not checked: core must stay up).
func healthz(pool *pgxpool.Pool) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                ctx, cancel := context.WithTimeout(r.Context(), 2*time.Second)
                defer cancel()
                if err := pool.Ping(ctx); err != nil {
                        httpx.WriteJSON(w, http.StatusServiceUnavailable, map[string]string{"status": "degraded", "service": "core"})
                        return
                }
                httpx.WriteJSON(w, http.StatusOK, map[string]string{"status": "ok", "service": "core"})
        }
}

// Run connects to Postgres (with retry while it starts) and Redis, starts the AI-cron and, with KAFKA_BROKERS, the
// outbox publisher, and serves until ctx is cancelled.
func Run(ctx context.Context, cfg Config, log *slog.Logger) error {
        if err := cfg.Validate(); err != nil {
                return err
        }
        if log == nil {
                log = slog.Default()
        }
        pool, err := pg.Open(ctx, cfg.PostgresURL, pg.Options{})
        if err != nil {
                return err
        }
        defer pool.Close()
        var rdb redis.Cmdable // stays a nil interface without Redis
        if cfg.RedisURL != "" {
                opts, err := redis.ParseURL(cfg.RedisURL)
                if err != nil {
                        return fmt.Errorf("REDIS_URL: %w", err)
                }
                client := redis.NewClient(opts)
                defer client.Close()
                rdb = client
        }
        a, err := New(cfg, pool, rdb, log)
        if err != nil {
                return err
        }
        ctx, cancel := context.WithCancel(ctx)
        defer cancel()
        go a.Cron.Run(ctx)
        if len(cfg.KafkaBrokers) > 0 {
                prod, err := analytics.NewKafkaProducer(cfg.KafkaBrokers)
                if err != nil {
                        return err
                }
                defer prod.Close()
                go analytics.NewPublisher(pool, prod, log).Run(ctx)
                log.Info("kafka publisher started", "brokers", cfg.KafkaBrokers)
        }
        return serve(ctx, fmt.Sprintf(":%d", cfg.Port), a.Handler, log)
}

func serve(ctx context.Context, addr string, h http.Handler, log *slog.Logger) error {
        srv := &http.Server{Addr: addr, Handler: h, ReadHeaderTimeout: 5 * time.Second}
        errc := make(chan error, 1)
        go func() { errc <- srv.ListenAndServe() }()
        log.Info("listening", "addr", addr)
        select {
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
                defer cancel()
                return srv.Shutdown(shutdownCtx)
        case err := <-errc:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        }
}

// Healthcheck performs GET url and fails on a non-200 answer (Docker HEALTHCHECK via `-healthcheck`).
func Healthcheck(url string) error {
        client := &http.Client{Timeout: 3 * time.Second}
        resp, err := client.Get(url)
        if err != nil {
                return err
        }
        defer resp.Body.Close()
        if resp.StatusCode != http.StatusOK {
                return fmt.Errorf("healthcheck: status %d", resp.StatusCode)
        }
        return nil
}

var (
        uuidRE       = regexp.MustCompile(`^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$`)
        approvalIDRE = regexp.MustCompile(`^[A-Za-z0-9_-]{1,64}$`)
)

// lead returns the caller when it is a lead with a token; other roles get forbidden (R71).
func lead(w http.ResponseWriter, r *http.Request) (authn.Claims, string, bool) {
        c, ok := authn.FromContext(r.Context())
        token, tok := authn.TokenFromContext(r.Context())
        switch {
        case !ok || !tok:
                apierr.WriteJSON(w, apierr.Unauthorized())
        case c.Role != "lead":
                apierr.WriteJSON(w, apierr.Forbidden())
        default:
                return c, token, true
        }
        return authn.Claims{}, "", false
}

func (h *Handler) list(w http.ResponseWriter, r *http.Request) {
        _, token, ok := lead(w, r)
        if !ok {
                return
        }
        items, err := h.mcp.ListPendingApprovals(r.Context(), token)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        out := make([]Approval, 0, len(items))
        for _, it := range items {
                a, err := h.toAPI(r.Context(), it)
                if err != nil {
                        apierr.WriteJSON(w, err)
                        return
                }
                out = append(out, a)
        }
        httpx.WriteJSON(w, http.StatusOK, out)
}

func (h *Handler) decide(w http.ResponseWriter, r *http.Request) {
        c, token, ok := lead(w, r)
        if !ok {
                return
        }
        key := strings.TrimSpace(r.Header.Get("Idempotency-Key"))
        if !uuidRE.MatchString(key) {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Нужен заголовок Idempotency-Key в формате UUID"))
                return
        }
        id := r.PathValue("approval_id")
        if !approvalIDRE.MatchString(id) {
                apierr.WriteJSON(w, apierr.NotFound())
                return
        }
        var body struct {
                Decision string `json:"decision"`
                Comment  string `json:"comment"`
        }
        if err := httpx.ReadJSONLimit(w, r, &body, 16<<10); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        if body.Decision != "approved" && body.Decision != "rejected" {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Решение: approved или rejected"))
                return
        }
        if utf8.RuneCountInString(body.Comment) > 1000 {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Комментарий: не больше 1000 символов"))
                return
        }
        res, err := h.mcp.DecideApproval(r.Context(), token, id, body.Decision, strings.TrimSpace(body.Comment), key)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        // The decision is recorded by the approvals service; mirror it in core even if the client has gone away.
        ctx, cancel := context.WithTimeout(context.WithoutCancel(r.Context()), 10*time.Second)
        defer cancel()
        if err := h.apply(ctx, c, res); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        out, err := h.toAPI(ctx, res)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        httpx.WriteJSON(w, http.StatusOK, out)
}

// apply updates the requester's action and plan task and notifies them, in one transaction. Requests that were
// not made through the chat have no action in core and are skipped.
func (h *Handler) apply(ctx context.Context, c authn.Claims, res mcpclient.Approval) error {
        tx, err := h.pool.Begin(ctx)
        if err != nil {
                return fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        // Only the requester's action whose request names the deciding lead as approver (R71).
        a, changed, err := h.actions.ApplyDecision(ctx, tx, res.Kind, res.RefID, res.Status,
                actions.Scope{RequesterID: res.RequesterID, ApproverID: c.Sub})
        switch {
        case errors.Is(err, actions.ErrNotFound):
                h.log.Info("decision for a request without a core action", "approval", res.ID, "ref", res.RefID)
                return nil
        case apierr.HasCode(err, apierr.CodeValidation):
                h.log.Warn("unexpected approval result", "approval", res.ID, "status", res.Status, "kind", res.Kind)
                return nil
        case err != nil:
                return fmt.Errorf("apply decision: %w", err)
        }
        if changed && h.emitter != nil {
                if err := h.emitter.Emit(ctx, tx, analytics.Event{Name: "action.status_changed", UserID: c.Sub, Role: c.Role, TeamID: c.TeamID,
                        Props: map[string]any{"kind": a.Kind, "action_id": a.ID, "status": a.Status}}); err != nil {
                        return fmt.Errorf("emit action.status_changed: %w", err)
                }
        }
        if err := tx.Commit(ctx); err != nil {
                return fmt.Errorf("commit: %w", err)
        }
        return nil
}

// toAPI maps an MCP approval onto the openapi shape, filling the requester's name from core when MCP omits it.
func (h *Handler) toAPI(ctx context.Context, it mcpclient.Approval) (Approval, error) {
        a := Approval{ID: it.ID, Kind: it.Kind, RequesterDisplayName: it.RequesterDisplayName, Summary: it.Summary,
                Status: it.Status, CreatedAt: it.CreatedAt}
        if it.Comment != "" {
                comment := it.Comment
                a.Comment = &comment
        }
        if a.RequesterDisplayName == "" {
                err := h.pool.QueryRow(ctx, `SELECT COALESCE((SELECT display_name FROM core.users WHERE id = $1), '')`, it.RequesterID).
                        Scan(&a.RequesterDisplayName)
                if err != nil {
                        return a, fmt.Errorf("load requester name: %w", err)
                }
        }
        return a, nil
}

// User is the openapi User schema.
type User struct {
        ID          string  `json:"id"`
        Login       string  `json:"login"`
        DisplayName string  `json:"display_name"`
        Role        string  `json:"role"`
        TeamID      string  `json:"team_id"`
        TeamName    string  `json:"team_name,omitempty"`
        CohortStart *string `json:"cohort_start,omitempty"`
}

const userSelect = `SELECT u.id, u.display_name, u.role, u.team_id, COALESCE(t.name, ''), u.cohort_start::text
        FROM core.users u LEFT JOIN directory.teams t ON t.id = u.team_id`

func scanUser(row pgx.Row) (User, error) {
        var u User
        err := row.Scan(&u.ID, &u.DisplayName, &u.Role, &u.TeamID, &u.TeamName, &u.CohortStart)
        u.Login = u.ID
        return u, err
}

func badLogin() *apierr.Error {
        return apierr.Unauthorized().WithMessage("Неверный логин или пароль")
}

type loginRequest struct {
        Login    string `json:"login"`
        Password string `json:"password"`
}

const maxCredLen = 128

func (h *Handler) login(w http.ResponseWriter, r *http.Request) {
        var req loginRequest
        if err := httpx.ReadJSONLimit(w, r, &req, 4<<10); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        if req.Login == "" || req.Password == "" || len(req.Login) > maxCredLen || len(req.Password) > maxCredLen {
                apierr.WriteJSON(w, badLogin())
                return
        }
        ctx := r.Context()
        // Password check happens in SQL; an unknown login and a wrong password are indistinguishable (generic 401, R52).
        u, err := scanUser(h.pool.QueryRow(ctx, userSelect+`
                WHERE u.id = $1 AND u.can_login AND u.password_hash = crypt($2, u.password_hash)`, req.Login, req.Password))
        if errors.Is(err, pgx.ErrNoRows) {
                // Spend comparable time for unknown logins.
                _, _ = h.pool.Exec(ctx, `SELECT crypt($1, gen_salt('bf', 8))`, req.Password)
                apierr.WriteJSON(w, badLogin())
                return
        }
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("login query: %w", err))
                return
        }
        claims := authn.Claims{Sub: u.ID, Role: u.Role, TeamID: u.TeamID}
        if u.CohortStart != nil {
                claims.CohortStart = *u.CohortStart
        }
        token, err := authn.Issue(h.secret, claims, h.ttl)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("issue token: %w", err))
                return
        }
        props := map[string]any{}
        if u.CohortStart != nil {
                if start, err := period.ParseDate(*u.CohortStart); err == nil {
                        props["period"] = string(period.Current(start, h.Now()))
                }
        }
        if err := h.emitter.Emit(ctx, h.pool, analytics.Event{Name: "session.started", UserID: u.ID, Role: u.Role, TeamID: u.TeamID, Props: props}); err != nil {
                h.log.Warn("session.started event failed", "err", err) // analytics must not block login
        }
        httpx.WriteJSON(w, http.StatusOK, map[string]any{"token": token, "user": u})
}

func (h *Handler) me(w http.ResponseWriter, r *http.Request) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        u, err := scanUser(h.pool.QueryRow(r.Context(), userSelect+` WHERE u.id = $1 AND u.can_login`, c.Sub))
        if errors.Is(err, pgx.ErrNoRows) {
                apierr.WriteJSON(w, apierr.Unauthorized()) // token of a user that no longer exists
                return
        }
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("me query: %w", err))
                return
        }
        fl, err := h.flags.ForTeam(r.Context(), u.TeamID)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("flags: %w", err))
                return
        }
        httpx.WriteJSON(w, http.StatusOK, map[string]any{"user": u, "flags": fl})
}

// Package chat serves the chat API (S2, S3, S4): sessions, messages streamed from the agent as SSE, and the
// confirmation of write proposals (contracts/agent-stream.md, contracts/openapi.yaml).

// Message turn: core saves the user message, builds user_context from verified JWT claims and its own DB (R71),
// calls the agent and relays its events to the browser unchanged and in order (unknown events too; tool_result and

// with it, action_proposal is stored in core.proposals (only for the allow-listed write tools). The agent's done is
// relayed with message_id set to the saved assistant message (core sends done{message_id} when the agent did not)

// caller returns the verified claims and raw token of the request, or answers 401.
func caller(w http.ResponseWriter, r *http.Request) (authn.Claims, string, bool) {
        c, ok := authn.FromContext(r.Context())
        token, tok := authn.TokenFromContext(r.Context())
        if !ok || !tok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return authn.Claims{}, "", false
        }
        return c, token, true
}

// ownSession checks that sessionID is a session of userID: a malformed id is not_found, an unknown or foreign one
// is forbidden (existence is not revealed, R71).
func (h *Handler) ownSession(ctx context.Context, userID, sessionID string) error {
        if !uuidRE.MatchString(sessionID) {
                return apierr.NotFound()
        }
        var owner string
        err := h.pool.QueryRow(ctx, `SELECT user_id FROM core.chat_sessions WHERE id = $1`, sessionID).Scan(&owner)
        if errors.Is(err, pgx.ErrNoRows) || (err == nil && owner != userID) {
                return apierr.Forbidden()
        }
        if err != nil {
                return fmt.Errorf("load session: %w", err)
        }
        return nil
}

func (h *Handler) createSession(w http.ResponseWriter, r *http.Request) {
        c, _, ok := caller(w, r)
        if !ok {
                return
        }
        var id string
        if err := h.pool.QueryRow(r.Context(), `INSERT INTO core.chat_sessions (user_id) VALUES ($1) RETURNING id::text`, c.Sub).Scan(&id); err != nil {
                apierr.WriteJSON(w, fmt.Errorf("create session: %w", err))
                return
        }
        httpx.WriteJSON(w, http.StatusCreated, map[string]string{"session_id": id})
}

// Message is the openapi ChatMessage.
type Message struct {
        ID        string            `json:"id"`
        Author    string            `json:"author"`
        Text      string            `json:"text"`
        Sources   []json.RawMessage `json:"sources"`
        CreatedAt string            `json:"created_at"`
}

const messagesLimit = 200

func (h *Handler) getSession(w http.ResponseWriter, r *http.Request) {
        c, _, ok := caller(w, r)
        if !ok {
                return
        }
        sid := r.PathValue("session_id")
        if err := h.ownSession(r.Context(), c.Sub, sid); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        rows, err := h.pool.Query(r.Context(), `SELECT id::text, author, text, sources,
                to_char(created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')
                FROM (SELECT * FROM core.chat_messages WHERE session_id = $1 ORDER BY created_at DESC, id DESC LIMIT $2) m
                ORDER BY created_at, id`, sid, messagesLimit)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("load messages: %w", err))
                return
        }
        defer rows.Close()
        out := []Message{}
        for rows.Next() {
                var m Message
                var sources []byte
                if err := rows.Scan(&m.ID, &m.Author, &m.Text, &sources, &m.CreatedAt); err != nil {
                        apierr.WriteJSON(w, fmt.Errorf("scan message: %w", err))
                        return
                }
                if err := json.Unmarshal(sources, &m.Sources); err != nil || m.Sources == nil {
                        m.Sources = []json.RawMessage{}
                }
                out = append(out, m)
        }
        if err := rows.Err(); err != nil {
                apierr.WriteJSON(w, fmt.Errorf("read messages: %w", err))
                return
        }
        httpx.WriteJSON(w, http.StatusOK, map[string]any{"session_id": sid, "messages": out})
}

// insertMessage stores a message with an explicit id (clock_timestamp keeps the order inside one request).
func insertMessage(ctx context.Context, q actions.DBTX, id, sessionID, author, text string, sources []json.RawMessage) error {
        if sources == nil {
                sources = []json.RawMessage{}
        }
        raw, err := json.Marshal(sources)
        if err != nil {
                return fmt.Errorf("encode sources: %w", err)
        }
        if _, err := q.Exec(ctx, `INSERT INTO core.chat_messages (id, session_id, author, text, sources, created_at)
                VALUES ($1, $2, $3, $4, $5, clock_timestamp()) ON CONFLICT (id) DO NOTHING`, id, sessionID, author, text, raw); err != nil {
                return fmt.Errorf("insert %s message: %w", author, err)
        }
        return nil
}

// emit records an analytics event; failures are logged, never shown to the user.
func (h *Handler) emit(ctx context.Context, c authn.Claims, name string, props map[string]any) {
        if h.emitter == nil {

        if err := h.emitter.Emit(ctx, h.pool, analytics.Event{Name: name, UserID: c.Sub, Role: c.Role, TeamID: c.TeamID, Props: props}); err != nil {
                h.log.Warn("analytics event failed", "name", name, "err", err)
        }
}

// newUUID returns a random UUID v4.
func newUUID() string {
        var b [16]byte
        _, _ = rand.Read(b[:]) // crypto/rand does not fail on supported platforms
        b[6] = b[6]&0x0f | 0x40
        b[8] = b[8]&0x3f | 0x80
        return fmt.Sprintf("%x-%x-%x-%x-%x", b[0:4], b[4:6], b[6:8], b[8:10], b[10:16])
}

        msgTooLarge         = "Ответ ассистента слишком большой, попробуйте ещё раз"
        msgBadResult        = "Не удалось подтвердить результат заявки, попробуйте ещё раз"
)

// agentError turns a failed agent call into the error event of the stream: an agent that answered 4xx with a JSON
// error keeps its code and message; everything else is upstream_unavailable, retryable.
func agentError(err error) streamError {
        var ue *agentclient.UnavailableError
        if errors.As(err, &ue) && ue.Status >= 400 && ue.Status < 500 && ue.Code != "" {
                msg := ue.Message
                if msg == "" {
                        msg = msgAgentUnavailable
                }
                return streamError{Code: ue.Code, Message: msg, Retryable: ue.Retryable}
        }
        return streamError{Code: string(apierr.CodeUpstreamUnavailable), Message: msgAgentUnavailable, Retryable: true}
}

// errTooLarge ends a turn whose content exceeds the bounds (R76).
var errTooLarge = streamError{Code: string(apierr.CodeUpstreamUnavailable), Message: msgTooLarge, Retryable: true}

// exactObject decodes a JSON object and rejects keys that differ only in case from one of want: Go's struct decoding
// would match them case-insensitively while the agent reads the exact key, so both sides must see the same value.
func exactObject(raw json.RawMessage, want ...string) (map[string]json.RawMessage, error) {
        var m map[string]json.RawMessage
        if err := json.Unmarshal(raw, &m); err != nil || m == nil {
                return nil, errors.New("not a JSON object")
        }
        for k := range m {
                for _, w := range want {
                        if k != w && strings.EqualFold(k, w) {
                                return nil, fmt.Errorf("key %q differs from %q only in case", k, w)
                        }
                }
        }
        return m, nil
}

// str returns the string value of key in m ("" when missing or not a string).
func str(m map[string]json.RawMessage, key string) string {
        var s string
        _ = json.Unmarshal(m[key], &s)
        return s
}

// Proposal lifecycle (core.proposals.status):
//
//        pending   → stored, never sent to the agent: can be confirmed, cancelled or expire.
//        confirmed → a confirm attempt is in flight.
//        failed    → an attempt reached the agent and did not complete. The write may have happened, so the proposal can
//                    no longer be cancelled or expire: a later confirm retries with the same idempotency key and replays.
//        executed  → saved (core.actions); a confirm replays the stored result (R79).
//        cancelled, expired → final; only reachable from pending.
//
// A failure before the agent is called puts the proposal back to the status it had.

// proposal is a row of core.proposals.
type proposal struct {
        id, kind, status, draftHash string
        draft                       json.RawMessage
        createdAt, expiresAt        time.Time
}

// loadProposal returns the caller's proposal in the session; a malformed id is not_found, a missing or foreign
// one forbidden.
func (h *Handler) loadProposal(ctx context.Context, q pgx.Row, id string) (proposal, error) {
        p := proposal{id: id}
        var draftText string
        err := q.Scan(&p.kind, &p.status, &p.draftHash, &draftText, &p.createdAt, &p.expiresAt)
        if errors.Is(err, pgx.ErrNoRows) {
                return p, apierr.Forbidden()
        }
        if err != nil {
                return p, fmt.Errorf("load proposal: %w", err)
        }
        p.draft = json.RawMessage(draftText)
        return p, nil
}

const proposalSelect = `SELECT kind, status, draft_hash, draft::text, created_at, expires_at FROM core.proposals
        WHERE id = $1 AND session_id = $2 AND user_id = $3`

// target checks the session and proposal ids of a confirm or cancel request and loads the proposal.
func (h *Handler) target(r *http.Request, c authn.Claims) (proposal, error) {
        sid, pid := r.PathValue("session_id"), r.PathValue("proposal_id")
        if err := h.ownSession(r.Context(), c.Sub, sid); err != nil {
                return proposal{}, err
        }
        if !uuidRE.MatchString(pid) {
                return proposal{}, apierr.NotFound()
        }
        return h.loadProposal(r.Context(), h.pool.QueryRow(r.Context(), proposalSelect, pid, sid, c.Sub), pid)
}

var (
        errCancelled = apierr.Conflict().WithMessage("Заявка отменена. Попросите ассистента подготовить новую")
        errExpired   = apierr.Expired().WithMessage("Черновик заявки устарел. Попросите ассистента подготовить его заново")
        errInFlight  = apierr.Conflict().WithMessage("Заявка уже отправлена")
)

// attempt is one confirmation of a proposal.
type attempt struct {
        c     authn.Claims
        token string
        p     proposal
        d     draft
        n     int // 1 for the first confirmation, 2 for a retry (analytics)
}

func (h *Handler) confirm(w http.ResponseWriter, r *http.Request) {
        ctx := r.Context()
        c, token, ok := caller(w, r)
        if !ok {
                return
        }
        p, err := h.target(r, c)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        sid := r.PathValue("session_id")
        switch {
        case p.status == "executed":
                h.replay(w, r, c, p)
                return
        case p.status == "cancelled":
                apierr.WriteJSON(w, errCancelled)
                return
        case p.status == "expired":
                apierr.WriteJSON(w, errExpired)
                return
        case p.status == "pending" && !h.Now().Before(p.expiresAt):
                // Only a proposal that never reached the agent expires; after an attempt the retry must stay possible (R66).
                if _, err := h.pool.Exec(ctx, `UPDATE core.proposals SET status = 'expired' WHERE id = $1 AND status = 'pending'`, p.id); err != nil {
                        h.log.Warn("mark proposal expired failed", "proposal", p.id, "err", err)
                }
                apierr.WriteJSON(w, errExpired)
                return
        }
        at := attempt{c: c, token: token, p: p, n: 1}
        if p.status != "pending" {
                at.n = 2
        }
        // A repeated confirm while one is in flight is allowed: the write is idempotent by proposal_id (R79).
        tag, err := h.pool.Exec(ctx, `UPDATE core.proposals SET status = 'confirmed' WHERE id = $1 AND status IN ('pending', 'failed', 'confirmed')`, p.id)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("claim proposal: %w", err))
                return
        }
        if tag.RowsAffected() != 1 {
                apierr.WriteJSON(w, errInFlight) // executed or cancelled in between
                return
        }
        if at.d, err = h.checkStored(c, p); err != nil {
                h.release(ctx, p)
                apierr.WriteJSON(w, err)
                return
        }
        confirmation, err := confirm.Issue(h.cfg.ConfirmationSecret, c.Sub, p.id, p.draftHash, h.cfg.ConfirmationTTL)
        if err != nil {
                h.release(ctx, p)
                apierr.WriteJSON(w, fmt.Errorf("issue confirmation token: %w", err))
                return
        }
        uc, flags, err := h.userContext(ctx, c)
        if err != nil {
                h.release(ctx, p)
                apierr.WriteJSON(w, err)
                return
        }
        hist, err := h.history(ctx, sid, newUUID())
        if err != nil {
                h.release(ctx, p)
                apierr.WriteJSON(w, err)
                return
        }
        sse, err := httpx.NewSSEWriter(ctx, w, h.cfg.KeepAlive)
        if err != nil {
                h.release(ctx, p)
                apierr.WriteJSON(w, err)
                return
        }
        defer sse.Close()
        // From here on the agent may have called the write tool: failures leave the proposal failed (retryable).
        assistantID := newUUID()
        stream, err := h.agent.Turn(ctx, token, agentclient.TurnRequest{SessionID: sid, MessageID: assistantID,
                Turn:        agentclient.Turn{Kind: agentclient.KindConfirm, ProposalID: p.id, Draft: p.draft, ConfirmationToken: confirmation},
                UserContext: uc, History: hist, Flags: flags})
        if err != nil {
                h.log.Warn("agent confirm turn failed", "proposal", p.id, "err", err)
                h.fail(ctx, c, p, sse, agentError(err))
                return
        }
        defer stream.Close()

        var (
                st    turnState
                saved bool
        )
loop:
        for {
                ev, err := stream.Next()
                switch {
                case errors.Is(err, io.EOF):
                        break loop
                case err != nil:
                        h.log.Warn("agent confirm stream failed", "proposal", p.id, "err", err)
                        if saved {
                                break loop // the write is saved; a late agent failure only loses the follow-up text
                        }
                        h.fail(ctx, c, p, sse, agentError(err))
                        return
                }
                st.collect(ev)
                if st.tooLarge {
                        h.log.Warn("agent confirm turn over the size bounds", "proposal", p.id, "event", ev.Name)
                        if !saved {
                                h.markFailed(ctx, p.id)
                        }
                        _ = sse.Send(agentclient.EventError, errTooLarge)
                        return
                }
                switch {
                case ev.Name == agentclient.EventToolResult:
                        if saved {
                                continue
                        }
                        res, err := parseToolResult(ev.Data, p, at.d)
                        if err != nil {
                                h.log.Warn("tool_result rejected", "proposal", p.id, "err", err)
                                h.fail(ctx, c, p, sse, streamError{Code: string(apierr.CodeUpstreamUnavailable), Message: msgBadResult, Retryable: true})
                                return
                        }
                        save := h.saveResult
                        if at.d.Tool == mcpclient.ToolDecideApproval {
                                save = h.saveDecision
                        }
                        a, msgID, err := save(ctx, at, res, assistantID)
                        if err != nil {
                                h.log.Error("save action failed", "proposal", p.id, "err", err)
                                h.fail(ctx, c, p, sse, streamError{Code: string(apierr.CodeInternal), Message: msgSaveFailed, Retryable: true})
                                return
                        }
                        saved, assistantID = true, msgID
                        if a != nil && sse.Send(agentclient.EventActionResult, a.Result()) != nil {
                                return
                        }
                        if res.Followup != "" {
                                if st.addText(res.Followup); st.tooLarge {
                                        _ = sse.Send(agentclient.EventError, errTooLarge)
                                        return
                                }
                                if sse.Send(agentclient.EventToken, map[string]string{"text": res.Followup}) != nil {
                                        return
                                }
                        }
                case ev.Name == agentclient.EventError:
                        if saved {
                                break loop
                        }
                        // The proposal stays confirmable: a retry is safe because the write is idempotent by proposal_id (R79).
                        var e streamError
                        _ = json.Unmarshal(ev.Data, &e)
                        h.markFailed(ctx, p.id)
                        h.emit(ctx, c, "action.failed", map[string]any{"kind": p.kind, "code": e.Code, "retryable": e.Retryable})
                        _ = sse.Send(ev.Name, ev.Data)
                        return
                case ev.Name == agentclient.EventDone:
                        break loop
                case internalEvents[ev.Name] || ev.Name == agentclient.EventActionProposal:
                        h.log.Warn("agent event dropped", "event", ev.Name, "proposal", p.id)
                default: // status, token, sources, next_step and events added later
                        if sse.Send(ev.Name, ev.Data) != nil {
                                if !saved {
                                        h.markFailed(ctx, p.id)
                                }
                                return
                        }
                }
        }
        if !saved {
                h.fail(ctx, c, p, sse, streamError{Code: string(apierr.CodeUpstreamUnavailable), Message: msgNoResult, Retryable: true})
                return
        }
        if text := st.text.String(); text != "" {
                uctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 10*time.Second)
                if _, err := h.pool.Exec(uctx, `UPDATE core.chat_messages SET text = $2 WHERE id = $1`, assistantID, text); err != nil {
                        h.log.Warn("update assistant message failed", "message", assistantID, "err", err)
                }
                cancel()
        }
        _ = sse.Send(agentclient.EventDone, st.doneEvent(assistantID))
}

// checkStored re-checks a stored proposal before it is signed: exact-key draft, the tool allowed for the stored kind
// (approval decisions only for leads) and draft_hash equal to the hash of draft.args.
func (h *Handler) checkStored(c authn.Claims, p proposal) (draft, error) {
        d, err := decodeDraft(p.draft)
        if err != nil {
                return draft{}, fmt.Errorf("proposal %s: %w: %w", p.id, apierr.Internal(), err)
        }
        if tool, ok := allowedTools[p.kind]; !ok || tool != d.Tool || (p.kind == kindApprovalDecision && (c.Role != "lead" || h.mcp == nil)) {
                return draft{}, fmt.Errorf("proposal %s: tool %q not allowed for kind %q: %w", p.id, d.Tool, p.kind, apierr.Forbidden().WithMessage(msgBadProposal))
        }
        if hash, err := canonjson.SHA256Hex(d.Args); err != nil || hash != p.draftHash {
                return draft{}, fmt.Errorf("proposal %s: stored draft does not match draft_hash: %w", p.id, apierr.Internal())
        }
        return d, nil
}

// replay answers a confirm of an executed proposal with the stored action_result and done, calling nobody (R79).
func (h *Handler) replay(w http.ResponseWriter, r *http.Request, c authn.Claims, p proposal) {
        var a *actions.Action
        var messageID string
        var stored struct {
                Tool string `json:"tool"`
                Core struct {
                        ActionID  string `json:"action_id"`
                        MessageID string `json:"message_id"`
                } `json:"_core"`
        }
        _ = json.Unmarshal(p.draft, &stored)
        if p.kind == kindApprovalDecision {
                messageID = stored.Core.MessageID
                if stored.Core.ActionID != "" {
                        got, err := h.actions.ByID(r.Context(), h.pool, stored.Core.ActionID)
                        if err != nil {
                                apierr.WriteJSON(w, fmt.Errorf("replay decision %s: %w", p.id, err))
                                return
                        }
                        a = &got
                }
        } else {
                got, err := h.actions.ByProposal(r.Context(), h.pool, c.Sub, p.id)
                if err != nil {
                        apierr.WriteJSON(w, fmt.Errorf("replay proposal %s: %w", p.id, err))
                        return
                }
                a, messageID = &got, got.MessageID
        }
        sse, err := httpx.NewSSEWriter(r.Context(), w, h.cfg.KeepAlive)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        defer sse.Close()
        if a != nil && sse.Send(agentclient.EventActionResult, a.Result()) != nil {
                return
        }
        _ = sse.Send(agentclient.EventDone, map[string]string{"message_id": messageID})
}

// saveResult saves the action, the assistant message announcing it and marks the proposal executed in one
// transaction. It survives a client disconnect: the write already happened and must be visible after reload (R66).
func (h *Handler) saveResult(ctx context.Context, at attempt, res toolResult, messageID string) (*actions.Action, string, error) {
        var out *actions.Action
        err := h.inProposalTx(ctx, at.p.id, func(ctx context.Context, tx pgx.Tx, sessionID string) (string, error) {
                a, err := h.actions.Save(ctx, tx, actions.SaveInput{Claims: at.c, ProposalID: at.p.id, Kind: at.p.kind, Tool: at.d.Tool,
                        Args: at.d.Args, Result: res.Result, MessageID: messageID, ProposedAt: at.p.createdAt, Attempt: at.n,
                        ChecklistItemID: res.ChecklistItemID, ApproverName: res.ApproverName})
                if err != nil {
                        return "", err
                }
                if a.MessageID != "" {
                        messageID = a.MessageID // a concurrent confirm saved it first
                }
                a.MessageID, out = messageID, &a
                if err := insertMessage(ctx, tx, messageID, sessionID, "assistant", a.Summary, nil); err != nil {
                        return "", err
                }
                return `UPDATE core.proposals SET status = 'executed' WHERE id = $1`, nil
        })
        return out, messageID, err
}

// saveDecision records a lead's approval decision made in chat (approval_decision). The agent's tool_result was
// already checked against the confirmed draft (approval id and decision); core then reads the decision back from the
// approvals service itself (approvals.decide with the same idempotency key replays the stored result and writes
// nothing new) and applies it only to the requester's action whose request names this lead as approver (R71):
// action and plan task follow the decision, the requester is notified, the assistant message is saved and the
// proposal is marked executed with a note of the action and message, which a later confirm replays. A decision on
// a request made outside the chat has no action in core: then only the message is saved.
func (h *Handler) saveDecision(ctx context.Context, at attempt, _ toolResult, messageID string) (*actions.Action, string, error) {
        var args map[string]any
        if err := json.Unmarshal(at.d.Args, &args); err != nil {
                return nil, "", fmt.Errorf("decode decision args: %w", err)
        }
        vctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 10*time.Second)
        raw, err := h.mcp.Execute(vctx, at.token, mcpclient.ToolDecideApproval, args, at.p.id)
        cancel()
        if err != nil {
                return nil, "", fmt.Errorf("read back approvals.decide: %w", err)
        }
        var decided mcpclient.Approval
        if err := mcpclient.Decode(raw, &decided); err != nil {
                return nil, "", err
        }
        if decided.ID != args["approval_id"] || decided.Status != args["decision"] || decided.RefID == "" || decided.RequesterID == "" {
                return nil, "", fmt.Errorf("approvals.decide answered %s/%s for %v/%v: %w", decided.ID, decided.Status,
                        args["approval_id"], args["decision"], apierr.Internal())
        }
        var out *actions.Action
        err = h.inProposalTx(ctx, at.p.id, func(ctx context.Context, tx pgx.Tx, sessionID string) (string, error) {
                a, changed, err := h.actions.ApplyDecision(ctx, tx, decided.Kind, decided.RefID, decided.Status,
                        actions.Scope{RequesterID: decided.RequesterID, ApproverID: at.c.Sub})
                switch {
                case errors.Is(err, actions.ErrNotFound):
                case err != nil:
                        return "", err
                default:
                        out = &a
                }
                text := decided.Summary
                actionID := ""
                if out != nil {
                        text, actionID = a.Summary, a.ID
                }
                if changed && h.emitter != nil {
                        if err := h.emitter.Emit(ctx, tx, analytics.Event{Name: "action.status_changed", UserID: at.c.Sub, Role: at.c.Role,
                                TeamID: at.c.TeamID, Props: map[string]any{"kind": a.Kind, "action_id": a.ID, "status": a.Status}}); err != nil {
                                return "", fmt.Errorf("emit action.status_changed: %w", err)
                        }
                }
                if err := insertMessage(ctx, tx, messageID, sessionID, "assistant", text, nil); err != nil {
                        return "", err
                }
                note, _ := json.Marshal(map[string]map[string]string{"_core": {"action_id": actionID, "message_id": messageID}})
                if _, err := tx.Exec(ctx, `UPDATE core.proposals SET draft = draft || $2::jsonb WHERE id = $1`, at.p.id, string(note)); err != nil {
                        return "", fmt.Errorf("note decision: %w", err)
                }
                return `UPDATE core.proposals SET status = 'executed' WHERE id = $1`, nil
        })
        return out, messageID, err
}

// inProposalTx runs fn in a transaction holding the proposal row lock and then executes the statement fn returns
// (with the proposal id). It survives a client disconnect: the write already happened and must be visible after
// reload (R66).
func (h *Handler) inProposalTx(ctx context.Context, id string, fn func(ctx context.Context, tx pgx.Tx, sessionID string) (string, error)) error {
        ctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 10*time.Second)
        defer cancel()
        tx, err := h.pool.Begin(ctx)
        if err != nil {
                return fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        var status, sessionID string
        if err := tx.QueryRow(ctx, `SELECT status, session_id::text FROM core.proposals WHERE id = $1 FOR UPDATE`, id).Scan(&status, &sessionID); err != nil {
                return fmt.Errorf("lock proposal: %w", err)
        }
        if status == "cancelled" || status == "expired" {
                return fmt.Errorf("proposal %s is %s: %w", id, status, apierr.Conflict())
        }
        final, err := fn(ctx, tx, sessionID)
        if err != nil {
                return err
        }
        if _, err := tx.Exec(ctx, final, id); err != nil {
                return fmt.Errorf("mark proposal executed: %w", err)
        }
        if err := tx.Commit(ctx); err != nil {
                return fmt.Errorf("commit: %w", err)
        }
        return nil
}

// fail marks the proposal failed (it can be confirmed again), records action.failed and emits the error event.
func (h *Handler) fail(ctx context.Context, c authn.Claims, p proposal, sse *httpx.SSEWriter, e streamError) {
        h.markFailed(ctx, p.id)
        h.emit(context.WithoutCancel(ctx), c, "action.failed", map[string]any{"kind": p.kind, "code": e.Code, "retryable": e.Retryable})
        _ = sse.Send(agentclient.EventError, e)
}

// markFailed moves a confirmed proposal to failed after an attempt reached the agent, even when the client has
// gone away.
func (h *Handler) markFailed(ctx context.Context, id string) {
        h.setStatus(ctx, id, "failed")
}

// release undoes the claim of an attempt that failed before the agent was called: the proposal gets back the
// status it had (pending stays cancellable and can expire; failed stays failed).
func (h *Handler) release(ctx context.Context, p proposal) {
        status := "failed"
        if p.status == "pending" {
                status = "pending"
        }
        h.setStatus(ctx, p.id, status)
}

func (h *Handler) setStatus(ctx context.Context, id, status string) {
        ctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 5*time.Second)
        defer cancel()
        if _, err := h.pool.Exec(ctx, `UPDATE core.proposals SET status = $2 WHERE id = $1 AND status = 'confirmed'`, id, status); err != nil {
                h.log.Warn("update proposal status failed", "proposal", id, "status", status, "err", err)
        }
}

func (h *Handler) cancel(w http.ResponseWriter, r *http.Request) {
        c, _, ok := caller(w, r)
        if !ok {
                return
        }
        p, err := h.target(r, c)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        // Nothing is sent anywhere: the proposal is only marked cancelled. Once an attempt reached the agent the write
        // may have happened, so cancelling is refused and the user can confirm again to see the result (R66).
        tag, err := h.pool.Exec(r.Context(), `UPDATE core.proposals SET status = 'cancelled' WHERE id = $1 AND status IN ('pending', 'expired', 'cancelled')`, p.id)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("cancel proposal: %w", err))
                return
        }
        if tag.RowsAffected() != 1 {
                msg := "Заявка уже отправлена, отменить её в чате нельзя"
                if p.status != "executed" {
                        msg = "Заявка уже могла быть отправлена, отменить её нельзя. Подтвердите ещё раз, чтобы увидеть результат"
                }
                apierr.WriteJSON(w, apierr.Conflict().WithMessage(msg))
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

// toolResult is what core takes from the agent's tool_result event.
type toolResult struct {
        Result          json.RawMessage // structured MCP result of draft.tool
        Followup        string          // optional text the agent wants shown after the result card
        ChecklistItemID string          // task_checklist_item_id: plan task to link (access only)
        ApproverName    string          // optional approver_display_name
}

// parseToolResult binds the agent's tool_result to the confirmed proposal (R71): proposal_id, kind and tool must be
// those of the proposal and the stored draft, the MCP result ("result") must describe the confirmed draft.args.
func parseToolResult(data json.RawMessage, p proposal, d draft) (toolResult, error) {
        obj, err := exactObject(data, "proposal_id", "kind", "tool", "result")
        if err != nil {
                return toolResult{}, fmt.Errorf("tool_result: %w", err)
        }
        for key, want := range map[string]string{"proposal_id": p.id, "kind": p.kind, "tool": d.Tool} {
                if got := str(obj, key); got != want {
                        return toolResult{}, fmt.Errorf("tool_result %s %q, want %q", key, got, want)
                }
        }
        result, err := exactObject(obj["result"], "id", "status")
        if err != nil {
                return toolResult{}, fmt.Errorf("tool_result result: %w", err)
        }
        if err := checkResult(d, result); err != nil {
                return toolResult{}, err
        }
        tr := toolResult{Result: obj["result"], Followup: str(obj, "followup"), ApproverName: str(obj, "approver_display_name")}
        if d.Tool == mcpclient.ToolRequestAccess {
                tr.ChecklistItemID = str(obj, "task_checklist_item_id")
        }
        return tr, nil
}

// checkResult verifies that an MCP result is the result of the confirmed draft: the same dates or resource, the
// approval and decision that were signed, and a status a new request can have.
func checkResult(d draft, result map[string]json.RawMessage) error {
        var args map[string]json.RawMessage
        _ = json.Unmarshal(d.Args, &args) // decodeDraft checked it is an object
        same := func(keys ...string) error {
                for _, k := range keys {
                        if str(result, k) != str(args, k) {
                                return fmt.Errorf("result %s %q differs from the confirmed %q", k, str(result, k), str(args, k))
                        }
                }
                return nil
        }
        id, status := str(result, "id"), str(result, "status")
        switch d.Tool {
        case mcpclient.ToolCreateAbsence:
                if err := same("type", "start_date", "end_date"); err != nil {
                        return err
                }
                if !strings.HasPrefix(id, "ABS-") || status != actions.StatusPendingApproval {
                        return fmt.Errorf("new absence request %q has status %q", id, status)
                }
        case mcpclient.ToolRequestAccess:
                if err := same("resource_id"); err != nil {
                        return err
                }
                if !strings.HasPrefix(id, "ACC-") || (status != actions.StatusPendingApproval && status != actions.StatusGranted) {
                        return fmt.Errorf("new access request %q has status %q", id, status)
                }
        case mcpclient.ToolDecideApproval:
                if id == "" || id != str(args, "approval_id") || status != str(args, "decision") || (status != "approved" && status != "rejected") {
                        return fmt.Errorf("decision %q/%q differs from the confirmed %q/%q", id, status, str(args, "approval_id"), str(args, "decision"))
                }
        default:
                return fmt.Errorf("tool %q cannot be confirmed in chat", d.Tool)
        }
        return nil
}

// userContext builds user_context for the agent. Identity (user_id, role, team_id) comes only from the verified
// claims (R71); the rest from core's DB: display name, current period, open top-level tasks of the caller's own
// plans, mentees of a mentor or lead and the runtime flags of the team. Mentees and flags are best effort.
func (h *Handler) userContext(ctx context.Context, c authn.Claims) (agentclient.UserContext, map[string]bool, error) {
        now := h.Now().UTC()
        uc := agentclient.UserContext{UserID: c.Sub, Role: c.Role, TeamID: c.TeamID, CurrentPeriod: string(period.Week1),
                OpenTasks: []agentclient.OpenTask{}}
        var name, cohort, planCohort, location *string
        err := h.pool.QueryRow(ctx, `SELECT u.display_name, u.cohort_start::text, u.location,
                (SELECT i.cohort_start::text FROM core.plans p JOIN core.users i ON i.id = p.intern_id
                 WHERE p.user_id = u.id ORDER BY (p.plan_role = 'intern') DESC, p.created_at, p.id LIMIT 1)
                FROM core.users u WHERE u.id = $1`, c.Sub).Scan(&name, &cohort, &location, &planCohort)
        if err != nil && !errors.Is(err, pgx.ErrNoRows) {
                return uc, nil, fmt.Errorf("load user: %w", err)
        }
        if name != nil {
                uc.DisplayName = *name
        }
        if location != nil {
                uc.Location = *location
        }
        // A mentor follows the periods of the intern of their (first) mentor plan.
        for _, s := range []*string{&c.CohortStart, cohort, planCohort} {
                if s == nil || *s == "" {
                        continue
                }
                if start, err := period.ParseDate(*s); err == nil {
                        uc.CurrentPeriod = string(period.Current(start, now))
                        break
                }
        }
        tasks, err := h.openTasks(ctx, c.Sub, now.Format("2006-01-02"))
        if err != nil {
                return uc, nil, err
        }
        uc.OpenTasks = tasks
        if c.Role == "mentor" || c.Role == "lead" {
                if mentees, err := h.mentees(ctx, c, now); err != nil {
                        h.log.Warn("user_context mentees failed", "user", c.Sub, "err", err)
                } else {
                        uc.Mentees = mentees
                }
        }
        var flags map[string]bool
        if h.flags != nil {
                if f, err := h.flags.ForTeam(ctx, c.TeamID); err != nil {
                        h.log.Warn("user_context flags failed", "team", c.TeamID, "err", err)
                } else {
                        flags = f
                }
        }
        return uc, flags, nil
}

// Mentee is an item of user_context.mentees: the GET /api/team/interns progress of an intern plus their open
// tasks and the accesses they requested through the chat (S4: the mentor sees what the intern requested).
type Mentee struct {
        plan.InternProgress
        OpenTasks []agentclient.OpenTask `json:"open_tasks"`
        Accesses  []MenteeAccess         `json:"accesses"`
}

// MenteeAccess is an access request of a mentee.
type MenteeAccess struct {
        Name   string `json:"name"`
        Status string `json:"status"`

func (h *Handler) mentees(ctx context.Context, c authn.Claims, now time.Time) ([]Mentee, error) {
        interns, err := plan.TeamInterns(ctx, h.pool, c, now)
        if err != nil {
                return nil, err
        }
        out := make([]Mentee, 0, len(interns))
        for _, in := range interns {
                m := Mentee{InternProgress: in, Accesses: []MenteeAccess{}}
                if m.OpenTasks, err = h.openTasks(ctx, in.UserID, now.Format("2006-01-02")); err != nil {
                        return nil, err
                }
                rows, err := h.pool.Query(ctx, `SELECT COALESCE(payload->'result'->>'resource_name', payload->'args'->>'resource_id', summary), status
                        FROM core.actions WHERE user_id = $1 AND kind = 'access_request' ORDER BY created_at, id`, in.UserID)
                if err != nil {
                        return nil, fmt.Errorf("load mentee accesses: %w", err)
                }
                for rows.Next() {
                        var a MenteeAccess
                        if err := rows.Scan(&a.Name, &a.Status); err != nil {
                                rows.Close()
                                return nil, fmt.Errorf("scan mentee access: %w", err)
                        }
                        m.Accesses = append(m.Accesses, a)
                }
                rows.Close()
                if err := rows.Err(); err != nil {
                        return nil, fmt.Errorf("read mentee accesses: %w", err)
                }
                out = append(out, m)
        }
        return out, nil
}

// openTasks returns the open top-level tasks of the caller's own plans in plan order. Status is reported like
// GET /api/plan: overdue when due_at is before today and the task is neither done nor waiting_approval.
func (h *Handler) openTasks(ctx context.Context, userID, today string) ([]agentclient.OpenTask, error) {
        rows, err := h.pool.Query(ctx, `SELECT t.id::text, t.checklist_item_id, t.title,
                CASE WHEN t.status NOT IN ('done', 'waiting_approval') AND t.due_at < $2::date THEN 'overdue' ELSE t.status END,
                t.period, t.due_at::text, t.human_only
                FROM core.plan_tasks t JOIN core.plans p ON p.id = t.plan_id
                WHERE p.user_id = $1 AND t.parent_task_id IS NULL AND t.status <> 'done'
                ORDER BY array_position(ARRAY['week_1','week_2','month_1','month_2','after_3_months'], t.period), t.position, t.id
                LIMIT $3`, userID, today, openTasksLimit)
        if err != nil {
                return nil, fmt.Errorf("load open tasks: %w", err)
        }
        defer rows.Close()
        out := []agentclient.OpenTask{}
        for rows.Next() {
                var t agentclient.OpenTask
                if err := rows.Scan(&t.TaskID, &t.ChecklistItemID, &t.Title, &t.Status, &t.Period, &t.DueAt, &t.HumanOnly); err != nil {
                        return nil, fmt.Errorf("scan open task: %w", err)
                }
                out = append(out, t)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("read open tasks: %w", err)
        }
        return out, nil
}

// history returns the last historyLimit non-empty messages of the session in chronological order, excluding
// exceptID (the message of the current turn).
func (h *Handler) history(ctx context.Context, sessionID, exceptID string) ([]agentclient.HistoryItem, error) {
        rows, err := h.pool.Query(ctx, `SELECT author, text FROM (
                        SELECT author, text, created_at, id FROM core.chat_messages
                        WHERE session_id = $1 AND id <> $2::uuid AND text <> ''
                        ORDER BY created_at DESC, id DESC LIMIT $3) m
                ORDER BY created_at, id`, sessionID, exceptID, historyLimit)
        if err != nil {
                return nil, fmt.Errorf("load history: %w", err)
        }
        defer rows.Close()
        out := []agentclient.HistoryItem{}
        for rows.Next() {
                var it agentclient.HistoryItem
                if err := rows.Scan(&it.Author, &it.Text); err != nil {
                        return nil, fmt.Errorf("scan history: %w", err)
                }
                out = append(out, it)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("read history: %w", err)
        }
        return out, nil
}

// kindApprovalDecision is the proposal kind of a lead's decision in chat (approvals.decide).
const kindApprovalDecision = "approval_decision"

// allowedTools maps a proposal kind to the only write tool a chat proposal of that kind may name (R71).
var allowedTools = map[string]string{
        "absence_request":    mcpclient.ToolCreateAbsence,
        "access_request":     mcpclient.ToolRequestAccess,
        kindApprovalDecision: mcpclient.ToolDecideApproval, // lead only
}

// internalEvents are agent events never forwarded to web: tool_result is internal and action_result is emitted by
// core only (after saving). The agent's done is replaced by core's (same data, core's message_id).
var internalEvents = map[string]bool{
        agentclient.EventToolResult: true, agentclient.EventActionResult: true,
}

// turnState collects what a turn persists and reports while its events are relayed.
type turnState struct {
        text          strings.Builder
        sources       []json.RawMessage
        summary       string          // summary of a stored proposal (assistant text fallback)
        refusal       string          // refusal.reason
        refusalText   string          // refusal.text (assistant text fallback)
        escalatedTo   string          // escalation_draft.to_role
        done          json.RawMessage // the agent's done, relayed with core's message_id
        doneProvider  string
        doneLatencyMS *int64
        doneEscalated bool
        tooLarge      bool // text or sources over the per-turn bounds
}

// addText appends assistant text unless the turn would exceed maxTurnText (then tooLarge is set).
func (s *turnState) addText(t string) {
        if s.text.Len()+len(t) > maxTurnText {
                s.tooLarge = true
                return
        }
        s.text.WriteString(t)
}

// collect records the parts of an event core persists or reports.
func (s *turnState) collect(ev agentclient.Event) {
        switch ev.Name {
        case agentclient.EventToken:
                var t struct {
                        Text string `json:"text"`
                }
                if json.Unmarshal(ev.Data, &t) == nil {
                        s.addText(t.Text)
                }
        case agentclient.EventSources:
                var src struct {
                        Items []json.RawMessage `json:"items"`
                }
                if json.Unmarshal(ev.Data, &src) == nil {
                        if len(s.sources)+len(src.Items) > maxSources {
                                s.tooLarge = true
                        } else {
                                s.sources = append(s.sources, src.Items...)
                        }
                }
        case agentclient.EventRefusal:
                var r struct {
                        Reason string `json:"reason"`
                        Text   string `json:"text"`
                }
                if json.Unmarshal(ev.Data, &r) == nil {
                        s.refusal, s.refusalText = r.Reason, r.Text
                }
        case agentclient.EventEscalationDraft:
                var d struct {
                        ToRole string `json:"to_role"`
                }
                if json.Unmarshal(ev.Data, &d) == nil {
                        s.escalatedTo = d.ToRole
                }
        case agentclient.EventDone:
                s.done = ev.Data
                var d struct {
                        Provider  string `json:"provider"`
                        LatencyMS *int64 `json:"latency_ms"`
                        Escalated bool   `json:"escalated"`
                }
                if json.Unmarshal(ev.Data, &d) == nil {
                        s.doneProvider, s.doneLatencyMS, s.doneEscalated = d.Provider, d.LatencyMS, d.Escalated
                }
        }
}

// doneEvent is the agent's done (if any) with message_id set to messageID; core adds nothing else.
func (s *turnState) doneEvent(messageID string) json.RawMessage {
        var m map[string]json.RawMessage
        if json.Unmarshal(s.done, &m) != nil || m == nil {
                m = map[string]json.RawMessage{}
        }
        id, _ := json.Marshal(messageID)
        m["message_id"] = id
        out, _ := json.Marshal(m) // values are valid JSON: cannot fail
        return out
}

func (h *Handler) postMessage(w http.ResponseWriter, r *http.Request) {
        ctx := r.Context()
        c, token, ok := caller(w, r)
        if !ok {
                return
        }
        sid := r.PathValue("session_id")
        if err := h.ownSession(ctx, c.Sub, sid); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        var body struct {
                Text string `json:"text"`
        }
        if err := httpx.ReadJSONLimit(w, r, &body, 64<<10); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        text := strings.TrimSpace(body.Text)
        if text == "" || utf8.RuneCountInString(text) > maxTextRunes {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Сообщение: от 1 до 4000 символов"))
                return
        }
        if h.limiter != nil && h.cfg.RatePerMin > 0 {
                allowed, err := h.limiter.Allow(ctx, "chat:"+c.Sub, h.cfg.RatePerMin, time.Minute)
                if err != nil {
                        h.log.Warn("rate limiter unavailable, allowing", "err", err)
                }
                if !allowed {
                        apierr.WriteJSON(w, apierr.RateLimited().WithMessage("Слишком много сообщений, подождите минуту"))
                        return
                }
        }
        userMsgID := newUUID()
        if err := insertMessage(ctx, h.pool, userMsgID, sid, "user", text, nil); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        h.emit(ctx, c, "chat.message_sent", map[string]any{"session_id": sid})
        uc, flags, err := h.userContext(ctx, c)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        hist, err := h.history(ctx, sid, userMsgID)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        sse, err := httpx.NewSSEWriter(ctx, w, h.cfg.KeepAlive)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        defer sse.Close()
        started := h.Now()
        assistantID := newUUID()
        stream, err := h.agent.Turn(ctx, token, agentclient.TurnRequest{SessionID: sid, MessageID: assistantID,
                Turn: agentclient.Turn{Kind: agentclient.KindMessage, Text: text}, UserContext: uc, History: hist, Flags: flags})
        if err != nil {
                h.log.Warn("agent turn failed", "session", sid, "err", err)
                _ = sse.Send(agentclient.EventError, agentError(err))
                return
        }
        defer stream.Close()

        var st turnState
loop:
        for {
                ev, err := stream.Next()
                if errors.Is(err, io.EOF) {
                        break
                }
                if err != nil {
                        h.log.Warn("agent stream failed", "session", sid, "err", err)
                        h.saveAssistant(ctx, assistantID, sid, &st)
                        _ = sse.Send(agentclient.EventError, agentError(err))
                        return
                }
                st.collect(ev)
                if st.tooLarge {
                        h.log.Warn("agent turn over the size bounds", "session", sid, "event", ev.Name)
                        h.saveAssistant(ctx, assistantID, sid, &st)
                        _ = sse.Send(agentclient.EventError, errTooLarge)
                        return
                }
                switch {
                case internalEvents[ev.Name]:
                        h.log.Warn("agent event dropped", "event", ev.Name, "session", sid)
                case ev.Name == agentclient.EventActionProposal:
                        summary, err := h.storeProposal(ctx, c, sid, ev.Data)
                        if err != nil {
                                h.log.Warn("action proposal rejected", "session", sid, "err", err)
                                h.saveAssistant(ctx, assistantID, sid, &st)
                                e := apierr.From(err)
                                _ = sse.Send(agentclient.EventError, streamError{Code: string(e.Code), Message: e.Message, Retryable: e.Retryable})
                                return
                        }
                        st.summary = summary
                        if sse.Send(ev.Name, ev.Data) != nil {
                                return
                        }
                case ev.Name == agentclient.EventError:
                        h.saveAssistant(ctx, assistantID, sid, &st)
                        _ = sse.Send(ev.Name, ev.Data)
                        return
                case ev.Name == agentclient.EventDone:
                        break loop
                default: // status, token, sources, escalation_draft, next_step, refusal and events added later
                        if sse.Send(ev.Name, ev.Data) != nil {
                                return // client went away; ctx cancels the agent call
                        }
                }
        }
        if !h.saveAssistant(ctx, assistantID, sid, &st) {
                _ = sse.Send(agentclient.EventError, streamError{Code: string(apierr.CodeInternal), Message: apierr.Internal().Message, Retryable: true})
                return
        }
        h.reportAnswer(ctx, c, sid, assistantID, started, &st)
        _ = sse.Send(agentclient.EventDone, st.doneEvent(assistantID))
}

// reportAnswer writes chat.answered, chat.refused and chat.escalated from the turn (contracts/events.md).
func (h *Handler) reportAnswer(ctx context.Context, c authn.Claims, sid, messageID string, started time.Time, st *turnState) {
        latency := h.Now().Sub(started).Milliseconds()
        if st.doneLatencyMS != nil {
                latency = *st.doneLatencyMS
        }
        props := map[string]any{"session_id": sid, "message_id": messageID, "source_count": len(st.sources), "latency_ms": latency}
        if st.doneProvider != "" {
                props["provider"] = st.doneProvider
        }
        h.emit(ctx, c, "chat.answered", props)
        if st.refusal != "" {
                h.emit(ctx, c, "chat.refused", map[string]any{"reason": st.refusal})
        }
        if st.doneEscalated {
                esc := map[string]any{}
                if st.escalatedTo != "" {
                        esc["to_role"] = st.escalatedTo
                }
                h.emit(ctx, c, "chat.escalated", esc)
        }
}

// saveAssistant stores the assistant message of a turn (tokens, else the proposal summary or refusal text) so
// done.message_id always names a stored message, and reports success.
func (h *Handler) saveAssistant(ctx context.Context, id, sessionID string, st *turnState) bool {
        text := st.text.String()
        for _, alt := range []string{st.summary, st.refusalText} {
                if text == "" {
                        text = alt
                }
        }
        ctx, cancel := context.WithTimeout(context.WithoutCancel(ctx), 10*time.Second)
        defer cancel()
        if err := insertMessage(ctx, h.pool, id, sessionID, "assistant", text, st.sources); err != nil {
                h.log.Error("save assistant message failed", "session", sessionID, "err", err)
                return false
        }
        return true
}

// draft is the part of a proposal draft core checks; the whole draft is stored and sent back unchanged.
type draft struct {
        Tool string
        Args json.RawMessage
}

// decodeDraft reads tool and args with exact keys (no case variants of tool, args or display); args must be an
// object without idempotency_key and confirmation_token.
func decodeDraft(raw json.RawMessage) (draft, error) {
        m, err := exactObject(raw, "tool", "args", "display")
        if err != nil {
                return draft{}, fmt.Errorf("draft: %w", err)
        }
        d := draft{Tool: str(m, "tool"), Args: m["args"]}
        if d.Tool == "" {
                return draft{}, errors.New("draft without tool")
        }
        args, err := exactObject(d.Args, confirm.KeyIdempotency, confirm.KeyConfirmation)
        if err != nil {
                return draft{}, fmt.Errorf("draft args: %w", err)
        }
        for _, k := range []string{confirm.KeyIdempotency, confirm.KeyConfirmation} {
                if _, ok := args[k]; ok {
                        return draft{}, fmt.Errorf("draft args carry %s", k)
                }
        }
        return d, nil
}

// storeProposal validates an action_proposal and stores it in core.proposals for the caller and session with
// draft_hash = canonical SHA-256 of draft.args and expiry ProposalTTL. Only allow-listed write tools are accepted;
// approval decisions only from leads.
func (h *Handler) storeProposal(ctx context.Context, c authn.Claims, sessionID string, data json.RawMessage) (string, error) {
        bad := apierr.Forbidden().WithMessage(msgBadProposal)
        ev, err := exactObject(data, "proposal_id", "kind", "draft", "summary")
        if err != nil {
                return "", fmt.Errorf("decode proposal: %w: %w", bad, err)
        }
        p := struct{ ProposalID, Kind, Summary string }{str(ev, "proposal_id"), str(ev, "kind"), str(ev, "summary")}
        rawDraft := ev["draft"]
        if len(rawDraft) > maxDraftBytes {
                return "", fmt.Errorf("draft of %d bytes: %w", len(rawDraft), apierr.UpstreamUnavailable().WithMessage(msgTooLarge))
        }
        d, err := decodeDraft(rawDraft)
        if err != nil {
                return "", fmt.Errorf("%w: %w", bad, err)
        }
        if !uuidRE.MatchString(p.ProposalID) {
                return "", fmt.Errorf("proposal id %q is not a uuid: %w", p.ProposalID, bad)
        }
        if tool, ok := allowedTools[p.Kind]; !ok || tool != d.Tool {
                return "", fmt.Errorf("tool %q of kind %q is not allowed: %w", d.Tool, p.Kind, bad)
        }
        if p.Kind == kindApprovalDecision && (c.Role != "lead" || h.mcp == nil) {
                return "", fmt.Errorf("approval decision proposed to role %q: %w", c.Role, bad)
        }
        hash, err := canonjson.SHA256Hex(d.Args)
        if err != nil {
                return "", fmt.Errorf("hash draft args: %w: %w", bad, err)
        }
        var compact bytes.Buffer
        if err := json.Compact(&compact, rawDraft); err != nil {
                return "", fmt.Errorf("compact draft: %w: %w", bad, err)
        }
        tag, err := h.pool.Exec(ctx, `INSERT INTO core.proposals (id, session_id, user_id, kind, draft, draft_hash, expires_at)
                VALUES ($1, $2, $3, $4, $5, $6, $7) ON CONFLICT (id) DO NOTHING`,
                p.ProposalID, sessionID, c.Sub, p.Kind, compact.Bytes(), hash, h.Now().Add(h.cfg.ProposalTTL))
        if err != nil {
                return "", fmt.Errorf("insert proposal: %w", err)
        }
        if tag.RowsAffected() != 1 {
                return "", fmt.Errorf("proposal %s already exists: %w", p.ProposalID, bad)
        }
        h.emit(ctx, c, "action.proposed", map[string]any{"kind": p.Kind, "proposal_id": p.ProposalID})
        return p.Summary, nil
}

// Package cron is the AI-cron: a periodic job that marks overdue plan tasks and sends reminders to interns and
// mentors (R54). Texts are fixed Russian templates, no LLM is involved.
//
// Per plan the job (1) stores status 'overdue' on open tasks whose due date has passed, (2) creates one notification
// per overdue top-level task (dedupe key overdue:<task_id>) and (3) reminds about every open top-level task of the
// intern's current period (dedupe key period:<checklist_item_id>:<period>). Survey items CL-15 and CL-16 get their own
// text. Repeated runs never duplicate notifications. human_only items are only reminded, never touched.

// DefaultInterval is CRON_INTERVAL's default.
const DefaultInterval = 24 * time.Hour

// Result counts what one run did.
type Result struct {
        Plans         int `json:"plans"`
        MarkedOverdue int `json:"marked_overdue"`
        Overdue       int `json:"overdue_notifications"`
        Reminders     int `json:"reminders"`
}

// Runner executes the job.
type Runner struct {
        pool     *pgxpool.Pool
        notify   notify.Store
        emitter  *analytics.Emitter
        log      *slog.Logger
        interval time.Duration
        // Now is the clock (tests); defaults to time.Now.
        Now func() time.Time
}

// NewRunner returns a Runner that fires every interval (<= 0 means DefaultInterval). emitter and log may be nil.
func NewRunner(pool *pgxpool.Pool, emitter *analytics.Emitter, interval time.Duration, log *slog.Logger) *Runner {
        if interval <= 0 {
                interval = DefaultInterval
        }
        if log == nil {
                log = slog.Default()
        }
        return &Runner{pool: pool, emitter: emitter, interval: interval, log: log, Now: time.Now}
}

// Run executes the job once at start and then every interval until ctx is cancelled. Errors are logged.
func (r *Runner) Run(ctx context.Context) {
        t := time.NewTicker(r.interval)
        defer t.Stop()
        for {
                if res, err := r.RunOnce(ctx); err != nil {
                        r.log.Error("ai-cron run failed", "err", err)
                } else {
                        r.log.Info("ai-cron run finished", "plans", res.Plans, "overdue", res.Overdue, "reminders", res.Reminders)
                }
                select {
                case <-ctx.Done():
                        return
                case <-t.C:
                }
        }
}

type planRow struct {
        id, userID, role string
        cohort           *string
}

type taskRow struct {
        id, title, period string
        item              string
        due               *string
        humanOnly         bool
}

// RunOnce processes every plan, each in its own transaction, and returns the totals. The first error stops the run
// (already processed plans stay committed; the job is idempotent, so the next run continues).
func (r *Runner) RunOnce(ctx context.Context) (Result, error) {
        var res Result
        rows, err := r.pool.Query(ctx, `SELECT p.id::text, p.user_id, p.plan_role, u.cohort_start::text
                FROM core.plans p JOIN core.users u ON u.id = p.intern_id ORDER BY p.created_at, p.id`)
        if err != nil {
                return res, fmt.Errorf("select plans: %w", err)
        }
        var plans []planRow
        for rows.Next() {
                var p planRow
                if err := rows.Scan(&p.id, &p.userID, &p.role, &p.cohort); err != nil {
                        rows.Close()
                        return res, fmt.Errorf("scan plan: %w", err)
                }
                plans = append(plans, p)
        }
        rows.Close()
        if err := rows.Err(); err != nil {
                return res, fmt.Errorf("read plans: %w", err)
        }
        for _, p := range plans {
                if err := r.runPlan(ctx, p, &res); err != nil {
                        return res, fmt.Errorf("plan %s: %w", p.id, err)
                }
                res.Plans++
        }
        return res, nil
}

func (r *Runner) runPlan(ctx context.Context, p planRow, res *Result) error {
        now := r.Now().UTC()
        today := now.Format("2006-01-02")
        tx, err := r.pool.Begin(ctx)
        if err != nil {
                return fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()

        tag, err := tx.Exec(ctx, `UPDATE core.plan_tasks SET status = 'overdue', updated_at = now()
                WHERE plan_id = $1 AND due_at < $2::date AND status IN ('not_started', 'in_progress')`, p.id, today)
        if err != nil {
                return fmt.Errorf("mark overdue: %w", err)
        }
        res.MarkedOverdue += int(tag.RowsAffected())

        overdue, err := r.topTasks(ctx, tx, p.id, `status = 'overdue'`, "")
        if err != nil {
                return err
        }
        for _, t := range overdue {
                n := notify.Notification{UserID: p.userID, Kind: notify.KindOverdue, ChecklistItemID: t.item, DedupeKey: "overdue:" + t.id,
                        Title: "Просрочено: «" + t.title + "»", Body: overdueBody(t)}
                created, err := r.notify.CreateIfNew(ctx, tx, n)
                if err != nil {
                        return err
                }
                if !created {
                        continue
                }
                res.Overdue++
                user := analytics.Event{UserID: p.userID, Role: p.role}
                if err := r.emit(ctx, tx, user, "plan.task_overdue", map[string]any{"checklist_item_id": t.item, "period": t.period}); err != nil {
                        return err
                }
                if err := r.emit(ctx, tx, user, "notification.created", map[string]any{"kind": notify.KindOverdue, "checklist_item_id": t.item}); err != nil {
                        return err
                }
        }

        cur := period.Week1
        if p.cohort != nil {
                if start, err := period.ParseDate(*p.cohort); err == nil {
                        cur = period.Current(start, now)
                }
        }
        open, err := r.topTasks(ctx, tx, p.id, `status IN ('not_started', 'in_progress', 'overdue') AND period = $2`, string(cur))
        if err != nil {
                return err
        }
        for _, t := range open {
                title, body := reminderText(t, p.role)
                n := notify.Notification{UserID: p.userID, Kind: notify.KindReminder, ChecklistItemID: t.item,
                        DedupeKey: "period:" + t.item + ":" + t.period, Title: title, Body: body}
                created, err := r.notify.CreateIfNew(ctx, tx, n)
                if err != nil {
                        return err
                }
                if !created {
                        continue
                }
                res.Reminders++
                if err := r.emit(ctx, tx, analytics.Event{UserID: p.userID, Role: p.role}, "notification.created",
                        map[string]any{"kind": notify.KindReminder, "checklist_item_id": t.item}); err != nil {
                        return err
                }
        }
        if err := tx.Commit(ctx); err != nil {
                return fmt.Errorf("commit: %w", err)
        }
        return nil
}

func (r *Runner) emit(ctx context.Context, tx analytics.DBTX, base analytics.Event, name string, props map[string]any) error {
        base.Name, base.Props = name, props
        return r.emitter.Emit(ctx, tx, base)
}

// topTasks returns the top-level checklist tasks of a plan matching cond ($1 = plan id, $2 = arg when not empty).
func (r *Runner) topTasks(ctx context.Context, tx notify.DBTX, planID, cond, arg string) ([]taskRow, error) {
        args := []any{planID}
        if arg != "" {
                args = append(args, arg)
        }
        rows, err := tx.Query(ctx, `SELECT id::text, title, period, checklist_item_id, due_at::text, human_only FROM core.plan_tasks
                WHERE plan_id = $1 AND parent_task_id IS NULL AND source = 'checklist' AND `+cond+` ORDER BY position, id`, args...)
        if err != nil {
                return nil, fmt.Errorf("select tasks: %w", err)
        }
        defer rows.Close()
        var out []taskRow
        for rows.Next() {
                var t taskRow
                if err := rows.Scan(&t.id, &t.title, &t.period, &t.item, &t.due, &t.humanOnly); err != nil {
                        return nil, fmt.Errorf("scan task: %w", err)
                }
                out = append(out, t)
        }
        return out, rows.Err()
}

// ruDate turns YYYY-MM-DD into DD.MM.YYYY.
func ruDate(iso *string) string {
        if iso == nil {
                return ""
        }
        t, err := time.Parse("2006-01-02", *iso)
        if err != nil {
                return *iso
        }
        return t.Format("02.01.2006")
}

func overdueBody(t taskRow) string {
        b := "Срок был " + ruDate(t.due) + ". Откройте план адаптации и отметьте прогресс или спросите ассистента, если нужна помощь."
        if t.humanOnly {
                b = "Срок был " + ruDate(t.due) + ". Этот пункт выполняют люди, ассистент только напоминает о нём."
        }
        return b
}

func reminderText(t taskRow, planRole string) (title, body string) {
        switch t.item {
        case "CL-15", "CL-16":
                return "Пройдите опрос о стажировке", "Ваши ответы помогут улучшить программу для следующих стажёров."
        }
        if planRole == "mentor" {
                title = "Напоминание наставнику: «" + t.title + "»"
        } else {
                title = "Пора заняться: «" + t.title + "»"
        }
        body = "Пункт относится к текущему периоду адаптации"
        if t.due != nil {
                body += ", срок — " + ruDate(t.due)
        }
        body += "."
        if t.humanOnly {
                body += " Этот пункт выполняют люди, ассистент только напоминает о нём."
        }
        return title, body
}

// Handler serves POST /api/cron/run.
type Handler struct {
        runner *Runner
        log    *slog.Logger
        // Demo enables the endpoint; it defaults to buildflags.Demo and is a field so tests can flip it.
        Demo bool
        // Sync makes the handler run the job before answering 202 (tests). By default it runs in the background.
        Sync bool
}

// New returns the cron HTTP handler.
func New(runner *Runner, log *slog.Logger) *Handler {
        if log == nil {
                log = slog.Default()
        }
        return &Handler{runner: runner, log: log, Demo: buildflags.Demo}
}

// Register adds POST /api/cron/run (auth required; answers 404 outside demo builds).
func (h *Handler) Register(mux *http.ServeMux) {
        mux.HandleFunc("POST /api/cron/run", h.run)
}

func (h *Handler) run(w http.ResponseWriter, r *http.Request) {
        if !h.Demo {
                apierr.WriteJSON(w, apierr.NotFound())
                return
        }
        if _, ok := authn.FromContext(r.Context()); !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        job := func(ctx context.Context) {
                if res, err := h.runner.RunOnce(ctx); err != nil {
                        h.log.Error("manual ai-cron run failed", "err", err)
                } else {
                        h.log.Info("manual ai-cron run finished", "plans", res.Plans, "overdue", res.Overdue, "reminders", res.Reminders)
                }
        }
        if h.Sync {
                job(r.Context())
        } else {
                // The request ends right after 202, so the job gets its own context.
                go func() {
                        ctx, cancel := context.WithTimeout(context.Background(), time.Minute)
                        defer cancel()
                        job(ctx)
                }()
        }
        w.WriteHeader(http.StatusAccepted)
}

// Package escalations lets an intern send a question drafted by buddy to a person (mentor, lead, HR BP, curator),

// FAQWriter publishes an answered question to the team knowledge base (kb.add_faq). Core's MCP client implements it
// and issues the confirmation token itself; mentorToken is the raw JWT of the answering user and idempotencyKey is
// the escalation id.
type FAQWriter interface {
        AddFAQ(ctx context.Context, mentorToken, title, body, idempotencyKey string) error
}

// FlagChecker tells whether a runtime flag is on for a team (flags.Service implements it).
type FlagChecker interface {
        Enabled(ctx context.Context, key, team string) bool
}

// Handler serves the escalation endpoints.
type Handler struct {
        pool    *pgxpool.Pool
        notify  notify.Store
        emitter *analytics.Emitter
        flags   FlagChecker
        faq     FAQWriter
        log     *slog.Logger
}

// New returns the escalation HTTP handler. emitter, faq and log may be nil (nil faq = KB publishing off).
func New(pool *pgxpool.Pool, emitter *analytics.Emitter, flags FlagChecker, faq FAQWriter, log *slog.Logger) *Handler {
        if log == nil {
                log = slog.Default()
        }
        return &Handler{pool: pool, emitter: emitter, flags: flags, faq: faq, log: log}
}

// Register adds the escalation routes (auth required).
func (h *Handler) Register(mux *http.ServeMux) {
        mux.HandleFunc("POST /api/escalations", h.create)

var uuidRE = regexp.MustCompile(`^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$`)

const maxText = 4000

// Escalation is the openapi Escalation.
type Escalation struct {
        ID              string  `json:"id"`
        FromDisplayName string  `json:"from_display_name"`
        ToRole          string  `json:"to_role"`
        Text            string  `json:"text"`
        Status          string  `json:"status"`
        Answer          *string `json:"answer"`
        AddedToKB       bool    `json:"added_to_kb"`
        CreatedAt       string  `json:"created_at"`
}

const selectCols = `e.id::text, u.display_name, e.to_role, e.text, e.status, e.answer, e.added_to_kb,
        to_char(e.created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')`

func scanEsc(row pgx.Row) (Escalation, error) {
        var e Escalation
        err := row.Scan(&e.ID, &e.FromDisplayName, &e.ToRole, &e.Text, &e.Status, &e.Answer, &e.AddedToKB, &e.CreatedAt)
        return e, err
}

func claims(w http.ResponseWriter, r *http.Request) (authn.Claims, bool) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
        }
        return c, ok
}

// target finds the addressee: the mentor of the sender, the lead of the sender's team, or the first user of the role.
func (h *Handler) target(ctx context.Context, c authn.Claims, role string) (string, error) {
        var id string
        var err error
        switch role {
        case "mentor":
                err = h.pool.QueryRow(ctx, `SELECT user_id FROM core.plans WHERE intern_id = $1 AND plan_role = 'mentor' ORDER BY created_at, user_id LIMIT 1`, c.Sub).Scan(&id)
        case "lead":
                err = h.pool.QueryRow(ctx, `SELECT id FROM core.users WHERE role = 'lead' AND team_id = $1 ORDER BY id LIMIT 1`, c.TeamID).Scan(&id)
        default: // hrbp, curator
                err = h.pool.QueryRow(ctx, `SELECT id FROM core.users WHERE role = $1 ORDER BY id LIMIT 1`, role).Scan(&id)
        }
        if errors.Is(err, pgx.ErrNoRows) {
                return "", apierr.NotFound().WithMessage("Адресат не найден")
        }
        if err != nil {
                return "", fmt.Errorf("find addressee: %w", err)
        }
        return id, nil
}

func (h *Handler) create(w http.ResponseWriter, r *http.Request) {
        c, ok := claims(w, r)
        if !ok {
                return
        }
        if c.Role != "intern" {
                apierr.WriteJSON(w, apierr.Forbidden())
                return
        }
        var body struct {
                ToRole    string  `json:"to_role"`
                Text      string  `json:"text"`
                SessionID *string `json:"session_id"`
        }
        if err := httpx.ReadJSON(w, r, &body); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        switch body.ToRole {
        case "mentor", "lead", "hrbp", "curator":
        default:
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Адресат: mentor, lead, hrbp или curator"))
                return
        }
        text := strings.TrimSpace(body.Text)
        if text == "" || utf8.RuneCountInString(text) > maxText {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Текст обращения: от 1 до 4000 символов"))
                return
        }
        ctx := r.Context()
        var session any
        if body.SessionID != nil {
                if !uuidRE.MatchString(*body.SessionID) {
                        apierr.WriteJSON(w, apierr.Validation().WithMessage("Некорректный session_id"))
                        return
                }
                var own bool
                err := h.pool.QueryRow(ctx, `SELECT true FROM core.chat_sessions WHERE id = $1 AND user_id = $2`, *body.SessionID, c.Sub).Scan(&own)
                if errors.Is(err, pgx.ErrNoRows) {
                        apierr.WriteJSON(w, apierr.Forbidden())
                        return
                }
                if err != nil {
                        apierr.WriteJSON(w, fmt.Errorf("check session: %w", err))
                        return
                }
                session = *body.SessionID
        }
        to, err := h.target(ctx, c, body.ToRole)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        out, err := h.insert(ctx, c, to, body.ToRole, text, session)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        httpx.WriteJSON(w, http.StatusCreated, out)
}

func (h *Handler) insert(ctx context.Context, c authn.Claims, to, toRole, text string, session any) (Escalation, error) {
        tx, err := h.pool.Begin(ctx)
        if err != nil {
                return Escalation{}, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        var id string
        if err := tx.QueryRow(ctx, `INSERT INTO core.escalations (from_user_id, to_user_id, to_role, session_id, text)
                VALUES ($1, $2, $3, $4, $5) RETURNING id::text`, c.Sub, to, toRole, session, text).Scan(&id); err != nil {
                return Escalation{}, fmt.Errorf("insert escalation: %w", err)
        }
        if err := h.notify.Create(ctx, tx, notify.Notification{UserID: to, Kind: notify.KindReminder,
                Title: "Новый вопрос от стажёра", Body: "Откройте раздел обращений, чтобы ответить.", DedupeKey: "escalation:" + id}); err != nil {
                return Escalation{}, err
        }
        if err := h.emitter.Emit(ctx, tx, analytics.Event{Name: "escalation.sent", UserID: c.Sub, Role: c.Role, TeamID: c.TeamID,
                Props: map[string]any{"to_role": toRole}}); err != nil {
                return Escalation{}, err
        }
        out, err := scanEsc(tx.QueryRow(ctx, `SELECT `+selectCols+` FROM core.escalations e JOIN core.users u ON u.id = e.from_user_id WHERE e.id = $1`, id))
        if err != nil {
                return Escalation{}, fmt.Errorf("reload escalation: %w", err)
        }
        if err := tx.Commit(ctx); err != nil {
                return Escalation{}, fmt.Errorf("commit: %w", err)
        }
        return out, nil
}

func (h *Handler) list(w http.ResponseWriter, r *http.Request) {
        c, ok := claims(w, r)
        if !ok {
                return
        }
        rows, err := h.pool.Query(r.Context(), `SELECT `+selectCols+` FROM core.escalations e JOIN core.users u ON u.id = e.from_user_id
                WHERE e.from_user_id = $1 OR e.to_user_id = $1 ORDER BY e.created_at DESC, e.id LIMIT 100`, c.Sub)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("list escalations: %w", err))
                return
        }
        defer rows.Close()
        out := []Escalation{}
        for rows.Next() {
                e, err := scanEsc(rows)
                if err != nil {
                        apierr.WriteJSON(w, fmt.Errorf("scan escalation: %w", err))
                        return
                }
                out = append(out, e)
        }
        if err := rows.Err(); err != nil {
                apierr.WriteJSON(w, fmt.Errorf("read escalations: %w", err))
                return
        }
        httpx.WriteJSON(w, http.StatusOK, out)
}

func (h *Handler) answer(w http.ResponseWriter, r *http.Request) {
        c, ok := claims(w, r)
        if !ok {
                return
        }
        id := r.PathValue("escalation_id")
        if !uuidRE.MatchString(id) {
                apierr.WriteJSON(w, apierr.NotFound())
                return
        }
        var body struct {
                Text    string `json:"text"`
                AddToKB bool   `json:"add_to_kb"`
        }
        if err := httpx.ReadJSON(w, r, &body); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        text := strings.TrimSpace(body.Text)
        if text == "" || utf8.RuneCountInString(text) > maxText {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Текст ответа: от 1 до 4000 символов"))
                return
        }
        question, err := h.saveAnswer(r.Context(), c, id, text)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        ctx := r.Context()
        added := false
        if body.AddToKB && h.faq != nil && h.flags != nil && h.flags.Enabled(ctx, flags.EscalationToKB, c.TeamID) {
                token, _ := authn.TokenFromContext(ctx)
                title := faqTitle(question)
                // The answer is already saved; a KB failure must not lose it, so it is logged and reported as added_to_kb=false.
                if err := h.faq.AddFAQ(ctx, token, title, "Вопрос: "+question+"\n\nОтвет: "+text, id); err != nil {
                        h.log.Warn("kb.add_faq failed", "escalation_id", id, "err", err)
                } else {
                        added = true
                }
        }
        if added {
                if _, err := h.pool.Exec(ctx, `UPDATE core.escalations SET added_to_kb = true WHERE id = $1`, id); err != nil {
                        h.log.Error("mark added_to_kb failed", "escalation_id", id, "err", err)
                }
        }
        if err := h.emitter.Emit(ctx, h.pool, analytics.Event{Name: "escalation.answered", UserID: c.Sub, Role: c.Role, TeamID: c.TeamID,
                Props: map[string]any{"added_to_kb": added}}); err != nil {
                h.log.Warn("escalation.answered event failed", "err", err)
        }
        out, err := scanEsc(h.pool.QueryRow(ctx, `SELECT `+selectCols+` FROM core.escalations e JOIN core.users u ON u.id = e.from_user_id WHERE e.id = $1`, id))
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("reload escalation: %w", err))
                return
        }
        httpx.WriteJSON(w, http.StatusOK, out)
}

// saveAnswer stores the answer and notifies the sender; it returns the question text.
func (h *Handler) saveAnswer(ctx context.Context, c authn.Claims, id, text string) (string, error) {
        tx, err := h.pool.Begin(ctx)
        if err != nil {
                return "", fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        var from, to, status, question string
        err = tx.QueryRow(ctx, `SELECT from_user_id, to_user_id, status, text FROM core.escalations WHERE id = $1 FOR UPDATE`, id).
                Scan(&from, &to, &status, &question)
        if errors.Is(err, pgx.ErrNoRows) || (err == nil && to != c.Sub) {
                return "", apierr.Forbidden() // only the addressee may answer; unknown ids look the same
        }
        if err != nil {
                return "", fmt.Errorf("select escalation: %w", err)
        }
        if status != "open" {
                return "", apierr.Conflict().WithMessage("На это обращение уже ответили")
        }
        if _, err := tx.Exec(ctx, `UPDATE core.escalations SET status = 'answered', answer = $2, answered_at = now() WHERE id = $1`, id, text); err != nil {
                return "", fmt.Errorf("update escalation: %w", err)
        }
        if err := h.notify.Create(ctx, tx, notify.Notification{UserID: from, Kind: notify.KindEscalationAnswer,
                Title: "Получен ответ на ваш вопрос", Body: "Откройте раздел обращений, чтобы прочитать ответ.", DedupeKey: "escalation_answer:" + id}); err != nil {
                return "", err
        }
        if err := tx.Commit(ctx); err != nil {
                return "", fmt.Errorf("commit: %w", err)
        }
        return question, nil
}

// faqTitle is the first line of the question, shortened to 80 characters.
func faqTitle(question string) string {
        line := strings.TrimSpace(strings.SplitN(strings.TrimSpace(question), "\n", 2)[0])
        r := []rune(line)
        if len(r) > 80 {
                return strings.TrimSpace(string(r[:79])) + "…"
        }
        return line
}

var knownKeys = []string{ScenarioS1, ScenarioS2, ScenarioS3, ScenarioS4, ProactiveDigest, EscalationToKB}

// CacheTTL is how long the flag table is cached in Redis.
const CacheTTL = 30 * time.Second

const cacheKey = "flags:v1"

// Flag is one row of core.feature_flags. Empty Teams means all teams.
type Flag struct {
        Key     string   `json:"key"`
        Enabled bool     `json:"enabled"`
        Teams   []string `json:"teams"`
}

// On reports whether the flag is on for team.
func (f Flag) On(team string) bool {
        return f.Enabled && (len(f.Teams) == 0 || slices.Contains(f.Teams, team))
}

// Service reads and writes flags.
type Service struct {
        pool    *pgxpool.Pool
        rdb     redis.Cmdable
        emitter *analytics.Emitter
        log     *slog.Logger
}

// NewService returns a flag service. rdb may be nil (no cache); it must not be a typed-nil pointer.
func NewService(pool *pgxpool.Pool, rdb redis.Cmdable, emitter *analytics.Emitter, log *slog.Logger) *Service {
        if log == nil {
                log = slog.Default()
        }
        return &Service{pool: pool, rdb: rdb, emitter: emitter, log: log}
}

// All returns every flag, from the Redis cache when it is available and fresh, otherwise from the database.
func (s *Service) All(ctx context.Context) ([]Flag, error) {
        if s.rdb != nil {
                raw, err := s.rdb.Get(ctx, cacheKey).Bytes()
                if err == nil {
                        var fl []Flag
                        if json.Unmarshal(raw, &fl) == nil {
                                return fl, nil
                        }
                } else if err != redis.Nil {
                        s.log.Warn("flags cache read failed, using database", "err", err)
                }
        }
        fl, err := s.load(ctx)
        if err != nil {
                return nil, err
        }
        if s.rdb != nil {
                if raw, err := json.Marshal(fl); err == nil {
                        if err := s.rdb.Set(ctx, cacheKey, raw, CacheTTL).Err(); err != nil {
                                s.log.Warn("flags cache write failed", "err", err)
                        }
                }
        }
        return fl, nil
}

func (s *Service) load(ctx context.Context) ([]Flag, error) {
        rows, err := s.pool.Query(ctx, `SELECT key, enabled, teams FROM core.feature_flags ORDER BY key`)
        if err != nil {
                return nil, fmt.Errorf("select flags: %w", err)
        }
        defer rows.Close()
        out := []Flag{}
        for rows.Next() {
                var f Flag
                if err := rows.Scan(&f.Key, &f.Enabled, &f.Teams); err != nil {
                        return nil, fmt.Errorf("scan flag: %w", err)
                }
                if f.Teams == nil {
                        f.Teams = []string{}
                }
                out = append(out, f)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("read flags: %w", err)
        }
        return out, nil
}

// ForTeam returns key -> enabled for team (targeting applied).
func (s *Service) ForTeam(ctx context.Context, team string) (map[string]bool, error) {
        fl, err := s.All(ctx)
        if err != nil {
                return nil, err
        }
        out := make(map[string]bool, len(fl))
        for _, f := range fl {
                out[f.Key] = f.On(team)
        }
        return out, nil
}

// Enabled reports whether key is on for team; unknown keys and read errors count as off.
func (s *Service) Enabled(ctx context.Context, key, team string) bool {
        fl, err := s.All(ctx)
        if err != nil {
                s.log.Error("flag lookup failed", "key", key, "err", err)
                return false
        }
        for _, f := range fl {
                if f.Key == key {
                        return f.On(team)
                }
        }
        return false
}

// Set upserts a flag, emits flag.changed and drops the cache. teams empty = all teams.
func (s *Service) Set(ctx context.Context, actor authn.Claims, key string, enabled bool, teams []string) error {
        if teams == nil {
                teams = []string{}
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        if _, err := tx.Exec(ctx, `INSERT INTO core.feature_flags (key, enabled, teams) VALUES ($1, $2, $3)
                ON CONFLICT (key) DO UPDATE SET enabled = EXCLUDED.enabled, teams = EXCLUDED.teams, updated_at = now()`,
                key, enabled, teams); err != nil {
                return fmt.Errorf("upsert flag: %w", err)
        }
        if err := s.emitter.Emit(ctx, tx, analytics.Event{Name: "flag.changed", UserID: actor.Sub, Role: actor.Role, TeamID: actor.TeamID,
                Props: map[string]any{"key": key, "enabled": enabled, "teams_count": len(teams)}}); err != nil {
                return err
        }
        if err := tx.Commit(ctx); err != nil {
                return fmt.Errorf("commit: %w", err)
        }
        if s.rdb != nil {
                if err := s.rdb.Del(ctx, cacheKey).Err(); err != nil {
                        s.log.Warn("flags cache invalidation failed; stale for up to 30s", "err", err)
                }
        }
        return nil
}

// Register adds GET /api/flags and PUT /api/flags/{key} (auth required; PUT answers 404 outside demo builds).
func (h *Handler) Register(mux *http.ServeMux) {
        mux.HandleFunc("GET /api/flags", h.get)
        mux.HandleFunc("PUT /api/flags/{key}", h.put)
}

func (h *Handler) get(w http.ResponseWriter, r *http.Request) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        m, err := h.svc.ForTeam(r.Context(), c.TeamID)
        if err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        httpx.WriteJSON(w, http.StatusOK, m)
}

var teamRE = regexp.MustCompile(`^[a-z0-9_-]{1,40}$`)

func (h *Handler) put(w http.ResponseWriter, r *http.Request) {
        if !h.Demo {
                apierr.WriteJSON(w, apierr.NotFound())
                return
        }
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        if c.Role != "lead" {
                apierr.WriteJSON(w, apierr.Forbidden())
                return
        }
        key := r.PathValue("key")
        if !slices.Contains(knownKeys, key) {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Неизвестный флаг"))
                return
        }
        var body struct {
                Enabled *bool    `json:"enabled"`
                Teams   []string `json:"teams"`
        }
        if err := httpx.ReadJSON(w, r, &body); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        if body.Enabled == nil || len(body.Teams) > 20 {
                apierr.WriteJSON(w, apierr.Validation().WithMessage("Укажите enabled и не более 20 команд"))
                return
        }
        for _, t := range body.Teams {
                if !teamRE.MatchString(t) {
                        apierr.WriteJSON(w, apierr.Validation().WithMessage("Недопустимый идентификатор команды"))
                        return
                }
        }
        if err := h.svc.Set(r.Context(), c, key, *body.Enabled, body.Teams); err != nil {
                apierr.WriteJSON(w, err)
                return
        }
        h.get(w, r)
}

// Package mcpclient is core's typed adapter over mcpx.Client for the mock MCP servers (contracts/mcp-tools.md).
//
// Core calls MCP itself only where a button press is the confirmation: the lead's decision buttons
// (approvals.list_pending, approvals.decide) and a mentor's answer published to the KB (kb.add_faq). Confirmed chat
// writes (absence.create_request, access.request_access) are executed by the agent in a confirm turn
// (contracts/agent-stream.md); their result types live here because core decodes and saves them. Every call carries
// the user's own JWT, so the MCP servers derive identity from it (R71). For write tools the client signs a
// confirmation token for the caller's verified subject, the idempotency key and the exact arguments it sends
// (mcpx.WithConfirmation).
//
// Errors: tool errors are returned as *apierr.Error with an MCP code (forbidden, not_found, validation_error,
// conflict, invalid_confirmation, unavailable); transport and protocol failures wrap an *apierr.Error
// (upstream_unavailable, unauthorized, internal). apierr.WriteJSON maps both onto openapi codes.
package mcpclient

import (
        "bytes"
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "net/http"
        "strings"
        "time"

        "aibuddy/services/internal/platform/apierr"
        "aibuddy/services/internal/platform/authn"
        "aibuddy/services/internal/platform/mcpx"
)

// DefaultBaseURL is MCP_BASE_URL when unset.
const DefaultBaseURL = "http://mcp-mocks:8100/mcp"

// DefaultConfirmationTTL is the lifetime of a confirmation token issued by the client (openapi: exp 5 min).
const DefaultConfirmationTTL = 5 * time.Minute

// Tool names in the "<server>.<tool>" form used by agent drafts.
const (
        ToolCreateAbsence  = "absence.create_request"
        ToolRequestAccess  = "access.request_access"
        ToolListApprovals  = "approvals.list_pending"
        ToolDecideApproval = "approvals.decide"
        ToolAddFAQ         = "kb.add_faq"
)

// Config configures a Client.
type Config struct {
        BaseURL            string        // MCP_BASE_URL; empty means DefaultBaseURL
        JWTSecret          []byte        // JWT_SECRET: the subject of a confirmation comes from the verified user token
        ConfirmationSecret []byte        // CONFIRMATION_HMAC_SECRET
        HTTPClient         *http.Client  // nil means http.DefaultTransport without a client timeout (use ctx deadlines)
        ConfirmationTTL    time.Duration // <= 0 means DefaultConfirmationTTL
}

// Client calls MCP tools on behalf of users. Safe for concurrent use.
type Client struct {
        mcp      *mcpx.Client
        verifier *authn.Verifier
        secret   []byte
        ttl      time.Duration
}

// New returns a Client; both secrets are required.
func New(cfg Config) (*Client, error) {
        if len(cfg.JWTSecret) == 0 || len(cfg.ConfirmationSecret) == 0 {
                return nil, errors.New("mcpclient: JWT and confirmation secrets are required")
        }
        base := cfg.BaseURL
        if base == "" {
                base = DefaultBaseURL
        }
        ttl := cfg.ConfirmationTTL
        if ttl <= 0 {
                ttl = DefaultConfirmationTTL
        }
        return &Client{
                mcp:      &mcpx.Client{BaseURL: base, HTTPClient: cfg.HTTPClient},
                verifier: authn.NewVerifier(cfg.JWTSecret),
                secret:   cfg.ConfirmationSecret,
                ttl:      ttl,
        }, nil
}

// SplitTool splits "<server>.<tool>" into its parts.
func SplitTool(tool string) (server, name string, ok bool) {
        server, name, ok = strings.Cut(tool, ".")
        if !ok || server == "" || name == "" || strings.ContainsAny(name, "./ ") || strings.ContainsAny(server, "/ ") {
                return "", "", false
        }
        return server, name, true
}

// Call invokes a read tool ("<server>.<tool>") with args (nil means {}) and returns its structured result.
func (c *Client) Call(ctx context.Context, userToken, tool string, args any) (json.RawMessage, error) {
        server, name, ok := SplitTool(tool)
        if !ok {
                return nil, fmt.Errorf("mcpclient: invalid tool name %q: %w", tool, apierr.Internal())
        }
        out, toolErr, err := c.mcp.Call(ctx, userToken, server, name, args)
        if err != nil {
                return nil, err
        }
        if toolErr != nil {
                return nil, toolErr
        }
        return out, nil
}

// Execute invokes a write tool ("<server>.<tool>") with args plus idempotency_key = key and a confirmation token
// signed for the subject of userToken. args must not contain idempotency_key or confirmation_token (they are
// overwritten). A repeated call with the same key returns the stored result (`replayed: true`).
func (c *Client) Execute(ctx context.Context, userToken, tool string, args map[string]any, key string) (json.RawMessage, error) {
        claims, err := c.verifier.Verify(userToken)
        if err != nil {
                return nil, fmt.Errorf("mcpclient: user token: %w: %w", apierr.Unauthorized(), err)
        }
        if args == nil {
                args = map[string]any{}
        }
        signed, err := mcpx.WithConfirmation(c.secret, claims.Sub, key, args, c.ttl)
        if err != nil {
                return nil, fmt.Errorf("mcpclient: sign %s: %w: %w", tool, apierr.Internal(), err)
        }
        return c.Call(ctx, userToken, tool, signed)
}

// Decode unmarshals a tool result into dst; a malformed result is an internal error.
func Decode(raw json.RawMessage, dst any) error {
        dec := json.NewDecoder(bytes.NewReader(raw))
        if err := dec.Decode(dst); err != nil {
                return fmt.Errorf("mcpclient: decode result: %w: %w", apierr.Internal(), err)
        }
        return nil
}

func execTyped[T any](ctx context.Context, c *Client, token, tool string, args map[string]any, key string) (T, error) {
        var out T
        raw, err := c.Execute(ctx, token, tool, args, key)
        if err != nil {
                return out, err
        }
        err = Decode(raw, &out)
        return out, err
}

// ListPendingApprovals calls approvals.list_pending with the lead's token.
func (c *Client) ListPendingApprovals(ctx context.Context, userToken string) ([]Approval, error) {
        raw, err := c.Call(ctx, userToken, ToolListApprovals, map[string]any{})
        if err != nil {
                return nil, err
        }
        var out struct {
                Items []Approval `json:"items"`
        }
        if err := Decode(raw, &out); err != nil {
                return nil, err
        }
        if out.Items == nil {
                out.Items = []Approval{}
        }
        return out.Items, nil
}

// DecideApproval calls approvals.decide with args {approval_id, decision[, comment]} (comment only when non-empty)

}

// New starts the fakes on an httptest server closed at test cleanup.
func New(t testing.TB, jwtSecret, confirmationSecret []byte) *Server {

        s := &Server{
                calls: map[string]int{}, lastArgs: map[string]json.RawMessage{}, fail: map[string][]*apierr.Error{},
                absence: map[string]*absenceRec{}, access: map[string]*accessRec{}, faqs: map[string]mcpclient.FAQ{},
                seqABS: 100, seqACC: 40,
        }
        cfg := mcpx.Config{JWTSecret: jwtSecret, ConfirmationSecret: confirmationSecret}
        mux := http.NewServeMux()
        var servers []*mcpx.Server
        for name, tools := range map[string]map[string]mcpx.HandlerFunc{
                "absence": {
                        "get_policy": s.getPolicy, "list_my_requests": s.listMyRequests, "validate_request": s.validateRequest,
                        "create_request": s.createRequest, "get_request": s.getRequest,
                },
                "access": {
                        "list_required": s.listRequired, "list_my_access": s.listMyAccess, "list_team_channels": s.listTeamChannels,
                        "request_access": s.requestAccess,
                },
                "approvals": {"list_pending": s.listPending, "decide": s.decide},
                "kb":        {"search": s.search, "add_faq": s.addFAQ},
        } {
                srv, err := mcpx.NewServer(name, cfg)
                if err != nil {
                        t.Fatalf("mcptest: %v", err)
                }
                for tool, h := range tools {
                        srv.Handle(tool, s.wrap(name+"."+tool, h))
                }
                servers = append(servers, srv)
        }
        if err := mcpx.Mount(mux, servers...); err != nil {
                t.Fatalf("mcptest: mount: %v", err)
        }
        ts := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                s.mu.Lock()
                s.requests++
                s.mu.Unlock()
                mux.ServeHTTP(w, r)
        }))
        t.Cleanup(ts.Close)
        s.URL = ts.URL + "/mcp"
        return s
}

// wrap counts handler calls, remembers the arguments and returns queued failures.
func (s *Server) wrap(tool string, h mcpx.HandlerFunc) mcpx.HandlerFunc {
        return func(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
                s.mu.Lock()
                s.calls[tool]++
                s.lastArgs[tool] = append(json.RawMessage(nil), args...)
                var queued *apierr.Error
                if q := s.fail[tool]; len(q) > 0 {
                        queued, s.fail[tool] = q[0], q[1:]
                }
                s.mu.Unlock()
                if queued != nil {
                        return nil, queued
                }
                return h(ctx, c, args)
        }
}

// Requests returns the number of HTTP requests received by all servers.

func (s *Server) decide(_ context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in struct {
                ApprovalID string `json:"approval_id"`
                Decision   string `json:"decision"`
                Comment    string `json:"comment"`
                Key        string `json:"idempotency_key"`
        }
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        s.mu.Lock()
        defer s.mu.Unlock()
        for _, a := range s.approvals {
                if a.res.ID != in.ApprovalID || a.approver != c.Sub {
                        continue
                }
                if a.key == in.Key {
                        out := a.res
                        out.Replayed = true
                        return out, nil
                }
                if a.res.Status != "pending" {
                        return nil, apierr.Conflict().WithMessage("Запрос уже рассмотрен")
                }
                a.key, a.res.Status, a.res.Comment = in.Key, in.Decision, in.Comment
                s.applyDecision(a.res)
                return a.res, nil
        }
        return nil, apierr.Forbidden()
}

        for _, r := range s.absence {
                if a.Kind == "absence_request" && r.res.ID == a.RefID {
                        r.res.Status = a.Status
                }
        }
        for _, r := range s.access {
                if a.Kind == "access_request" && r.res.ID == a.RefID {
                        r.res.Status = map[string]string{"approved": "granted", "rejected": "rejected"}[a.Status]
                }
        }
}

// DBTX is the subset of pgxpool.Pool and pgx.Tx used here.
type DBTX interface {
        Exec(ctx context.Context, sql string, args ...any) (pgconn.CommandTag, error)
        Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)
        QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
}

// Notification kinds (openapi Notification.kind).
const (
        KindReminder         = "reminder"
        KindOverdue          = "overdue"
        KindStatusChange     = "status_change"
        KindEscalationAnswer = "escalation_answer"
)

// ErrInvalid is returned by Create for a notification that cannot be stored (empty user or title, unknown kind).
var ErrInvalid = errors.New("notify: invalid notification")

// Notification is the input of Store.Create. Empty Body, ChecklistItemID and DedupeKey are stored as NULL.
type Notification struct {
        UserID          string
        Kind            string
        Title           string
        Body            string
        ChecklistItemID string
        // DedupeKey makes the notification unique per (UserID, DedupeKey): a second Create with the same pair is a no-op.
        DedupeKey string
}

// Store writes notifications. The zero value is ready to use.
type Store struct{}

// Create inserts n through q. A repeated (UserID, DedupeKey) is silently ignored (ON CONFLICT DO NOTHING).
func (s Store) Create(ctx context.Context, q DBTX, n Notification) error {
        _, err := s.CreateIfNew(ctx, q, n)
        return err
}

// CreateIfNew is Create that also reports whether a row was inserted (false = dedupe hit).
func (Store) CreateIfNew(ctx context.Context, q DBTX, n Notification) (bool, error) {
        switch n.Kind {
        case KindReminder, KindOverdue, KindStatusChange, KindEscalationAnswer:
        default:
                return false, fmt.Errorf("%w: kind %q", ErrInvalid, n.Kind)
        }
        if n.UserID == "" || n.Title == "" {
                return false, fmt.Errorf("%w: user and title are required", ErrInvalid)
        }
        tag, err := q.Exec(ctx, `INSERT INTO core.notifications (user_id, kind, title, body, checklist_item_id, dedupe_key)
                VALUES ($1, $2, $3, NULLIF($4, ''), NULLIF($5, ''), NULLIF($6, ''))
                ON CONFLICT DO NOTHING`, n.UserID, n.Kind, n.Title, n.Body, n.ChecklistItemID, n.DedupeKey)
        if err != nil {
                return false, fmt.Errorf("insert notification: %w", err)
        }
        return tag.RowsAffected() == 1, nil
}

        Kind            string  `json:"kind"`
        Title           string  `json:"title"`
        Body            *string `json:"body"`
        ChecklistItemID *string `json:"checklist_item_id"`
        Read            bool    `json:"read"`
        CreatedAt       string  `json:"created_at"`
}

const listLimit = 100

func (h *Handler) list(w http.ResponseWriter, r *http.Request) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        rows, err := h.pool.Query(r.Context(), `SELECT id::text, kind, title, body, checklist_item_id, read_at IS NOT NULL,
                to_char(created_at AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"')
                FROM core.notifications WHERE user_id = $1 ORDER BY created_at DESC, id LIMIT $2`, c.Sub, listLimit)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("list notifications: %w", err))
                return
        }
        defer rows.Close()
        out := []item{}
        for rows.Next() {
                var it item
                if err := rows.Scan(&it.ID, &it.Kind, &it.Title, &it.Body, &it.ChecklistItemID, &it.Read, &it.CreatedAt); err != nil {
                        apierr.WriteJSON(w, fmt.Errorf("scan notification: %w", err))
                        return
                }
                out = append(out, it)
        }
        if err := rows.Err(); err != nil {
                apierr.WriteJSON(w, fmt.Errorf("read notifications: %w", err))
                return
        }
        httpx.WriteJSON(w, http.StatusOK, out)
}

var uuidRE = regexp.MustCompile(`^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$`)

// markRead marks an own notification as read (idempotent). Foreign or unknown ids answer 403 so existence is not revealed.
func (h *Handler) markRead(w http.ResponseWriter, r *http.Request) {
        c, ok := authn.FromContext(r.Context())
        if !ok {
                apierr.WriteJSON(w, apierr.Unauthorized())
                return
        }
        id := r.PathValue("notification_id")
        if !uuidRE.MatchString(id) {
                apierr.WriteJSON(w, apierr.NotFound())
                return
        }
        tag, err := h.pool.Exec(r.Context(), `UPDATE core.notifications SET read_at = COALESCE(read_at, now())
                WHERE id = $1 AND user_id = $2`, id, c.Sub)
        if err != nil {
                apierr.WriteJSON(w, fmt.Errorf("mark notification read: %w", err))
                return
        }
        if tag.RowsAffected() == 0 {
                apierr.WriteJSON(w, apierr.Forbidden())
                return
        }
        w.WriteHeader(http.StatusNoContent)
}

// effectiveStatus applies the "overdue on read" rule. today and due are YYYY-MM-DD.
func effectiveStatus(status string, due *string, today string) string {
        if status == "done" || status == "waiting_approval" || status == "overdue" {
                return status
        }
        if due != nil && *due < today {
                return "overdue"
        }
        return status
}

type planRef struct {

        internID string

const planSelect = `SELECT p.id::text, p.user_id, p.plan_role, p.intern_id, u.cohort_start::text
        FROM core.plans p JOIN core.users u ON u.id = p.intern_id`

func (h *Handler) internInTeam(ctx context.Context, id, team string) (bool, error) {
        var ok bool
        err := h.pool.QueryRow(ctx, `SELECT true FROM core.users WHERE id = $1 AND role = 'intern' AND team_id = $2`, id, team).Scan(&ok)
        if errors.Is(err, pgx.ErrNoRows) {
                return false, nil
        }
        return ok, err
}

// resolve picks the plan the caller asked for or returns a 400/403/404 apierr.
func (h *Handler) resolve(ctx context.Context, c authn.Claims, userID, internID string) (planRef, error) {
        one := func(where string, args ...any) (planRef, error) {
                p, err := scanPlan(h.pool.QueryRow(ctx, planSelect+" WHERE "+where+" ORDER BY p.created_at, p.intern_id LIMIT 1", args...))
                if errors.Is(err, pgx.ErrNoRows) {
                        return p, apierr.NotFound().WithMessage("План не найден")
                }
                return p, err
        }
        switch c.Role {
        case "intern":
                if (userID != "" && userID != c.Sub) || (internID != "" && internID != c.Sub) {
                        return planRef{}, apierr.Forbidden()
                }
                return one("p.user_id = $1 AND p.plan_role = 'intern'", c.Sub)
        case "mentor":
                if userID != "" {
                        ok, err := h.internInTeam(ctx, userID, c.TeamID)
                        if err != nil {
                                return planRef{}, err
                        }
                        if !ok {
                                return planRef{}, apierr.Forbidden()
                        }
                        return one("p.user_id = $1 AND p.plan_role = 'intern'", userID)
                }
                if internID != "" {
                        p, err := one("p.user_id = $1 AND p.plan_role = 'mentor' AND p.intern_id = $2", c.Sub, internID)
                        if apierr.HasCode(err, apierr.CodeNotFound) {
                                return planRef{}, apierr.Forbidden()
                        }
                        return p, err
                }
                return one("p.user_id = $1 AND p.plan_role = 'mentor'", c.Sub)
        case "lead":
                target := userID
                if target == "" {
                        target = internID
                }
                if target == "" {
                        return planRef{}, apierr.Validation().WithMessage("Укажите user_id стажёра")
                }
                ok, err := h.internInTeam(ctx, target, c.TeamID)
                if err != nil {
                        return planRef{}, err
                }
                if !ok {
                        return planRef{}, apierr.Forbidden()
                }
                if userID != "" {
                        return one("p.user_id = $1 AND p.plan_role = 'intern'", userID)
                }
                return one("p.plan_role = 'mentor' AND p.intern_id = $1", internID)
        default:
                return planRef{}, apierr.Forbidden()
        }
}

        err := row.Scan(&t.ID, &t.ParentTaskID, &t.Source, &t.ChecklistItemID, &t.Period, &t.Title, &t.Description,

// loadTasks returns the top-level tasks of a plan grouped by period, in checklist order, with subtasks nested.
func (h *Handler) loadTasks(ctx context.Context, planID, today string) ([]PeriodTasks, []TopTask, error) {
        rows, err := h.pool.Query(ctx, `SELECT `+taskCols+` FROM core.plan_tasks t WHERE t.plan_id = $1
                ORDER BY (t.source <> 'checklist'), t.position, t.title, t.id`, planID)
        if err != nil {
                return nil, nil, fmt.Errorf("select tasks: %w", err)
        }
        defer rows.Close()
        var tops []TopTask
        subs := map[string][]Task{}
        for rows.Next() {
                t, err := scanTask(rows, today)
                if err != nil {
                        return nil, nil, fmt.Errorf("scan task: %w", err)
                }
                if t.ParentTaskID != nil {
                        subs[*t.ParentTaskID] = append(subs[*t.ParentTaskID], t)
                } else {
                        tops = append(tops, TopTask{Task: t})
                }
        }
        if err := rows.Err(); err != nil {
                return nil, nil, fmt.Errorf("read tasks: %w", err)
        }
        periods := make([]PeriodTasks, 0, len(period.All()))
        idx := map[string]int{}
        for _, p := range period.All() {
                idx[string(p)] = len(periods)
                periods = append(periods, PeriodTasks{ID: string(p), Label: period.Label(p), Tasks: []TopTask{}})
        }
        for i := range tops {
                tops[i].Subtasks = subs[tops[i].ID]
                if tops[i].Subtasks == nil {
                        tops[i].Subtasks = []Task{}
                }
                if j, ok := idx[tops[i].Period]; ok {
                        periods[j].Tasks = append(periods[j].Tasks, tops[i])
                }
        }
        return periods, tops, nil
}

}

// ---- next step (R20) ----

// statusRank: overdue first, then in progress, then not started. waiting_approval and done are never next.
var statusRank = map[string]int{"overdue": 0, "in_progress": 1, "not_started": 2}

var askPrompts = map[string]string{
        "CL-07": "Расскажи, какие документы и мануалы по работе в моей команде мне нужно изучить",
        "CL-08": "Расскажи, что нужно изучить по защите персональных данных и другим инструкциям",
        "CL-09": "Расскажи, как закрывать практику и писать учебные работы по стажировке",
        "CL-10": "Помоги подготовиться к экваторной встрече: что нужно сделать и где взять материалы",
}

// nextStep picks the first not-done checklist task of the current or an earlier period: tasks that people do
// (human_only) come last; within a group overdue > in progress > not started. Among equals a task that nobody has
// started yet goes before one stored as in_progress (work on it is already under way), then period and position.
// Tasks created by actions (source = action) are skipped.
func nextStep(tops []TopTask, cur period.Period) *NextStep {
        var best *TopTask
        bestKey := [5]int{}
        curOrder := period.Order(cur)
        for i := range tops {
                t := &tops[i]
                rank, ok := statusRank[t.Status]
                if !ok || t.Source != "checklist" || period.Order(period.Period(t.Period)) > curOrder {
                        continue
                }
                human := 0
                if t.HumanOnly {
                        human = 1
                }
                started := 0
                if t.stored == "in_progress" {
                        started = 1
                }
                key := [5]int{human, rank, started, period.Order(period.Period(t.Period)), i}
                if best == nil || less(key, bestKey) {
                        best, bestKey = t, key
                }
        }
        if best == nil {
                return nil
        }
        ns := &NextStep{TaskID: best.ID, Title: best.Title}
        item := ""
        if best.ChecklistItemID != nil {
                item = *best.ChecklistItemID
        }
        switch {
        case item == "CL-05" || item == "CL-01":
                ns.Action = "request_access"
        case item == "CL-04":
                ns.Action = "draft_message"
        case askPrompts[item] != "":
                ns.Action, ns.Prompt = "ask_buddy", askPrompts[item]
        case best.HumanOnly:
                ns.Action = "mark_done"
        default:
                ns.Action, ns.Prompt = "ask_buddy", "Помоги выполнить пункт плана: «"+best.Title+"»"
        }
        return ns
}

func less(a, b [5]int) bool {
        for i := range a {
                if a[i] != b[i] {
                        return a[i] < b[i]
                }
        }
        return false
}

func (h *Handler) setStatus(ctx context.Context, c authn.Claims, id, status string) (any, error) {
        tx, err := h.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        var t taskRow
        err = tx.QueryRow(ctx, `SELECT t.plan_id::text, t.source, t.period, t.status, t.parent_task_id::text, t.checklist_item_id,
                        t.due_at::text, p.user_id
                FROM core.plan_tasks t JOIN core.plans p ON p.id = t.plan_id WHERE t.id = $1 FOR UPDATE OF t`, id).
                Scan(&t.planID, &t.source, &t.period, &t.status, &t.parent, &t.item, &t.due, &t.owner)
        if errors.Is(err, pgx.ErrNoRows) || (err == nil && t.owner != c.Sub) {
                return nil, apierr.Forbidden() // unknown and foreign tasks look the same (R71)
        }
        if err != nil {
                return nil, fmt.Errorf("select task: %w", err)
        }
        if t.source != "checklist" || t.status == "waiting_approval" {
                return nil, apierr.Conflict().WithMessage("Статус этой задачи меняется автоматически после решения по заявке")
        }
        if _, err := tx.Exec(ctx, `UPDATE core.plan_tasks SET status = $2, updated_at = now() WHERE id = $1`, id, status); err != nil {
                return nil, fmt.Errorf("update task: %w", err)
        }
        becameDone := status == "done" && t.status != "done"
        if t.parent == nil {
                if becameDone {
                        if _, err := tx.Exec(ctx, `UPDATE core.plan_tasks SET status = 'done', updated_at = now()
                                WHERE parent_task_id = $1 AND status <> 'done'`, id); err != nil {
                                return nil, fmt.Errorf("complete subtasks: %w", err)
                        }
                        if err := h.emitDone(ctx, tx, c, t); err != nil {
                                return nil, err
                        }
                }
        } else if err := h.syncParent(ctx, tx, c, *t.parent, status, t.status); err != nil {
                return nil, err
        }
        var out any
        if t.parent == nil {
                out, err = h.loadTop(ctx, tx, id)
        } else {
                out, err = scanTask(tx.QueryRow(ctx, `SELECT `+taskCols+` FROM core.plan_tasks t WHERE t.id = $1`, id), h.today())
        }
        if err != nil {
                return nil, fmt.Errorf("reload task: %w", err)
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        return out, nil
}

// syncParent completes the parent when all its subtasks are done and reopens a done parent when a subtask is reopened.
func (h *Handler) syncParent(ctx context.Context, tx pgx.Tx, c authn.Claims, parentID, newStatus, oldStatus string) error {
        if newStatus == "done" {
                var all bool
                if err := tx.QueryRow(ctx, `SELECT bool_and(status = 'done') FROM core.plan_tasks WHERE parent_task_id = $1`, parentID).Scan(&all); err != nil {
                        return fmt.Errorf("check subtasks: %w", err)
                }
                if !all {
                        return nil
                }
                var p taskRow
                err := tx.QueryRow(ctx, `SELECT status, period, checklist_item_id, due_at::text FROM core.plan_tasks WHERE id = $1 FOR UPDATE`, parentID).
                        Scan(&p.status, &p.period, &p.item, &p.due)
                if err != nil {
                        return fmt.Errorf("select parent: %w", err)
                }
                if p.status == "done" {
                        return nil
                }
                if _, err := tx.Exec(ctx, `UPDATE core.plan_tasks SET status = 'done', updated_at = now() WHERE id = $1`, parentID); err != nil {
                        return fmt.Errorf("complete parent: %w", err)
                }
                return h.emitDone(ctx, tx, c, p)
        }
        if oldStatus == "done" {
                if _, err := tx.Exec(ctx, `UPDATE core.plan_tasks SET status = 'in_progress', updated_at = now()
                        WHERE id = $1 AND status = 'done'`, parentID); err != nil {
                        return fmt.Errorf("reopen parent: %w", err)
                }
        }
        return nil
}

func (h *Handler) loadTop(ctx context.Context, q pgx.Tx, id string) (TopTask, error) {
        today := h.today()
        top, err := scanTask(q.QueryRow(ctx, `SELECT `+taskCols+` FROM core.plan_tasks t WHERE t.id = $1`, id), today)
        if err != nil {
                return TopTask{}, err
        }
        rows, err := q.Query(ctx, `SELECT `+taskCols+` FROM core.plan_tasks t WHERE t.parent_task_id = $1 ORDER BY t.position, t.id`, id)
        if err != nil {
                return TopTask{}, err
        }
        defer rows.Close()
        out := TopTask{Task: top, Subtasks: []Task{}}
        for rows.Next() {
                s, err := scanTask(rows, today)
                if err != nil {
                        return TopTask{}, err
                }
                out.Subtasks = append(out.Subtasks, s)
        }
        return out, rows.Err()
}

                return nil, fmt.Errorf("begin: %w", err)
        }

        var planID string
        err = tx.QueryRow(ctx, `SELECT id::text FROM core.plans WHERE user_id = $1 AND plan_role = 'intern'`, internID).Scan(&planID)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.NotFound().WithMessage("План не найден")
        }
        if err != nil {
                return nil, fmt.Errorf("select plan: %w", err)
        }

                return nil, fmt.Errorf("delete goals: %w", err)
        }

                err := tx.QueryRow(ctx, `INSERT INTO core.goals (plan_id, title, criteria, created_by, created_at)

                Props: map[string]any{"count": len(out), "by_role": c.Role}}); err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        return out, nil
}

                arg = c.TeamID
        }

        }
        return out, nil
}

// Package ratelimit is a Redis fixed-window rate limiter (used for chat messages per user, R76-R79).
// It fails open: when Redis is unreachable the request is allowed and the error is returned for logging.
package ratelimit

import (
        "context"
        "fmt"
        "time"

        "github.com/redis/go-redis/v9"
)

// Allower is what callers depend on, so tests can fake the limiter.
type Allower interface {
        // Allow counts one hit for key and reports whether it is within limit hits per window.
        // On error the returned bool is true (fail-open).
        Allow(ctx context.Context, key string, limit int, window time.Duration) (bool, error)
}

// Limiter implements Allower on Redis.
type Limiter struct {
        rdb redis.Cmdable
        now func() time.Time
}

// New returns a Limiter. A nil rdb means "no limiting".
func New(rdb redis.Cmdable) *Limiter {
        return &Limiter{rdb: rdb, now: time.Now}
}

// Allow implements Allower. The window is aligned to the Unix epoch (fixed window); a non-positive limit or
// window disables limiting for the call.
func (l *Limiter) Allow(ctx context.Context, key string, limit int, window time.Duration) (bool, error) {
        if l == nil || l.rdb == nil || limit <= 0 || window <= 0 {
                return true, nil
        }
        bucket := l.now().UnixNano() / int64(window)
        rkey := fmt.Sprintf("rl:%s:%d", key, bucket)
        pipe := l.rdb.TxPipeline()
        incr := pipe.Incr(ctx, rkey)
        pipe.Expire(ctx, rkey, window+time.Second)
        if _, err := pipe.Exec(ctx); err != nil {
                return true, fmt.Errorf("rate limit %q: %w", key, err)
        }
        return incr.Val() <= int64(limit), nil
}
