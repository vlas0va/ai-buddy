// Пакет absence реализует мок-сервер MCP «absence»: правила, проверку и заявки на отгул, отпуск, отсутствие и декрет
// (S3; чекпоинт 11.11). Сначала согласует руководитель, затем HR-администратор; пока HR-администратор не согласовал,
// заявку можно перенести на другие даты. Данные синтетические (R68). Личность берётся только из проверенного JWT (R71);
// заявку видит только её автор. create_request записывает заявку и её пункт согласования в одной транзакции и
// идемпотентна по ключу (R66, R79).

// Правила по видам (тестовые правила прототипа): самая длинная заявка в календарных днях и минимальный срок подачи в
// днях.
const (
        maxDays         = 3  // отгул
        maxVacationDays = 28 // отпуск
        vacationNotice  = 14 // отпуск: подаётся не меньше чем за две недели
        maxAbsenceDays  = 5  // отсутствие по личным причинам
        maxDocuments    = 3
        maxDocumentName = 200
)

// typeLabels называют каждый вид в сводках и сообщениях.
var typeLabels = map[string]string{"day_off": "Отгул", "vacation": "Отпуск", "absence": "Отсутствие", "parental_leave": "Декрет"}

// moscow — рабочий часовой пояс для «завтра». Europe/Moscow — UTC+3 без перехода на летнее время с 2014 года;
// фиксированная зона избавляет от зависимости от tzdata в образе контейнера.
var moscow = time.FixedZone("MSK", 3*60*60)

// Register привязывает каждый инструмент контракта absence к srv.
func Register(srv *mcpx.Server, pool *pgxpool.Pool) error {
        return register(srv, pool, time.Now)
}

// register — это Register с подменяемыми часами (для тестов).
func register(srv *mcpx.Server, pool *pgxpool.Pool, now func() time.Time) error {
        if srv == nil || pool == nil || now == nil {
                return errors.New("absence: server, pool and clock are required")
        }
        if srv.Name() != "absence" {
                return fmt.Errorf("absence: cannot register on server %q", srv.Name())
        }
        s := &service{pool: pool, now: now}
        srv.Handle("get_policy", s.getPolicy)
        srv.Handle("list_my_requests", s.listMyRequests)
        srv.Handle("validate_request", s.validateRequest)
        srv.Handle("create_request", s.createRequest)
        srv.Handle("get_request", s.getRequest)
        srv.Handle("reschedule_request", s.rescheduleRequest)
        srv.Handle("cancel_request", s.cancelRequest)
        return nil
}

type service struct {
        pool *pgxpool.Pool
        now  func() time.Time
}

// querier удовлетворяют *pgxpool.Pool и pgx.Tx.
type querier interface {
        Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)
        QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
}

// request — одна заявка на отсутствие в том виде, как её возвращают инструменты.
type request struct {
        ID                  string     `json:"id"`
        Type                string     `json:"type"`
        StartDate           string     `json:"start_date"`
        EndDate             string     `json:"end_date"`
        Comment             string     `json:"comment,omitempty"`
        Status              string     `json:"status"`
        ApproverID          string     `json:"approver_id"`
        ApproverDisplayName string     `json:"approver_display_name,omitempty"`
        CreatedAt           string     `json:"created_at"`
        Documents           []document `json:"documents"`
        Steps               []step     `json:"steps"`
        Replayed            *bool      `json:"replayed,omitempty"`
        // ReplacedID: несогласованная заявка внутри новых дат, которую отменил create_request (отзыв 12.13)
        ReplacedID string `json:"replaced_id,omitempty"`

        employeeID     string
        idempotencyKey string
        rescheduleKey  string
        cancelKey      string
}

// document — метаданные файла, приложенного к заявке; его содержимое в сервис не попадает.
type document struct {
        Name string `json:"name"`
        Mime string `json:"mime,omitempty"`
        Size int64  `json:"size,omitempty"`
}

// step — один этап отгула после его создания (ответы организаторов 10.11): руководитель согласует, HR-администратор
// оформляет согласованный отгул в HR-системе, HR BP получает уведомление.
type step struct {
        Role        string `json:"role"`
        DisplayName string `json:"display_name,omitempty"`
        Status      string `json:"status"`
        At          string `json:"at,omitempty"`
}

const selectRequest = `
SELECT r.id, r.type, r.start_date, r.end_date, coalesce(r.comment, ''), r.status, r.approver_id,
       coalesce(a.display_name, ''), r.created_at, r.employee_id, r.idempotency_key::text,
       r.decided_at, coalesce(ha.display_name, ''), r.registered_at, coalesce(hb.display_name, ''), r.hrbp_notified_at,
       r.documents, r.lead_decided_at, coalesce(r.reschedule_key::text, ''), coalesce(r.cancel_key::text, '')
FROM absence.requests r
LEFT JOIN directory.employees a ON a.id = r.approver_id
LEFT JOIN directory.employees emp ON emp.id = r.employee_id
LEFT JOIN directory.employees ha ON ha.id = coalesce(r.hr_admin_id,
    (SELECT x.id FROM directory.employees x WHERE x.role = 'hr_admin' ORDER BY x.id LIMIT 1))
LEFT JOIN directory.employees hb ON hb.id = coalesce(r.hrbp_id, emp.hrbp_id)`

func scanRequest(row pgx.CollectableRow) (request, error) {
        var (
                r                             request
                start, end                    time.Time
                created                       time.Time
                decided, registered, notified *time.Time
                leadDecided                   *time.Time
                hrAdmin, hrbp                 string
                docs                          []byte
        )
        err := row.Scan(&r.ID, &r.Type, &start, &end, &r.Comment, &r.Status, &r.ApproverID, &r.ApproverDisplayName,
                &created, &r.employeeID, &r.idempotencyKey, &decided, &hrAdmin, &registered, &hrbp, &notified,
                &docs, &leadDecided, &r.rescheduleKey, &r.cancelKey)
        r.StartDate, r.EndDate = start.Format(time.DateOnly), end.Format(time.DateOnly)
        r.CreatedAt = created.UTC().Format(time.RFC3339)
        if json.Unmarshal(docs, &r.Documents) != nil || r.Documents == nil {
                r.Documents = []document{}
        }
        r.Steps = steps(r.Status, r.ApproverDisplayName, hrAdmin, hrbp, leadDecided, decided, registered, notified)
        return r, err
}

// steps выводит этапы заявки из её статуса и отметок времени: решает руководитель (leadDecided), затем решает
// HR-администратор (decided) и оформляет её, затем уведомляется HR BP.
func steps(status, lead, hrAdmin, hrbp string, leadDecided, decided, registered, notified *time.Time) []step {
        at := func(t *time.Time) string {
                if t == nil {
                        return ""
                }
                return t.UTC().Format(time.RFC3339)
        }
        out := []step{{Role: "lead", DisplayName: lead, Status: "waiting"}, {Role: "hr_admin", DisplayName: hrAdmin, Status: "waiting"},
                {Role: "hrbp", DisplayName: hrbp, Status: "waiting"}}
        switch {
        case status == "pending_hr", status == "approved":
                out[0].Status, out[0].At = "approved", at(leadDecided)
                if leadDecided == nil {
                        out[0].At = at(decided)
                }
        case (status == "rejected" || status == "cancelled") && leadDecided != nil:
                out[0].Status, out[0].At = "approved", at(leadDecided)
                out[1].Status, out[1].At = "rejected", at(decided)
                out[2].Status = "skipped"
        case status == "rejected", status == "cancelled":
                out[0].Status, out[0].At = "rejected", at(decided)
                out[1].Status, out[2].Status = "skipped", "skipped"
        }
        if registered != nil {
                out[1].Status, out[1].At = "registered", at(registered)
        }
        if notified != nil {
                out[2].Status, out[2].At = "notified", at(notified)
        }
        return out
}

// findRequest возвращает заявку, подходящую под where (условие на псевдоним r с параметром $1), или ok=false.
func findRequest(ctx context.Context, q querier, where string, arg any) (request, bool, error) {
        rows, err := q.Query(ctx, selectRequest+" WHERE "+where, arg)
        if err != nil {
                return request{}, false, fmt.Errorf("query absence request: %w", err)
        }
        r, err := pgx.CollectOneRow(rows, scanRequest)
        if errors.Is(err, pgx.ErrNoRows) {
                return request{}, false, nil
        }
        if err != nil {
                return request{}, false, fmt.Errorf("scan absence request: %w", err)
        }
        return r, true, nil
}

func forbiddenRequest() *apierr.Error {
        return apierr.Forbidden().WithMessage("Заявка недоступна: можно просматривать только свои заявки")
}

type policy struct {
        DocID      string `json:"doc_id"`
        Title      string `json:"title"`
        Text       string `json:"text"`
        SourceType string `json:"source_type"`
        URL        string `json:"url,omitempty"`

// getPolicy возвращает тестовую инструкцию команды (absence-policy-<team>) или общую (absence-policy). ACL базы знаний
// применяется как в kb.search, поэтому документ, который вызывающему читать нельзя, никогда не возвращается (R71).
func (s *service) getPolicy(ctx context.Context, c mcpx.Caller, _ json.RawMessage) (any, error) {
        var p policy
        err := s.pool.QueryRow(ctx, `
SELECT doc_id, title, body, source_type, coalesce(url, '')
FROM kb.documents
WHERE doc_id IN ($1, 'absence-policy')
  AND owner_id IS NULL
  AND acl_roles @> ARRAY[$2::text]
  AND ('*' = ANY(acl_teams) OR $3 = ANY(acl_teams))
ORDER BY doc_id = $1 DESC
LIMIT 1`, "absence-policy-"+c.TeamID, c.Role, c.TeamID).Scan(&p.DocID, &p.Title, &p.Text, &p.SourceType, &p.URL)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.NotFound().WithMessage("Тестовая инструкция по оформлению отгула не найдена в базе знаний")
        }
        if err != nil {
                return nil, fmt.Errorf("load absence policy: %w", err)
        }
        return p, nil
}

type listOut struct {
        Items []request `json:"items"`
}

func (s *service) listMyRequests(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in struct {
                Status string `json:"status"`
        }
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        rows, err := s.pool.Query(ctx, selectRequest+`
WHERE r.employee_id = $1 AND ($2 = '' OR r.status = $2)
ORDER BY r.start_date DESC, r.id DESC
LIMIT 100`, c.Sub, in.Status)
        if err != nil {
                return nil, fmt.Errorf("list absence requests: %w", err)
        }
        items, err := pgx.CollectRows(rows, scanRequest)
        if err != nil {
                return nil, fmt.Errorf("scan absence requests: %w", err)
        }
        if items == nil {
                items = []request{}
        }
        return listOut{Items: items}, nil
}

func (s *service) getRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in struct {
                ID string `json:"id"`
        }
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        r, ok, err := findRequest(ctx, s.pool, "r.id = $1", in.ID)
        if err != nil {
                return nil, err
        }
        if !ok || r.employeeID != c.Sub {
                return nil, forbiddenRequest() // никогда не раскрывать, существует ли заявка другого пользователя (R71)
        }
        return r, nil
}

type dates struct {
        Type      string     `json:"type"`
        StartDate string     `json:"start_date"`
        EndDate   string     `json:"end_date"`
        Documents []document `json:"documents"`
        ExcludeID string     `json:"exclude_id"`
}

// validate проверяет даты новой заявки для c и находит согласующего (руководителя сотрудника).
func (s *service) validate(ctx context.Context, q querier, c mcpx.Caller, in dates) (validateOut, error) {
        out := validateOut{Problems: []string{}}
        start, errStart := time.Parse(time.DateOnly, in.StartDate)
        end, errEnd := time.Parse(time.DateOnly, in.EndDate)
        if errStart != nil || errEnd != nil {
                return out, apierr.Validation().WithMessage("Даты должны быть в формате ГГГГ-ММ-ДД")
        }
        var leadID, leadName *string
        err := q.QueryRow(ctx, `
SELECT e.lead_id, l.display_name
FROM directory.employees e LEFT JOIN directory.employees l ON l.id = e.lead_id
WHERE e.id = $1`, c.Sub).Scan(&leadID, &leadName)
        if errors.Is(err, pgx.ErrNoRows) {
                return out, apierr.Forbidden().WithMessage("Сотрудник не найден в справочнике")
        }
        if err != nil {
                return out, fmt.Errorf("load employee: %w", err)
        }

        if _, ok := typeLabels[in.Type]; !ok {
                return out, apierr.Validation().WithMessage("Вид заявки: отгул, отпуск, отсутствие или декрет")
        }
        if len(in.Documents) > maxDocuments {
                return out, apierr.Validation().WithMessage(fmt.Sprintf("Можно приложить не больше %d документов", maxDocuments))
        }
        for _, d := range in.Documents {
                if strings.TrimSpace(d.Name) == "" || len([]rune(d.Name)) > maxDocumentName {
                        return out, apierr.Validation().WithMessage("Имя документа: от 1 до 200 символов")
                }
        }
        out.Problems = append(out.Problems, typeProblems(in.Type, start, end, len(in.Documents), s.now())...)
        if !end.Before(start) {
                overlaps, err := overlapProblems(ctx, q, c.Sub, in.StartDate, in.EndDate, in.ExcludeID)
                if err != nil {
                        return out, err
                }
                out.Problems = append(out.Problems, overlaps...)
        }
        if leadID == nil {
                out.Problems = append(out.Problems,
                        "Не удалось определить согласующего: в справочнике не указан руководитель. Обратитесь к HR BP")
        } else {
                out.ApproverID, out.ApproverDisplayName = *leadID, deref(leadName)
        }
        out.Valid = len(out.Problems) == 0
        return out, nil
}

// typeProblems применяет правила каждого вида; у отгула остаются правила по рабочим дням (dateProblems).
func typeProblems(kind string, start, end time.Time, docs int, now time.Time) []string {
        y, m, d := now.In(moscow).Date()
        today := time.Date(y, m, d, 0, 0, 0, 0, time.UTC)
        days := int(end.Sub(start).Hours()/24) + 1
        var problems []string
        switch kind {
        case "day_off":
                return dateProblems(start, end, now)
        case "vacation":
                if earliest := today.AddDate(0, 0, vacationNotice); start.Before(earliest) {
                        problems = append(problems, fmt.Sprintf("Заявку на отпуск подают не позже чем за %d дней: начало не раньше %s",
                                vacationNotice, ru(earliest)))
                }
                if end.Before(start) {
                        return append(problems, "Дата окончания не может быть раньше даты начала")
                }
                if days > maxVacationDays {
                        problems = append(problems, fmt.Sprintf("Отпуск за один раз — не больше %d календарных дней, выбрано %d", maxVacationDays, days))
                }
        case "absence":
                if start.Before(today.AddDate(0, 0, 1)) {
                        problems = append(problems, fmt.Sprintf("Дата начала должна быть не раньше завтрашнего дня (%s)", ru(today.AddDate(0, 0, 1))))
                }
                if end.Before(start) {
                        return append(problems, "Дата окончания не может быть раньше даты начала")
                }
                if days > maxAbsenceDays {
                        problems = append(problems, fmt.Sprintf("Отсутствие оформляется не больше чем на %d дней, выбрано %d", maxAbsenceDays, days))
                }
        case "parental_leave":
                if start.Before(today) {
                        problems = append(problems, "Дата начала уже прошла")
                }
                if end.Before(start) {
                        return append(problems, "Дата окончания не может быть раньше даты начала")
                }
                if docs == 0 {
                        problems = append(problems, "Для декрета нужен подтверждающий документ — приложите заявление")
                }
        }
        return problems
}

// dateProblems применяет правила отгула: начало не раньше завтра (московская дата now), конец не раньше начала, только
// рабочие дни, не больше maxDays дней подряд.
func dateProblems(start, end, now time.Time) []string {
        var problems []string
        y, m, d := now.In(moscow).Date()
        tomorrow := time.Date(y, m, d+1, 0, 0, 0, 0, time.UTC)
        if start.Before(tomorrow) {
                problems = append(problems, fmt.Sprintf("Дата начала должна быть не раньше завтрашнего дня (%s)", ru(tomorrow)))
        }
        if end.Before(start) {
                return append(problems, "Дата окончания не может быть раньше даты начала")
        }
        days := int(end.Sub(start).Hours()/24) + 1
        var weekends []string
        for i := 0; i < days && i <= 31; i++ {
                day := start.AddDate(0, 0, i)
                switch day.Weekday() {
                case time.Saturday:
                        weekends = append(weekends, ru(day)+" (суббота)")
                case time.Sunday:
                        weekends = append(weekends, ru(day)+" (воскресенье)")
                }
        }
        if len(weekends) > 4 {
                weekends = append(weekends[:4], fmt.Sprintf("и ещё %d", len(weekends)-4))
        }
        if len(weekends) > 0 {
                problems = append(problems, "Отгул оформляется только на рабочие дни, выходные: "+strings.Join(weekends, ", "))
        }
        if days > maxDays {
                problems = append(problems, fmt.Sprintf("Отгул можно оформить не больше чем на %d дня подряд, выбрано %d %s",
                        maxDays, days, plural(days, "день", "дня", "дней")))
        }
        return problems
}

// overlapProblems сообщает о несогласованных или согласованных заявках вызывающего, пересекающих [start, end], кроме
// excludeID (переносимой заявки).
func overlapProblems(ctx context.Context, q querier, sub, start, end, excludeID string) ([]string, error) {
        rows, err := q.Query(ctx, `
SELECT id, start_date, end_date, status
FROM absence.requests
WHERE employee_id = $1 AND status IN ('pending_approval', 'pending_hr', 'approved') AND id <> $4
  AND start_date <= $3::date AND end_date >= $2::date
ORDER BY start_date, id`, sub, start, end, excludeID)
        if err != nil {
                return nil, fmt.Errorf("query overlapping requests: %w", err)
        }
        problems, err := pgx.CollectRows(rows, func(row pgx.CollectableRow) (string, error) {
                var (
                        id, status string
                        s, e       time.Time
                )
                err := row.Scan(&id, &s, &e, &status)
                label := "согласована"
                if status != "approved" {
                        label = "на согласовании"
                }
                return fmt.Sprintf("Даты пересекаются с заявкой %s на %s (%s)", id, period(s, e), label), err
        })
        if err != nil {
                return nil, fmt.Errorf("scan overlapping requests: %w", err)
        }
        return problems, nil
}

// --- create_request ---

type createIn struct {
        dates
        Comment        string `json:"comment"`
        IdempotencyKey string `json:"idempotency_key"`
        Replaces       string `json:"replaces"` // несогласованная заявка вызывающего внутри новых дат: отменяется тем же шагом
}

// createRequest повторно проверяет и в одной транзакции вставляет заявку (pending_approval, согласующий = руководитель)
// и её пункт согласования. Тот же idempotency_key возвращает сохранённую заявку с replayed=true (R79).
func (s *service) createRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in createIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        // Сериализуем записи вызывающего: поиск повтора и проверка пересечений должны видеть коммиты друг друга.
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "absence:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        prev, found, err := findRequest(ctx, tx, "r.idempotency_key = $1::uuid", in.IdempotencyKey)
        if err != nil {
                return nil, err
        }
        if found {
                return replay(prev, c, in)
        }
        if in.ExcludeID != "" {
                return nil, apierr.Validation().WithMessage("exclude_id используется только при проверке переноса")
        }
        d := in.dates
        if in.Replaces != "" {
                // отзыв 12.13: отпуск или декрет, накрывающий несогласованный отгул, заменяет его
                old, ok, err := findRequest(ctx, tx, "r.id = $1", in.Replaces)
                if err != nil {
                        return nil, err
                }
                if !ok || old.employeeID != c.Sub {
                        return nil, forbiddenRequest()
                }
                if old.Status != "pending_approval" && old.Status != "pending_hr" {
                        return nil, apierr.Conflict().WithMessage("Заменить можно только заявку, которую HR-администратор ещё не согласовал")
                }
                if old.StartDate < in.StartDate || old.EndDate > in.EndDate {
                        return nil, apierr.Validation().WithMessage("Заменяемая заявка должна целиком попадать в новые даты")
                }
                d.ExcludeID = old.ID
        }
        check, err := s.validate(ctx, tx, c, d)
        if err != nil {
                return nil, err
        }
        if !check.Valid {
                return nil, apierr.Validation().WithMessage("Заявку нельзя создать: " + strings.Join(check.Problems, ". "))
        }
        if in.Replaces != "" {
                if err := cancelInTx(ctx, tx, in.Replaces, ""); err != nil {
                        return nil, err
                }
        }
        docs, _ := json.Marshal(nonNilDocs(in.Documents))
        var id string
        err = tx.QueryRow(ctx, `
INSERT INTO absence.requests (employee_id, type, start_date, end_date, comment, approver_id, idempotency_key, documents)
VALUES ($1, $2, $3::date, $4::date, nullif($5, ''), $6, $7::uuid, $8)
RETURNING id`, c.Sub, in.Type, in.StartDate, in.EndDate, in.Comment, check.ApproverID, in.IdempotencyKey, docs).Scan(&id)
        if isUniqueViolation(err) {
                return nil, keyConflict() // ключ принадлежит заявке другого сотрудника
        }
        if err != nil {
                return nil, fmt.Errorf("insert absence request: %w", err)
        }
        start, _ := time.Parse(time.DateOnly, in.StartDate)
        end, _ := time.Parse(time.DateOnly, in.EndDate)
        if _, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary)
VALUES ('absence_request', $1, $2, $3, $4)`, id, c.Sub, check.ApproverID, typeLabels[in.Type]+" "+period(start, end)); err != nil {
                return nil, fmt.Errorf("insert approval item: %w", err)
        }
        created, _, err := findRequest(ctx, tx, "r.id = $1", id)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        created.Replayed = new(bool)
        created.ReplacedID = in.Replaces
        return created, nil
}

// cancelRequest отзывает заявку вызывающего, пока HR-администратор её не согласовал (отзыв 12.13): заявка становится
// cancelled, а её открытые пункты согласования исчезают из «Согласований». Повтор ключа снова возвращает её.
func (s *service) cancelRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in cancelIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "absence:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        r, ok, err := findRequest(ctx, tx, "r.id = $1", in.ID)
        if err != nil {
                return nil, err
        }
        if !ok || r.employeeID != c.Sub {
                return nil, forbiddenRequest()
        }
        if r.cancelKey == in.IdempotencyKey && r.Status == "cancelled" {
                replayed := true
                r.Replayed = &replayed
                return r, nil
        }
        if r.Status != "pending_approval" && r.Status != "pending_hr" {
                return nil, apierr.Conflict().WithMessage("Отменить можно только заявку, которую HR-администратор ещё не согласовал. " +
                        "Согласованную заявку отменяет HR-администратор — напиши ему.")
        }
        if err := cancelInTx(ctx, tx, r.ID, in.IdempotencyKey); err != nil {
                return nil, err
        }
        out, _, err := findRequest(ctx, tx, "r.id = $1", r.ID)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        out.Replayed = new(bool)
        return out, nil
}

func cancelInTx(ctx context.Context, tx pgx.Tx, id, key string) error {
        if _, err := tx.Exec(ctx, `UPDATE absence.requests SET status = 'cancelled', decided_at = now(),
    cancel_key = coalesce(nullif($2, '')::uuid, cancel_key) WHERE id = $1`, id, key); err != nil {
                if isUniqueViolation(err) {
                        return keyConflict()
                }
                return fmt.Errorf("cancel absence request: %w", err)
        }
        if _, err := tx.Exec(ctx, `UPDATE approvals.items SET status = 'cancelled', decided_at = now()
WHERE kind = 'absence_request' AND ref_id = $1 AND status = 'pending'`, id); err != nil {
                return fmt.Errorf("cancel approval items: %w", err)
        }
        return nil
}

// replay возвращает сохранённую заявку при повторе ключа или conflict, если ключ использовался с другими аргументами.
func replay(prev request, c mcpx.Caller, in createIn) (any, error) {
        if prev.employeeID != c.Sub || prev.Type != in.Type || prev.StartDate != in.StartDate ||
                prev.EndDate != in.EndDate || prev.Comment != in.Comment || len(prev.Documents) != len(in.Documents) {
                return nil, keyConflict()
        }
        replayed := true
        prev.Replayed = &replayed
        return prev, nil
}

// rescheduleRequest переносит заявку вызывающего на другие даты, пока HR-администратор её не согласовал (чекпоинт
// 11.11). Согласование начинается заново: открытые пункты согласования отменяются, руководитель получает новый. Тот же
// ключ возвращает перенесённую заявку с replayed=true (R79).
func (s *service) rescheduleRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in rescheduleIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "absence:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        r, ok, err := findRequest(ctx, tx, "r.id = $1", in.ID)
        if err != nil {
                return nil, err
        }
        if !ok || r.employeeID != c.Sub {
                return nil, forbiddenRequest()
        }
        if r.rescheduleKey == in.IdempotencyKey {
                if r.StartDate != in.StartDate || r.EndDate != in.EndDate {
                        return nil, keyConflict()
                }
                replayed := true
                r.Replayed = &replayed
                return r, nil
        }
        if r.Status != "pending_approval" && r.Status != "pending_hr" {
                return nil, apierr.Conflict().WithMessage("Перенести можно только заявку, которую HR-администратор ещё не согласовал")
        }
        check, err := s.validate(ctx, tx, c, dates{Type: r.Type, StartDate: in.StartDate, EndDate: in.EndDate,
                Documents: r.Documents, ExcludeID: r.ID})
        if err != nil {
                return nil, err
        }
        if !check.Valid {
                return nil, apierr.Validation().WithMessage("Перенести нельзя: " + strings.Join(check.Problems, ". "))
        }
        _, err = tx.Exec(ctx, `
UPDATE absence.requests SET start_date = $2::date, end_date = $3::date, status = 'pending_approval',
    approver_id = $4, lead_decided_at = NULL, decided_at = NULL, reschedule_key = $5::uuid
WHERE id = $1`, r.ID, in.StartDate, in.EndDate, check.ApproverID, in.IdempotencyKey)
        if isUniqueViolation(err) {
                return nil, keyConflict()
        }
        if err != nil {
                return nil, fmt.Errorf("move absence request: %w", err)
        }
        if _, err := tx.Exec(ctx, `UPDATE approvals.items SET status = 'cancelled', decided_at = now()
WHERE kind = 'absence_request' AND ref_id = $1 AND status = 'pending'`, r.ID); err != nil {
                return nil, fmt.Errorf("cancel approval items: %w", err)
        }
        start, _ := time.Parse(time.DateOnly, in.StartDate)
        end, _ := time.Parse(time.DateOnly, in.EndDate)
        if _, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary)
VALUES ('absence_request', $1, $2, $3, $4)`, r.ID, c.Sub, check.ApproverID,
                typeLabels[r.Type]+" "+period(start, end)+" (перенос)"); err != nil {
                return nil, fmt.Errorf("insert approval item: %w", err)
        }
        moved, _, err := findRequest(ctx, tx, "r.id = $1", r.ID)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        moved.Replayed = new(bool)
        return moved, nil
}

// --- форматирование ---

// ru форматирует дату как DD.MM.YYYY.
func ru(t time.Time) string { return t.Format("02.01.2006") }

// period форматирует диапазон дат: одна дата для одного дня, иначе «начало–конец».
func period(start, end time.Time) string {
        if start.Equal(end) {
                return ru(start)
        }
        return ru(start) + "–" + ru(end)
}

// plural выбирает русскую форму множественного числа для n: 1 день, 2 дня, 5 дней.
func plural(n int, one, few, many string) string {
        switch n10, n100 := n%10, n%100; {
        case n10 == 1 && n100 != 11:
                return one
        case n10 >= 2 && n10 <= 4 && (n100 < 12 || n100 > 14):
                return few
        default:
                return many
        }
}

func deref(s *string) string {
        if s == nil {
                return ""
        }
        return *s
}

// querier удовлетворяют pgx.Tx и *pgxpool.Pool.
type querier interface {
        Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)
}

// findRequest возвращает заявку на доступ, подходящую под where (условие на псевдоним q с параметрами $1, $2), или
// ok=false.
func findRequest(ctx context.Context, db querier, where string, args ...any) (requestOut, bool, error) {
        rows, err := db.Query(ctx, `
SELECT q.id, q.resource_id, r.name, q.status, q.approver_id, r.checklist_item_id, q.environment, q.level, q.created_at,
       q.employee_id, coalesce(q.reason, '')
FROM access.requests q JOIN access.resources r ON r.id = q.resource_id
WHERE `+where+`
ORDER BY q.created_at DESC, q.id DESC LIMIT 1`, args...)
        if err != nil {
                return requestOut{}, false, fmt.Errorf("query access request: %w", err)
        }
        out, err := pgx.CollectOneRow(rows, func(row pgx.CollectableRow) (requestOut, error) {
                var (
                        o       requestOut
                        created time.Time
                )
                err := row.Scan(&o.ID, &o.ResourceID, &o.ResourceName, &o.Status, &o.ApproverID, &o.ChecklistItemID, &o.Environment,
                        &o.Level, &created, &o.employeeID, &o.reason)
                o.CreatedAt = created.UTC().Format(time.RFC3339)
                return o, err
        })
        if errors.Is(err, pgx.ErrNoRows) {
                return requestOut{}, false, nil
        }
        if err != nil {
                return requestOut{}, false, fmt.Errorf("scan access request: %w", err)
        }
        return out, true, nil
}

// requestAccess создаёт заявку на доступ для вызывающего. Ресурс вне команды вызывающего запрещён; обычный ресурс
// выдаётся сразу; ресурс с повышенными правами ждёт руководителя вызывающего (пункт согласования). Повтор
// idempotency_key возвращает сохранённую заявку с replayed=true.
func (s *service) requestAccess(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in requestIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        // Сериализуем записи вызывающего, чтобы поиск повтора и проверка дубликатов видели коммиты друг друга.
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "access:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        prev, found, err := findRequest(ctx, tx, "q.idempotency_key = $1::uuid", in.IdempotencyKey)
        if err != nil {
                return nil, err
        }
        if in.Level == "" {
                in.Level = "read"
        }
        if found {
                if prev.employeeID != c.Sub || prev.ResourceID != in.ResourceID || prev.reason != in.Reason || prev.Level != in.Level ||
                        deref(prev.Environment) != in.Environment {
                        return nil, keyConflict()
                }
                prev.Replayed = true
                return prev, nil
        }

        var (
                name         string
                elevated     bool
                environments []string
                levels       []string
        )
        err = tx.QueryRow(ctx, `
SELECT name, elevated, environments, levels FROM access.resources WHERE id = $1 AND (team_id IS NULL OR team_id = $2)`,
                in.ResourceID, c.TeamID).Scan(&name, &elevated, &environments, &levels)
        if errors.Is(err, pgx.ErrNoRows) {
                // Неизвестные ресурсы и ресурсы других команд выглядят одинаково (R71).
                return nil, apierr.Forbidden().WithMessage("Ресурс недоступен для запроса из вашей команды")
        }
        if err != nil {
                return nil, fmt.Errorf("load resource: %w", err)
        }
        if err := checkScope(name, environments, levels, in); err != nil {
                return nil, err
        }
        // чекпоинт 11.11: права на запись и администрирование и всё на prod согласуются; чтение dev/stage выдаётся сразу
        elevated = elevated || in.Level != "read" || in.Environment == "prod"
        last, found, err := findRequest(ctx, tx, "q.employee_id = $1 AND q.resource_id = $2 AND coalesce(q.environment, '') = $3",
                c.Sub, in.ResourceID, in.Environment)
        if err != nil {
                return nil, err
        }
        if found && last.Status == "granted" && last.Level == in.Level {
                return nil, apierr.Conflict().WithMessage(fmt.Sprintf("Доступ к ресурсу «%s» уже выдан (заявка %s)", name, last.ID))
        }
        if found && last.Status == "pending_approval" {
                return nil, apierr.Conflict().WithMessage(
                        fmt.Sprintf("Заявка на доступ к ресурсу «%s» уже на согласовании (%s)", name, last.ID))
        }

        var lead *string
        err = tx.QueryRow(ctx, `SELECT lead_id FROM directory.employees WHERE id = $1`, c.Sub).Scan(&lead)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.Forbidden().WithMessage("Сотрудник не найден в справочнике")
        }
        if err != nil {
                return nil, fmt.Errorf("load employee: %w", err)
        }
        status, approver := "granted", (*string)(nil)
        if elevated {
                if lead == nil {
                        return nil, apierr.Validation().WithMessage(
                                "Не удалось определить согласующего: в справочнике не указан руководитель. Обратитесь к наставнику")
                }
                status, approver = "pending_approval", lead
        }
        var id string
        err = tx.QueryRow(ctx, `
INSERT INTO access.requests (employee_id, resource_id, reason, status, approver_id, idempotency_key, decided_at, environment, level)
VALUES ($1, $2, nullif($3, ''), $4, $5, $6::uuid, CASE WHEN $4 = 'granted' THEN now() END, nullif($7, ''), $8)
RETURNING id`, c.Sub, in.ResourceID, in.Reason, status, approver, in.IdempotencyKey, in.Environment, in.Level).Scan(&id)
        if isUniqueViolation(err) {
                return nil, keyConflict() // ключ принадлежит заявке другого сотрудника
        }
        if err != nil {
                return nil, fmt.Errorf("insert access request: %w", err)
        }
        if elevated {
                if _, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary)
VALUES ('access_request', $1, $2, $3, $4)`, id, c.Sub, *approver, "Доступ: "+name+scopeLabel(in)); err != nil {
                        return nil, fmt.Errorf("insert approval item: %w", err)
                }
        }
        created, _, err := findRequest(ctx, tx, "q.id = $1", id)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        return created, nil
}

// checkScope проверяет окружение и уровень по ресурсу: базе данных нужно окружение, у инструмента его нет; уровень
// должен быть из предлагаемых ресурсом.
func checkScope(name string, environments, levels []string, in requestIn) error {
        switch {
        case len(environments) > 0 && in.Environment == "":
                return apierr.Validation().WithMessage(fmt.Sprintf("Для ресурса «%s» укажите окружение: %v", name, environments))
        case len(environments) == 0 && in.Environment != "":
                return apierr.Validation().WithMessage(fmt.Sprintf("У ресурса «%s» нет окружений", name))
        case in.Environment != "" && !slices.Contains(environments, in.Environment):
                return apierr.Validation().WithMessage(fmt.Sprintf("Окружение %q недоступно для ресурса «%s»", in.Environment, name))
        case !slices.Contains(levels, in.Level):
                return apierr.Validation().WithMessage(fmt.Sprintf("Права «%s» недоступны для ресурса «%s»", levelLabels[in.Level], name))
        }
        return nil
}

// scopeLabel — « (stage, редактирование)» для сводки согласования; пусто для обычного чтения инструмента.
func scopeLabel(in requestIn) string {
        switch {
        case in.Environment != "":
                return fmt.Sprintf(" (%s, %s)", in.Environment, levelLabels[in.Level])
        case in.Level != "read":
                return fmt.Sprintf(" (%s)", levelLabels[in.Level])
        }
        return ""
}

// NewHandler собирает HTTP-обработчик: /healthz плюс /mcp/<server> для всех восьми серверов.
func NewHandler(cfg Config, pool *pgxpool.Pool, log *slog.Logger) (http.Handler, error) {
        if err := cfg.Validate(); err != nil {
                return nil, err
        }
        mux := http.NewServeMux()
        mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, r *http.Request) {
                ctx, cancel := context.WithTimeout(r.Context(), 2*time.Second)
                defer cancel()
                if err := pool.Ping(ctx); err != nil {
                        httpx.WriteJSON(w, http.StatusServiceUnavailable, map[string]string{"status": "degraded", "service": "mcp-mocks"})
                        return
                }
                httpx.WriteJSON(w, http.StatusOK, map[string]string{"status": "ok", "service": "mcp-mocks"})
        })
        mcfg := mcpx.Config{JWTSecret: []byte(cfg.JWTSecret), ConfirmationSecret: []byte(cfg.ConfirmationSecret), Logger: log}
        servers := make([]*mcpx.Server, 0, len(registrars))
        for _, r := range registrars {
                srv, err := mcpx.NewServer(r.name, mcfg)
                if err != nil {
                        return nil, fmt.Errorf("mcp server %s: %w", r.name, err)
                }
                if err := r.register(srv, pool); err != nil {
                        return nil, fmt.Errorf("register %s: %w", r.name, err)
                }
                servers = append(servers, srv)
        }
        if err := mcpx.Mount(mux, servers...); err != nil {
                return nil, err
        }
        return httpx.Chain(httpx.RequestID, httpx.Recover(log), httpx.AccessLog(log))(mux), nil
}

// Run подключается к Postgres, загружает базу знаний из KB_DIR и работает, пока ctx не отменён.
func Run(ctx context.Context, cfg Config, log *slog.Logger) error {
        if err := cfg.Validate(); err != nil {
                return err
        }
        pool, err := pg.Open(ctx, cfg.PostgresURL, pg.Options{})
        if err != nil {
                return err
        }
        defer pool.Close()
        n, err := kb.LoadDir(ctx, pool, cfg.KBDir)
        if err != nil {
                return fmt.Errorf("load knowledge base from %s: %w", cfg.KBDir, err)
        }
        log.Info("knowledge base loaded", "dir", cfg.KBDir, "documents", n)
        h, err := NewHandler(cfg, pool, log)
        if err != nil {
                return err
        }
        return serve(ctx, fmt.Sprintf(":%d", cfg.Port), h, log)
}

func serve(ctx context.Context, addr string, h http.Handler, log *slog.Logger) error {
        srv := &http.Server{Addr: addr, Handler: h, ReadHeaderTimeout: 5 * time.Second}
        errc := make(chan error, 1)
        go func() { errc <- srv.ListenAndServe() }()
        log.Info("listening", "addr", addr)
        select {
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
                defer cancel()
                return srv.Shutdown(shutdownCtx)
        case err := <-errc:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        }
}

// item — один пункт согласования в том виде, как его возвращают инструменты.
type item struct {
        ID                   string `json:"id"`
        Kind                 string `json:"kind"`
        RefID                string `json:"ref_id"`
        RequesterID          string `json:"requester_id"`
        RequesterDisplayName string `json:"requester_display_name,omitempty"`
        Summary              string `json:"summary"`
        Status               string `json:"status"`
        Comment              string `json:"comment,omitempty"`
        Stage                string `json:"stage"`

        Replayed             *bool  `json:"replayed,omitempty"`
        // TeamAbsent: кто ещё из команды заявителя отсутствует в эти дни (отзыв 12.13), только для пунктов об отсутствии
        TeamAbsent []string `json:"team_absent,omitempty"`

        idempotencyKey string

const selectItem = `
SELECT i.id, i.kind, i.ref_id, i.requester_id, coalesce(e.display_name, ''), i.summary, i.status,
       coalesce(i.comment, ''), i.created_at, i.approver_id, coalesce(i.idempotency_key::text, ''), i.stage
FROM approvals.items i
LEFT JOIN directory.employees e ON e.id = i.requester_id`

func scanItem(row pgx.CollectableRow) (item, error) {
        var (
                it      item
                created time.Time
        )
        err := row.Scan(&it.ID, &it.Kind, &it.RefID, &it.RequesterID, &it.RequesterDisplayName, &it.Summary, &it.Status,
                &it.Comment, &created, &it.approverID, &it.idempotencyKey, &it.Stage)
        it.CreatedAt = created.UTC().Format(time.RFC3339)
        return it, err
}

type listOut struct {
        Items []item `json:"items"`
}

// listPending возвращает ожидающие пункты, где согласующий — вызывающий, самые старые сначала.
func (s *service) listPending(ctx context.Context, c mcpx.Caller, _ json.RawMessage) (any, error) {
        rows, err := s.pool.Query(ctx, selectItem+`
WHERE i.approver_id = $1 AND i.status = 'pending'
ORDER BY i.created_at, i.id
LIMIT 200`, c.Sub)
        if err != nil {
                return nil, fmt.Errorf("list pending approvals: %w", err)
        }
        items, err := pgx.CollectRows(rows, scanItem)
        if err != nil {
                return nil, fmt.Errorf("scan pending approvals: %w", err)
        }
        if items == nil {
                items = []item{}
        }
        for i := range items {
                if items[i].Kind != "absence_request" {
                        continue
                }
                absent, err := s.teamAbsent(ctx, items[i].RefID)
                if err != nil {
                        return nil, err
                }
                items[i].TeamAbsent = absent
        }
        return listOut{Items: items}, nil
}

// teamAbsent перечисляет коллег заявителя с несогласованным или согласованным отсутствием, пересекающимся с заявкой:
// руководитель сразу видит, что половины команды не будет (отзыв 12.13).
func (s *service) teamAbsent(ctx context.Context, requestID string) ([]string, error) {
        rows, err := s.pool.Query(ctx, `
SELECT e.display_name, o.type, o.start_date, o.end_date
FROM absence.requests r
JOIN directory.employees me ON me.id = r.employee_id
JOIN absence.requests o ON o.employee_id <> r.employee_id AND o.status IN ('pending_approval', 'pending_hr', 'approved')
     AND o.start_date <= r.end_date AND o.end_date >= r.start_date
JOIN directory.employees e ON e.id = o.employee_id AND e.team_id = me.team_id
WHERE r.id = $1
ORDER BY o.start_date, e.display_name
LIMIT 5`, requestID)
        if err != nil {
                return nil, fmt.Errorf("team absences: %w", err)
        }
        return pgx.CollectRows(rows, func(row pgx.CollectableRow) (string, error) {
                var name, kind string
                var start, end time.Time
                err := row.Scan(&name, &kind, &start, &end)
                label := map[string]string{"day_off": "отгул", "vacation": "отпуск", "absence": "отсутствие", "parental_leave": "декрет"}[kind]
                when := start.Format("02.01")
                if !end.Equal(start) {
                        when += "–" + end.Format("02.01")
                }
                return fmt.Sprintf("%s: %s %s", name, label, when), err
        })
}

type decideIn struct {
        ApprovalID     string `json:"approval_id"`
        Decision       string `json:"decision"`
        Comment        string `json:"comment"`
        IdempotencyKey string `json:"idempotency_key"`
}

// underlying сопоставляет виду пункта и решению таблицу заявок и её новый статус.
var underlying = map[string]struct {
        table    string
        statuses map[string]string
}{
        "absence_request": {"absence.requests", map[string]string{"approved": "approved", "rejected": "rejected"}},
        "access_request":  {"access.requests", map[string]string{"approved": "granted", "rejected": "rejected"}},
        "trip_request":    {"trips.requests", map[string]string{"approved": "approved", "rejected": "rejected"}},
}

// decide согласует или отклоняет один из ожидающих пунктов вызывающего и распространяет решение на заявку.
func (s *service) decide(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in decideIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()

        rows, err := tx.Query(ctx, selectItem+" WHERE i.id = $1 FOR UPDATE OF i", in.ApprovalID)
        if err != nil {
                return nil, fmt.Errorf("query approval item: %w", err)
        }
        it, err := pgx.CollectOneRow(rows, scanItem)
        if errors.Is(err, pgx.ErrNoRows) || (err == nil && it.approverID != c.Sub) {
                // Пункты других команд и неизвестные id выглядят одинаково (R71).
                return nil, apierr.Forbidden().WithMessage("Запрос недоступен: решение принимает только назначенный согласующий")
        }
        if err != nil {
                return nil, fmt.Errorf("scan approval item: %w", err)
        }
        if it.idempotencyKey != "" && it.idempotencyKey == in.IdempotencyKey {
                if it.Status != in.Decision || it.Comment != in.Comment {
                        return nil, keyConflict()
                }
                replayed := true
                it.Replayed = &replayed
                return it, nil
        }
        if it.Status != "pending" {
                return nil, apierr.Conflict().WithMessage("Запрос уже рассмотрен, решение изменить нельзя")
        }

        _, err = tx.Exec(ctx, `
UPDATE approvals.items SET status = $2, comment = nullif($3, ''), idempotency_key = $4::uuid, decided_at = now()
WHERE id = $1`, it.ID, in.Decision, in.Comment, in.IdempotencyKey)
        if isUniqueViolation(err) {
                return nil, keyConflict() // ключ уже использован для другого решения
        }
        if err != nil {
                return nil, fmt.Errorf("update approval item: %w", err)
        }
        if it.Kind == "absence_request" {
                if err := decideAbsence(ctx, tx, it, in.Decision, c.Sub); err != nil {
                        return nil, err
                }
        } else {
                // kind ограничен CHECK-ограничением ключами underlying; имя таблицы никогда не берётся из ввода.
                target := underlying[it.Kind]
                tag, err := tx.Exec(ctx, `UPDATE `+target.table+` SET status = $2, decided_at = now()
WHERE id = $1 AND status = 'pending_approval'`, it.RefID, target.statuses[in.Decision])
                if err != nil {
                        return nil, fmt.Errorf("update %s: %w", target.table, err)
                }
                if tag.RowsAffected() != 1 {
                        return nil, apierr.Conflict().WithMessage("Заявка больше не ожидает согласования")
                }
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        it.Status, it.Comment, it.Replayed = in.Decision, in.Comment, new(bool)
        return it, nil
}

// decideAbsence применяет решение к заявке на отсутствие в два этапа (чекпоинт 11.11): руководитель согласует → заявка
// ждёт HR-администратора (pending_hr), и HR-администратор получает пункт согласования; HR-администратор согласует →
// заявка согласована, оформлена в HR-системе, а HR BP сотрудника получает уведомление. Отказ на любом этапе отклоняет
// заявку.
func decideAbsence(ctx context.Context, tx pgx.Tx, it item, decision, approver string) error {
        var (
                sql  string
                args []any
        )
        switch {
        case it.Stage == "lead" && decision == "approved":
                sql = `UPDATE absence.requests SET status = 'pending_hr', lead_decided_at = now()
WHERE id = $1 AND status = 'pending_approval'`
                args = []any{it.RefID}
        case it.Stage == "lead":
                sql = `UPDATE absence.requests SET status = 'rejected', decided_at = now()
WHERE id = $1 AND status = 'pending_approval'`
                args = []any{it.RefID}
        case decision == "approved":
                sql = `UPDATE absence.requests r SET status = 'approved', decided_at = now(), hr_admin_id = $2,
    registered_at = now(), hrbp_id = (SELECT e.hrbp_id FROM directory.employees e WHERE e.id = r.employee_id),
    hrbp_notified_at = now()
WHERE r.id = $1 AND r.status = 'pending_hr'`
                args = []any{it.RefID, approver}
        default:
                sql = `UPDATE absence.requests SET status = 'rejected', decided_at = now(), hr_admin_id = $2
WHERE id = $1 AND status = 'pending_hr'`
                args = []any{it.RefID, approver}
        }
        tag, err := tx.Exec(ctx, sql, args...)
        if err != nil {
                return fmt.Errorf("update absence request: %w", err)
        }
        if tag.RowsAffected() != 1 {
                return apierr.Conflict().WithMessage("Заявка больше не ожидает согласования")
        }
        if it.Stage == "lead" && decision == "approved" {
                tag, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary, stage)
SELECT 'absence_request', $1::text, $2::text, x.id, $3::text, 'hr_admin'
FROM directory.employees x WHERE x.role = 'hr_admin' ORDER BY x.id LIMIT 1`, it.RefID, it.RequesterID, it.Summary)
                if err != nil {
                        return fmt.Errorf("insert hr admin approval item: %w", err)
                }
                if tag.RowsAffected() != 1 {
                        return apierr.Conflict().WithMessage("В справочнике нет HR-администратора")
                }
        }
        return nil
}

func template(ctx context.Context, pool *pgxpool.Pool, role string) (any, error) {
        rows, err := pool.Query(ctx, `
                SELECT id, role, period, title, coalesce(description, ''), steps, scenarios, human_only
                FROM checklist.items WHERE role IN ($1, 'both') ORDER BY position, id`, role)
        if err != nil {
                return nil, fmt.Errorf("loading checklist items: %w", err)
        }
        defer rows.Close()
        items := []itemOut{}
        for rows.Next() {
                var it itemOut
                var steps []byte
                if err := rows.Scan(&it.ID, &it.Role, &it.Period, &it.Title, &it.Description, &steps, &it.Scenarios, &it.HumanOnly); err != nil {
                        return nil, fmt.Errorf("scanning checklist item: %w", err)
                }
                if len(steps) == 0 || string(steps) == "null" {
                        steps = []byte("[]")
                }
                it.Steps = steps
                if it.Scenarios == nil {
                        it.Scenarios = []string{}
                }
                items = append(items, it)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("reading checklist items: %w", err)
        }
        periods := []periodOut{}
        for _, p := range period.All() {
                periods = append(periods, periodOut{ID: string(p), Label: period.Label(p)})
        }
        return map[string]any{"periods": periods, "items": items}, nil
}

func findSignature(ctx context.Context, q pgx.Tx, where string, args ...any) (signature, bool, error) {
        var (
                g  signature
                at time.Time
        )
        err := q.QueryRow(ctx, `

// sign записывает добровольное согласие вызывающего. Согласия, которые подписываются только в HR-портале, отклоняются;
// уже подписанное согласие — conflict; тот же ключ возвращает сохранённую подпись с replayed=true.
func (s *service) sign(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in signIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "consents:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        prev, found, err := findSignature(ctx, tx, "g.idempotency_key = $1::uuid", in.IdempotencyKey)
        if err != nil {
                return nil, err
        }
        if found {
                if prev.employeeID != c.Sub || prev.ConsentID != in.ConsentID {
                        return nil, keyConflict()
                }
                replayed := true
                prev.Replayed = &replayed
                return prev, nil
        }
        var (
                title    string
                signable bool
        )
        err = tx.QueryRow(ctx, `SELECT title, signable FROM consents.types WHERE id = $1`, in.ConsentID).Scan(&title, &signable)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.NotFound().WithMessage("Такого согласия нет")
        }
        if err != nil {
                return nil, fmt.Errorf("load consent: %w", err)
        }
        if !signable {
                return nil, apierr.Forbidden().WithMessage("«" + title + "» подписывается только в кадровом портале при оформлении")
        }
        if done, ok, err := findSignature(ctx, tx, "g.employee_id = $1 AND g.consent_id = $2", c.Sub, in.ConsentID); err != nil {
                return nil, err
        } else if ok {
                return nil, apierr.Conflict().WithMessage("Согласие «" + title + "» уже подписано (" + done.ID + ")")
        }
        var id string
        err = tx.QueryRow(ctx, `
INSERT INTO consents.signatures (employee_id, consent_id, via, idempotency_key)
VALUES ($1, $2, 'buddy', $3::uuid) RETURNING id`, c.Sub, in.ConsentID, in.IdempotencyKey).Scan(&id)
        if isUniqueViolation(err) {
                return nil, keyConflict()
        }
        if err != nil {
                return nil, fmt.Errorf("insert signature: %w", err)
        }
        out, _, err := findSignature(ctx, tx, "g.id = $1", id)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        out.Replayed = new(bool)
        return out, nil
}

func keyConflict() *apierr.Error {
        return apierr.Conflict().WithMessage("Ключ идемпотентности уже использован, подтвердите действие ещё раз")
}

func isUniqueViolation(err error) bool {
        var pgErr *pgconn.PgError
        return errors.As(err, &pgErr) && pgErr.Code == "23505"
}

// aclRole сопоставляет HR-администратора аудитории HR BP тестовых документов (HR-администратор решает заявки на

// search возвращает документы, которые вызывающему можно видеть: ACL по роли и команде и личные заметки только самого
// вызывающего.
func search(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller, query string, k int) (any, error) {
        rows, err := pool.Query(ctx, `
                SELECT d.doc_id, d.title, d.source_type, coalesce(d.url, ''),
                       ts_headline('russian', d.body, q.q, 'StartSel="", StopSel="", MinWords=12, MaxWords=35, ShortWord=2'),
                       ts_rank(d.tsv, q.q)::float8
                FROM kb.documents d, websearch_to_tsquery('russian', $1) AS q(q)
                WHERE d.tsv @@ q.q
                  AND $2 = ANY(d.acl_roles)
                  AND ('*' = ANY(d.acl_teams) OR $3 = ANY(d.acl_teams))
                  AND (d.owner_id IS NULL OR d.owner_id = $4)
                ORDER BY 6 DESC, d.doc_id
                LIMIT $5`, query, aclRole(c.Role), c.TeamID, c.Sub, k)
        if err != nil {
                return nil, fmt.Errorf("searching knowledge base: %w", err)
        }
        defer rows.Close()
        items := []hit{}
        for rows.Next() {
                var h hit
                if err := rows.Scan(&h.DocID, &h.Title, &h.SourceType, &h.URL, &h.Snippet, &h.Score); err != nil {
                        return nil, fmt.Errorf("scanning search hit: %w", err)
                }
                h.Snippet = truncateRunes(h.Snippet, maxSnippet)
                items = append(items, h)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("reading search hits: %w", err)
        }
        return map[string]any{"items": items}, nil
}

// get возвращает один документ, который вызывающему можно видеть (тот же ACL, что в search); ответ наставника,
// добавленный в базу знаний, находится здесь, поэтому ссылка на источник под ответом его открывает (отзыв 12.11).
// Недоступный и отсутствующий выглядят одинаково: not_found.
func get(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller, docID string) (any, error) {
        if docID == "" {
                return nil, apierr.Validation().WithMessage("Укажите doc_id")
        }
        var d struct {
                DocID      string `json:"doc_id"`
                Title      string `json:"title"`
                Body       string `json:"body"`
                SourceType string `json:"source_type"`
                Kind       string `json:"kind"`
        }
        err := pool.QueryRow(ctx, `SELECT doc_id, title, body, source_type, kind FROM kb.documents d
                WHERE d.doc_id = $1 AND $2 = ANY(d.acl_roles) AND ('*' = ANY(d.acl_teams) OR $3 = ANY(d.acl_teams))
                  AND (d.owner_id IS NULL OR d.owner_id = $4)`, docID, aclRole(c.Role), c.TeamID, c.Sub).
                Scan(&d.DocID, &d.Title, &d.Body, &d.SourceType, &d.Kind)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.NotFound().WithMessage("Документ не найден или недоступен")
        }
        if err != nil {
                return nil, fmt.Errorf("reading document: %w", err)
        }
        return d, nil
}

func truncateRunes(s string, n int) string {
        r := []rune(s)
        if len(r) <= n {
                return s
        }
        return string(r[:n-1]) + "…"
}

        DocID      string `json:"doc_id"`

// addFAQ сохраняет ответ наставника как документ FAQ: для команды наставника (scope "team", по умолчанию) или, если
// ответ общий, для всех команд (scope "all", отзыв 12.11). Идемпотентно по ключу (R79). В id документа в любом случае
// остаётся команда наставника.
func addFAQ(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller, title, body, key, scope string) (any, error) {
        teams := []string{c.TeamID}
        if scope == "all" {
                teams = []string{"*"}
        }
        suffix := make([]byte, 4)
        if _, err := rand.Read(suffix); err != nil {
                return nil, fmt.Errorf("generating doc id: %w", err)
        }
        docID := fmt.Sprintf("faq-%s-%s", c.TeamID, hex.EncodeToString(suffix))
        var out faqOut
        err := pool.QueryRow(ctx, `
                INSERT INTO kb.documents (doc_id, title, body, source_type, acl_roles, acl_teams, kind, created_by, idempotency_key)
                VALUES ($1, $2, $3, 'test', ARRAY['intern','mentor','lead'], $4::text[], 'faq', $5, $6)
                ON CONFLICT (idempotency_key) DO NOTHING
                RETURNING doc_id, title, source_type`, docID, title, body, teams, c.Sub, key).
                Scan(&out.DocID, &out.Title, &out.SourceType)
        if err == nil {
                return out, nil
        }
        if !errors.Is(err, pgx.ErrNoRows) {
                return nil, fmt.Errorf("inserting faq: %w", err)
        }
        // Ключ уже использовался: воспроизводим сохранённый результат, если это тот же запрос, иначе conflict.
        var createdBy, storedBody string
        err = pool.QueryRow(ctx, `SELECT doc_id, title, body, coalesce(created_by, '') FROM kb.documents WHERE idempotency_key = $1`, key).
                Scan(&out.DocID, &out.Title, &storedBody, &createdBy)
        if err != nil {
                return nil, fmt.Errorf("loading stored faq: %w", err)
        }
        if createdBy != c.Sub || out.Title != title || storedBody != body {
                return nil, apierr.Conflict().WithMessage("Этот ключ идемпотентности уже использован с другими данными")
        }
        out.SourceType = "test"
        out.Replayed = true
        return out, nil
}

        Source string `yaml:"source_type"`
        ACL    struct {
                Roles []string `yaml:"roles"`
                Teams []string `yaml:"teams"`
        } `yaml:"acl"`
        ChecklistItemID string    `yaml:"checklist_item_id"`
        URL             string    `yaml:"url"`
        Updated         yaml.Node `yaml:"updated"` // любой скаляр; используется только дата YYYY-MM-DD
}

type document struct {
        frontmatter
        body    string
        updated *time.Time
}

// LoadDir вставляет или обновляет каждый файл *.md из dir (кроме README.md) в kb.documents (kind doc, без владельца) и
// возвращает, сколько загружено. Файлам нужен YAML-frontmatter с doc_id, title, source_type: test (R69) и acl {roles,
// teams}; checklist_item_id необязателен; другие поля frontmatter (checklist, tags, updated, origin, ...) игнорируются,
// только корректный `updated` в формате YYYY-MM-DD становится датой документа. Любой некорректный файл валит всю
// загрузку с ошибкой, в которой он назван, и ничего не записывается. Другие файлы игнорируются. Отсутствующий dir
// возвращает 0 без ошибки.
func LoadDir(ctx context.Context, pool *pgxpool.Pool, dir string) (int, error) {
        entries, err := os.ReadDir(dir)
        if errors.Is(err, os.ErrNotExist) {
                return 0, nil
        }
        if err != nil {
                return 0, fmt.Errorf("reading kb dir %s: %w", dir, err)
        }
        var docs []document
        seen := map[string]string{}
        for _, e := range entries {
                if e.IsDir() || !strings.EqualFold(filepath.Ext(e.Name()), ".md") || strings.EqualFold(e.Name(), "README.md") {
                        continue
                }
                raw, err := os.ReadFile(filepath.Join(dir, e.Name()))
                if err != nil {
                        return 0, fmt.Errorf("reading %s: %w", e.Name(), err)
                }
                d, err := parseDoc(raw)
                if err != nil {
                        return 0, fmt.Errorf("kb file %s: %w", e.Name(), err)
                }
                if prev, dup := seen[d.DocID]; dup {
                        return 0, fmt.Errorf("kb file %s: doc_id %q already used by %s", e.Name(), d.DocID, prev)
                }
                seen[d.DocID] = e.Name()
                docs = append(docs, d)
        }
        if len(docs) == 0 {
                return 0, nil
        }
        tx, err := pool.Begin(ctx)
        if err != nil {
                return 0, fmt.Errorf("starting kb load: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        for _, d := range docs {
                if err := upsert(ctx, tx, d); err != nil {
                        return 0, fmt.Errorf("kb file %s: %w", seen[d.DocID], err)
                }
        }
        if err := tx.Commit(ctx); err != nil {
                return 0, fmt.Errorf("committing kb load: %w", err)
        }
        return len(docs), nil
}

func upsert(ctx context.Context, tx pgx.Tx, d document) error {
        tag, err := tx.Exec(ctx, `
                INSERT INTO kb.documents (doc_id, title, body, source_type, acl_roles, acl_teams, kind, checklist_item_id, url, updated)
                VALUES ($1, $2, $3, 'test', $4, $5, 'doc', nullif($6, ''), nullif($7, ''), coalesce($8::date, current_date))
                ON CONFLICT (doc_id) DO UPDATE SET title = EXCLUDED.title, body = EXCLUDED.body,
                        acl_roles = EXCLUDED.acl_roles, acl_teams = EXCLUDED.acl_teams,
                        checklist_item_id = EXCLUDED.checklist_item_id, url = EXCLUDED.url, updated = EXCLUDED.updated
                WHERE kb.documents.kind = 'doc' AND kb.documents.owner_id IS NULL`,
                d.DocID, d.Title, d.body, d.ACL.Roles, d.ACL.Teams, d.ChecklistItemID, d.URL, d.updated)
        if err != nil {
                return fmt.Errorf("upserting %s: %w", d.DocID, err)
        }
        if tag.RowsAffected() == 0 {
                return fmt.Errorf("doc_id %q belongs to a non-document record", d.DocID)
        }
        return nil
}

func parseDoc(raw []byte) (document, error) {
        var d document
        raw = bytes.ReplaceAll(bytes.TrimPrefix(raw, []byte("\xef\xbb\xbf")), []byte("\r\n"), []byte("\n"))
        if !bytes.HasPrefix(raw, []byte("---\n")) {
                return d, errors.New("no YAML frontmatter (file must start with ---)")
        }
        rest := raw[4:]
        end := bytes.Index(rest, []byte("\n---"))
        if end < 0 {
                return d, errors.New("frontmatter is not closed with ---")
        }
        if err := yaml.Unmarshal(rest[:end], &d.frontmatter); err != nil {
                return d, fmt.Errorf("invalid frontmatter: %w", err)
        }
        d.body = strings.TrimSpace(string(rest[end+4:]))
        var missing []string
        for name, empty := range map[string]bool{
                "doc_id": d.DocID == "", "title": strings.TrimSpace(d.Title) == "", "source_type": d.Source == "",
                "acl.roles": len(d.ACL.Roles) == 0, "acl.teams": len(d.ACL.Teams) == 0, "body": d.body == "",
        } {
                if empty {
                        missing = append(missing, name)
                }
        }
        if len(missing) > 0 {
                slices.Sort(missing)
                return d, fmt.Errorf("missing required fields: %s", strings.Join(missing, ", "))
        }
        if d.Source != "test" {
                return d, fmt.Errorf("source_type must be \"test\", got %q (R69)", d.Source)
        }
        if !docIDRe.MatchString(d.DocID) {
                return d, fmt.Errorf("invalid doc_id %q", d.DocID)
        }
        for _, r := range d.ACL.Roles {
                if !slices.Contains(known, r) {
                        return d, fmt.Errorf("unknown role %q in acl.roles", r)
                }
        }
        if d.ChecklistItemID != "" && !itemIDRe.MatchString(d.ChecklistItemID) {
                return d, fmt.Errorf("invalid checklist_item_id %q", d.ChecklistItemID)
        }
        if d.Updated.Kind == yaml.ScalarNode {
                if t, err := time.Parse("2006-01-02", strings.TrimSpace(d.Updated.Value)); err == nil {
                        d.updated = &t
                }
        }
        return d, nil
}

func register(srv *mcpx.Server, pool *pgxpool.Pool, now func() time.Time) error {
        if srv == nil || pool == nil || now == nil {
                return errors.New("trips: server, pool and clock are required")
        }
        if srv.Name() != "trips" {
                return fmt.Errorf("trips: cannot register on server %q", srv.Name())
        }
        s := &service{pool: pool, now: now}
        srv.Handle("get_policy", s.getPolicy)
        srv.Handle("list_my_requests", s.listMyRequests)
        srv.Handle("validate_request", s.validateRequest)
        srv.Handle("create_request", s.createRequest)
        return nil
}

type querier interface {

type request struct {

const selectRequest = `
SELECT r.id, r.city, r.start_date, r.end_date, r.purpose, r.status, r.approver_id, coalesce(a.display_name, ''),
       r.created_at, r.employee_id
FROM trips.requests r
LEFT JOIN directory.employees a ON a.id = r.approver_id`

func scanRequest(row pgx.CollectableRow) (request, error) {
        var (
                r          request
                start, end time.Time
                created    time.Time
        )
        err := row.Scan(&r.ID, &r.City, &start, &end, &r.Purpose, &r.Status, &r.ApproverID, &r.ApproverDisplayName,
                &created, &r.employeeID)
        r.StartDate, r.EndDate = start.Format(time.DateOnly), end.Format(time.DateOnly)
        r.CreatedAt = created.UTC().Format(time.RFC3339)
        return r, err
}

func findRequest(ctx context.Context, q querier, where string, arg any) (request, bool, error) {
        rows, err := q.Query(ctx, selectRequest+" WHERE "+where, arg)
        if err != nil {
                return request{}, false, fmt.Errorf("query trip request: %w", err)
        }
        r, err := pgx.CollectOneRow(rows, scanRequest)
        if errors.Is(err, pgx.ErrNoRows) {
                return request{}, false, nil
        }
        if err != nil {
                return request{}, false, fmt.Errorf("scan trip request: %w", err)
        }
        return r, true, nil
}

// validate проверяет новую командировку c: известный офис не в родном городе сотрудника, даты, срок подачи,
// длительность, без пересечения с другой командировкой; согласующий — руководитель сотрудника.
func (s *service) validate(ctx context.Context, q querier, c mcpx.Caller, in tripIn) (validateOut, error) {
        out := validateOut{Problems: []string{}}
        start, errStart := time.Parse(time.DateOnly, in.StartDate)
        end, errEnd := time.Parse(time.DateOnly, in.EndDate)
        if errStart != nil || errEnd != nil {
                return out, apierr.Validation().WithMessage("Даты должны быть в формате ГГГГ-ММ-ДД")
        }
        var leadID, leadName, home *string
        err := q.QueryRow(ctx, `
SELECT e.lead_id, l.display_name, e.location
FROM directory.employees e LEFT JOIN directory.employees l ON l.id = e.lead_id
WHERE e.id = $1`, c.Sub).Scan(&leadID, &leadName, &home)
        if errors.Is(err, pgx.ErrNoRows) {
                return out, apierr.Forbidden().WithMessage("Сотрудник не найден в справочнике")
        }
        if err != nil {
                return out, fmt.Errorf("load employee: %w", err)
        }
        city := strings.TrimSpace(in.City)
        switch {
        case !slices.Contains(Offices, city):
                out.Problems = append(out.Problems, "Офиса в этом городе нет. Доступно: "+strings.Join(Offices, ", "))
        case home != nil && *home == city:
                out.Problems = append(out.Problems, "Это ваш город: командировка оформляется только в другой офис")
        }
        y, m, d := s.now().In(moscow).Date()
        earliest := time.Date(y, m, d+minNoticeDays, 0, 0, 0, 0, time.UTC)
        if start.Before(earliest) {
                out.Problems = append(out.Problems, fmt.Sprintf("Заявку подают не позже чем за %d дней: начало не раньше %s",
                        minNoticeDays, earliest.Format("02.01.2006")))
        }
        if end.Before(start) {
                out.Problems = append(out.Problems, "Дата окончания не может быть раньше даты начала")
        } else {
                if days := int(end.Sub(start).Hours()/24) + 1; days > maxDays {
                        out.Problems = append(out.Problems, fmt.Sprintf("Командировка длится не больше %d дней, у вас %d", maxDays, days))
                }
                var other string
                err := q.QueryRow(ctx, `
SELECT id FROM trips.requests
WHERE employee_id = $1 AND status IN ('pending_approval', 'approved') AND start_date <= $3::date AND end_date >= $2::date
ORDER BY start_date LIMIT 1`, c.Sub, in.StartDate, in.EndDate).Scan(&other)
                if err == nil {
                        out.Problems = append(out.Problems, "Даты пересекаются с другой командировкой ("+other+")")
                } else if !errors.Is(err, pgx.ErrNoRows) {
                        return out, fmt.Errorf("check overlap: %w", err)
                }
        }
        if strings.TrimSpace(in.Purpose) == "" {
                out.Problems = append(out.Problems, "Укажите цель командировки")
        }
        if leadID == nil {
                out.Problems = append(out.Problems, "Не удалось определить согласующего: в справочнике не указан руководитель")
        } else {
                out.ApproverID, out.ApproverDisplayName = *leadID, deref(leadName)
        }
        out.Valid = len(out.Problems) == 0
        return out, nil
}

type createIn struct {
        tripIn
        IdempotencyKey string `json:"idempotency_key"`
}

// createRequest повторно проверяет и в одной транзакции вставляет заявку (pending_approval, согласующий = руководитель)
// и её пункт согласования. Тот же idempotency_key возвращает сохранённую заявку с replayed=true (R79).
func (s *service) createRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in createIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "trips:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        prev, found, err := findRequest(ctx, tx, "r.idempotency_key = $1::uuid", in.IdempotencyKey)
        if err != nil {
                return nil, err
        }
        if found {
                if prev.employeeID != c.Sub || prev.City != strings.TrimSpace(in.City) || prev.StartDate != in.StartDate ||
                        prev.EndDate != in.EndDate || prev.Purpose != strings.TrimSpace(in.Purpose) {
                        return nil, keyConflict()
                }
                replayed := true
                prev.Replayed = &replayed
                return prev, nil
        }
        check, err := s.validate(ctx, tx, c, in.tripIn)
        if err != nil {
                return nil, err
        }
        if !check.Valid {
                return nil, apierr.Validation().WithMessage("Заявку нельзя создать: " + strings.Join(check.Problems, "; "))
        }
        var id string
        err = tx.QueryRow(ctx, `
INSERT INTO trips.requests (employee_id, city, start_date, end_date, purpose, approver_id, idempotency_key)
VALUES ($1, $2, $3::date, $4::date, $5, $6, $7::uuid)
RETURNING id`, c.Sub, strings.TrimSpace(in.City), in.StartDate, in.EndDate, strings.TrimSpace(in.Purpose),
                check.ApproverID, in.IdempotencyKey).Scan(&id)
        if isUniqueViolation(err) {
                return nil, keyConflict()
        }
        if err != nil {
                return nil, fmt.Errorf("insert trip request: %w", err)
        }
        start, _ := time.Parse(time.DateOnly, in.StartDate)
        end, _ := time.Parse(time.DateOnly, in.EndDate)
        summary := fmt.Sprintf("Командировка: %s, %s", strings.TrimSpace(in.City), period(start, end))
        if _, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary)
VALUES ('trip_request', $1, $2, $3, $4)`, id, c.Sub, check.ApproverID, summary); err != nil {
                return nil, fmt.Errorf("insert approval item: %w", err)
        }
        created, _, err := findRequest(ctx, tx, "r.id = $1", id)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        created.Replayed = new(bool)
        return created, nil
}

func keyConflict() *apierr.Error {
        return apierr.Conflict().WithMessage("Ключ идемпотентности уже использован для другой заявки, подтвердите действие ещё раз")
}

func isUniqueViolation(err error) bool {
        var pgErr *pgconn.PgError
        return errors.As(err, &pgErr) && pgErr.Code == "23505"
}

func period(start, end time.Time) string {
        if start.Equal(end) {
                return start.Format("02.01.2006")
        }
        return start.Format("02.01.2006") + "–" + end.Format("02.01.2006")
}

func deref(s *string) string {
        if s == nil {
                return ""
        }
        return *s
}
