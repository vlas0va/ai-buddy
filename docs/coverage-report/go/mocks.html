// maxDays is the longest day off in consecutive calendar days.
const maxDays = 3

// moscow is the business time zone for "tomorrow". Europe/Moscow is UTC+3 without DST since 2014; a fixed zone
// avoids depending on tzdata in the container image.
var moscow = time.FixedZone("MSK", 3*60*60)

// Register binds every tool of the absence contract to srv.
func Register(srv *mcpx.Server, pool *pgxpool.Pool) error {
        return register(srv, pool, time.Now)
}

// register is Register with an injectable clock (tests).
func register(srv *mcpx.Server, pool *pgxpool.Pool, now func() time.Time) error {
        if srv == nil || pool == nil || now == nil {
                return errors.New("absence: server, pool and clock are required")
        }
        if srv.Name() != "absence" {
                return fmt.Errorf("absence: cannot register on server %q", srv.Name())
        }
        s := &service{pool: pool, now: now}
        srv.Handle("get_policy", s.getPolicy)
        srv.Handle("list_my_requests", s.listMyRequests)
        srv.Handle("validate_request", s.validateRequest)
        srv.Handle("create_request", s.createRequest)
        srv.Handle("get_request", s.getRequest)
        return nil
}

type service struct {
        pool *pgxpool.Pool
        now  func() time.Time
}

// querier is satisfied by *pgxpool.Pool and pgx.Tx.
type querier interface {
        Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)
        QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
}

// request is one absence request as returned by the tools.
type request struct {
        ID                  string `json:"id"`
        Type                string `json:"type"`
        StartDate           string `json:"start_date"`
        EndDate             string `json:"end_date"`
        Comment             string `json:"comment,omitempty"`
        Status              string `json:"status"`
        ApproverID          string `json:"approver_id"`
        ApproverDisplayName string `json:"approver_display_name,omitempty"`
        CreatedAt           string `json:"created_at"`
        Replayed            *bool  `json:"replayed,omitempty"`

        employeeID     string
        idempotencyKey string
}

const selectRequest = `
SELECT r.id, r.type, r.start_date, r.end_date, coalesce(r.comment, ''), r.status, r.approver_id,
       coalesce(a.display_name, ''), r.created_at, r.employee_id, r.idempotency_key::text
FROM absence.requests r
LEFT JOIN directory.employees a ON a.id = r.approver_id`

func scanRequest(row pgx.CollectableRow) (request, error) {
        var (
                r          request
                start, end time.Time
                created    time.Time
        )
        err := row.Scan(&r.ID, &r.Type, &start, &end, &r.Comment, &r.Status, &r.ApproverID, &r.ApproverDisplayName,
                &created, &r.employeeID, &r.idempotencyKey)
        r.StartDate, r.EndDate = start.Format(time.DateOnly), end.Format(time.DateOnly)
        r.CreatedAt = created.UTC().Format(time.RFC3339)
        return r, err
}

// findRequest returns the request matching where (a condition on alias r with parameter $1), or ok=false.
func findRequest(ctx context.Context, q querier, where string, arg any) (request, bool, error) {
        rows, err := q.Query(ctx, selectRequest+" WHERE "+where, arg)
        if err != nil {
                return request{}, false, fmt.Errorf("query absence request: %w", err)
        }
        r, err := pgx.CollectOneRow(rows, scanRequest)
        if errors.Is(err, pgx.ErrNoRows) {
                return request{}, false, nil
        }
        if err != nil {
                return request{}, false, fmt.Errorf("scan absence request: %w", err)
        }
        return r, true, nil
}

func forbiddenRequest() *apierr.Error {
        return apierr.Forbidden().WithMessage("Заявка недоступна: можно просматривать только свои заявки")
}

// getPolicy returns the team's test instruction (absence-policy-<team>) or the common one (absence-policy).
// The KB ACL is applied as in kb.search, so a document the caller may not read is never returned (R71).
func (s *service) getPolicy(ctx context.Context, c mcpx.Caller, _ json.RawMessage) (any, error) {
        var p policy
        err := s.pool.QueryRow(ctx, `
SELECT doc_id, title, body, source_type, coalesce(url, '')
FROM kb.documents
WHERE doc_id IN ($1, 'absence-policy')
  AND owner_id IS NULL
  AND acl_roles @> ARRAY[$2::text]
  AND ('*' = ANY(acl_teams) OR $3 = ANY(acl_teams))
ORDER BY doc_id = $1 DESC
LIMIT 1`, "absence-policy-"+c.TeamID, c.Role, c.TeamID).Scan(&p.DocID, &p.Title, &p.Text, &p.SourceType, &p.URL)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.NotFound().WithMessage("Тестовая инструкция по оформлению отгула не найдена в базе знаний")
        }
        if err != nil {
                return nil, fmt.Errorf("load absence policy: %w", err)
        }
        return p, nil
}

func (s *service) getRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in struct {

        Type      string `json:"type"`

// validate checks the dates of a new request for c and finds the approver (the employee's lead).
func (s *service) validate(ctx context.Context, q querier, c mcpx.Caller, in dates) (validateOut, error) {
        out := validateOut{Problems: []string{}}
        start, errStart := time.Parse(time.DateOnly, in.StartDate)
        end, errEnd := time.Parse(time.DateOnly, in.EndDate)
        if errStart != nil || errEnd != nil {
                return out, apierr.Validation().WithMessage("Даты должны быть в формате ГГГГ-ММ-ДД")
        }
        var leadID, leadName *string
        err := q.QueryRow(ctx, `
SELECT e.lead_id, l.display_name
FROM directory.employees e LEFT JOIN directory.employees l ON l.id = e.lead_id
WHERE e.id = $1`, c.Sub).Scan(&leadID, &leadName)
        if errors.Is(err, pgx.ErrNoRows) {
                return out, apierr.Forbidden().WithMessage("Сотрудник не найден в справочнике")
        }
        if err != nil {
                return out, fmt.Errorf("load employee: %w", err)
        }

        out.Problems = append(out.Problems, dateProblems(start, end, s.now())...)
        if !end.Before(start) {
                overlaps, err := overlapProblems(ctx, q, c.Sub, in.StartDate, in.EndDate)
                if err != nil {
                        return out, err
                }
                out.Problems = append(out.Problems, overlaps...)
        }
        if leadID == nil {
                out.Problems = append(out.Problems,
                        "Не удалось определить согласующего: в справочнике не указан руководитель. Обратитесь к HR BP")
        } else {
                out.ApproverID, out.ApproverDisplayName = *leadID, deref(leadName)
        }
        out.Valid = len(out.Problems) == 0
        return out, nil
}

// dateProblems applies the calendar rules: start not before tomorrow (Moscow date of now), end not before start,
// working days only, at most maxDays consecutive days.
func dateProblems(start, end, now time.Time) []string {
        var problems []string
        y, m, d := now.In(moscow).Date()
        tomorrow := time.Date(y, m, d+1, 0, 0, 0, 0, time.UTC)
        if start.Before(tomorrow) {
                problems = append(problems, fmt.Sprintf("Дата начала должна быть не раньше завтрашнего дня (%s)", ru(tomorrow)))
        }
        if end.Before(start) {
                return append(problems, "Дата окончания не может быть раньше даты начала")
        }
        days := int(end.Sub(start).Hours()/24) + 1
        var weekends []string
        for i := 0; i < days && i <= 31; i++ {
                day := start.AddDate(0, 0, i)
                switch day.Weekday() {
                case time.Saturday:
                        weekends = append(weekends, ru(day)+" (суббота)")
                case time.Sunday:
                        weekends = append(weekends, ru(day)+" (воскресенье)")
                }
        }
        if len(weekends) > 4 {
                weekends = append(weekends[:4], fmt.Sprintf("и ещё %d", len(weekends)-4))
        }
        if len(weekends) > 0 {
                problems = append(problems, "Отгул оформляется только на рабочие дни, выходные: "+strings.Join(weekends, ", "))
        }
        if days > maxDays {
                problems = append(problems, fmt.Sprintf("Отгул можно оформить не больше чем на %d дня подряд, выбрано %d %s",
                        maxDays, days, plural(days, "день", "дня", "дней")))
        }
        return problems
}

// overlapProblems reports the caller's pending or approved requests that intersect [start, end].
func overlapProblems(ctx context.Context, q querier, sub, start, end string) ([]string, error) {
        rows, err := q.Query(ctx, `
SELECT id, start_date, end_date, status
FROM absence.requests
WHERE employee_id = $1 AND status IN ('pending_approval', 'approved')
  AND start_date <= $3::date AND end_date >= $2::date
ORDER BY start_date, id`, sub, start, end)
        if err != nil {
                return nil, fmt.Errorf("query overlapping requests: %w", err)
        }
        problems, err := pgx.CollectRows(rows, func(row pgx.CollectableRow) (string, error) {
                var (
                        id, status string
                        s, e       time.Time
                )
                err := row.Scan(&id, &s, &e, &status)
                label := "согласована"
                if status == "pending_approval" {
                        label = "на согласовании"
                }
                return fmt.Sprintf("Даты пересекаются с заявкой %s на %s (%s)", id, period(s, e), label), err
        })
        if err != nil {
                return nil, fmt.Errorf("scan overlapping requests: %w", err)
        }
        return problems, nil
}

// --- create_request ---

type createIn struct {
        dates
        Comment        string `json:"comment"`
        IdempotencyKey string `json:"idempotency_key"`
}

// createRequest re-validates and, in one transaction, inserts the request (pending_approval, approver = lead)
// and its approval item. The same idempotency_key returns the stored request with replayed=true (R79).
func (s *service) createRequest(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in createIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        // Serialise the caller's writes: the replay lookup and the overlap check must see each other's commits.
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "absence:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        prev, found, err := findRequest(ctx, tx, "r.idempotency_key = $1::uuid", in.IdempotencyKey)
        if err != nil {
                return nil, err
        }
        if found {
                return replay(prev, c, in)
        }
        check, err := s.validate(ctx, tx, c, in.dates)
        if err != nil {
                return nil, err
        }
        if !check.Valid {
                return nil, apierr.Validation().WithMessage("Заявку нельзя создать: " + strings.Join(check.Problems, "; "))
        }
        var id string
        err = tx.QueryRow(ctx, `
INSERT INTO absence.requests (employee_id, type, start_date, end_date, comment, approver_id, idempotency_key)
VALUES ($1, $2, $3::date, $4::date, nullif($5, ''), $6, $7::uuid)
RETURNING id`, c.Sub, in.Type, in.StartDate, in.EndDate, in.Comment, check.ApproverID, in.IdempotencyKey).Scan(&id)
        if isUniqueViolation(err) {
                return nil, keyConflict() // the key belongs to another employee's request
        }
        if err != nil {
                return nil, fmt.Errorf("insert absence request: %w", err)
        }
        start, _ := time.Parse(time.DateOnly, in.StartDate)
        end, _ := time.Parse(time.DateOnly, in.EndDate)
        if _, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary)
VALUES ('absence_request', $1, $2, $3, $4)`, id, c.Sub, check.ApproverID, "Отгул "+period(start, end)); err != nil {
                return nil, fmt.Errorf("insert approval item: %w", err)
        }
        created, _, err := findRequest(ctx, tx, "r.id = $1", id)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        created.Replayed = new(bool)
        return created, nil
}

// replay returns the stored request for a repeated key, or conflict when the key was used for other arguments.
func replay(prev request, c mcpx.Caller, in createIn) (any, error) {
        if prev.employeeID != c.Sub || prev.Type != in.Type || prev.StartDate != in.StartDate ||
                prev.EndDate != in.EndDate || prev.Comment != in.Comment {
                return nil, keyConflict()
        }
        replayed := true
        prev.Replayed = &replayed
        return prev, nil
}

func keyConflict() *apierr.Error {
        return apierr.Conflict().WithMessage("Ключ идемпотентности уже использован для другой заявки, подтвердите действие ещё раз")
}

func isUniqueViolation(err error) bool {
        var pgErr *pgconn.PgError
        return errors.As(err, &pgErr) && pgErr.Code == "23505"
}

// --- formatting ---

// ru formats a date as DD.MM.YYYY.
func ru(t time.Time) string { return t.Format("02.01.2006") }

// period formats a date range: one date for a single day, otherwise "start–end".
func period(start, end time.Time) string {
        if start.Equal(end) {
                return ru(start)
        }
        return ru(start) + "–" + ru(end)
}

// plural picks the Russian plural form for n: 1 день, 2 дня, 5 дней.
func plural(n int, one, few, many string) string {
        switch n10, n100 := n%10, n%100; {
        case n10 == 1 && n100 != 11:
                return one
        case n10 >= 2 && n10 <= 4 && (n100 < 12 || n100 > 14):
                return few
        default:
                return many
        }
}

func deref(s *string) string {
        if s == nil {
                return ""
        }
        return *s
}

// latestStatus is a LATERAL subquery: the caller's ($1) most recent request for resource r.
const latestStatus = `
LATERAL (SELECT q.status FROM access.requests q
         WHERE q.resource_id = r.id AND q.employee_id = $1
         ORDER BY q.created_at DESC, q.id DESC LIMIT 1) q`

// resourceStatus is a resource with the caller's status for it.

// listRequired returns the tools and spaces required for the caller's team with the caller's status.

// querier is satisfied by pgx.Tx and *pgxpool.Pool.
type querier interface {
        Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)

// findRequest returns the access request matching where (a condition on alias q with parameters $1, $2), or ok=false.
func findRequest(ctx context.Context, db querier, where string, args ...any) (requestOut, bool, error) {
        rows, err := db.Query(ctx, `
SELECT q.id, q.resource_id, r.name, q.status, q.approver_id, r.checklist_item_id, q.created_at, q.employee_id,
       coalesce(q.reason, '')
FROM access.requests q JOIN access.resources r ON r.id = q.resource_id
WHERE `+where+`
ORDER BY q.created_at DESC, q.id DESC LIMIT 1`, args...)
        if err != nil {
                return requestOut{}, false, fmt.Errorf("query access request: %w", err)
        }
        out, err := pgx.CollectOneRow(rows, func(row pgx.CollectableRow) (requestOut, error) {
                var (
                        o       requestOut
                        created time.Time
                )
                err := row.Scan(&o.ID, &o.ResourceID, &o.ResourceName, &o.Status, &o.ApproverID, &o.ChecklistItemID, &created,
                        &o.employeeID, &o.reason)
                o.CreatedAt = created.UTC().Format(time.RFC3339)
                return o, err
        })
        if errors.Is(err, pgx.ErrNoRows) {
                return requestOut{}, false, nil
        }
        if err != nil {
                return requestOut{}, false, fmt.Errorf("scan access request: %w", err)
        }
        return out, true, nil
}

// requestAccess creates an access request for the caller. A resource outside the caller's team is forbidden;
// a plain resource is granted at once; an elevated one waits for the caller's lead (approvals item). A repeated
// idempotency_key returns the stored request with replayed=true.
func (s *service) requestAccess(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in requestIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()
        // Serialise the caller's writes so that the replay lookup and the duplicate check see each other's commits.
        if _, err := tx.Exec(ctx, `SELECT pg_advisory_xact_lock(hashtextextended($1::text, 0))`, "access:"+c.Sub); err != nil {
                return nil, fmt.Errorf("lock employee: %w", err)
        }
        prev, found, err := findRequest(ctx, tx, "q.idempotency_key = $1::uuid", in.IdempotencyKey)
        if err != nil {
                return nil, err
        }
        if found {
                if prev.employeeID != c.Sub || prev.ResourceID != in.ResourceID || prev.reason != in.Reason {
                        return nil, keyConflict()
                }
                prev.Replayed = true
                return prev, nil
        }

        var (
                name     string
                elevated bool
        )
        err = tx.QueryRow(ctx, `
SELECT name, elevated FROM access.resources WHERE id = $1 AND (team_id IS NULL OR team_id = $2)`,
                in.ResourceID, c.TeamID).Scan(&name, &elevated)
        if errors.Is(err, pgx.ErrNoRows) {
                // Unknown resources and other teams' resources look the same (R71).
                return nil, apierr.Forbidden().WithMessage("Ресурс недоступен для запроса из вашей команды")
        }
        if err != nil {
                return nil, fmt.Errorf("load resource: %w", err)
        }
        last, found, err := findRequest(ctx, tx, "q.employee_id = $1 AND q.resource_id = $2", c.Sub, in.ResourceID)
        if err != nil {
                return nil, err
        }
        if found && last.Status == "granted" {
                return nil, apierr.Conflict().WithMessage(fmt.Sprintf("Доступ к ресурсу «%s» уже выдан (заявка %s)", name, last.ID))
        }
        if found && last.Status == "pending_approval" {
                return nil, apierr.Conflict().WithMessage(
                        fmt.Sprintf("Заявка на доступ к ресурсу «%s» уже на согласовании (%s)", name, last.ID))
        }

        var lead *string
        err = tx.QueryRow(ctx, `SELECT lead_id FROM directory.employees WHERE id = $1`, c.Sub).Scan(&lead)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.Forbidden().WithMessage("Сотрудник не найден в справочнике")
        }
        if err != nil {
                return nil, fmt.Errorf("load employee: %w", err)
        }
        status, approver := "granted", (*string)(nil)
        if elevated {
                if lead == nil {
                        return nil, apierr.Validation().WithMessage(
                                "Не удалось определить согласующего: в справочнике не указан руководитель. Обратитесь к наставнику")
                }
                status, approver = "pending_approval", lead
        }
        var id string
        err = tx.QueryRow(ctx, `
INSERT INTO access.requests (employee_id, resource_id, reason, status, approver_id, idempotency_key, decided_at)
VALUES ($1, $2, nullif($3, ''), $4, $5, $6::uuid, CASE WHEN $4 = 'granted' THEN now() END)
RETURNING id`, c.Sub, in.ResourceID, in.Reason, status, approver, in.IdempotencyKey).Scan(&id)
        if isUniqueViolation(err) {
                return nil, keyConflict() // the key belongs to another employee's request
        }
        if err != nil {
                return nil, fmt.Errorf("insert access request: %w", err)
        }
        if elevated {
                if _, err := tx.Exec(ctx, `
INSERT INTO approvals.items (kind, ref_id, requester_id, approver_id, summary)
VALUES ('access_request', $1, $2, $3, $4)`, id, c.Sub, *approver, "Доступ: "+name); err != nil {
                        return nil, fmt.Errorf("insert approval item: %w", err)
                }
        }
        created, _, err := findRequest(ctx, tx, "q.id = $1", id)
        if err != nil {
                return nil, err
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        return created, nil
}

func keyConflict() *apierr.Error {
        return apierr.Conflict().WithMessage("Ключ идемпотентности уже использован для другой заявки, подтвердите действие ещё раз")
}

func isUniqueViolation(err error) bool {
        var pgErr *pgconn.PgError
        return errors.As(err, &pgErr) && pgErr.Code == "23505"
}

// Config holds mcp-mocks settings (docs/backend/IMPLEMENTATION.md §7).

// LoadConfig reads configuration from getenv; Validate reports missing secrets.
func LoadConfig(getenv func(string) string) Config {
        env := config.New(getenv)
        return Config{
                Port:               env.Int("MCP_MOCKS_PORT", 8100),
                PostgresURL:        env.String("POSTGRES_URL", "postgres://buddy:buddy@postgres:5432/buddy?sslmode=disable"),
                JWTSecret:          env.String("JWT_SECRET", ""),

        }

// Validate fails when a required secret is missing: the service must not start with empty secrets.
func (c Config) Validate() error {
        var missing []string
        if c.JWTSecret == "" {
                missing = append(missing, "JWT_SECRET")
        }
        if c.ConfirmationSecret == "" {
                missing = append(missing, "CONFIRMATION_HMAC_SECRET")
        }
        if len(missing) > 0 {
                return fmt.Errorf("missing required environment variables: %v", missing)
        }
        return nil
}

// registrars lists every mock server and its registration function, in a stable order.

        register func(*mcpx.Server, *pgxpool.Pool) error
}{

        {"approvals", approvals.Register},
        {"access", access.Register},
        {"directory", directory.Register},

// NewHandler builds the HTTP handler: /healthz plus /mcp/<server> for all six servers.
func NewHandler(cfg Config, pool *pgxpool.Pool, log *slog.Logger) (http.Handler, error) {
        if err := cfg.Validate(); err != nil {
                return nil, err
        }
        mux := http.NewServeMux()
        mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, r *http.Request) {
                ctx, cancel := context.WithTimeout(r.Context(), 2*time.Second)
                defer cancel()
                if err := pool.Ping(ctx); err != nil {
                        httpx.WriteJSON(w, http.StatusServiceUnavailable, map[string]string{"status": "degraded", "service": "mcp-mocks"})
                        return
                }
                httpx.WriteJSON(w, http.StatusOK, map[string]string{"status": "ok", "service": "mcp-mocks"})
        })
        mcfg := mcpx.Config{JWTSecret: []byte(cfg.JWTSecret), ConfirmationSecret: []byte(cfg.ConfirmationSecret), Logger: log}
        servers := make([]*mcpx.Server, 0, len(registrars))
        for _, r := range registrars {
                srv, err := mcpx.NewServer(r.name, mcfg)
                if err != nil {
                        return nil, fmt.Errorf("mcp server %s: %w", r.name, err)
                }
                if err := r.register(srv, pool); err != nil {
                        return nil, fmt.Errorf("register %s: %w", r.name, err)
                }
                servers = append(servers, srv)
        }
        if err := mcpx.Mount(mux, servers...); err != nil {
                return nil, err
        }
        return httpx.Chain(httpx.RequestID, httpx.Recover(log), httpx.AccessLog(log))(mux), nil
}

// Run connects to Postgres, loads the knowledge base from KB_DIR and serves until ctx is cancelled.
func Run(ctx context.Context, cfg Config, log *slog.Logger) error {
        if err := cfg.Validate(); err != nil {
                return err
        }
        pool, err := pg.Open(ctx, cfg.PostgresURL, pg.Options{})
        if err != nil {
                return err
        }
        defer pool.Close()
        n, err := kb.LoadDir(ctx, pool, cfg.KBDir)
        if err != nil {
                return fmt.Errorf("load knowledge base from %s: %w", cfg.KBDir, err)
        }
        log.Info("knowledge base loaded", "dir", cfg.KBDir, "documents", n)
        h, err := NewHandler(cfg, pool, log)
        if err != nil {
                return err
        }
        return serve(ctx, fmt.Sprintf(":%d", cfg.Port), h, log)
}

func serve(ctx context.Context, addr string, h http.Handler, log *slog.Logger) error {
        srv := &http.Server{Addr: addr, Handler: h, ReadHeaderTimeout: 5 * time.Second}
        errc := make(chan error, 1)
        go func() { errc <- srv.ListenAndServe() }()
        log.Info("listening", "addr", addr)
        select {
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
                defer cancel()
                return srv.Shutdown(shutdownCtx)
        case err := <-errc:
                if errors.Is(err, http.ErrServerClosed) {
                        return nil
                }
                return err
        }
}

// Healthcheck performs GET url and fails on a non-200 answer (Docker HEALTHCHECK via `-healthcheck`).
func Healthcheck(url string) error {
        client := &http.Client{Timeout: 3 * time.Second}
        resp, err := client.Get(url)
        if err != nil {
                return err
        }
        defer resp.Body.Close()
        if resp.StatusCode != http.StatusOK {
                return fmt.Errorf("healthcheck: status %d", resp.StatusCode)
        }
        return nil
}

type service struct {

// item is one approval item as returned by the tools.
type item struct {
        ID                   string `json:"id"`
        Kind                 string `json:"kind"`
        RefID                string `json:"ref_id"`
        RequesterID          string `json:"requester_id"`
        RequesterDisplayName string `json:"requester_display_name,omitempty"`
        Summary              string `json:"summary"`
        Status               string `json:"status"`
        Comment              string `json:"comment,omitempty"`
        CreatedAt            string `json:"created_at"`
        Replayed             *bool  `json:"replayed,omitempty"`

        approverID     string
        idempotencyKey string
}

const selectItem = `
SELECT i.id, i.kind, i.ref_id, i.requester_id, coalesce(e.display_name, ''), i.summary, i.status,
       coalesce(i.comment, ''), i.created_at, i.approver_id, coalesce(i.idempotency_key::text, '')
FROM approvals.items i
LEFT JOIN directory.employees e ON e.id = i.requester_id`

func scanItem(row pgx.CollectableRow) (item, error) {
        var (
                it      item
                created time.Time
        )
        err := row.Scan(&it.ID, &it.Kind, &it.RefID, &it.RequesterID, &it.RequesterDisplayName, &it.Summary, &it.Status,
                &it.Comment, &created, &it.approverID, &it.idempotencyKey)
        it.CreatedAt = created.UTC().Format(time.RFC3339)
        return it, err
}

type listOut struct {
        Items []item `json:"items"`
}

// listPending returns the pending items whose approver is the caller, oldest first.
func (s *service) listPending(ctx context.Context, c mcpx.Caller, _ json.RawMessage) (any, error) {
        rows, err := s.pool.Query(ctx, selectItem+`
WHERE i.approver_id = $1 AND i.status = 'pending'
ORDER BY i.created_at, i.id
LIMIT 200`, c.Sub)
        if err != nil {
                return nil, fmt.Errorf("list pending approvals: %w", err)
        }
        items, err := pgx.CollectRows(rows, scanItem)
        if err != nil {
                return nil, fmt.Errorf("scan pending approvals: %w", err)
        }
        if items == nil {
                items = []item{}
        }
        return listOut{Items: items}, nil
}

type decideIn struct {
        ApprovalID     string `json:"approval_id"`
        Decision       string `json:"decision"`
        Comment        string `json:"comment"`
        IdempotencyKey string `json:"idempotency_key"`
}

// underlying maps an item kind and decision to the request table and its new status.
var underlying = map[string]struct {
        table    string
        statuses map[string]string
}{
        "absence_request": {"absence.requests", map[string]string{"approved": "approved", "rejected": "rejected"}},
        "access_request":  {"access.requests", map[string]string{"approved": "granted", "rejected": "rejected"}},
}

// decide approves or rejects one of the caller's pending items and propagates the decision to the request.
func (s *service) decide(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
        var in decideIn
        if err := mcpx.DecodeArgs(args, &in); err != nil {
                return nil, err
        }
        tx, err := s.pool.Begin(ctx)
        if err != nil {
                return nil, fmt.Errorf("begin: %w", err)
        }
        defer func() { _ = tx.Rollback(context.WithoutCancel(ctx)) }()

        rows, err := tx.Query(ctx, selectItem+" WHERE i.id = $1 FOR UPDATE OF i", in.ApprovalID)
        if err != nil {
                return nil, fmt.Errorf("query approval item: %w", err)
        }
        it, err := pgx.CollectOneRow(rows, scanItem)
        if errors.Is(err, pgx.ErrNoRows) || (err == nil && it.approverID != c.Sub) {
                // Other teams' items and unknown ids look the same (R71).
                return nil, apierr.Forbidden().WithMessage("Запрос недоступен: решение принимает только назначенный согласующий")
        }
        if err != nil {
                return nil, fmt.Errorf("scan approval item: %w", err)
        }
        if it.idempotencyKey != "" && it.idempotencyKey == in.IdempotencyKey {
                if it.Status != in.Decision || it.Comment != in.Comment {
                        return nil, keyConflict()
                }
                replayed := true
                it.Replayed = &replayed
                return it, nil
        }
        if it.Status != "pending" {
                return nil, apierr.Conflict().WithMessage("Запрос уже рассмотрен, решение изменить нельзя")
        }

        _, err = tx.Exec(ctx, `
UPDATE approvals.items SET status = $2, comment = nullif($3, ''), idempotency_key = $4::uuid, decided_at = now()
WHERE id = $1`, it.ID, in.Decision, in.Comment, in.IdempotencyKey)
        if isUniqueViolation(err) {
                return nil, keyConflict() // the key was already used for another decision
        }
        if err != nil {
                return nil, fmt.Errorf("update approval item: %w", err)
        }
        // kind is limited by a CHECK constraint to the keys of underlying; the table name never comes from input.
        target := underlying[it.Kind]
        tag, err := tx.Exec(ctx, `UPDATE `+target.table+` SET status = $2, decided_at = now()
WHERE id = $1 AND status = 'pending_approval'`, it.RefID, target.statuses[in.Decision])
        if err != nil {
                return nil, fmt.Errorf("update %s: %w", target.table, err)
        }
        if tag.RowsAffected() != 1 {
                return nil, apierr.Conflict().WithMessage("Заявка больше не ожидает согласования")
        }
        if err := tx.Commit(ctx); err != nil {
                return nil, fmt.Errorf("commit: %w", err)
        }
        it.Status, it.Comment, it.Replayed = in.Decision, in.Comment, new(bool)
        return it, nil
}

func keyConflict() *apierr.Error {
        return apierr.Conflict().WithMessage("Ключ идемпотентности уже использован для другого решения, повторите действие")
}

func isUniqueViolation(err error) bool {
        var pgErr *pgconn.PgError
        return errors.As(err, &pgErr) && pgErr.Code == "23505"
}

type periodOut struct {
        ID    string `json:"id"`
        Label string `json:"label"`
}

type itemOut struct {
        ID          string          `json:"id"`
        Role        string          `json:"role"`
        Period      string          `json:"period"`
        Title       string          `json:"title"`
        Description string          `json:"description,omitempty"`
        Steps       json.RawMessage `json:"steps"`
        Scenarios   []string        `json:"scenarios"`
        HumanOnly   bool            `json:"human_only"`
}

func template(ctx context.Context, pool *pgxpool.Pool, role string) (any, error) {
        rows, err := pool.Query(ctx, `
                SELECT id, role, period, title, coalesce(description, ''), steps, scenarios, human_only
                FROM checklist.items WHERE role IN ($1, 'both') ORDER BY position, id`, role)
        if err != nil {
                return nil, fmt.Errorf("loading checklist items: %w", err)
        }
        defer rows.Close()
        items := []itemOut{}
        for rows.Next() {
                var it itemOut
                var steps []byte
                if err := rows.Scan(&it.ID, &it.Role, &it.Period, &it.Title, &it.Description, &steps, &it.Scenarios, &it.HumanOnly); err != nil {
                        return nil, fmt.Errorf("scanning checklist item: %w", err)
                }
                if len(steps) == 0 || string(steps) == "null" {
                        steps = []byte("[]")
                }
                it.Steps = steps
                if it.Scenarios == nil {
                        it.Scenarios = []string{}
                }
                items = append(items, it)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("reading checklist items: %w", err)
        }
        periods := []periodOut{}
        for _, p := range period.All() {
                periods = append(periods, periodOut{ID: string(p), Label: period.Label(p)})
        }
        return map[string]any{"periods": periods, "items": items}, nil
}

func whoami(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller, now time.Time) (any, error) {
        var (
                out   whoamiOut
                start *time.Time
        )
        err := pool.QueryRow(ctx, `
                SELECT e.id, e.display_name, e.role, e.team_id, coalesce(t.name, ''), e.cohort_start
                FROM directory.employees e LEFT JOIN directory.teams t ON t.id = e.team_id
                WHERE e.id = $1`, c.Sub).Scan(&out.EmployeeID, &out.DisplayName, &out.Role, &out.TeamID, &out.TeamName, &start)
        if errors.Is(err, pgx.ErrNoRows) {
                return nil, apierr.Forbidden()
        }
        if err != nil {
                return nil, fmt.Errorf("loading employee: %w", err)
        }
        var cohort time.Time
        switch {
        case start != nil:
                cohort = *start
        case c.CohortStart != "":
                if d, perr := period.ParseDate(c.CohortStart); perr == nil {
                        cohort = d
                }
        }
        if !cohort.IsZero() {
                out.CohortStart = cohort.Format("2006-01-02")
                out.CurrentPeriod = string(period.Current(cohort, now))
        }
        return out, nil
}

func contacts(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller) (any, error) {
        rows, err := pool.Query(ctx, `
                SELECT r.role, p.id, p.display_name, coalesce(p.channel, '')
                FROM directory.employees e
                CROSS JOIN LATERAL (VALUES (1, 'mentor', e.mentor_id), (2, 'lead', e.lead_id),
                        (3, 'hrbp', e.hrbp_id), (4, 'curator', e.curator_id)) AS r(ord, role, pid)
                JOIN directory.employees p ON p.id = r.pid
                WHERE e.id = $1
                ORDER BY r.ord`, c.Sub)
        if err != nil {
                return nil, fmt.Errorf("loading contacts: %w", err)
        }
        defer rows.Close()
        items := []contact{}
        for rows.Next() {
                var ct contact
                if err := rows.Scan(&ct.Role, &ct.EmployeeID, &ct.DisplayName, &ct.Channel); err != nil {
                        return nil, fmt.Errorf("scanning contact: %w", err)
                }
                ct.AskAbout = askAbout[ct.Role]
                items = append(items, ct)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("reading contacts: %w", err)
        }
        return map[string]any{"items": items}, nil
}

// Package kb implements the mock "kb" MCP server: the synthetic knowledge base with ACL applied before anything
// reaches the LLM (R71). Every document is a test instruction of the prototype (R69).
package kb

import (
        "context"
        "crypto/rand"
        "encoding/hex"
        "encoding/json"
        "errors"
        "fmt"

        "github.com/jackc/pgx/v5"

        "aibuddy/services/internal/platform/apierr"
        "aibuddy/services/internal/platform/mcpx"
)

const (
        defaultK   = 5
        maxSnippet = 300
)

// Register binds every tool of the kb contract to srv.
func Register(srv *mcpx.Server, pool *pgxpool.Pool) error {
        if srv == nil || pool == nil {
                return errors.New("kb: server and pool are required")
        }
        srv.Handle("search", func(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
                var in struct {
                        Query string `json:"query"`
                        K     int    `json:"k"`
                }
                if err := mcpx.DecodeArgs(args, &in); err != nil {
                        return nil, err
                }
                if in.K <= 0 {
                        in.K = defaultK
                }
                return search(ctx, pool, c, in.Query, in.K)
        })
        srv.Handle("add_faq", func(ctx context.Context, c mcpx.Caller, args json.RawMessage) (any, error) {
                var in struct {
                        Title string `json:"title"`
                        Body  string `json:"body"`
                        Key   string `json:"idempotency_key"`
                }
                if err := mcpx.DecodeArgs(args, &in); err != nil {
                        return nil, err
                }
                return addFAQ(ctx, pool, c, in.Title, in.Body, in.Key)
        })
        return nil
}

type hit struct {
        DocID      string  `json:"doc_id"`
        Title      string  `json:"title"`
        Snippet    string  `json:"snippet"`
        SourceType string  `json:"source_type"`
        URL        string  `json:"url"`
        Score      float64 `json:"score"`
}

// search returns documents the caller may see: role and team ACL, and private notes of the caller only.
func search(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller, query string, k int) (any, error) {
        rows, err := pool.Query(ctx, `
                SELECT d.doc_id, d.title, d.source_type, coalesce(d.url, ''),
                       ts_headline('russian', d.body, q.q, 'StartSel="", StopSel="", MinWords=12, MaxWords=35, ShortWord=2'),
                       ts_rank(d.tsv, q.q)::float8
                FROM kb.documents d, websearch_to_tsquery('russian', $1) AS q(q)
                WHERE d.tsv @@ q.q
                  AND $2 = ANY(d.acl_roles)
                  AND ('*' = ANY(d.acl_teams) OR $3 = ANY(d.acl_teams))
                  AND (d.owner_id IS NULL OR d.owner_id = $4)
                ORDER BY 6 DESC, d.doc_id
                LIMIT $5`, query, c.Role, c.TeamID, c.Sub, k)
        if err != nil {
                return nil, fmt.Errorf("searching knowledge base: %w", err)
        }
        defer rows.Close()
        items := []hit{}
        for rows.Next() {
                var h hit
                if err := rows.Scan(&h.DocID, &h.Title, &h.SourceType, &h.URL, &h.Snippet, &h.Score); err != nil {
                        return nil, fmt.Errorf("scanning search hit: %w", err)
                }
                h.Snippet = truncateRunes(h.Snippet, maxSnippet)
                items = append(items, h)
        }
        if err := rows.Err(); err != nil {
                return nil, fmt.Errorf("reading search hits: %w", err)
        }
        return map[string]any{"items": items}, nil
}

func truncateRunes(s string, n int) string {
        r := []rune(s)
        if len(r) <= n {
                return s
        }
        return string(r[:n-1]) + "…"
}

type faqOut struct {
        DocID      string `json:"doc_id"`
        Title      string `json:"title"`
        SourceType string `json:"source_type"`
        Replayed   bool   `json:"replayed"`
}

// addFAQ stores a mentor answer as an FAQ document of the mentor's team; idempotent by key (R79).
func addFAQ(ctx context.Context, pool *pgxpool.Pool, c mcpx.Caller, title, body, key string) (any, error) {
        suffix := make([]byte, 4)
        if _, err := rand.Read(suffix); err != nil {
                return nil, fmt.Errorf("generating doc id: %w", err)
        }
        docID := fmt.Sprintf("faq-%s-%s", c.TeamID, hex.EncodeToString(suffix))
        var out faqOut
        err := pool.QueryRow(ctx, `
                INSERT INTO kb.documents (doc_id, title, body, source_type, acl_roles, acl_teams, kind, created_by, idempotency_key)
                VALUES ($1, $2, $3, 'test', ARRAY['intern','mentor','lead'], ARRAY[$4::text], 'faq', $5, $6)
                ON CONFLICT (idempotency_key) DO NOTHING
                RETURNING doc_id, title, source_type`, docID, title, body, c.TeamID, c.Sub, key).
                Scan(&out.DocID, &out.Title, &out.SourceType)
        if err == nil {
                return out, nil
        }
        if !errors.Is(err, pgx.ErrNoRows) {
                return nil, fmt.Errorf("inserting faq: %w", err)
        }
        // The key was used before: replay the stored result when it is the same request, otherwise conflict.
        var createdBy, storedBody string
        err = pool.QueryRow(ctx, `SELECT doc_id, title, body, coalesce(created_by, '') FROM kb.documents WHERE idempotency_key = $1`, key).
                Scan(&out.DocID, &out.Title, &storedBody, &createdBy)
        if err != nil {
                return nil, fmt.Errorf("loading stored faq: %w", err)
        }
        if createdBy != c.Sub || out.Title != title || storedBody != body {
                return nil, apierr.Conflict().WithMessage("Этот ключ идемпотентности уже использован с другими данными")
        }
        out.SourceType = "test"
        out.Replayed = true
        return out, nil
}

package kb

import (
        "bytes"
        "context"
        "errors"
        "fmt"
        "os"
        "path/filepath"
        "regexp"
        "slices"
        "strings"
        "time"

        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgxpool"
        "gopkg.in/yaml.v3"
)

var (
        docIDRe  = regexp.MustCompile(`^[a-z0-9][a-z0-9_-]*$`)
        itemIDRe = regexp.MustCompile(`^CL-[0-9]{2}$`)
        known    = []string{"intern", "mentor", "lead", "hrbp", "alumni"}
)

type frontmatter struct {
        DocID  string `yaml:"doc_id"`
        Title  string `yaml:"title"`
        Source string `yaml:"source_type"`
        ACL    struct {
                Roles []string `yaml:"roles"`
                Teams []string `yaml:"teams"`
        } `yaml:"acl"`
        ChecklistItemID string    `yaml:"checklist_item_id"`
        URL             string    `yaml:"url"`
        Updated         yaml.Node `yaml:"updated"` // any scalar; only a YYYY-MM-DD date is used
}

type document struct {
        frontmatter
        body    string
        updated *time.Time
}

// LoadDir upserts every *.md file of dir (except README.md) into kb.documents (kind doc, no owner) and returns how
// many were loaded. Files need YAML frontmatter with doc_id, title, source_type: test (R69) and acl {roles, teams};
// checklist_item_id is optional; other frontmatter fields (checklist, tags, updated, origin, ...) are ignored, except
// that a valid YYYY-MM-DD `updated` becomes the document date. Any invalid file fails the whole load with an error
// naming it, and nothing is written. Other files are ignored. A missing dir returns 0 and no error.
func LoadDir(ctx context.Context, pool *pgxpool.Pool, dir string) (int, error) {
        entries, err := os.ReadDir(dir)
        if errors.Is(err, os.ErrNotExist) {
                return 0, nil
        }
        if err != nil {
                return 0, fmt.Errorf("reading kb dir %s: %w", dir, err)
        }
        var docs []document
        seen := map[string]string{}
        for _, e := range entries {
                if e.IsDir() || !strings.EqualFold(filepath.Ext(e.Name()), ".md") || strings.EqualFold(e.Name(), "README.md") {
                        continue
                }
                raw, err := os.ReadFile(filepath.Join(dir, e.Name()))
                if err != nil {
                        return 0, fmt.Errorf("reading %s: %w", e.Name(), err)
                }
                d, err := parseDoc(raw)
                if err != nil {
                        return 0, fmt.Errorf("kb file %s: %w", e.Name(), err)
                }
                if prev, dup := seen[d.DocID]; dup {
                        return 0, fmt.Errorf("kb file %s: doc_id %q already used by %s", e.Name(), d.DocID, prev)
                }
                seen[d.DocID] = e.Name()
                docs = append(docs, d)
        }
        if len(docs) == 0 {
                return 0, nil
        }
        tx, err := pool.Begin(ctx)
        if err != nil {
                return 0, fmt.Errorf("starting kb load: %w", err)
        }
        defer func() { _ = tx.Rollback(ctx) }()
        for _, d := range docs {
                if err := upsert(ctx, tx, d); err != nil {
                        return 0, fmt.Errorf("kb file %s: %w", seen[d.DocID], err)
                }
        }
        if err := tx.Commit(ctx); err != nil {
                return 0, fmt.Errorf("committing kb load: %w", err)
        }
        return len(docs), nil
}

func upsert(ctx context.Context, tx pgx.Tx, d document) error {
        tag, err := tx.Exec(ctx, `
                INSERT INTO kb.documents (doc_id, title, body, source_type, acl_roles, acl_teams, kind, checklist_item_id, url, updated)
                VALUES ($1, $2, $3, 'test', $4, $5, 'doc', nullif($6, ''), nullif($7, ''), coalesce($8::date, current_date))
                ON CONFLICT (doc_id) DO UPDATE SET title = EXCLUDED.title, body = EXCLUDED.body,
                        acl_roles = EXCLUDED.acl_roles, acl_teams = EXCLUDED.acl_teams,
                        checklist_item_id = EXCLUDED.checklist_item_id, url = EXCLUDED.url, updated = EXCLUDED.updated
                WHERE kb.documents.kind = 'doc' AND kb.documents.owner_id IS NULL`,
                d.DocID, d.Title, d.body, d.ACL.Roles, d.ACL.Teams, d.ChecklistItemID, d.URL, d.updated)
        if err != nil {
                return fmt.Errorf("upserting %s: %w", d.DocID, err)
        }
        if tag.RowsAffected() == 0 {
                return fmt.Errorf("doc_id %q belongs to a non-document record", d.DocID)
        }
        return nil
}

func parseDoc(raw []byte) (document, error) {
        var d document
        raw = bytes.ReplaceAll(bytes.TrimPrefix(raw, []byte("\xef\xbb\xbf")), []byte("\r\n"), []byte("\n"))
        if !bytes.HasPrefix(raw, []byte("---\n")) {
                return d, errors.New("no YAML frontmatter (file must start with ---)")
        }
        rest := raw[4:]
        end := bytes.Index(rest, []byte("\n---"))
        if end < 0 {
                return d, errors.New("frontmatter is not closed with ---")
        }
        if err := yaml.Unmarshal(rest[:end], &d.frontmatter); err != nil {
                return d, fmt.Errorf("invalid frontmatter: %w", err)
        }
        d.body = strings.TrimSpace(string(rest[end+4:]))
        var missing []string
        for name, empty := range map[string]bool{
                "doc_id": d.DocID == "", "title": strings.TrimSpace(d.Title) == "", "source_type": d.Source == "",
                "acl.roles": len(d.ACL.Roles) == 0, "acl.teams": len(d.ACL.Teams) == 0, "body": d.body == "",
        } {
                if empty {
                        missing = append(missing, name)
                }
        }
        if len(missing) > 0 {
                slices.Sort(missing)
                return d, fmt.Errorf("missing required fields: %s", strings.Join(missing, ", "))
        }
        if d.Source != "test" {
                return d, fmt.Errorf("source_type must be \"test\", got %q (R69)", d.Source)
        }
        if !docIDRe.MatchString(d.DocID) {
                return d, fmt.Errorf("invalid doc_id %q", d.DocID)
        }
        for _, r := range d.ACL.Roles {
                if !slices.Contains(known, r) {
                        return d, fmt.Errorf("unknown role %q in acl.roles", r)
                }
        }
        if d.ChecklistItemID != "" && !itemIDRe.MatchString(d.ChecklistItemID) {
                return d, fmt.Errorf("invalid checklist_item_id %q", d.ChecklistItemID)
        }
        if d.Updated.Kind == yaml.ScalarNode {
                if t, err := time.Parse("2006-01-02", strings.TrimSpace(d.Updated.Value)); err == nil {
                        d.updated = &t
                }
        }
        return d, nil
}
